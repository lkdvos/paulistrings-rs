1//! [`SupportIndex`], a bit-sliced index of which terms of each bucket act on each qubit, so a Liouvillian visits only the pairs whose supports meet.
2
3use rayon::prelude::*;
4
5use crate::coefficient::Coefficient;
6use crate::pauli_sum::storage::{BucketColumns, PauliSum};
7
8/// Above this weight a term's candidate mask costs more to assemble than scanning the bucket.
9pub(crate) const SCAN_ABOVE_WEIGHT: u32 = 64;
10
11/// Per bucket, row `q` is a bitset over the bucket's terms with bit `i` set iff term `i` acts on qubit `q`.
12pub(crate) struct SupportIndex {
13    /// Bucket `b`'s rows, `num_qubits` of `words[b]` words each.
14    rows: Vec<Vec<u64>>,
15    words: Vec<usize>,
16}
17
18fn bucket_rows<const W: usize, C: Coefficient>(
19    columns: &BucketColumns<W, C>,
20    num_qubits: usize,
21) -> (Vec<u64>, usize) {
22    let words = columns.len().div_ceil(64);
23    let mut rows = vec![0u64; num_qubits * words];
24    for i in 0..columns.len() {
25        for w in 0..W {
26            let mut support = columns.x[i][w] | columns.z[i][w];
27            while support != 0 {
28                let q = w * 64 + support.trailing_zeros() as usize;
29                rows[q * words + i / 64] |= 1 << (i % 64);
30                support &= support - 1;
31            }
32        }
33    }
34    (rows, words)
35}
36
37impl SupportIndex {
38    pub(crate) fn new<const W: usize, C: Coefficient>(sum: &PauliSum<W, C>) -> Self {
39        let (rows, words) =
40            sum.buckets().par_iter().map(|columns| bucket_rows(columns, sum.num_qubits())).unzip();
41        Self { rows, words }
42    }
43
44    /// Overwrite `candidates` with the bitset of bucket `b`'s terms acting on some qubit of `support`.
45    #[inline]
46    pub(crate) fn candidates<const W: usize>(
47        &self,
48        b: usize,
49        support: &[u64; W],
50        candidates: &mut Vec<u64>,
51    ) {
52        let words = self.words[b];
53        let rows = &self.rows[b];
54        candidates.clear();
55        candidates.resize(words, 0);
56        for (w, &mask) in support.iter().enumerate() {
57            let mut mask = mask;
58            while mask != 0 {
59                let q = w * 64 + mask.trailing_zeros() as usize;
60                let row = &rows[q * words..(q + 1) * words];
61                for (c, r) in candidates.iter_mut().zip(row) {
62                    *c |= r;
63                }
64                mask &= mask - 1;
65            }
66        }
67    }
68}
