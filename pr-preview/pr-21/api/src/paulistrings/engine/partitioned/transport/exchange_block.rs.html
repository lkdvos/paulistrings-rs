1//! The exchange wire unit: [`ExchangeBlock`], one per remote delta, and the [`PartnerPayload`] that carries a partner's blocks.
2
3use std::mem::size_of;
4
5use num_complex::Complex64;
6
7use super::{ChunkMap, Payload};

9/// Fixed-size wire prefix of one [`ExchangeBlock`]: four `u32`s, no padding, so it casts to bytes.
10#[repr(C)]
11#[derive(Clone, Copy, Debug, Default, PartialEq, Eq, bytemuck::Pod, bytemuck::Zeroable)]
12pub(crate) struct BlockHeader {
13    /// Destination positions, the group's agreed bucket count; `offsets` has one more entry.
14    pub num_buckets: u32,
15    /// Live rows, `offsets[num_buckets]`.
16    pub rows: u32,
17    /// The width `W` the block was built at, checked on receive.
18    pub width: u32,
19    /// The layer plan's remote-delta index this block carries.
20    pub entry: u32,
21}
22
23/// The rows one remote delta moves to one partner, CSR-indexed by the receiver's destination position ([`ChunkMap`]).
24///
25/// The columns are grow-only and may be longer than `header.rows`, which alone says how much is live: read through `columns` or `segment`, never `x.len()`.
26#[derive(Clone, Debug, Default)]
27pub(crate) struct ExchangeBlock<const W: usize> {
28    pub header: BlockHeader,
29    /// CSR offsets by destination position, `num_buckets + 1` entries.
30    pub offsets: Vec<u32>,
31    pub x: Vec<[u64; W]>,
32    pub z: Vec<[u64; W]>,
33    pub coeff: Vec<Complex64>,
34}
35
36/// Compares the live rows only, so a reused block equals the same block built fresh.
37impl<const W: usize> PartialEq for ExchangeBlock<W> {
38    fn eq(&self, other: &Self) -> bool {
39        self.header == other.header
40            && self.offsets == other.offsets
41            && self.columns() == other.columns()
42    }
43}
44
45/// Everything one partner receives from this partition for one layer: one block per remote delta, ascending by [`BlockHeader::entry`].
46#[derive(Clone, Debug, Default, PartialEq)]
47pub(crate) struct PartnerPayload<const W: usize> {
48    pub blocks: Vec<ExchangeBlock<W>>,
49}
50
51impl<const W: usize> ExchangeBlock<W> {
52    /// `set_counts` on a fresh block.
53    #[cfg(test)]
54    pub fn with_counts(entry: u32, counts: &[u32]) -> Self {
55        let mut block = Self::default();
56        block.set_counts(entry, counts);
57        block
58    }
59
60    /// Re-aim the block at `counts[p]` rows per destination position under remote-delta index `entry`, growing but never shrinking or re-zeroing the columns.
61    pub(crate) fn set_counts(&mut self, entry: u32, counts: &[u32]) {
62        self.offsets.clear();
63        self.offsets.reserve(counts.len() + 1);
64        self.offsets.push(0u32);
65        let mut rows = 0u32;
66        for &count in counts {
67            rows = rows
68                .checked_add(count)
69                .expect("exchange block exceeds u32::MAX rows");
70            self.offsets.push(rows);
71        }
72        self.header = BlockHeader {
73            num_buckets: counts.len() as u32,
74            rows,
75            width: W as u32,
76            entry,
77        };
78        self.grow_columns(rows as usize);
79    }
80
81    fn grow_columns(&mut self, rows: usize) {
82        if self.coeff.len() < rows {
83            self.x.resize(rows, [0u64; W]);
84            self.z.resize(rows, [0u64; W]);
85            self.coeff.resize(rows, Complex64::new(0.0, 0.0));
86        }
87    }
88
89    /// The live rows of the three columns.
90    pub(crate) fn columns(&self) -> (&[[u64; W]], &[[u64; W]], &[Complex64]) {
91        let rows = self.rows();
92        (&self.x[..rows], &self.z[..rows], &self.coeff[..rows])
93    }
94
95    /// The rows destined for position `p`; output bucket `β′` reads `segment(map.position_of(β′))`.
96    pub(crate) fn segment(&self, p: u32) -> (&[[u64; W]], &[[u64; W]], &[Complex64]) {
97        let lo = self.offsets[p as usize] as usize;
98        let hi = self.offsets[p as usize + 1] as usize;
99        (&self.x[lo..hi], &self.z[lo..hi], &self.coeff[lo..hi])
100    }
101
102    /// Live rows.
103    pub(crate) fn rows(&self) -> usize {
104        self.header.rows as usize
105    }
106
107    /// Destination positions the block is indexed by.
108    #[cfg(any(test, debug_assertions))]
109    pub(crate) fn num_buckets(&self) -> u32 {
110        self.header.num_buckets
111    }
112
113    /// Wire footprint in bytes, exactly what `byte_parts` hands out.
114    pub(crate) fn bytes(&self) -> usize {
115        size_of::<BlockHeader>()
116            + self.offsets.len() * size_of::<u32>()
117            + 2 * self.rows() * W * size_of::<u64>()
118            + self.rows() * size_of::<Complex64>()
119    }
120}
121
122/// Parts per block on the wire: header, offsets, x, z, coeff.
123pub(super) const PARTS_PER_BLOCK: usize = 5;
124
125/// The CSR `offsets` read at `map`'s chunk boundaries, `chunks + 1` entries.
126pub(crate) fn chunk_rows_of(offsets: &[u32], map: &ChunkMap) -> Vec<usize> {
127    debug_assert_eq!(
128        offsets.len(),
129        map.positions() + 1,
130        "the chunk map is built for a different bucket count than the block",
131    );
132    (0..=map.chunks())
133        .map(|k| offsets[map.bound(k) as usize] as usize)
134        .collect()
135}
136
137/// Cut `col` at `bounds` (row indices) and view each piece as bytes.
138fn chunk_slices<'s, T, F>(col: &'s [T], bounds: &[usize], as_bytes: F) -> Vec<&'s [u8]>
139where
140    F: Fn(&'s [T]) -> &'s [u8],
141{
142    bounds
143        .windows(2)
144        .map(|w| as_bytes(&col[w[0]..w[1]]))
145        .collect()
146}
147
148/// [`chunk_slices`] for the receive side: disjoint mutable pieces, in order.
149fn chunk_slices_mut<'s, T, F>(col: &'s mut [T], bounds: &[usize], as_bytes: F) -> Vec<&'s mut [u8]>
150where
151    F: Fn(&'s mut [T]) -> &'s mut [u8],
152{
153    let mut rest = col;
154    let mut out = Vec::with_capacity(bounds.len().saturating_sub(1));
155    let mut at = bounds[0];
156    for &edge in &bounds[1..] {
157        let (head, tail) = rest.split_at_mut(edge - at);
158        out.push(as_bytes(head));
159        rest = tail;
160        at = edge;
161    }
162    out
163}
164
165/// Panic unless a declared part length is a whole number of `stride`-byte elements.
166pub(super) fn check_stride(len: usize, stride: usize, what: &str) {
167    assert_eq!(
168        len % stride,
169        0,
170        "exchange block {what}: {len} bytes is not a whole number of {stride}-byte entries",
171    );
172}

174impl<const W: usize> Payload for PartnerPayload<W> {
175    fn byte_parts(&self) -> Vec<&[u8]> {
176        let mut parts = Vec::with_capacity(PARTS_PER_BLOCK * self.blocks.len());
177        for block in &self.blocks {
178            let (x, z, coeff) = block.columns();
179            parts.push(bytemuck::bytes_of(&block.header));
180            parts.push(bytemuck::cast_slice(&block.offsets));
181            parts.push(bytemuck::cast_slice(x.as_flattened()));
182            parts.push(bytemuck::cast_slice(z.as_flattened()));
183            parts.push(bytemuck::cast_slice(coeff));
184        }
185        parts
186    }
187
188    fn receive_into(&mut self, lens: &[usize]) -> Vec<&mut [u8]> {
189        assert_eq!(
190            lens.len() % PARTS_PER_BLOCK,
191            0,
192            "partner payload: {} parts is not a whole number of {PARTS_PER_BLOCK}-part blocks",
193            lens.len(),
194        );
195        let block_count = lens.len() / PARTS_PER_BLOCK;
196        let key_stride = W * size_of::<u64>();
197        self.blocks.truncate(block_count);
198        if self.blocks.len() < block_count {
199            self.blocks
200                .resize_with(block_count, ExchangeBlock::<W>::default);
201        }
202        let mut parts = Vec::with_capacity(lens.len());
203        for (block, lens) in self
204            .blocks
205            .iter_mut()
206            .zip(lens.chunks_exact(PARTS_PER_BLOCK))
207        {
208            assert_eq!(
209                lens[0],
210                size_of::<BlockHeader>(),
211                "partner payload: block header is {} bytes, expected {}",
212                lens[0],
213                size_of::<BlockHeader>(),
214            );
215            check_stride(lens[1], size_of::<u32>(), "offsets");
216            check_stride(lens[2], key_stride, "x column");
217            check_stride(lens[3], key_stride, "z column");
218            check_stride(lens[4], size_of::<Complex64>(), "coeff column");
219            assert_eq!(
220                lens[2], lens[3],
221                "partner payload: x column is {} bytes and z column {}",
222                lens[2], lens[3],
223            );
224            let rows = lens[2] / key_stride;
225            assert_eq!(
226                lens[4] / size_of::<Complex64>(),
227                rows,
228                "partner payload: coeff column carries {} rows where the keys carry {rows}",
229                lens[4] / size_of::<Complex64>(),
230            );
231            block.offsets.clear();
232            block.offsets.resize(lens[1] / size_of::<u32>(), 0);
233            block.grow_columns(rows);
234            // Overwritten by the arriving header, which `finish_receive` checks against the columns.
235            block.header.rows = rows as u32;
236            let ExchangeBlock {
237                header,
238                offsets,
239                x,
240                z,
241                coeff,
242            } = block;
243            parts.push(bytemuck::bytes_of_mut(header));
244            parts.push(bytemuck::cast_slice_mut(&mut offsets[..]));
245            parts.push(bytemuck::cast_slice_mut(x[..rows].as_flattened_mut()));
246            parts.push(bytemuck::cast_slice_mut(z[..rows].as_flattened_mut()));
247            parts.push(bytemuck::cast_slice_mut(&mut coeff[..rows]));
248        }
249        parts
250    }
251
252    fn finish_receive(&mut self) {
253        for block in &self.blocks {
254            assert_eq!(
255                block.header.width as usize, W,
256                "partner payload: block encoded at width W={} decoded at W={W}",
257                block.header.width,
258            );
259            // A garbled header must fail here rather than as an out-of-bounds read.
260            assert!(
261                block.header.rows as usize <= block.coeff.len(),
262                "partner payload: a block header claims {} rows but only {} arrived",
263                block.header.rows,
264                block.coeff.len(),
265            );
266            assert_eq!(
267                block.offsets.len(),
268                block.header.num_buckets as usize + 1,
269                "partner payload: a block indexed by {} buckets arrived with {} offsets",
270                block.header.num_buckets,
271                block.offsets.len(),
272            );
273            assert_eq!(
274                block.offsets.last().copied().unwrap_or(0),
275                block.header.rows,
276                "partner payload: offsets end at {:?}, the columns carry {} rows",
277                block.offsets.last(),
278                block.header.rows,
279            );
280        }
281    }
282
283    /// The header and the CSR offsets of every block.
284    fn early_parts(&self) -> Vec<&[u8]> {
285        let mut parts = Vec::with_capacity(2 * self.blocks.len());
286        for block in &self.blocks {
287            parts.push(bytemuck::bytes_of(&block.header));
288            parts.push(bytemuck::cast_slice(&block.offsets));
289        }
290        parts
291    }
292
293    /// The three columns of each block, each cut at the chunk boundaries.
294    fn bulk_parts(&self, map: &ChunkMap) -> Vec<Vec<&[u8]>> {
295        let mut parts = Vec::with_capacity(3 * self.blocks.len());
296        for block in &self.blocks {
297            let rows = block.rows();
298            let (x, z, coeff) = (&block.x[..rows], &block.z[..rows], &block.coeff[..rows]);
299            let bounds = chunk_rows_of(&block.offsets, map);
300            parts.push(chunk_slices(x, &bounds, |s| {
301                bytemuck::cast_slice(s.as_flattened())
302            }));
303            parts.push(chunk_slices(z, &bounds, |s| {
304                bytemuck::cast_slice(s.as_flattened())
305            }));
306            parts.push(chunk_slices(coeff, &bounds, bytemuck::cast_slice));
307        }
308        parts
309    }
310
311    fn early_receive_into(&mut self, lens: &[usize]) -> Vec<&mut [u8]> {
312        let mut parts = self.receive_into(lens);
313        let mut early = Vec::with_capacity(2 * parts.len() / PARTS_PER_BLOCK);
314        for (i, view) in parts.drain(..).enumerate() {
315            if i % PARTS_PER_BLOCK < 2 {
316                early.push(view);
317            }
318        }
319        early
320    }
321
322    fn bulk_receive_into(&mut self, map: &ChunkMap) -> Vec<Vec<&mut [u8]>> {
323        let mut parts = Vec::with_capacity(3 * self.blocks.len());
324        for block in &mut self.blocks {
325            let rows = block.header.rows as usize;
326            let bounds = chunk_rows_of(&block.offsets, map);
327            let ExchangeBlock { x, z, coeff, .. } = block;
328            parts.push(chunk_slices_mut(&mut x[..rows], &bounds, |s| {
329                bytemuck::cast_slice_mut(s.as_flattened_mut())
330            }));
331            parts.push(chunk_slices_mut(&mut z[..rows], &bounds, |s| {
332                bytemuck::cast_slice_mut(s.as_flattened_mut())
333            }));
334            parts.push(chunk_slices_mut(
335                &mut coeff[..rows],
336                &bounds,
337                bytemuck::cast_slice_mut,
338            ));
339        }
340        parts
341    }
