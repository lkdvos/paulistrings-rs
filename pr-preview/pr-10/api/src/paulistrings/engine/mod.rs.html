1//! The propagation engine: front door [`propagate`], layer loop in [`bucketed`].
2//!
3//! One layer is a coset walk over the GF(2)-linear bucket partition (the crate-private `coset` module), with the per-run sort and fused merge kernels in the crate-private `merge` module.
4//! See ARCHITECTURE.md §Engine for the layer and this module's propagation loop.
5//!
6//! The crate-private `direct` module is an **additive** second layer path for sums small enough that the bucketed layer's per-layer fixed cost dominates: off unless a caller asks for it through [`propagate_with_options`], never reached by [`propagate`], and canonical for nothing.
7//! See `research/FINDINGS.md`.

29/// Propagation direction.
30///
31/// [`Direction::Forward`] applies channels in order; [`Direction::Heisenberg`] iterates in reverse and applies adjoints (for backpropagating observables).
32#[derive(Clone, Copy, Debug, PartialEq, Eq)]
33pub enum Direction {
34    /// Apply channels in the order they were pushed onto the [`Circuit`].
35    Forward,
36    /// Apply channels in reverse order, using each channel's [`Channel::apply_adjoint`].
37    Heisenberg,
38}

40/// Which layer engine [`propagate_with_options`] uses.
41///
42/// The bucketed sorting engine ([`bucketed`]) is canonical at every term count; the alternative is a strictly additive small-sum path (`engine::direct`) that applies a layer through [`Channel::apply`] into a hash map, skipping [`Channel::prepare`] and the bucketed machinery.
43/// See `research/FINDINGS.md`.
44///
45/// The default is [`EngineSelection::SortedOnly`]: today's behaviour, unchanged, for every caller that does not ask for anything else.
46#[derive(Clone, Copy, Debug, Default, PartialEq, Eq)]
47pub enum EngineSelection {
48    /// The bucketed sorting engine for every layer. The default.
49    #[default]
50    SortedOnly,
51    /// The small-sum direct path while it is expected to be faster — the sum is at most [`PropagateOptions::small_sum_threshold`] terms *and* the policy reports no layer finalization ([`TruncationPolicy::finalizes_layer`]) — then the sorting engine for the rest of the circuit.
52    ///
53    /// The policy condition is a performance decision, not a correctness one: a finalizing policy would make the direct path pay a materialize → finalize → re-ingest round trip per layer, which is what the direct path exists to avoid.
54    /// Use [`EngineSelection::SmallSumDirect`] to take the direct path anyway.
55    Auto,
56    /// The small-sum direct path whenever the sum is at most [`PropagateOptions::small_sum_threshold`] terms, whatever the policy.
57    ///
58    /// A policy with a layer pass still gets it, once per layer, on a materialized sum — the results match [`EngineSelection::SortedOnly`] to floating-point tolerance either way.
59    /// The only way to measure the direct path against a `TopN`-style policy.
60    SmallSumDirect,
61}
62
63/// Default for [`PropagateOptions::small_sum_threshold`].
64///
65/// The direct path saves the bucketed layer's fixed per-layer cost but pays a per-term cost that rises as its hash map leaves cache, where the sorting engine's is flat in the term count; the crossover is workload-dependent.
66/// 2048 is the largest value, in a measured sweep across representative workloads, at which no configuration regresses; see `research/FINDINGS.md` for the sweep.
67/// It also sits below `desired_bits`'s `worth_splitting` floor (`DEFAULT_MIN_BUCKETS × MIN_TERMS_PER_TASK = 8192`), so a sum on this path is one the sorting engine would have run in few buckets anyway.
68pub const DEFAULT_SMALL_SUM_THRESHOLD: usize = 2048;
69
70/// Tuning knobs for [`propagate_with_options`].
71///
72/// [`Default`] is exactly today's behaviour — the sorting engine for every layer — so `PropagateOptions::default()` and [`propagate`] agree bit for bit.
73///
74/// # Examples
75///
76/// ```
77/// use paulistrings::{EngineSelection, PropagateOptions};
78///
79/// let opts = PropagateOptions {
80///     engine: EngineSelection::Auto,
81///     ..PropagateOptions::default()
82/// };
83/// assert_eq!(PropagateOptions::default().engine, EngineSelection::SortedOnly);
84/// # let _ = opts;
85/// ```
86#[derive(Clone, Copy, Debug, PartialEq, Eq)]
87pub struct PropagateOptions {
88    /// Which layer engine to use. Default [`EngineSelection::SortedOnly`].
89    pub engine: EngineSelection,
90    /// Resident term count up to which the small-sum direct path is used.
91    /// Ignored under [`EngineSelection::SortedOnly`]. Default [`DEFAULT_SMALL_SUM_THRESHOLD`].
92    pub small_sum_threshold: usize,
93    /// Terms per bucket the per-layer partition targets. Default [`DEFAULT_TARGET_BUCKET_LEN`].
94    ///
95    /// A measurement lever, not a tuning parameter: the default is the measured optimum (ARCHITECTURE.md §Bucket-Policy).
96    /// What has to stay resident is the *gather run*, not the bucket, so the headroom is not the bucket size alone.
97    ///
98    /// Raising this alone does nothing above the `min_buckets` floor: with `len >= min_buckets * MIN_TERMS_PER_TASK`, [`desired_bits`](crate::bucket::desired_bits) clamps the bucket count below at `min_buckets` whatever the target says.
99    /// Both fields have to move together to get *fewer* buckets, and [`PauliSum::rebucket`](crate::PauliSum::rebucket) is grow-only, so lowering either mid-run never coarsens a partition already grown.
100    pub target_bucket_len: usize,
101    /// Floor on the per-layer bucket count once the sum is worth splitting.
102    /// Default [`DEFAULT_MIN_BUCKETS`].
103    ///
104    /// Must be `>= 16`: below that `desired_bits`'s "worth splitting" gate is non-monotone, and the invariant "a sum of at most `target_bucket_len` terms gets one bucket" stops holding.
105    /// Lowering it also lowers the parallel task count — the coset is the unit of work — so a coarse partition is a single-thread measurement tool first.
106    ///
107    /// Applies to the sorting engine's per-layer `rebucket` only. The small-sum direct path's materialize (`engine::direct`) still sizes its partition from the defaults; it is a small-`n` path where the knob has nothing to measure, and the asymmetry is deliberate.
108    pub min_buckets: usize,
109}
110
111impl Default for PropagateOptions {
112    fn default() -> Self {
113        Self {
114            engine: EngineSelection::SortedOnly,
115            small_sum_threshold: DEFAULT_SMALL_SUM_THRESHOLD,
116            target_bucket_len: DEFAULT_TARGET_BUCKET_LEN,
117            min_buckets: DEFAULT_MIN_BUCKETS,
118        }
119    }
120}

122impl PropagateOptions {
123    /// Whether a propagation starting at `len` terms under a policy that answers `policy_finalizes` to [`TruncationPolicy::finalizes_layer`] starts on the direct path.
124    ///
125    /// Evaluated once per [`propagate_with_options`] call, outside the layer loop.
126    /// The direct path is entered only here: once the sum outgrows the threshold the run continues on the sorting engine and never comes back (see [`propagate_with_options`]).
127    fn starts_direct(&self, len: usize, policy_finalizes: bool) -> bool {
128        match self.engine {
129            EngineSelection::SortedOnly => false,
130            EngineSelection::Auto => len <= self.small_sum_threshold && !policy_finalizes,
131            EngineSelection::SmallSumDirect => len <= self.small_sum_threshold,
132        }
133    }
134}

136/// Propagate `initial` through `circuit` under `policy`.
137///
138/// Iterates the circuit's channels — in order for [`Direction::Forward`], in reverse for [`Direction::Heisenberg`], calling [`Channel::apply_adjoint`] in the latter case (default = self-adjoint; overridden on [`PauliRotation`](crate::channel::PauliRotation) and [`Clifford1Q`](crate::channel::Clifford1Q)).
139///
140/// The sum is propagated in its bucketed form throughout — there is no conversion at either end, so calling this repeatedly on the same sum (a Trotter driver stepping an observable) costs nothing beyond the layers themselves; per-bucket storage capacity is retained inside the returned sum across calls.
141/// The bucket count is re-normalized against [`desired_bits`](crate::bucket::desired_bits) before every layer.
142///
143/// # Progress logging
144///
145/// Progress is reported through the [`log`] facade under the target `paulistrings::propagate`: one `INFO` line on entry (term count, channel count, direction), one `INFO` line on exit (layers applied, terms in → out, elapsed seconds), and one `DEBUG` line per layer (`layer k/n [name]: before -> after terms, ms`).
146/// Per-layer is `DEBUG` rather than `INFO` because a Trotter driver calls this hundreds of times.
147/// The layer name comes from [`Channel::debug_name`].
148///
149/// With no logger installed each site costs one relaxed atomic load and allocates nothing — in particular the per-layer clock read is itself behind the level check.
150/// To see the lines, install any `log` implementation, e.g. `env_logger::init()` in `main` plus `RUST_LOG=paulistrings=debug` in the environment.
151///
152/// # Examples
153///
154/// ```
155/// use paulistrings::{
156///     BuildAccumulator, Circuit, Direction, PauliString, Phase, TruncationPolicy,
157///     channel::Clifford1Q, propagate,
158/// };
159/// use num_complex::Complex64;
160///
161/// let mut acc = BuildAccumulator::<1>::new(1);
162/// acc.add_term(PauliString::<1>::z(0), Phase::ONE, Complex64::new(1.0, 0.0));
163/// let observable = acc.finalize();
164///
165/// let mut circuit = Circuit::<1>::new(1);
166/// circuit.push(Clifford1Q::h(0));
167///
168/// struct KeepAll;
169/// impl<const W: usize> TruncationPolicy<W> for KeepAll {}
170///
171/// // H conjugates Z to X, so propagating Z₀ through H gives X₀.
172/// let evolved = propagate(&circuit, observable, &KeepAll, Direction::Heisenberg);
173/// assert_eq!(evolved.len(), 1);
174/// let (x, z, _c) = evolved.bucket(0);
175/// assert_eq!(x[0], [1]);
176/// assert_eq!(z[0], [0]);
177/// ```
178pub fn propagate<const W: usize, T>(
179    circuit: &Circuit<W>,
180    sum: PauliSum<W>,
181    policy: &T,
182    direction: Direction,
183) -> PauliSum<W>
184where
185    T: TruncationPolicy<W> + ?Sized,
186{
187    let mut scratch = LayerScratch::<W>::new();
188    propagate_with_scratch(circuit, sum, policy, direction, &mut scratch)
189}

191/// [`propagate`] with a caller-held [`LayerScratch`].
192///
193/// Behaves identically to [`propagate`] — it *is* the implementation — but lets the caller retain the scratch's high-water buffer capacity across calls (a Trotter driver stepping an observable), and, under the `phase-timing` feature, read the per-phase counters afterwards through `LayerScratch::take_stats` (the method — and so a resolvable doc link to it — exists only when that feature is enabled).
194///
195/// It is also the only entry point that records a [`TermTrace`](bucketed::TermTrace) — the always-compiled, opt-in per-layer term counts.
196/// Call [`LayerScratch::enable_term_trace`] before propagating and [`LayerScratch::take_term_trace`] afterwards; [`propagate`], which owns its scratch, never enables it.
197///
198/// # Progress logging
199///
200/// This is where the events described under [`propagate`] are emitted: target `paulistrings::propagate`, `INFO` on entry and exit, `DEBUG` per layer.
201/// Every event is emitted on the calling thread, outside the engine's Rayon region, so a logger implementation never runs inside a parallel layer.
202pub fn propagate_with_scratch<const W: usize, T>(
203    circuit: &Circuit<W>,
204    sum: PauliSum<W>,
205    policy: &T,
206    direction: Direction,
207    scratch: &mut LayerScratch<W>,
208) -> PauliSum<W>
209where
210    T: TruncationPolicy<W> + ?Sized,
211{
212    propagate_with_scratch_and_options(
213        circuit,
214        sum,
215        policy,
216        direction,
217        scratch,
218        PropagateOptions::default(),
219    )
220}

222/// [`propagate`] with [`PropagateOptions`].
223///
224/// `PropagateOptions::default()` is [`propagate`] exactly — same code path, same bits — so this is only interesting for opting into a non-default [`EngineSelection`].
225///
226/// # Examples
227///
228/// ```
229/// use paulistrings::{
230///     BuildAccumulator, Circuit, Direction, EngineSelection, PauliString, Phase,
231///     PropagateOptions, TruncationPolicy, channel::Clifford1Q, propagate_with_options,
232/// };
233/// use num_complex::Complex64;
234///
235/// struct KeepAll;
236/// impl<const W: usize> TruncationPolicy<W> for KeepAll {
237///     // The direct path skips the finalize round trip only for a policy that
238///     // says it has no layer pass; the trait's default answer is the
239///     // conservative `true`.
240///     fn finalizes_layer(&self) -> bool { false }
241/// }
242///
243/// let mut acc = BuildAccumulator::<1>::new(1);
244/// acc.add_term(PauliString::<1>::z(0), Phase::ONE, Complex64::new(1.0, 0.0));
245/// let mut circuit = Circuit::<1>::new(1);
246/// circuit.push(Clifford1Q::h(0));
247///
248/// let opts = PropagateOptions {
249///     engine: EngineSelection::Auto,
250///     ..PropagateOptions::default()
251/// };
252/// let evolved = propagate_with_options(
253///     &circuit, acc.finalize(), &KeepAll, Direction::Heisenberg, opts,
254/// );
255/// // H conjugates Z to X, whichever engine ran the layer.
256/// assert_eq!(evolved.get(&[1], &[0]), Some(Complex64::new(1.0, 0.0)));
257/// ```
258pub fn propagate_with_options<const W: usize, T>(
259    circuit: &Circuit<W>,
260    sum: PauliSum<W>,
261    policy: &T,
262    direction: Direction,
263    options: PropagateOptions,
264) -> PauliSum<W>
265where
266    T: TruncationPolicy<W> + ?Sized,
267{
268    let mut scratch = LayerScratch::<W>::new();
269    propagate_with_scratch_and_options(circuit, sum, policy, direction, &mut scratch, options)
270}

272/// [`propagate_with_scratch`] with [`PropagateOptions`] — the implementation every other entry point delegates to.
273///
274/// # The small-sum path, when selected
275///
276/// Under [`EngineSelection::Auto`] or [`EngineSelection::SmallSumDirect`] and a starting sum within [`PropagateOptions::small_sum_threshold`], the leading layers run on the direct path (`engine::direct`): the sum is held in a hash map across those layers, and materialized back into a [`PauliSum`] once — when a layer leaves it above the threshold, or when the circuit ends.
277/// From that point the sorting engine runs every remaining layer.
278///
279/// The transition is **one-way**. Re-entering the direct path when a later truncation drops the sum back under the threshold is deliberately not done: each crossing costs an `O(n)` ingest plus an `O(n log n)` materialize, and a sum oscillating around the threshold would pay them per layer for an upside bounded by the fixed cost the direct path saves.
280/// One crossing per call is also trivially reasoned about: the per-layer term counts and the [`TermTrace`](bucketed::TermTrace) are the same records in the same order regardless of where it happened.
281///
282/// Truncation is applied identically on both sides of the transition: `keep_term` runs per layer on summed coefficients (where the merge phase runs it), and `finalize_layer` runs per layer on a materialized sum whenever [`TruncationPolicy::finalizes_layer`] says there is one.
283/// Progress logging and the term trace emit the same records from either path.
284///
285/// # Channels wider than the sorting engine can prepare
286///
287/// The direct path calls only [`Channel::apply`], so it applies a channel of any support width — including the `> MAX_LOCAL_SUPPORT` channels for which `Channel::prepare` returns `None` and the sorting engine panics.
288/// It is a wider path, not a narrower one, and that asymmetry is visible: a circuit containing such a channel propagates while the sum is under the threshold and panics on the layer after it grows past it, exactly as it panics today under [`EngineSelection::SortedOnly`].
289/// The generalization design for the sorting engine is tracked in `research/FINDINGS.md`.
290pub fn propagate_with_scratch_and_options<const W: usize, T>(
291    circuit: &Circuit<W>,
292    mut sum: PauliSum<W>,
293    policy: &T,
294    direction: Direction,
295    scratch: &mut LayerScratch<W>,
296    options: PropagateOptions,
297) -> PauliSum<W>
298where
299    T: TruncationPolicy<W> + ?Sized,
300{
301    let n = circuit.channels.len();
302    let adjoint = matches!(direction, Direction::Heisenberg);
303    // The bucket count `B` is a deterministic function of the *history* of term counts (`rebucket` is grow-only, so `B` is the running max of `desired_bits` over every layer so far, not the instantaneous length), which is itself fully determined by the circuit and the starting sum.
304    // Agreement across `B` is only required to floating-point tolerance (ARCHITECTURE.md §Determinism).
305
306    // Entry/exit INFO pair. One unconditional `Instant` pair per `propagate` call is negligible next to a single layer; the *per-layer* clock reads below are the ones that have to be gated.
307    let terms_in = sum.len();
308    let started = std::time::Instant::now();
309    log::info!(
310        target: LOG_TARGET,
311        "propagate: {terms_in} terms through {n} channels ({direction:?})",
312    );
313
314    // Hoisted out of the layer loop: nothing inside it can enable or disable the trace, so the per-layer test is on a register rather than a load through `scratch`.
315    let tracing = scratch.term_trace.is_some();
316    let gate_tracing = scratch.gate_trace.is_some();
317
318    // The engine choice is made once, here, outside every loop: a layer's own code path cannot change it, and under the default `SortedOnly` this is one not-taken branch before the loop and nothing inside it.
319    // `n > 0` keeps a zero-layer call off the direct path entirely, so it stays a no-op rather than an ingest/materialize round trip.
320    let mut start = 0usize;
321    if n > 0 && options.starts_direct(terms_in, policy.finalizes_layer()) {
322        let (out, applied) =
323            direct::run_direct_prefix(circuit, sum, policy, direction, scratch, options);
324        sum = out;
325        start = applied;
326    }
327
328    for k in start..n {
329        let idx = match direction {
330            Direction::Forward => k,
331            Direction::Heisenberg => n - 1 - k,
332        };
333        let ch: &dyn Channel<W> = circuit.channels[idx].as_ref();
334
335        // Per-layer DEBUG progress, and the opt-in gate trace, share one clock read: `log_enabled!` is a relaxed atomic load plus a compare, so a disabled logger and a disabled gate trace together cost one branch per layer and never read the clock; `terms_before` is the cached length field.
336        let debug_on = log::log_enabled!(target: LOG_TARGET, log::Level::Debug);
337        let want_timer = gate_tracing || debug_on;
338        let layer_t0 = want_timer.then(std::time::Instant::now);
339        let terms_before = sum.len();
340
341        #[cfg(feature = "phase-timing")]
342        let mut st = stats::Stamp::now();
343        #[cfg(feature = "phase-timing")]
344        {
345            scratch.stats.layers += 1;
346            scratch.stats.terms_in += sum.len() as u64;
347        }
348
349        sum.rebucket(options.target_bucket_len, options.min_buckets);
350        #[cfg(feature = "phase-timing")]
351        st.lap(&mut scratch.stats.rebucket_ns);
352
353        let prep = ch.prepare(sum.hash(), adjoint);
354        #[cfg(feature = "phase-timing")]
355        st.lap(&mut scratch.stats.prepare_ns);
356
357        match prep {
358            Some(prep) => {
359                apply_layer_bucketed(&mut sum, &prep, policy, scratch);
360                #[cfg(feature = "phase-timing")]
361                st.rearm();
362            }
363            None => {
364                // No built-in channel can reach this arm: every one of them has support ≤ MAX_LOCAL_SUPPORT except `PauliRotation`, which overrides `prepare` and returns `Prepared::Rotation` at any generator weight, and none writes outside its declared support.
365                // So this is only reachable from a user-supplied `Channel` impl, and it is a hard error rather than a slow path: there is no whole-sum fallback to absorb it.
366                let weight: u32 = ch.support().iter().map(|w| w.count_ones()).sum();
367                panic!(
368                    "layer {idx}: Channel::prepare declined, so this channel cannot be \
369                     propagated. The engine tabulates channels of support ≤ \
370                     {MAX_LOCAL_SUPPORT} qubits (this one declares {weight}), and a \
371                     channel must not write outside its declared support. See \
372                     research/FINDINGS.md",
373                );
374            }
375        }
376
377        policy.finalize_layer(&mut sum);
378        #[cfg(feature = "phase-timing")]
379        {
380            st.lap(&mut scratch.stats.finalize_ns);
381            scratch.stats.terms_out += sum.len() as u64;
382        }
383        // Opt-in term trace (`LayerScratch::enable_term_trace`), both counts in one call: `terms_before` from the read above, `sum.len()` post-truncation, so the pair says what the layer actually left resident.
384        // Placed last so it is outside every measured phase, and behind a hoisted flag + a `#[cold]` callee so the untraced layer body grows by a register test and a not-taken branch — this loop inlines `apply_layer_bucketed`, whose merge kernels are sensitive to a few bytes of code motion (CLAUDE.md §Performance discipline).
385        if tracing {
386            record_layer_terms(scratch, terms_before, sum.len());
387        }
388
389        if want_timer {
390            let dt = layer_t0
391                .expect("want_timer implies layer_t0 is Some")
392                .elapsed();
393            if gate_tracing {
394                record_gate_trace(
395                    scratch,
396                    idx as u32,
397                    k as u32,
398                    ch.debug_name(),
399                    terms_before,
400                    sum.len(),
401                    dt,
402                );
403            }
404            if debug_on {
405                log::debug!(
406                    target: LOG_TARGET,
407                    "layer {}/{} [{}]: {} -> {} terms, {:.1} ms",
408                    k + 1,
409                    n,
410                    ch.debug_name(),
411                    terms_before,
412                    sum.len(),
413                    dt.as_secs_f64() * 1e3,
414                );
415            }
416        }
417    }
418
419    log::info!(
420        target: LOG_TARGET,
421        "propagate: {} layers applied, {} -> {} terms, {:.3} s",
422        n,
423        terms_in,
424        sum.len(),
425        started.elapsed().as_secs_f64(),
426    );
427
428    sum
429}

431/// Append one layer's `(terms_in, terms_out)` to the scratch's trace.
432///
433/// Deliberately `#[cold]` + `#[inline(never)]`: the layer loop is the caller that inlines [`apply_layer_bucketed`] and, through it, the merge kernels, whose throughput is sensitive to a few bytes of code motion (CLAUDE.md §Performance discipline).
434/// Keeping the pushes out-of-line and marked cold puts the whole trace off the layer's code path; the tracing caller pays a call, which is nothing next to a layer.
435#[cold]
436#[inline(never)]
437fn record_layer_terms<const W: usize>(
438    scratch: &mut LayerScratch<W>,
439    terms_in: usize,
440    terms_out: usize,
441) {
442    if let Some(trace) = scratch.term_trace.as_mut() {
443        trace.terms_in.push(terms_in);
444        trace.terms_out.push(terms_out);
445    }
446}
447
448/// Append one layer's full [`GateTrace`] record.
449///
450/// Deliberately `#[cold]` + `#[inline(never)]` for the same reason as [`record_layer_terms`]: keeps the trace off the layer's inlined code path.
451#[cold]
452#[inline(never)]
453#[allow(clippy::too_many_arguments)]
454fn record_gate_trace<const W: usize>(
455    scratch: &mut LayerScratch<W>,
456    circuit_index: u32,
457    application_index: u32,
458    gate_name: &'static str,
459    terms_in: usize,
460    terms_out: usize,
461    elapsed: std::time::Duration,
462) {
463    if let Some(trace) = scratch.gate_trace.as_mut() {
464        trace.circuit_index.push(circuit_index);
465        trace.application_index.push(application_index);
466        trace.gate_name.push(gate_name);
467        trace.terms_in.push(terms_in);
468        trace.terms_out.push(terms_out);
469        trace.nanos.push(elapsed.as_nanos() as u64);
470    }
471}

473/// Bucket-count floor: enough buckets that Rayon has slack to load-balance.
474///
475/// Fixed, not derived from `rayon::current_num_threads`: see [`crate::bucket::sum::DEFAULT_MIN_BUCKETS`] for why a thread-independent floor is what we want here (ARCHITECTURE.md §Bucket-Policy).
476/// Combined with the grow-only `rebucket` policy, this floor is a lower bound on `B` throughout a `propagate` call, not just at the layer where it was first crossed.
477pub fn default_min_buckets() -> usize {
478    crate::bucket::sum::DEFAULT_MIN_BUCKETS
479}
