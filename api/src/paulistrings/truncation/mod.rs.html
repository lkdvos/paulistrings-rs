1//! [`TruncationPolicy<W>`] — composable per-term and per-layer term filters.
2//!
3//! Built-ins: [`CoefficientThreshold`] drops terms below a magnitude, [`WeightCutoff`] drops terms above a Pauli weight, [`TopN`] keeps at most `n` largest-magnitude terms without splitting a tie group, and [`ApproxTopN`] trades the exact count for a cheaper histogram threshold.
4//! Compose with [`And`] (both must accept) or [`Or`] (either accepts).
5//!
6//! # Examples
7//!
8//! ```
9//! use paulistrings::truncation::{And, CoefficientThreshold, WeightCutoff};
10//! use paulistrings::TruncationPolicy;
11//! use num_complex::Complex64;
12//!
13//! // Keep terms with |coeff| > 1e-9 AND weight ≤ 4.
14//! let policy = And(CoefficientThreshold(1e-9), WeightCutoff(4));
15//!
16//! // A weight-1 term with coeff 0.5: passes both.
17//! assert!(<_ as TruncationPolicy<1>>::keep_term(
18//!     &policy, &[1], &[0], Complex64::new(0.5, 0.0),
19//! ));
20//! ```
21
22pub mod builtin;
23
24pub use builtin::{And, ApproxTopN, CoefficientThreshold, Or, TopN, WeightCutoff};
25
26use crate::pauli_sum::PauliSum;
27use num_complex::Complex64;

29/// A truncation strategy. Both methods have sensible defaults so users only
30/// implement the one they need.
31///
32/// # Implementing
33///
34/// Override [`keep_term`](Self::keep_term) for per-output decisions (runs inside the merge phase, must inline).
35/// Override [`finalize_layer`](Self::finalize_layer) for global decisions that depend on the whole layer's output (e.g. partial sort for selecting the top `n` terms).
36///
37/// ```
38/// use paulistrings::TruncationPolicy;
39/// use num_complex::Complex64;
40///
41/// /// Drop the imaginary half: keep only terms whose coefficient is real.
42/// struct RealOnly;
43/// impl<const W: usize> TruncationPolicy<W> for RealOnly {
44///     #[inline]
45///     fn keep_term(&self, _x: &[u64; W], _z: &[u64; W], c: Complex64) -> bool {
46///         c.im == 0.0
47///     }
48/// }
49/// ```
50pub trait TruncationPolicy<const W: usize>: Send + Sync {
51    /// Cheap per-term filter applied during the merge phase. Must inline.
52    ///
53    /// `c` is the summed coefficient at the key `(x, z)`: `keep_term` runs after all scratch entries with this key have been reduced, not on individual scratch entries.
54    #[inline]
55    fn keep_term(&self, _x: &[u64; W], _z: &[u64; W], _c: Complex64) -> bool {
56        true
57    }
58
59    /// Optional global pass after each circuit layer. May be non-local (e.g. partial sort for [`TopN`]).
60    ///
61    /// The sum arrives in its bucketed working form.
62    /// A filter-shaped policy is easiest written with [`PauliSum::retain`] (per-bucket parallel, in place, preserves every invariant); a policy needing a global view first can read terms via [`PauliSum::iter`] or per bucket via [`PauliSum::bucket`].
63    fn finalize_layer(&self, _sum: &mut PauliSum<W>) {}
64
65    /// Whether [`finalize_layer`](Self::finalize_layer) does anything at all.
66    ///
67    /// Purely an optimization hint read only by the engine: it never changes what a policy means.
68    /// The engine's small-sum direct path holds the sum in a hash map between layers, so honouring a layer pass there costs a full materialize → finalize → re-ingest round trip; this method lets a policy say that round trip is pointless.
69    ///
70    /// The default is `true`, the conservative answer.
71    /// Override with `false` only if `finalize_layer` is genuinely a no-op: returning `false` while `finalize_layer` does something makes the direct path skip it, which is a wrong answer, not a slow one.
72    ///
73    /// ```
74    /// use paulistrings::TruncationPolicy;
75    /// use paulistrings::truncation::{CoefficientThreshold, TopN};
76    ///
77    /// assert!(!<_ as TruncationPolicy<1>>::finalizes_layer(&CoefficientThreshold(1e-9)));
78    /// assert!(<_ as TruncationPolicy<1>>::finalizes_layer(&TopN(1000)));
79    /// ```
80    fn finalizes_layer(&self) -> bool {
81        true
82    }
83}
