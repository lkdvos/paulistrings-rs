1//! [`PauliString<W>`] — symplectic-encoded Pauli operator on up to `64·W` qubits. See ARCHITECTURE.md §Data-Model and §Width.
2//!
3//! Multiplication XORs the `(x, z)` parts and returns the `i^k` phase factor as a [`Phase`]; the phase is not stored on the type, callers fold it into a coefficient at the boundary.
4//!
5//! # Examples
6//!
7//! `X · Z = -i·Y`: the XOR gives the Y bits and the returned [`Phase`] is the
8//! `-i` factor.
9//!
10//! ```
11//! use paulistrings::{PauliString, Phase};
12//!
13//! let mut p = PauliString::<1>::x(0);
14//! let phase = p.mul_assign(&PauliString::<1>::z(0));
15//! assert_eq!(p, PauliString::<1>::y(0));
16//! assert_eq!(phase, Phase::MINUS_I);
17//! ```
18//!
19//! [`PauliSum`]: crate::PauliSum
20//! [`BuildAccumulator`]: crate::BuildAccumulator
21//! [`Channel::apply`]: crate::Channel::apply
22//! [`Phase`]: crate::Phase
23

31/// A Pauli operator on up to `64 · W` qubits. See ARCHITECTURE.md §Data-Model.
32///
33/// `#[repr(C)]` so the type is `Pod` and can be reinterpreted as bytes for serialization or GPU upload. [`Ord`] is the load-bearing trait — the engine is sort-based, not hashmap-based.
34///
35/// # Examples
36///
37/// ```
38/// use paulistrings::PauliString;
39///
40/// let p = PauliString::<1>::x(3);
41/// assert_eq!(p.weight(), 1);
42/// assert!(p.commutes_with(&PauliString::<1>::x(0)));
43/// ```
44#[derive(Clone, Copy, PartialEq, Eq, Debug)]
45#[repr(C)]
46pub struct PauliString<const W: usize> {
47    /// X-part bitmask: bit `q` set iff qubit `q` is `X` or `Y`.
48    pub x: [u64; W],
49    /// Z-part bitmask: bit `q` set iff qubit `q` is `Z` or `Y`.
50    pub z: [u64; W],
51}
52
53unsafe impl<const W: usize> Zeroable for PauliString<W> {}
54unsafe impl<const W: usize> Pod for PauliString<W> {}
55
56impl<const W: usize> PauliString<W> {
57    /// Identity Pauli string (all qubits `I`).
58    ///
59    /// # Examples
60    ///
61    /// ```
62    /// use paulistrings::PauliString;
63    ///
64    /// let id = PauliString::<1>::identity();
65    /// assert_eq!(id.weight(), 0);
66    /// ```
67    pub const fn identity() -> Self {
68        Self {
69            x: [0u64; W],
70            z: [0u64; W],
71        }
72    }
73
74    /// Single-qubit `X` Pauli on `qubit`.
75    ///
76    /// # Panics
77    ///
78    /// Panics in debug builds if `qubit >= 64 · W`.
79    ///
80    /// # Examples
81    ///
82    /// ```
83    /// use paulistrings::PauliString;
84    ///
85    /// let p = PauliString::<1>::x(2);
86    /// assert_eq!(p.x, [0b100]);
87    /// assert_eq!(p.z, [0]);
88    /// ```
89    #[inline]
90    pub fn x(qubit: u32) -> Self {
91        debug_assert!((qubit as usize) < 64 * W);
92        let mut p = Self::identity();
93        p.x[(qubit / 64) as usize] = 1u64 << (qubit % 64);
94        p
95    }
96
97    /// Canonical Pauli `Y = (1, 1)` on `qubit`. The `i` factor in `Y = i · X · Z` is the caller's concern.
98    ///
99    /// # Panics
100    ///
101    /// Panics in debug builds if `qubit >= 64 · W`.
102    #[inline]
103    pub fn y(qubit: u32) -> Self {
104        debug_assert!((qubit as usize) < 64 * W);
105        let mut p = Self::identity();
106        let w = (qubit / 64) as usize;
107        let bit = 1u64 << (qubit % 64);
108        p.x[w] = bit;
109        p.z[w] = bit;
110        p
111    }
112
113    /// Single-qubit `Z` Pauli on `qubit`.
114    ///
115    /// # Panics
116    ///
117    /// Panics in debug builds if `qubit >= 64 · W`.
118    #[inline]
119    pub fn z(qubit: u32) -> Self {
120        debug_assert!((qubit as usize) < 64 * W);
121        let mut p = Self::identity();
122        p.z[(qubit / 64) as usize] = 1u64 << (qubit % 64);
123        p
124    }
125
126    /// Number of non-identity qubits (Hamming weight of `x | z`).
127    ///
128    /// # Examples
129    ///
130    /// ```
131    /// use paulistrings::PauliString;
132    ///
133    /// let mut p = PauliString::<1>::x(0);
134    /// p.mul_assign(&PauliString::<1>::z(1));
135    /// assert_eq!(p.weight(), 2);
136    /// ```
137    #[inline]
138    pub fn weight(&self) -> u32 {
139        (0..W).map(|i| (self.x[i] | self.z[i]).count_ones()).sum()
140    }
141
142    /// Multiply `self * other` in place. Returns the `i^k` phase factor such that the true product is `phase · self_after_xor`.
143    ///
144    /// # Examples
145    ///
146    /// `X · Z = -i·Y`: the bits XOR to `Y` and the phase is `-i`.
147    ///
148    /// ```
149    /// use paulistrings::{PauliString, Phase};
150    ///
151    /// let mut p = PauliString::<1>::x(0);
152    /// let phase = p.mul_assign(&PauliString::<1>::z(0));
153    /// assert_eq!(p, PauliString::<1>::y(0));
154    /// assert_eq!(phase, Phase::MINUS_I);
155    /// ```
156    #[inline]
157    pub fn mul_assign(&mut self, other: &Self) -> Phase {
158        // Per-qubit: P(a,b) · P(c,d) = i^δ · P(a⊕c, b⊕d) where
159        //   δ = 2·(b·c) + a·b + c·d − (a⊕c)·(b⊕d)   (mod 4)
160        // (derived from P(a,b) = i^{a·b} X^a Z^b and ZX = -XZ).
161        let mut delta: u32 = 0;
162        for i in 0..W {
163            let a = self.x[i];
164            let b = self.z[i];
165            let c = other.x[i];
166            let d = other.z[i];
167            let zc_x = (b & c).count_ones();
168            let y_self = (a & b).count_ones();
169            let y_other = (c & d).count_ones();
170            let y_result = ((a ^ c) & (b ^ d)).count_ones();
171            delta = delta.wrapping_add(zc_x.wrapping_mul(2));
172            delta = delta.wrapping_add(y_self);
173            delta = delta.wrapping_add(y_other);
174            delta = delta.wrapping_sub(y_result);
175            self.x[i] ^= c;
176            self.z[i] ^= d;
177        }
178        Phase::new(delta as u8)
179    }
180
181    /// Value-returning multiply: `(self * other, phase)`.
182    ///
183    /// Not `std::ops::Mul` because the operation also returns a `Phase`, which calling sites need to fold into a coefficient at the boundary.
184    #[allow(clippy::should_implement_trait)]
185    #[inline]
186    pub fn mul(mut self, other: &Self) -> (Self, Phase) {
187        let phase = self.mul_assign(other);
188        (self, phase)
189    }
190
191    /// `true` iff every set bit lies on a qubit index `< num_qubits`.
192    ///
193    /// The engine and built-in channels preserve this bound by construction, so this check is not on the hot path; used in `debug_assert!` at boundaries with custom [`Channel`] impls, in `PauliSum::assert_invariants`, and in tests.
194    ///
195    /// # Panics
196    ///
197    /// Panics in debug builds if `num_qubits > 64 · W`.
198    ///
199    /// [`Channel`]: crate::Channel
200    /// [`Channel::support`]: crate::Channel::support
201    /// [`Circuit`]: crate::Circuit
202    #[inline]
203    pub fn is_within(&self, num_qubits: usize) -> bool {
204        debug_assert!(num_qubits <= 64 * W);
205        let mut leak: u64 = 0;
206        for i in 0..W {
207            let lo = 64 * i;
208            let in_bounds: u64 = if num_qubits >= lo + 64 {
209                !0u64
210            } else if num_qubits <= lo {
211                0
212            } else {
213                (1u64 << (num_qubits - lo)) - 1
214            };
215            leak |= (self.x[i] | self.z[i]) & !in_bounds;
216        }
217        leak == 0
218    }

220    /// `true` iff `self` and `other` commute as Pauli operators.
221    ///
222    /// # Examples
223    ///
224    /// ```
225    /// use paulistrings::PauliString;
226    ///
227    /// // X and Z anticommute.
228    /// assert!(!PauliString::<1>::x(0).commutes_with(&PauliString::<1>::z(0)));
229    /// // X on disjoint qubits commutes with anything on the other qubit.
230    /// assert!(PauliString::<1>::x(0).commutes_with(&PauliString::<1>::z(1)));
231    /// ```
232    ///
233    /// Only the low bit of the symplectic form survives, and popcount parity is GF(2)-linear (`parity(a) ^ parity(b) == parity(a ^ b)`), so the masked words are XOR-folded first and reduced by a single `count_ones` instead of one per word.
234    #[inline]
235    pub fn commutes_with(&self, other: &Self) -> bool {
236        let mut acc: u64 = 0;
237        for i in 0..W {
238            acc ^= (self.x[i] & other.z[i]) ^ (self.z[i] & other.x[i]);
239        }
240        acc.count_ones() & 1 == 0
241    }
242
243    /// Commutator `[self, other] = self·other − other·self`, as `(product, coefficient)`.
244    ///
245    /// Two Pauli strings either commute or anticommute, so the difference is either exactly `0` or twice the product: the coefficient is `2·i^k` when they anticommute and `0` when they commute. The returned string is [`Self::mul`]'s product either way, so a `0` coefficient is an exact algebraic zero rather than a missing term.
246    ///
247    /// # Examples
248    ///
249    /// ```
250    /// use paulistrings::PauliString;
251    /// use num_complex::Complex64;
252    ///
253    /// // [X, Z] = 2·X·Z = -2i·Y.
254    /// let (product, coeff) = PauliString::<1>::x(0).commutator(&PauliString::<1>::z(0));
255    /// assert_eq!(product, PauliString::<1>::y(0));
256    /// assert_eq!(coeff, Complex64::new(0.0, -2.0));
257    /// ```
258    #[inline]
259    pub fn commutator(self, other: &Self) -> (Self, Complex64) {
260        let vanishes = self.commutes_with(other);
261        let (product, phase) = self.mul(other);
262        (product, scaled_phase(phase, vanishes))
263    }
264
265    /// Anticommutator `{self, other} = self·other + other·self`, as `(product, coefficient)`.
266    ///
267    /// [`Self::commutator`]'s mirror: the coefficient is `2·i^k` when the two strings commute and `0` when they anticommute, so the two together decompose `2·self·other`.
268    ///
269    /// # Examples
270    ///
271    /// ```
272    /// use paulistrings::PauliString;
273    /// use num_complex::Complex64;
274    ///
275    /// // {X, X} = 2·I.
276    /// let (product, coeff) = PauliString::<1>::x(0).anticommutator(&PauliString::<1>::x(0));
277    /// assert_eq!(product, PauliString::<1>::identity());
278    /// assert_eq!(coeff, Complex64::new(2.0, 0.0));
279    /// ```
280    #[inline]
281    pub fn anticommutator(self, other: &Self) -> (Self, Complex64) {
282        let vanishes = !self.commutes_with(other);
283        let (product, phase) = self.mul(other);
284        (product, scaled_phase(phase, vanishes))
285    }
286}
287
288/// `0` when the bracket vanishes, `2·i^k` otherwise — the coefficient both brackets carry.
289#[inline]
290fn scaled_phase(phase: Phase, vanishes: bool) -> Complex64 {
291    if vanishes {
292        Complex64::new(0.0, 0.0)
293    } else {
294        phase.to_complex() * 2.0
295    }
296}

304impl<const W: usize> Ord for PauliString<W> {
305    /// Lexicographic compare on `(x, z)`, low-to-high word order.
306    fn cmp(&self, other: &Self) -> Ordering {
307        for i in 0..W {
308            match self.x[i].cmp(&other.x[i]) {
309                Ordering::Equal => continue,
310                ord => return ord,
311            }
312        }
313        for i in 0..W {
314            match self.z[i].cmp(&other.z[i]) {
315                Ordering::Equal => continue,
316                ord => return ord,
317            }
318        }
319        Ordering::Equal
320    }
321}

323impl<const W: usize> PartialOrd for PauliString<W> {
324    fn partial_cmp(&self, other: &Self) -> Option<Ordering> {
325        Some(self.cmp(other))
326    }
327}

344    /// `X` and `Z` anticommute, so `[X, Z] = 2·X·Z = -2i·Y` and `{X, Z} = 0`.
345    #[test]
346    fn commutator_of_x_and_z_is_minus_two_i_y() {
347        fn check<const W: usize>() {
348            let x = PauliString::<W>::x(0);
349            let z = PauliString::<W>::z(0);
350
351            let (product, coeff) = x.commutator(&z);
352            assert_eq!(product, PauliString::<W>::y(0));
353            assert_eq!(coeff, Complex64::new(0.0, -2.0));
354
355            let (product, coeff) = x.anticommutator(&z);
356            assert_eq!(product, PauliString::<W>::y(0));
357            assert_eq!(coeff, ZERO);
358        }
359        check::<1>();
360        check::<2>();
361    }

363    /// Reversing the operands flips the commutator's sign: `[Z, X] = 2·Z·X = +2i·Y`.
364    #[test]
365    fn commutator_is_antisymmetric_on_x_and_z() {
366        let (product, coeff) = PauliString::<1>::z(0).commutator(&PauliString::<1>::x(0));
367        assert_eq!(product, PauliString::<1>::y(0));
368        assert_eq!(coeff, Complex64::new(0.0, 2.0));

371    /// `X⊗I` and `I⊗X` commute, so `[P, Q] = 0` and `{P, Q} = 2·P·Q = 2·X⊗X`.
372    #[test]
373    fn commuting_strings_have_a_zero_commutator_and_twice_the_product() {
374        fn check<const W: usize>() {
375            let a = PauliString::<W>::x(0);
376            let b = PauliString::<W>::x(1);
377            let mut xx = PauliString::<W>::x(0);
378            xx.x[0] |= 1u64 << 1;
379
380            let (product, coeff) = a.commutator(&b);
381            assert_eq!(product, xx);
382            assert_eq!(coeff, ZERO);
383
384            let (product, coeff) = a.anticommutator(&b);
385            assert_eq!(product, xx);
386            assert_eq!(coeff, Complex64::new(2.0, 0.0));
387        }
388        check::<1>();
389        check::<2>();
390    }

392    /// Identity commutes with everything, so its anticommutator carries the whole `2·P`.
393    #[test]

419        fn arb_pauli_w2() -> impl Strategy<Value = PauliString<2>> {
420            (any::<u64>(), any::<u64>(), any::<u64>(), any::<u64>()).prop_map(|(x0, x1, z0, z1)| {

422                    x: [x0, x1],

428        proptest! {
429            /// `[P, Q] + {P, Q} = 2·P·Q` — the defining decomposition, and the reason exactly one of the two is nonzero.
430            #[test]
431            fn commutator_plus_anticommutator_is_twice_the_product(
432                a in arb_pauli_w2(),
433                b in arb_pauli_w2(),
434            ) {
435                let (product, phase) = a.mul(&b);
436                let (comm_product, comm) = a.commutator(&b);
437                let (anti_product, anti) = a.anticommutator(&b);
438
439                prop_assert_eq!(comm_product, product);
440                prop_assert_eq!(anti_product, product);
441                prop_assert_eq!(comm + anti, phase.to_complex() * 2.0);
442                prop_assert!((comm == ZERO) != (anti == ZERO));
443            }
444        }
