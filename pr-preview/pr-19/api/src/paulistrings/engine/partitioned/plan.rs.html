1//! Per-layer classification of a prepared channel's deltas into *local* and
2//! *remote*. See ARCHITECTURE.md §Engine for the delta set itself.
3//!
4//! A partitioning splits keys by `part(v) = P·v` ([`PartitionRows`]) on top of
5//! the bucket split `loc(v) = H·v` ([`Gf2Hash`]). Both are GF(2)-linear, so a
6//! prepared channel's key delta `d` moves a term by a *constant* partition
7//! delta `pd = part(d)` and bucket delta `bd = h(d)`, whatever the term. That
8//! is what makes this a per-layer plan rather than a per-term decision:
9//!
10//! * `pd == 0` — the delta is **local**: it lands in the same partition, and the ordinary coset loop handles it with no communication.
11//! * `pd != 0` — the delta is **remote**: every row it produces belongs to partition `rank ^ pd`, so the layer exports one stream per such delta to that one partner.
12//!
13//! The identity delta has mask `0` and `part(0) = 0`, so it is always local: a partition never has to ship a term to itself.
14
15use crate::bucket::hash::{Gf2Hash, PartitionRows};
16use crate::channel::prepared::Prepared;
17use crate::circuit::Circuit;
18
19/// One delta of a prepared channel that crosses a partition boundary.
20#[derive(Clone, Debug, PartialEq, Eq)]
21pub(crate) struct RemoteDelta {
22    /// Index into `ptm.deltas()`.
23    /// For [`Prepared::Rotation`] the two implicit entries are numbered the way the engine's plan numbers them: the identity pass is `0`, the generator pass is `1`.
24    pub entry: usize,
25    /// The partition every row of this delta belongs to, `rank ^ partition_delta`.
26    pub partner: u32,
27    /// `h(d)` — the entry's bucket delta, carried so the export pass can name
28    /// the destination bucket without re-hashing.
29    pub bucket_delta: u32,
30    /// `part(d)`, nonzero by construction.
31    pub partition_delta: u32,
32}
33
34/// How one prepared channel's deltas split under a partitioning, from the
35/// point of view of one partition (`rank`).
36#[derive(Clone, Debug)]
37pub(crate) struct PartitionPlan {
38    /// Per entry, `true` if the entry's partition delta is zero.
39    ///
40    /// Length is `ptm.deltas().len()` for [`Prepared::Local`] and 2 for [`Prepared::Rotation`] (identity pass, generator pass).
41    /// Indices line up with [`RemoteDelta::entry`], so the two together cover every entry exactly once.
42    pub local_entries: Vec<bool>,
43    /// The local deltas' bucket deltas, ascending and deduplicated — the input to `Gf2Span::new` for the coset loop that runs inside this partition.
44    ///
45    /// Always contains `0`: the identity delta is local, and `0` is in every span regardless, so including it unconditionally cannot change the span it generates.
46    pub local_bucket_deltas: Vec<u32>,
47    /// The remote deltas, ascending by [`RemoteDelta::entry`].
48    pub remote: Vec<RemoteDelta>,
49    /// Non-identity *realized* deltas, local and remote together.
50    ///
51    /// The gather's sort-kernel choice (`merge::RADIX_MIN_REST_STREAMS`) turns on how wide a channel's fanout is, which is a property of the channel, not of how this partition happens to see it — so the gate reads the total, not the local count.
52    pub rest_streams_total: usize,
53}
54
55impl PartitionPlan {
56    /// Classify `prep`'s deltas under `rows`, as seen from partition `rank`.
57    ///
58    /// With [`PartitionRows::none`] every delta is local and `local_bucket_deltas` is exactly `prep.bucket_deltas()`.
59    ///
60    /// # Panics
61    ///
62    /// Panics in debug builds if `rank` is not a partition of `rows`, or if the identity delta somehow classifies as remote (it cannot: `part(0) = 0`).
63    pub(crate) fn new<const W: usize>(
64        prep: &Prepared<W>,
65        rows: &PartitionRows<W>,
66        rank: u32,
67    ) -> Self {
68        debug_assert!(
69            rows.num_partitions() > rank as usize,
70            "PartitionPlan: rank {rank} outside {} partitions",
71            rows.num_partitions(),
72        );
73
74        // `(mask_x, mask_z, bucket_delta)` per entry, in entry order.
75        // Built once so the classification below is one loop for both variants; this runs once per layer, so the small allocation is free.
76        let entries: Vec<([u64; W], [u64; W], u32)> = match prep {
77            Prepared::Local(ptm) => ptm
78                .deltas()
79                .iter()
80                .map(|d| {
81                    let (mx, mz) = d.mask();
82                    (mx, mz, d.bucket_delta)
83                })
84                .collect(),
85            // The rotation's two implicit entries: the identity pass (mask 0) and the generator pass (mask `P`).
86            Prepared::Rotation(r) => {
87                let (gx, gz) = r.gen_mask();
88                vec![
89                    ([0u64; W], [0u64; W], r.bucket_delta_identity),
90                    (gx, gz, r.bucket_delta_gen),
91                ]
92            }
93        };
94
95        let mut local_entries: Vec<bool> = Vec::with_capacity(entries.len());
96        let mut local_bucket_deltas: Vec<u32> = vec![0];
97        let mut remote: Vec<RemoteDelta> = Vec::new();
98
99        for (entry, (mask_x, mask_z, bucket_delta)) in entries.iter().enumerate() {
100            let partition_delta = rows.partition_of(mask_x, mask_z);
101            local_entries.push(partition_delta == 0);
102            if partition_delta == 0 {
103                local_bucket_deltas.push(*bucket_delta);
104            } else {
105                remote.push(RemoteDelta {
106                    entry,
107                    partner: rank ^ partition_delta,
108                    bucket_delta: *bucket_delta,
109                    partition_delta,
110                });
111            }
112        }
113
114        // `deltas()` is the *realized* delta set (ARCHITECTURE.md §Bucketing), so its length minus the identity entry is the number of streams a gather concatenates into a run's rest columns — the quantity `engine::bucketed`'s own plan calls `rest_streams`.
115        // The rotation has exactly one non-identity pass at any generator weight.
116        let rest_streams_total = match prep {
117            Prepared::Local(ptm) => {
118                let has_identity = ptm.deltas().first().is_some_and(|d| d.local_delta == 0);
119                debug_assert!(
120                    !has_identity || local_entries[0],
121                    "the identity delta must be local: part(0) = 0",
122                );
123                ptm.deltas().len() - has_identity as usize
124            }
125            Prepared::Rotation(_) => {
126                debug_assert!(local_entries[0], "the identity pass must be local");
127                1
128            }
129        };
130
131        local_bucket_deltas.sort_unstable();
132        local_bucket_deltas.dedup();
133
134        Self {
135            local_entries,
136            local_bucket_deltas,
137            remote,
138            rest_streams_total,
139        }
140    }
141
142    /// `true` if this layer moves anything across a partition boundary.
143    pub(crate) fn has_remote(&self) -> bool {
144        !self.remote.is_empty()
145    }
146
147    /// The remote deltas destined for partition `q`, ascending by entry.
148    pub(crate) fn remote_for_partner(&self, q: u32) -> impl Iterator<Item = &RemoteDelta> {
149        self.remote.iter().filter(move |r| r.partner == q)
150    }
151}

153/// Per layer, `(local delta count, remote delta count)` under `rows`, for `circuit`'s channels prepared against `hash`.
154///
155/// Layers are reported in **application order**: circuit order for `adjoint == false`, reverse order for `adjoint == true`, matching [`Direction::Heisenberg`](crate::Direction::Heisenberg).
156///
157/// A diagnostic answering "how much of this circuit crosses a partition boundary?" without running a layer.
158/// The counts do not depend on `rank`, so it reports from partition 0.
159///
160/// # Panics
161///
162/// Panics if any channel declines [`Channel::prepare`](crate::Channel::prepare), the same condition on which `propagate` panics.
163pub fn count_remote_deltas<const W: usize>(
164    circuit: &Circuit<W>,
165    hash: &Gf2Hash<W>,
166    rows: &PartitionRows<W>,
167    adjoint: bool,
168) -> Vec<(usize, usize)> {
169    let n = circuit.channels.len();
170    (0..n)
171        .map(|k| if adjoint { n - 1 - k } else { k })
172        .map(|idx| {
173            let prep = circuit.channels[idx]
174                .prepare(hash, adjoint)
175                .unwrap_or_else(|| {
176                    panic!("count_remote_deltas: layer {idx} declined Channel::prepare")
177                });
178            let plan = PartitionPlan::new(&prep, rows, 0);
179            let local = plan.local_entries.iter().filter(|b| **b).count();
180            (local, plan.remote.len())
181        })
182        .collect()
183}

185#[cfg(test)]
186impl PartitionPlan {
187    /// The partitions this layer exports to, distinct and ascending.
188    ///
189    /// The engine routes by [`Self::remote`] directly; this is the tests' way of asking the same question as a set.
190    fn partners(&self) -> impl Iterator<Item = u32> + '_ {
191        let mut v: Vec<u32> = self.remote.iter().map(|r| r.partner).collect();
192        v.sort_unstable();
193        v.dedup();
194        v.into_iter()
195    }
196}

210    fn hash() -> Gf2Hash<2> {
211        Gf2Hash::<2>::new(NQ, 8, 0xB00C)

214    /// One partition row that is `Z` on `qubit` and nothing else: `part(v)` is then the z-bit of `v` on that qubit (the row's z-mask is what meets the key's z-word in `partition_of`).
215    fn z_row(qubit: u32) -> PartitionRows<2> {
216        let mut rz = [0u64; 2];
217        rz[qubit as usize / 64] = 1u64 << (qubit % 64);
218        PartitionRows::<2>::from_rows(NQ, vec![[0u64; 2]], vec![rz])
219    }

221    fn prep_of<C: Channel<2>>(ch: &C) -> Prepared<2> {
222        ch.prepare(&hash(), false).unwrap()
223    }

227    #[test]
228    fn without_partition_rows_nothing_is_remote() {
229        let rows = PartitionRows::<2>::none(NQ);
230        for (name, ch) in [
231            ("h", Box::new(Clifford1Q::h(3)) as Box<dyn Channel<2>>),
232            ("cnot", Box::new(Clifford2Q::cnot(1, 4))),
233            (
234                "haar",
235                Box::new(GeneralUnitary2Q::from_matrix(1, 5, haar_su4_matrix())),
236            ),
237        ] {
238            let prep = ch.prepare(&hash(), false).unwrap();
239            let plan = PartitionPlan::new(&prep, &rows, 0);
240            assert!(!plan.has_remote(), "{name}: unexpected remote deltas");
241            assert!(plan.remote.is_empty(), "{name}");
242            assert_eq!(plan.partners().count(), 0, "{name}");
243            assert!(plan.local_entries.iter().all(|b| *b), "{name}");
244            assert_eq!(
245                plan.local_bucket_deltas,
246                prep.bucket_deltas(),
247                "{name}: local span input differs from the channel's own",
248            );
249            let Prepared::Local(ptm) = &prep else {
250                panic!("{name}: expected Local")
251            };
252            assert_eq!(plan.rest_streams_total, ptm.num_deltas() - 1, "{name}");
253        }
254    }

273    #[test]
274    fn a_delta_whose_mask_sets_the_partition_bit_is_remote() {
275        // H(3)'s delta set is {0, XZ on qubit 3}. A single partition row that is Z on qubit 3 reads the z-bit there, which that mask sets, so `part(XZ_3) = 1`.
276        let rows = z_row(3);
277        assert_eq!(rows.num_partitions(), 2);
278        let prep = prep_of(&Clifford1Q::h(3));
279        let Prepared::Local(ptm) = &prep else {
280            panic!("expected Local")
281        };
282        assert_eq!(ptm.num_deltas(), 2);
283
284        for rank in 0..2u32 {
285            let plan = PartitionPlan::new(&prep, &rows, rank);
286            assert_eq!(plan.local_entries, vec![true, false], "rank {rank}");
287            assert!(plan.has_remote(), "rank {rank}");
288            assert_eq!(
289                plan.remote,
290                vec![RemoteDelta {
291                    entry: 1,
292                    partner: rank ^ 1,
293                    bucket_delta: ptm.deltas()[1].bucket_delta,
294                    partition_delta: 1,
295                }],
296                "rank {rank}",
297            );
298            // Only the identity delta stays behind, so the local coset loop is over a single bucket.
299            assert_eq!(plan.local_bucket_deltas, vec![0], "rank {rank}");
300            // ...but the sort-kernel gate still sees the channel's full fanout.
301            assert_eq!(plan.rest_streams_total, 1, "rank {rank}");
302            assert_eq!(
303                plan.partners().collect::<Vec<_>>(),
304                vec![rank ^ 1],
305                "rank {rank}",
306            );
307            assert_eq!(
308                plan.remote_for_partner(rank ^ 1).count(),
309                1,
310                "rank {rank}: partner stream",
311            );
312            assert_eq!(plan.remote_for_partner(rank).count(), 0, "rank {rank}");
313        }
314    }

346    #[test]
347    fn local_and_remote_cover_every_entry_exactly_once() {
348        let mut rng = Xs64::new(0xA11CE);
349        let prep = prep_of(&GeneralUnitary2Q::from_matrix(1, 5, haar_su4_matrix()));
350        let Prepared::Local(ptm) = &prep else {
351            panic!("expected Local")
352        };
353        // The dense fixture realizes all sixteen deltas.
354        assert_eq!(ptm.num_deltas(), 16);
355
356        for trial in 0..32 {
357            let bits = 1 + (trial % 3);
358            let rows_x: Vec<[u64; 2]> = (0..bits).map(|_| rng.next_array::<2>()).collect();
359            let rows_z: Vec<[u64; 2]> = (0..bits).map(|_| rng.next_array::<2>()).collect();
360            let rows = PartitionRows::<2>::from_rows(NQ, rows_x, rows_z);
361            let rank = (rng.next_u64() as u32) % rows.num_partitions() as u32;
362            let plan = PartitionPlan::new(&prep, &rows, rank);
363
364            assert_eq!(plan.local_entries.len(), ptm.num_deltas());
365            let n_local = plan.local_entries.iter().filter(|b| **b).count();
366            assert_eq!(n_local + plan.remote.len(), ptm.num_deltas());
367            // Remote entries are exactly the false slots, in ascending order.
368            let remote_entries: Vec<usize> = plan.remote.iter().map(|r| r.entry).collect();
369            let expected: Vec<usize> = plan
370                .local_entries
371                .iter()
372                .enumerate()
373                .filter(|(_, b)| !**b)
374                .map(|(e, _)| e)
375                .collect();
376            assert_eq!(remote_entries, expected, "trial {trial}");
377            for r in &plan.remote {
378                assert_ne!(r.partition_delta, 0);
379                assert_eq!(r.partner, rank ^ r.partition_delta);
380                assert_ne!(r.partner, rank, "a partition never ships to itself");
381                assert_eq!(r.bucket_delta, ptm.deltas()[r.entry].bucket_delta);
382            }
383            // Partners: distinct and ascending, and every remote delta's partner appears.
384            let partners: Vec<u32> = plan.partners().collect();
385            assert!(partners.windows(2).all(|w| w[0] < w[1]), "trial {trial}");
386            for r in &plan.remote {
387                assert!(partners.contains(&r.partner), "trial {trial}");
388            }
389            assert_eq!(
390                partners
391                    .iter()
392                    .map(|&q| plan.remote_for_partner(q).count())
393                    .sum::<usize>(),
394                plan.remote.len(),
395                "trial {trial}",
396            );
397            // The local span input is exactly the local entries' bucket deltas (plus 0, which the identity always contributes anyway).
398            let mut want: Vec<u32> = vec![0];
399            for (e, keep) in plan.local_entries.iter().enumerate() {
400                if *keep {
401                    want.push(ptm.deltas()[e].bucket_delta);
402                }
403            }
404            want.sort_unstable();
405            want.dedup();
406            assert_eq!(plan.local_bucket_deltas, want, "trial {trial}");
407            assert_eq!(plan.rest_streams_total, 15, "trial {trial}");
408        }
409    }

413    #[test]
414    fn count_remote_deltas_reports_layers_in_application_order() {
415        // Two layers with hand-checkable delta sets under a Z-row on qubit 3:
416        //   H(3):  deltas {0, XZ_3} -> part = {0, 1}  -> (1 local, 1 remote)
417        //   S(9):  deltas {0,  Z_9} -> part = {0, 0}  -> (2 local, 0 remote)
418        // (S maps X -> Y and Y -> -X, so its only nonzero delta is a z-bit.)
419        let rows = z_row(3);
420        let h = hash();
421        let mut circuit = Circuit::<2>::new(NQ);
422        circuit.push(Clifford1Q::h(3));
423        circuit.push(Clifford1Q::s(9));
424
425        assert_eq!(
426            count_remote_deltas(&circuit, &h, &rows, false),
427            vec![(1, 1), (2, 0)],
428        );
429        // Heisenberg order is the reverse. `S` is not self-adjoint, but its adjoint's delta set is the same subspace, so the counts do not move.
430        assert_eq!(
431            count_remote_deltas(&circuit, &h, &rows, true),
432            vec![(2, 0), (1, 1)],
433        );
434    }
