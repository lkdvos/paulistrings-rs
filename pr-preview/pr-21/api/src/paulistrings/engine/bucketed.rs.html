27/// Reusable layer working set for [`propagate_with`](crate::propagate_with), which keeps its capacity across layers and calls and carries the opt-in traces.
28///
29/// A task's output cannot depend on which worker slot it drew: every write destination is cleared before use.
30#[derive(Debug, Default)]
31pub struct LayerScratch<const W: usize> {
32    /// The serial path's coset working set.
33    pub(super) task: CosetScratch<W>,
34    /// The layer's handle permutation, `permutation[β] = span.permuted_index(β)`.
35    pub(super) permutation: Vec<u32>,
36    /// The inverse of [`Self::permutation`], filled only under [`ExtraRows::NEEDS_BETA`]; empty means the identity.
37    pub(super) inverse_permutation: Vec<u32>,
38    /// Bucket handles in coset-contiguous order while a layer runs.
39    pub(super) staging: Vec<BucketColumns<W>>,
40    /// One coset working set per Rayon worker, indexed by `rayon::current_thread_index()`, so each mutex is uncontended.
41    pub(super) workers: Vec<Mutex<CosetScratch<W>>>,
42    /// Layer-level wall-clock counters; per-coset busy time lives in each `CosetScratch`.
43    #[cfg(feature = "phase-timing")]
44    pub(crate) stats: PhaseStats,
45    /// `None` unless [`Self::enable_term_trace`] was called.
46    pub(crate) term_trace: Option<TermTrace>,
47    /// `None` unless [`Self::enable_gate_trace`] was called.
48    pub(crate) gate_trace: Option<GateTrace>,
49}
50
51impl<const W: usize> LayerScratch<W> {
52    /// An empty scratch.
53    pub fn new() -> Self {
54        Self::default()
55    }
56
57    /// Drain the phase counters accumulated since the last drain, across layers and calls.
58    #[cfg(feature = "phase-timing")]
59    pub fn take_stats(&mut self) -> PhaseStats {
60        let mut total = std::mem::take(&mut self.stats);
61        total.absorb_coset(&std::mem::take(&mut self.task.stats));
62        for slot in &self.workers {
63            let mut worker_scratch = slot.lock().unwrap();
64            total.absorb_coset(&std::mem::take(&mut worker_scratch.stats));
65        }
66        total
67    }
68
69    /// Record a [`TermTrace`] on every later call driven by this scratch; idempotent.
70    pub fn enable_term_trace(&mut self) {
71        self.term_trace.get_or_insert_with(TermTrace::default);
72    }
73
74    /// Drain the term trace, or `None` if it was never enabled; tracing stays enabled.
75    pub fn take_term_trace(&mut self) -> Option<TermTrace> {
76        self.term_trace.as_mut().map(std::mem::take)
77    }
78
79    /// Record a [`GateTrace`] on every later call driven by this scratch; idempotent.
80    pub fn enable_gate_trace(&mut self) {
81        self.gate_trace.get_or_insert_with(GateTrace::default);
82    }
83
84    /// Drain the gate trace, or `None` if it was never enabled; tracing stays enabled.
85    pub fn take_gate_trace(&mut self) -> Option<GateTrace> {
86        self.gate_trace.as_mut().map(std::mem::take)
87    }
88}

90/// Resident term counts between layers, one entry per layer in application order.
91///
92/// Counts are post-truncation; the transient expansion inside a layer is not captured.
93#[derive(Clone, Debug, Default, PartialEq, Eq)]
94pub struct TermTrace {
95    /// Resident term count before each layer.
96    pub terms_in: Vec<usize>,
97    /// Resident term count after each layer's `finalize_layer`.
98    pub terms_out: Vec<usize>,
99}
100
101impl TermTrace {
102    /// Peak resident term count between layers, or `None` for a zero-layer trace.
103    pub fn peak_terms(&self) -> Option<usize> {
104        self.terms_in
105            .first()
106            .copied()
107            .into_iter()
108            .chain(self.terms_out.iter().copied())
109            .max()
110    }
111}

114#[derive(Clone, Debug, Default, PartialEq)]

119    pub application_index: Vec<u32>,

121    pub gate_name: Vec<&'static str>,
122    /// Resident term count before the layer.
123    pub terms_in: Vec<usize>,
124    /// Resident term count after the layer, i.e. after `finalize_layer`.
125    pub terms_out: Vec<usize>,
126    /// Wall-clock nanoseconds for the whole layer, from `rebucket` through `finalize_layer`.
127    pub nanos: Vec<u64>,

130/// A prepared channel's delta set annotated with each entry's coset coordinate, computed once per layer.
131pub(super) enum DeltaPlan<'p, const W: usize> {
132    /// Tabulated deltas; `coords[e]` pairs with `ptm.deltas()[e]`.
133    Local {
134        ptm: &'p LocalPtm<W>,
135        coords: Vec<u32>,
136        /// Whether `deltas()[0]` is the identity delta, whose stream goes to the run's pre-sorted `id` columns.
137        has_identity: bool,
138        /// Whether the identity amplitude is nonzero on every active support pattern, so the merge borrows the id keys from the source bucket (ARCHITECTURE.md §Engine).
139        dense_identity: bool,
140        /// Whether this layer's runs use the radix sort kernel, decided once per layer (see `RADIX_MIN_REST_STREAMS`).
141        radix_sort: bool,
142    },
143    /// Wide rotation: two implicit entries, the identity pass and the generator pass.
144    Rotation {
145        rotation: &'p PreparedRotation<W>,
146        coord_identity: u32,
147        generator_coordinate: u32,
148        /// Whether the generator pass emits here ([`LayerKnobs::generator_local`]); `generator_coordinate` is 0 when not.
149        generator_local: bool,
150    },
151}

153/// Per-layer overrides the partitioned engine hands the coset loop; [`Default`] is the unpartitioned answer.
154#[derive(Clone, Copy, Debug)]
155pub(crate) struct LayerKnobs<'k> {
156    /// Bucket deltas to build the coset span from, instead of [`Prepared::bucket_deltas`].
157    pub bucket_deltas: Option<&'k [u32]>,
158    /// The channel's total rest-stream count, so every partition picks the sort kernel the unpartitioned run would.
159    pub rest_streams: Option<usize>,
160    /// The unrestricted channel's [`rest_rows_per_key`], since a PTM cut down to local entries can look more disjoint than the channel is.
161    pub rows_per_key: Option<f64>,
162    /// Whether a wide rotation's generator pass emits here; false when every generator row belongs to a partner.
163    pub generator_local: bool,
164}
165
166impl Default for LayerKnobs<'_> {
167    fn default() -> Self {
168        Self {
169            bucket_deltas: None,
170            rest_streams: None,
171            rows_per_key: None,
172            generator_local: true,
173        }
174    }
175}

177/// Mean rest rows per output key that gets any, the radix gate's second arm; ask it of the channel's own PTM, never a restricted one.
178pub(super) fn rest_rows_per_key<const W: usize>(ptm: &LocalPtm<W>) -> f64 {
179    let rest_start = usize::from(ptm.deltas().first().is_some_and(|d| d.local_delta == 0));
180    let dim = 1usize << (2 * ptm.k());
181    let (mut rows, mut keys) = (0u32, 0u32);
182    for output in 0..dim {
183        let rows_here = ptm.deltas()[rest_start..]
184            .iter()
185            .filter(|delta| delta.amplitude[output ^ delta.local_delta as usize] != ZERO)
186            .count() as u32;
187        rows += rows_here;
188        keys += u32::from(rows_here > 0);
189    }
190    if keys == 0 {
191        0.0
192    } else {
193        f64::from(rows) / f64::from(keys)
194    }
195}

197impl<'p, const W: usize> DeltaPlan<'p, W> {
198    pub(super) fn new(prepared: &'p Prepared<W>, span: &Gf2Span, knobs: LayerKnobs<'_>) -> Self {
199        match prepared {
200            Prepared::Local(ptm) => {
201                let coords: Vec<u32> = ptm
202                    .deltas()
203                    .iter()
204                    .map(|d| span.coord_of(d.bucket_delta))
205                    .collect();
206                let has_identity = ptm.deltas().first().is_some_and(|d| d.local_delta == 0);
207                debug_assert!(!has_identity || coords[0] == 0);
208                // `amplitude` is sized LOCAL_DIM but only `4^k` entries are populated.
209                let dim = 1usize << (2 * ptm.k());
210                let dense_identity =
211                    has_identity && ptm.deltas()[0].amplitude[..dim].iter().all(|a| *a != ZERO);
212                let rest_streams = knobs
213                    .rest_streams
214                    .unwrap_or(ptm.deltas().len() - has_identity as usize);
215                let radix_sort = rest_streams >= RADIX_MIN_REST_STREAMS
216                    || (rest_streams >= RADIX_MIN_DISJOINT_STREAMS
217                        && knobs.rows_per_key.unwrap_or_else(|| rest_rows_per_key(ptm))
218                            < RADIX_MAX_REST_ROWS_PER_KEY);
219                DeltaPlan::Local {
220                    ptm,
221                    coords,
222                    has_identity,
223                    dense_identity,
224                    radix_sort,
225                }
226            }
227            Prepared::Rotation(rotation) => DeltaPlan::Rotation {
228                rotation,
229                coord_identity: span.coord_of(rotation.bucket_delta_identity),
230                // `coord_of` demands its argument be in the span, and a remote generator's bucket delta is not.
231                generator_coordinate: if knobs.generator_local {
232                    span.coord_of(rotation.bucket_delta_generator)
233                } else {
234                    0
235                },
236                generator_local: knobs.generator_local,
237            },
238        }
239    }
240}

242/// Rows received from other partitions, appended to an output bucket's rest stream before the sort so `keep_term` sees the complete sum (ARCHITECTURE.md §Truncation).
243/// An implementation that can return rows must set `NEEDS_BETA = true`; under `false` every call site compiles away.
244pub(crate) trait ExtraRows<const W: usize> {
245    /// Whether the engine recovers original bucket indices and calls this source at all.
246    const NEEDS_BETA: bool = false;
247
248    /// Rows destined for original bucket `beta`; must equal what [`append_into`](Self::append_into) pushes.
249    fn count(&self, beta: u32) -> usize {
250        let _ = beta;
251        0
252    }
253
254    /// Append original bucket `beta`'s rows onto the run's rest columns.
255    fn append_into(
256        &self,
257        beta: u32,
258        x: &mut Vec<[u64; W]>,
259        z: &mut Vec<[u64; W]>,
260        c: &mut Vec<Complex64>,
261    ) {
262        let _ = (beta, x, z, c);
263    }
264}

272pub fn apply_layer_bucketed<const W: usize, T>(
273    sum: &mut PauliSum<W>,
274    prepared: &Prepared<W>,
275    policy: &T,
276    scratch: &mut LayerScratch<W>,

281        sum,
282        prepared,
283        policy,
284        scratch,
285        &NoExtra,
286        LayerKnobs::default(),
287    )

290/// [`apply_layer_bucketed`] with the partitioned engine's [`ExtraRows`] source and [`LayerKnobs`].
291pub(crate) fn apply_layer_bucketed_with<const W: usize, T, X>(
292    sum: &mut PauliSum<W>,
293    prepared: &Prepared<W>,
294    policy: &T,
295    scratch: &mut LayerScratch<W>,
296    extra: &X,
297    knobs: LayerKnobs<'_>,
298) where
299    T: TruncationPolicy<W> + ?Sized,
300    X: ExtraRows<W> + Sync,
301{
302    #[cfg(feature = "phase-timing")]
303    let mut stamp = Stamp::now();
304
305    // A partitioned layer's local table can look key-preserving while received rows still need merging.
306    if let Prepared::Local(ptm) = prepared {
307        if !X::NEEDS_BETA && ptm.is_key_preserving() {
308            rescale_in_place(sum, ptm, policy);
309            #[cfg(feature = "phase-timing")]
310            stamp.lap(&mut scratch.stats.rescale_ns);
311            return;
312        }
313    }
314
315    let own_deltas;
316    let deltas: &[u32] = match knobs.bucket_deltas {
317        Some(deltas) => deltas,
318        None => {
319            own_deltas = prepared.bucket_deltas();
320            &own_deltas
321        }
322    };
323    let span = Gf2Span::new(deltas, sum.hash().bits());
324    let plan = DeltaPlan::new(prepared, &span, knobs);
325    let coset_size = span.coset_size();
326    let num_cosets = span.num_cosets();
327    #[cfg(feature = "phase-timing")]
328    stamp.lap(&mut scratch.stats.span_plan_ns);
329
330    // Coset `c` owns `staging[c·2^r .. (c+1)·2^r]`; at `r = 0` the permutation is the identity and the handle passes are skipped.
331    let identity_perm = span.r() == 0;
332    if !identity_perm {
333        let buckets = sum.buckets_mut();
334        scratch.permutation.clear();
335        scratch
336            .permutation
337            .extend((0..buckets.len() as u32).map(|beta| span.permuted_index(beta)));
338        scratch
339            .staging
340            .resize_with(buckets.len(), BucketColumns::default);
341        for (beta, columns) in buckets.iter_mut().enumerate() {
342            scratch.staging[scratch.permutation[beta] as usize] = std::mem::take(columns);
343        }
344    }
345    if X::NEEDS_BETA {
346        scratch.inverse_permutation.clear();
347        if !identity_perm {
348            scratch
349                .inverse_permutation
350                .resize(scratch.permutation.len(), 0);
351            for (beta, &position) in scratch.permutation.iter().enumerate() {
352                scratch.inverse_permutation[position as usize] = beta as u32;
353            }
354        }
355    }
356    #[cfg(feature = "phase-timing")]
357    stamp.lap(&mut scratch.stats.permute_ns);
358
359    // Each coset reads and writes only its own chunk (ARCHITECTURE.md §Parallelism).
360    {
361        if num_cosets >= MIN_COSETS_FOR_PARALLEL {
362            let pool = rayon::current_num_threads().max(1);
363            if scratch.workers.len() < pool {
364                scratch.workers.resize_with(pool, Mutex::default);
365            }
366        }
367        let workers = &scratch.workers;
368        let inverse_permutation: &[u32] = &scratch.inverse_permutation;
369        let chunks: &mut [BucketColumns<W>] = if identity_perm {
370            sum.buckets_mut()
371        } else {
372            scratch.staging.as_mut_slice()
373        };
374        if num_cosets < MIN_COSETS_FOR_PARALLEL {
375            for (coset_index, chunk) in chunks.chunks_mut(coset_size).enumerate() {
376                let base = coset_index * coset_size;
377                fill_coset::<W, T, X>(
378                    chunk,
379                    &plan,
380                    policy,
381                    &mut scratch.task,
382                    extra,
383                    base,
384                    inverse_permutation,
385                );
386            }
387        } else {
388            chunks
389                .par_chunks_mut(coset_size)
390                .enumerate()
391                .for_each(|(coset_index, chunk)| {
392                    let base = coset_index * coset_size;
393                    // The fresh-scratch arm is a defensive fallback; a pool worker always has an index.
394                    match rayon::current_thread_index() {
395                        Some(worker) if worker < workers.len() => {
396                            let mut worker_scratch = workers[worker].lock().unwrap();
397                            fill_coset::<W, T, X>(
398                                chunk,
399                                &plan,
400                                policy,
401                                &mut worker_scratch,
402                                extra,
403                                base,
404                                inverse_permutation,
405                            );
406                        }
407                        _ => {
408                            let mut worker_scratch = CosetScratch::<W>::default();
409                            fill_coset::<W, T, X>(
410                                chunk,
411                                &plan,
412                                policy,
413                                &mut worker_scratch,
414                                extra,
415                                base,
416                                inverse_permutation,
417                            );
418                        }
419                    }
420                });
421        }
422    }
423    #[cfg(feature = "phase-timing")]
424    stamp.lap(&mut scratch.stats.coset_loop_ns);
425
426    if !identity_perm {
427        let buckets = sum.buckets_mut();
428        for (beta, columns) in buckets.iter_mut().enumerate() {
429            *columns = std::mem::take(&mut scratch.staging[scratch.permutation[beta] as usize]);
430        }
431    }
432    #[cfg(feature = "phase-timing")]
433    stamp.lap(&mut scratch.stats.unpermute_ns);
434    sum.recount();
435    #[cfg(feature = "phase-timing")]
436    stamp.lap(&mut scratch.stats.recount_ns);
437
438    #[cfg(debug_assertions)]
439    sum.assert_invariants();
440}

442/// In-place coefficient rescale for a key-preserving channel, with the general path's zero drop and `keep_term`.
443fn rescale_in_place<const W: usize, T>(sum: &mut PauliSum<W>, ptm: &LocalPtm<W>, policy: &T)
444where
445    T: TruncationPolicy<W> + ?Sized,
446{
447    let amplitude = &ptm.deltas()[0].amplitude;
448    sum.buckets_mut().par_iter_mut().for_each(|columns| {
449        let len = columns.len();
450        let mut keep = 0usize;
451        for i in 0..len {
452            let pattern = ptm.support_bits(&columns.x[i], &columns.z[i]);
453            let coeff = columns.coeff[i] * amplitude[pattern];
454            if coeff == ZERO || !policy.keep_term(&columns.x[i], &columns.z[i], coeff) {
455                continue;
456            }
457            columns.x[keep] = columns.x[i];
458            columns.z[keep] = columns.z[i];
459            columns.coeff[keep] = coeff;
460            keep += 1;
461        }
462        columns.x.truncate(keep);
463        columns.z.truncate(keep);
464        columns.coeff.truncate(keep);
465    });
466    sum.recount();
467
468    #[cfg(debug_assertions)]
469    sum.assert_invariants();
470}
