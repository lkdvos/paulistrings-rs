1//! One layer of the bucketed engine, applied coset by coset, and the reusable `LayerScratch`.
2//!
3//! A coset task swaps its bucket columns into worker scratch and writes its outputs back into the emptied columns, so the layer runs in place.
4//! Identity-delta rows keep their keys, so a run's identity stream inherits the source bucket's order and is never sorted; every other row, received rows included, joins the rest stream.
5//! A table shaped like a rotation skips both streams and runs `rotation_layer` on each bucket in place.

39/// Reusable layer working set for [`propagate_with`](crate::propagate_with), which keeps its capacity across layers and calls and carries the opt-in traces.
40///
41/// A task's output cannot depend on which worker slot it drew: every write destination is cleared before use.
42#[derive(Debug, Default)]
43pub struct LayerScratch<const W: usize, C: Coefficient = num_complex::Complex64> {
44    /// The serial path's coset working set.
45    pub(super) task: CosetScratch<W, C>,
46    /// The partitioned layer's coset order, kept here for its allocations; the coset loop itself enumerates cosets from the span.
47    pub(super) order: CosetOrder,
48    /// One coset working set per Rayon worker, indexed by `rayon::current_thread_index()`, so each mutex is uncontended.
49    pub(super) workers: Vec<Mutex<CosetScratch<W, C>>>,
50    /// Layer-level wall-clock counters; per-coset busy time lives in each `CosetScratch`.
51    #[cfg(feature = "phase-timing")]
52    pub(crate) stats: PhaseStats,
53    /// `None` unless [`Self::enable_trace`] was called.
54    pub(crate) trace: Option<PartitionTrace>,
55    /// The octave counts an `ApproxTopN` layer pass reads instead of the sum.
56    pub(crate) octaves: LayerOctaves,
57}

59/// Per-bucket octave counts, taken by the coset loop as it writes each bucket, and the cut they decide, applied as the next coset loop reads each bucket.
60///
61/// Contract: a cut is pending only between a layer pass and the next layer's coset loop, and anything else that reads or rebuckets the sum first applies it ([`settle`](Self::settle)), except an exchanging layer's export, which reads around it ([`pending`](Self::pending)).
62/// Every partition's window is set from the agreed boundary octave, so all count alike.
63#[derive(Debug, Default)]
64pub(crate) struct LayerOctaves {
65    /// The window the next layer counts, centred on the last boundary octave; `None` before the first.
66    window: Option<OctaveWindow>,
67    per_bucket: Vec<BucketOctaves>,
68    /// Whether `per_bucket` counts the sum as the last layer left it, under `window`.
69    counted: bool,
70    /// The last layer pass's cut, not yet applied.
71    pending: Option<BoundaryCut>,
72    /// An upper bound on the whole sum's length, agreed by every partition: set by a layer pass, scaled by each layer's [`max_fanout`], reset at the end of a call.
73    bound: Option<usize>,
74    /// Whether this layer's coset loop takes counts, set by [`begin_layer`](Self::begin_layer).
75    armed: bool,
76    /// Whether this layer's output is known to fit in `n`, so its layer pass is skipped on every partition alike.
77    within: bool,
78    /// Per bucket, the counts ahead of the next layer's export.
79    ahead: Vec<BucketAhead>,
80    /// The window `ahead` is binned under; `None` when there are no counts ahead.
81    ahead_window: Option<OctaveWindow>,
82}
83
84impl LayerOctaves {
85    /// Decide what the coming layer does for an `ApproxTopN(n)` policy (`None` for any other), from its [`max_fanout`].
86    pub(crate) fn begin_layer(&mut self, fanout: usize, n: Option<usize>) {
87        let Some(n) = n else {
88            (self.armed, self.within) = (false, false);
89            return;
90        };
91        self.bound = self.bound.map(|bound| bound.saturating_mul(fanout));
92        self.within = self.bound.is_some_and(|bound| bound <= n);
93        self.armed = !self.within && self.window.is_some();
94    }
95
96    /// `ApproxTopN(n)`'s layer pass over `sum`, from the counts when the last layer took them, leaving an in-window cut pending.
97    pub(crate) fn finalize<const W: usize, C: Coefficient>(
98        &mut self,
99        sum: &mut PauliSum<W, C>,
100        collectives: &dyn Collectives,
101        n: usize,
102    ) {
103        debug_assert!(self.pending.is_none(), "the layer did not apply the last cut");
104        if std::mem::take(&mut self.within) {
105            return;
106        }
107        let decided = match self.window {
108            Some(window) if std::mem::take(&mut self.counted) => {
109                decide_from_window(sum.len(), collectives, n, window, &self.per_bucket)
110            }
111            _ => None,
112        };
113        let (edge, total) = match decided {
114            Some((edge, total, cut)) => {
115                if edge == EdgeDecision::Clear {
116                    sum.clear();
117                }
118                self.pending = cut;
119                (edge, total)
120            }
121            None => approx_top_n_pass(sum, collectives, n),
122        };
123        // A cut not left pending has changed the sum under the counts ahead.
124        if edge != EdgeDecision::KeepAll && self.pending.is_none() {
125            self.drop_ahead();
126        }
127        self.bound = Some(match edge {
128            EdgeDecision::KeepAll => total,
129            EdgeDecision::Clear => 0,
130            EdgeDecision::Boundary { bin, .. } => {
131                self.window = Some(OctaveWindow::around(bin));
132                n
133            }
134        });
135    }
136
137    pub(crate) fn len_after_cut(&self, len: usize) -> usize {
138        self.pending.as_ref().map_or(len, |cut| cut.kept)
139    }
140
141    /// The cut the next coset loop applies, which an export before it must honour.
142    pub(crate) fn pending(&self) -> Option<&BoundaryCut> {
143        self.pending.as_ref()
144    }
145
146    /// Apply the pending cut and drop the counts ahead, before anything outside a coset loop that changes the sum or its buckets.
147    pub(crate) fn apply_pending<const W: usize, C: Coefficient>(
148        &mut self,
149        sum: &mut PauliSum<W, C>,
150    ) {
151        self.drop_ahead();
152        if let Some(cut) = self.pending.take() {
153            cut.apply(sum);
154        }
155    }
156
157    /// Leave `cut` pending, as a layer pass would.
158    #[cfg(test)]
159    pub(crate) fn defer(&mut self, cut: BoundaryCut) {
160        self.pending = Some(cut);
161    }
162
163    /// End a call: apply the pending cut and forget the bound, since the sum may change before the next.
164    pub(crate) fn settle<const W: usize, C: Coefficient>(&mut self, sum: &mut PauliSum<W, C>) {
165        self.apply_pending(sum);
166        self.bound = None;
167    }
168}

170impl<const W: usize, C: Coefficient> LayerScratch<W, C> {
171    /// An empty scratch.
172    pub fn new() -> Self {
173        Self::default()
174    }
175
176    /// Drain the phase counters since the last drain, zeroing them; `per_partition` has one entry.
177    #[cfg(feature = "phase-timing")]
178    pub fn take_stats(&mut self) -> PartitionPhaseStats {
179        let stats = self.take_phase_stats();
180        PartitionPhaseStats {
181            layers: stats.layers,
182            per_partition: vec![stats],
183            scatter_ns: 0,
184            gather_ns: 0,
185        }
186    }
187
188    /// Drain this scratch's own counters, one partition's entry of a driver's [`PartitionPhaseStats`].
189    #[cfg(feature = "phase-timing")]
190    pub(crate) fn take_phase_stats(&mut self) -> PhaseStats {
191        let mut total = std::mem::take(&mut self.stats);
192        total.absorb_coset(&std::mem::take(&mut self.task.stats));
193        for slot in &self.workers {
194            let mut worker_scratch = slot.lock().unwrap();
195            total.absorb_coset(&std::mem::take(&mut worker_scratch.stats));
196        }
197        total
198    }
199
200    /// The counters [`Self::take_phase_stats`] would drain, left in place.
201    #[cfg(feature = "phase-timing")]
202    pub(crate) fn peek_phase_stats(&self) -> PhaseStats {
203        let mut total = self.stats;
204        total.absorb_coset(&self.task.stats);
205        for slot in &self.workers {
206            total.absorb_coset(&slot.lock().unwrap().stats);
207        }
208        total
209    }
210
211    /// Start recording a one-partition [`PartitionTrace`] on every later propagation driven by this scratch; idempotent.
212    pub fn enable_trace(&mut self) {
213        self.trace.get_or_insert_with(PartitionTrace::default);
214    }
215
216    /// Drain the per-layer records, or `None` if tracing was never enabled; tracing stays enabled.
217    pub fn take_trace(&mut self) -> Option<PartitionTrace> {
218        self.trace.as_mut().map(std::mem::take)
219    }

222/// A prepared channel's delta set annotated with each entry's coset coordinate, computed once per layer.
223pub(super) enum DeltaPlan<'p, const W: usize, C: Coefficient = num_complex::Complex64> {
224    /// Tabulated deltas; `coordinates[e]` pairs with `ptm.deltas()[e]`.
225    Local {
226        ptm: &'p LocalPtm<W, C>,
227        coordinates: Vec<u32>,
228        /// Whether `deltas()[0]` is the identity delta, whose stream goes to the run's pre-sorted `id` columns.
229        has_identity: bool,
230        /// Whether the identity amplitude is nonzero on every active support pattern, so the identity stream is one-to-one with the source bucket and the merge borrows its key columns.
231        dense_identity: bool,
232        /// This layer's per-run sort kernel.
233        sort_kernel: SortKernel,
234    },
235    /// Wide rotation: two implicit entries, the identity pass and the generator pass.
236    Rotation {
237        rotation: &'p PreparedRotation<W, C::Real>,
238        coordinate_identity: u32,
239        generator_coordinate: u32,
240    },
241}

243/// Per-layer overrides the partitioned engine hands the coset loop; [`Default`] is the unpartitioned answer.
244#[derive(Clone, Copy, Debug)]
245pub(crate) struct LayerKnobs<'k> {
246    /// Bucket deltas to build the coset span from, instead of [`Prepared::bucket_deltas`].
247    pub bucket_deltas: Option<&'k [u32]>,
248    /// The channel's total rest-stream count, so every partition picks the sort kernel the unpartitioned run would.
249    pub rest_streams: Option<usize>,
250    /// The unrestricted channel's [`rest_rows_per_key`], since a PTM cut down to local entries can look more disjoint than the channel is.
251    pub rows_per_key: Option<f64>,
252    /// Whether a rotation's generator pass emits here; false when the generator crosses to a partner, which sends the layer, tabulated or wide, to `fill_received_rotation_bucket`.
253    pub generator_local: bool,
254    /// The coset order a partitioned layer exported and chunks by, whose coset `c` must be the loop's coset `c` ([`Gf2Span::representative_at`]); checked in debug builds.
255    pub order: Option<&'k CosetOrder>,
256}
257
258impl Default for LayerKnobs<'_> {
259    fn default() -> Self {
260        Self {
261            bucket_deltas: None,
262            rest_streams: None,
263            rows_per_key: None,
264            generator_local: true,
265            order: None,
266        }
267    }

270impl<'p, const W: usize, C: Coefficient> DeltaPlan<'p, W, C> {
271    pub(super) fn new(prepared: &'p Prepared<W, C>, span: &Gf2Span, knobs: LayerKnobs<'_>) -> Self {
272        match prepared {
273            Prepared::Local(ptm) => {
274                let coordinates: Vec<u32> =
275                    ptm.deltas().iter().map(|d| span.coordinate_of(d.bucket_delta)).collect();
276                let has_identity = ptm.has_identity();
277                debug_assert!(!has_identity || coordinates[0] == 0);
278                let dense_identity = has_identity && ptm.entry_is_dense(0);
279                let rest_streams = knobs.rest_streams.unwrap_or(ptm.rest_streams());
280                let sort_kernel = SortKernel::for_layer(rest_streams, || {
281                    knobs.rows_per_key.unwrap_or_else(|| rest_rows_per_key(ptm))
282                });
283                DeltaPlan::Local { ptm, coordinates, has_identity, dense_identity, sort_kernel }
284            }
285            Prepared::Rotation(rotation) => {
286                debug_assert!(
287                    knobs.generator_local,
288                    "a remote generator runs on `ReceivedRotation`"
289                );
290                DeltaPlan::Rotation {
291                    rotation,
292                    coordinate_identity: span.coordinate_of(rotation.bucket_delta_identity),
293                    generator_coordinate: span.coordinate_of(rotation.bucket_delta_generator),
294                }
295            }
296        }
297    }
298}

300/// Rows received from other partitions, appended to an output bucket's rest stream before the sort so `keep_term` sees the complete sum; never the identity stream, which the merge requires strictly ascending.
301/// An implementation that can return rows must set `NEEDS_BETA = true`; under `false` every call site compiles away.
302pub(crate) trait ExtraRows<const W: usize, C: Coefficient = num_complex::Complex64> {
303    /// Whether the engine recovers original bucket indices and calls this source at all.
304    const NEEDS_BETA: bool = false;
305
306    /// Rows destined for original bucket `beta`; must equal what [`append_into`](Self::append_into) pushes.
307    fn count(&self, beta: u32) -> usize {
308        let _ = beta;
309        0
310    }
311
312    /// Append original bucket `beta`'s rows onto the run's rest columns.
313    fn append_into(&self, beta: u32, rest: &mut BucketColumns<W, C>) {
314        let _ = (beta, rest);
315    }
316}

336/// [`apply_layer_bucketed`] with the partitioned engine's [`ExtraRows`] source and [`LayerKnobs`], counting the rows `ahead` emits for the next layer's export.
337pub(crate) fn apply_layer_bucketed_with<const W: usize, C: Coefficient, T, X>(
338    sum: &mut PauliSum<W, C>,
339    prepared: &Prepared<W, C>,
340    policy: &T,
341    scratch: &mut LayerScratch<W, C>,
342    extra: &X,
343    knobs: LayerKnobs<'_>,
344    ahead: Option<&RowEmitter<'_, W, C>>,
345) where
346    T: TruncationPolicy<W, C> + ?Sized,
347    X: ExtraRows<W, C> + Sync,
348{
349    stamp!(stamp);
350    scratch.octaves.counted = false;
351    scratch.octaves.drop_ahead();
352    let armed = std::mem::take(&mut scratch.octaves.armed);
353    let cut = scratch.octaves.pending.take();
354
355    // A partitioned layer's local table can look key-preserving while received rows still need merging.
356    let rotation = match prepared {
357        Prepared::Local(ptm) if !X::NEEDS_BETA => {
358            if ptm.is_key_preserving() {
359                if let Some(cut) = cut {
360                    cut.apply(sum);
361                }
362                rescale_in_place(sum, ptm, policy);
363                lap!(stamp, scratch.stats.rescale_ns);
364                return;
365            }
366            RotationLayer::of(ptm)
367        }
368        _ => None,
369    };
370    // A partitioned rotation whose generator pass is off exported every partner.
371    let received_rotation = match prepared {
372        _ if !X::NEEDS_BETA || knobs.generator_local => None,
373        Prepared::Local(ptm) => Some(ReceivedRotation::Table(
374            RotationLayer::of(ptm).expect("only a rotation table runs with its generator pass off"),
375        )),
376        Prepared::Rotation(rotation) => Some(ReceivedRotation::wide(rotation)),
377    };
378
379    let own_deltas;
380    let deltas: &[u32] = match knobs.bucket_deltas {
381        Some(deltas) => deltas,
382        None => {
383            own_deltas = prepared.bucket_deltas();
384            &own_deltas
385        }
386    };
387    let span = Gf2Span::new(deltas, sum.hash().bits());
388    debug_assert!(knobs.order.is_none_or(|order| order.dimension() == span.dimension()));
389
390    let window = scratch.octaves.window.filter(|_| armed);
391    let mut per_bucket = std::mem::take(&mut scratch.octaves.per_bucket);
392    if window.is_some() {
393        per_bucket.resize(sum.num_buckets(), [0; _]);
394    }
395    let octaves = window.map(|window| (window, DisjointSlots::new(&mut per_bucket)));
396    // Any window bins the counts ahead when no cut can follow, which takes an armed window.
397    let ahead_window = window.or(scratch.octaves.window).unwrap_or(OctaveWindow::around(0));
398    let mut ahead_counts = std::mem::take(&mut scratch.octaves.ahead);
399    if ahead.is_some() {
400        ahead_counts.resize_with(sum.num_buckets(), BucketAhead::default);
401    }
402    let ahead_slots = ahead.map(|emitter| (emitter, DisjointSlots::new(&mut ahead_counts)));
403    let octaves = LayerOctaveIo {
404        cut: cut.as_ref(),
405        count: octaves.as_ref().map(|(window, slots)| (*window, slots)),
406        ahead: ahead_slots.as_ref().map(|(emitter, slots)| (ahead_window, *emitter, slots)),
407    };
408    let len = match (received_rotation, rotation) {
409        (Some(layer), _) => {
410            debug_assert_eq!(span.dimension(), 0, "only the identity delta is local");
411            lap!(stamp, scratch.stats.span_plan_ns);
412            coset_loop(sum, &span, scratch, octaves, |members, betas, worker_scratch| {
413                fill_received_rotation_bucket(members, betas, &layer, policy, worker_scratch, extra)
414            })
415        }
416        (None, Some(layer)) => {
417            debug_assert!(knobs.order.is_none());
418            lap!(stamp, scratch.stats.span_plan_ns);
419            coset_loop(sum, &span, scratch, octaves, |members, _, worker_scratch| {
420                fill_rotation_coset(members, &layer, policy, worker_scratch)
421            })
422        }
423        (None, None) => {
424            let plan = DeltaPlan::new(prepared, &span, knobs);
425            lap!(stamp, scratch.stats.span_plan_ns);
426            coset_loop(sum, &span, scratch, octaves, |members, betas, worker_scratch| {
427                fill_coset::<W, C, T, X>(members, &plan, policy, worker_scratch, extra, betas)
428            })
429        }
430    };
431    scratch.octaves.per_bucket = per_bucket;
432    scratch.octaves.counted = window.is_some();
433    scratch.octaves.ahead = ahead_counts;
434    scratch.octaves.ahead_window = ahead.map(|_| ahead_window);
435    lap!(stamp, scratch.stats.coset_loop_ns);
436    sum.set_len_after_layer(len);
437    lap!(stamp, scratch.stats.recount_ns);
438
439    #[cfg(debug_assertions)]
440    sum.assert_invariants();
441}

443/// What a coset loop does for [`LayerOctaves`]: the cut it applies to each bucket before the fill, and the octave counts and counts ahead it takes after, with their windows.
444#[derive(Clone, Copy)]
445struct LayerOctaveIo<'s, 'a, 't, const W: usize, C: Coefficient> {
446    cut: Option<&'s BoundaryCut>,
447    count: Option<(OctaveWindow, &'s DisjointSlots<'a, BucketOctaves>)>,
448    ahead: Option<(OctaveWindow, &'s RowEmitter<'t, W, C>, &'s DisjointSlots<'a, BucketAhead>)>,
449}

451/// Run `fill(members, betas, scratch)` on every coset of `span`, serially or on the Rayon pool, each over its swapped-out bucket columns with `octaves`' cut and counts; returns the output terms.
452fn coset_loop<const W: usize, C: Coefficient, F>(
453    sum: &mut PauliSum<W, C>,
454    span: &Gf2Span,
455    scratch: &mut LayerScratch<W, C>,
456    octaves: LayerOctaveIo<'_, '_, '_, W, C>,
457    fill: F,
458) -> usize
459where
460    F: Fn(&mut [BucketColumns<W, C>], &[u32], &mut CosetScratch<W, C>) + Sync,
461{
462    let num_cosets = span.num_cosets();
463    // Each coset reads and writes only its own buckets (engine/coset.rs).
464    if num_cosets < MIN_COSETS_FOR_PARALLEL {
465        let buckets = DisjointSlots::new(sum.buckets_mut());
466        (0..num_cosets as u32)
467            .map(|coset| run_coset(coset, span, &buckets, octaves, &mut scratch.task, &fill))
468            .sum()
469    } else {
470        let pool = rayon::current_num_threads().max(1);
471        if scratch.workers.len() < pool {
472            scratch.workers.resize_with(pool, Mutex::default);
473        }
474        let workers = &scratch.workers;
475        let buckets = DisjointSlots::new(sum.buckets_mut());
476        (0..num_cosets as u32)
477            .into_par_iter()
478            .map(|coset| {
479                // `workers` was sized to `current_num_threads()` above, and this closure runs on that pool.
480                let worker = rayon::current_thread_index()
481                    .expect("the coset loop runs on a Rayon pool worker");
482                let mut worker_scratch = workers[worker].lock().unwrap();
483                run_coset(coset, span, &buckets, octaves, &mut worker_scratch, &fill)
484            })
485            .sum()
486    }
487}

489/// Swap coset `coset`'s buckets into the task's member slots, cut, fill and count them while they are in cache, and swap the outputs back; returns the coset's output terms.
490fn run_coset<const W: usize, C: Coefficient, F>(
491    coset: u32,
492    span: &Gf2Span,
493    buckets: &DisjointSlots<'_, BucketColumns<W, C>>,
494    octaves: LayerOctaveIo<'_, '_, '_, W, C>,
495    scratch: &mut CosetScratch<W, C>,
496    fill: &F,
497) -> usize
498where
499    F: Fn(&mut [BucketColumns<W, C>], &[u32], &mut CosetScratch<W, C>),
500{
501    let representative = span.representative_at(coset);
502    let mut members = std::mem::take(&mut scratch.members);
503    let mut betas = std::mem::take(&mut scratch.betas);
504    betas.clear();
505    betas.extend((0..span.coset_size() as u32).map(|j| span.member(representative, j)));
506    members.resize_with(betas.len(), BucketColumns::default);
507    for (slot, &beta) in members.iter_mut().zip(&betas) {
508        // SAFETY: the cosets partition the buckets, so no other task reaches bucket `beta`.
509        std::mem::swap(slot, unsafe { buckets.get(beta as usize) });
510        if let Some(cut) = octaves.cut {
511            cut.apply_to_bucket(beta as usize, slot);
512        }
513    }
514    fill(&mut members, &betas, scratch);
515    let mut len = 0;
516    for (slot, &beta) in members.iter_mut().zip(&betas) {
517        len += slot.len();
518        // SAFETY (both arms): as for the buckets, which the counts are indexed alike with.
519        match (octaves.count, octaves.ahead) {
520            (count, Some((window, emitter, ahead))) => {
521                debug_assert!(count.is_none_or(|(counted, _)| counted == window));
522                let counts = count.map(|(_, counts)| unsafe { counts.get(beta as usize) });
523                window.count_emitting(slot, emitter, counts, unsafe { ahead.get(beta as usize) });
524            }
525            (Some((window, counts)), None) => {
526                window.count(&slot.coeff, unsafe { counts.get(beta as usize) });
527            }
528            (None, None) => {}
529        }
530        // SAFETY: as above.
531        std::mem::swap(slot, unsafe { buckets.get(beta as usize) });
532    }
533    scratch.members = members;
534    scratch.betas = betas;
535    len
536}

538/// A slice shared by tasks that each touch pairwise distinct elements.
539struct DisjointSlots<'a, T> {
540    base: *mut T,
541    len: usize,
542    _slice: std::marker::PhantomData<&'a mut [T]>,
543}
544
545// SAFETY: `get`'s contract keeps every element reached by at most one task at a time, so sharing the wrapper shares no element.
546unsafe impl<T: Send> Sync for DisjointSlots<'_, T> {}
547
548impl<'a, T> DisjointSlots<'a, T> {
549    fn new(slice: &'a mut [T]) -> Self {
550        Self { base: slice.as_mut_ptr(), len: slice.len(), _slice: std::marker::PhantomData }
551    }
552
553    /// Element `index`.
554    ///
555    /// # Safety
556    ///
557    /// No other reference to element `index` may be live while the returned one is.
558    #[allow(clippy::mut_from_ref)]
559    unsafe fn get(&self, index: usize) -> &mut T {
560        assert!(index < self.len, "DisjointSlots: index {index} out of {}", self.len);
561        // SAFETY: in bounds, and exclusive by the caller's contract.
562        unsafe { &mut *self.base.add(index) }
563    }
564}

566/// Most output strings one input string can have under `prepared`, as whichever layer path runs it emits them.
567pub(crate) fn max_fanout<const W: usize, C: Coefficient>(prepared: &Prepared<W, C>) -> usize {
568    match prepared {
569        // The rotation kernels drop the identity residue of a Clifford rotation.
570        Prepared::Local(ptm) if RotationLayer::of(ptm).is_some_and(|layer| layer.is_clifford()) => {
571            1
572        }
573        Prepared::Local(ptm) => (0..LOCAL_DIM)
574            .map(|s| ptm.deltas().iter().filter(|delta| delta.amplitude[s].is_nonzero()).count())
575            .max()
576            .unwrap_or(0)
577            .max(1),
578        Prepared::Rotation(_) => 2,
579    }
580}

583pub(crate) fn is_rotation_table<const W: usize, C: Coefficient>(ptm: &LocalPtm<W, C>) -> bool {
584    RotationLayer::of(ptm).is_some()

587/// In-place coefficient rescale for a key-preserving channel, with the general path's zero drop and `keep_term`.
588fn rescale_in_place<const W: usize, C: Coefficient, T>(
589    sum: &mut PauliSum<W, C>,
590    ptm: &LocalPtm<W, C>,
591    policy: &T,
592) where
593    T: TruncationPolicy<W, C> + ?Sized,
594{
595    let amplitude = &ptm.deltas()[0].amplitude;
596    sum.update_per_bucket(|_| {
597        |x: &[u64; W], z: &[u64; W], c: C| {
598            let coeff = c * amplitude[ptm.support_bits(x, z)];
599            (coeff != C::ZERO && policy.keep_term(x, z, coeff)).then_some(coeff)
600        }
601    });
602
603    #[cfg(debug_assertions)]
604    sum.assert_invariants();
605}
