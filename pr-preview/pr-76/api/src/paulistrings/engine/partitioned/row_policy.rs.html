12/// The qubits per row [`PartitionRowPolicy::for_run`] gives parity rows, the best balance on four Blackwell GPUs (`research/FINDINGS.md` §X-parity rows on device groups).
13pub const FOR_RUN_PARITY_GROUP: usize = 2;
14
15/// Which partition rows a scatter splits by: a seeded GF(2)-random draw, a qubit cut that keeps a channel local when its qubits share a block, parity rows near a site, or explicit rows.
16#[derive(Clone, Debug, PartialEq, Eq)]
17pub enum PartitionRowPolicy {
18    /// [`PartitionRows::from_seed`] with this seed, or the sum's own hash seed at `None`; the default.
19    Seeded(Option<u64>),
20    /// [`PartitionRows::cut`]: one disjoint qubit block per rank, in rank order; the rows are z-only.
21    Cut(Vec<Vec<usize>>),
22    /// [`PartitionRows::from_seed_excluding`]: a seeded draw whose rows read neither the x-bits of `exclude_x` nor the z-bits of `exclude_z`, so keys differing only there share a partition.
23    SeededExcluding {
24        /// As in [`Seeded`](Self::Seeded).
25        seed: Option<u64>,
26        /// Qubits whose x-bit no row reads.
27        exclude_x: Vec<usize>,
28        /// Qubits whose z-bit no row reads.
29        exclude_z: Vec<usize>,
30    },
31    /// [`PartitionRows::from_matrix`]: one row per partition bit as a flag matrix over the `2 · num_qubits` coordinates, x-bits first.
32    Matrix(Vec<Vec<bool>>),
33    /// [`parity`](Self::parity): row `i` reads the `axis` bits of the qubits `order[i · group..(i + 1) · group]`.
34    Parity {
35        /// Which bits a row reads: x-bits for `X`, z-bits for `Z`, both for `Y`.
36        axis: PauliAxis,
37        /// The qubits, nearest the observable first; the rows read its first `log2(P) · group`.
38        order: Vec<usize>,
39        /// Qubits per row.
40        group: usize,
41    },
42    /// [`blocks`](Self::blocks): `order` cut into `P` contiguous runs, position `i` in run `⌊i · P / len⌋`, labelled as [`Cut`](Self::Cut) labels its blocks, the rows reading `axis` bits.
43    Blocks {
44        /// Which bits a row reads, as in [`Parity`](Self::Parity).
45        axis: PauliAxis,
46        /// Every qubit at most once; run `b` is partition `b`'s block.
47        order: Vec<usize>,
48    },
49}
50
51impl PartitionRowPolicy {
52    /// Parity rows: row `i` is the parity of the `axis` bits of the `i`-th `group` qubits of `order`, so a channel moves a term between partitions exactly when one of its key shifts sets an odd number of those bits in some row's group.
53    ///
54    /// `X` rows read x-bits only, so a channel whose shifts set no x-bit (`R_Z`, `R_ZZ`, any diagonal gate or dephasing) is local whatever its qubits, and only a channel setting x-bits on the rows' `log2(P) · group` qubits is remote; `Z` rows mirror that with `R_X` and `R_XX` local; `Y` rows read both bits, which keeps exactly the `Y`-type shifts local.
55    /// They balance an observable on few qubits whose terms grow outward from `order`'s head, since every term then has content on the rows' qubits; an observable spread over the register leaves most terms with none there and in partition 0 (`research/FINDINGS.md` §Parity rows balance a local observable, block rows an extensive one).
56    ///
57    /// # Panics
58    ///
59    /// When resolved, if `group` is zero, if the rows need more qubits than `order` holds, or if `order` repeats a qubit or names one outside the register.
60    pub fn parity(axis: PauliAxis, order: Vec<usize>, group: usize) -> Self {
61        Self::Parity { axis, order, group }
62    }
63
64    /// [`parity`](Self::parity) rows over the qubits nearest `sites` on the interaction graph `edges`, in [`nearest_first`](Self::nearest_first) order.
65    ///
66    /// # Panics
67    ///
68    /// As [`nearest_first`](Self::nearest_first), and as [`parity`](Self::parity) when resolved.
69    pub fn parity_around(
70        axis: PauliAxis,
71        num_qubits: usize,
72        edges: &[(usize, usize)],
73        sites: &[usize],
74        group: usize,
75    ) -> Self {
76        Self::parity(axis, Self::nearest_first(num_qubits, edges, sites), group)
77    }
78
79    /// Block rows: `order` cut into `P` contiguous runs, position `i` in run `⌊i · P / len⌋`, run `b` labelled `b` as [`PartitionRows::cut`] labels its blocks, every row reading `axis` bits.
80    ///
81    /// A term's label is the XOR of the labels of the runs it has odd `axis`-weight in, so under `Z` rows every `R_X` is local and a `ZZ` bond is remote exactly when it crosses runs; `X` rows mirror that.
82    /// They balance an observable spread over the register, whose terms occupy every run; a circuit that conserves the total `axis` parity starves half the partitions once the terms occupy few runs, as a local observable's do (`research/FINDINGS.md` §Row choice on heavy-hex: x-parity rows beat the z-only cut).
83    ///
84    /// # Panics
85    ///
86    /// When resolved, as [`PartitionRows::cut`] on the runs, and if `order` repeats a qubit.
87    pub fn blocks(axis: PauliAxis, order: Vec<usize>) -> Self {
88        Self::Blocks { axis, order }
89    }
90
91    /// Every qubit of a `num_qubits` register in breadth-first order over `edges` from all of `sites` at once, `sites` first in ascending order, each qubit's neighbours in ascending order, and the qubits `edges` never reach last in ascending order.
92    ///
93    /// # Panics
94    ///
95    /// If a site or an edge's qubit is outside the register.
96    pub fn nearest_first(
97        num_qubits: usize,
98        edges: &[(usize, usize)],
99        sites: &[usize],
100    ) -> Vec<usize> {
101        let mut adjacent = vec![Vec::new(); num_qubits];
102        for &(a, b) in edges {
103            assert!(
104                a < num_qubits && b < num_qubits,
105                "edge ({a}, {b}) is outside the {num_qubits}-qubit register"
106            );
107            adjacent[a].push(b);
108            adjacent[b].push(a);
109        }
110        let mut seen = vec![false; num_qubits];
111        let mut starts = sites.to_vec();
112        starts.sort_unstable();
113        starts.dedup();
114        let mut queue = VecDeque::new();
115        for &site in &starts {
116            assert!(site < num_qubits, "site {site} is outside the {num_qubits}-qubit register");
117            seen[site] = true;
118            queue.push_back(site);
119        }
120        let mut order = Vec::with_capacity(num_qubits);
121        while let Some(q) = queue.pop_front() {
122            order.push(q);
123            adjacent[q].sort_unstable();
124            for &next in &adjacent[q] {
125                if !seen[next] {
126                    seen[next] = true;
127                    queue.push_back(next);
128                }
129            }
130        }
131        order.extend((0..num_qubits).filter(|&q| !seen[q]));
132        order
133    }

135    /// The rows `circuit` and its `observable` suggest, from which bits the circuit's entanglers set and how far the observable spreads.
136    ///
137    /// When every channel on two or more qubits shifts only z-bits (`ZZ`-type entanglers) the parity axis is `X`, when only x-bits it is `Z`; any other circuit gets [`Seeded`](Self::Seeded) rows.
138    /// An observable on at most [`FOR_RUN_PARITY_GROUP`] qubits, under a circuit some channel of which sets parity-axis bits, gets [`parity_around`](Self::parity_around) rows at that axis from its support over the entanglers' graph, which keep every entangler local.
139    /// Any other observable gets [`blocks`](Self::blocks) over the qubits in index order, reading the bits the entanglers set, which keeps every single-qubit rotation about the parity axis local.
140    ///
141    /// Partitioned results agree with one partition's to rounding under any rows for an exact or threshold truncation; a partitioned `ApproxTopN` hands its boundary quota out in rank order and a collapse draws over the partitions in rank order, so their results depend on the rows (`research/FINDINGS.md` §The rank-order boundary quota biases cut-row splits), which is why this is not the scatters' default.
142    ///
143    /// # Panics
144    ///
145    /// When resolved, as [`parity`](Self::parity) or [`blocks`](Self::blocks): parity rows for `P` partitions need `2 · log2(P)` qubits.
146    pub fn for_run<const W: usize, C: Coefficient>(
147        observable: &PauliSum<W, C>,
148        circuit: &Circuit<W>,
149    ) -> Self {
150        let num_qubits = circuit.num_qubits();
151        let mut edges = Vec::new();
152        let (mut entangler_x, mut entangler_z, mut sets_x, mut sets_z) =
153            (false, false, false, false);
154        for (channel, shifts) in circuit.channels().iter().zip(circuit.deltas()) {
155            let Some(shifts) = shifts else { continue };
156            let shifts_x = shifts.iter().any(|(x, _)| x.iter().any(|&w| w != 0));
157            let shifts_z = shifts.iter().any(|(_, z)| z.iter().any(|&w| w != 0));
158            sets_x |= shifts_x;
159            sets_z |= shifts_z;
160            let qubits = mask_qubits(&channel.support());
161            if qubits.len() >= 2 {
162                entangler_x |= shifts_x;
163                entangler_z |= shifts_z;
164                edges.extend(qubits.iter().zip(&qubits[1..]).map(|(&a, &b)| (a, b)));
165            }
166        }
167        let (parity_axis, block_axis, kicked) = match (entangler_x, entangler_z) {
168            (false, true) => (PauliAxis::X, PauliAxis::Z, sets_x),
169            (true, false) => (PauliAxis::Z, PauliAxis::X, sets_z),
170            _ => return Self::Seeded(None),
171        };
172        let mut support = [0u64; W];
173        for (x, z, _) in observable.iter() {
174            for w in 0..W {
175                support[w] |= x[w] | z[w];
176            }
177        }
178        let sites = mask_qubits(&support);
179        if kicked && sites.len() <= FOR_RUN_PARITY_GROUP {
180            Self::parity_around(parity_axis, num_qubits, &edges, &sites, FOR_RUN_PARITY_GROUP)
181        } else {
182            Self::blocks(block_axis, (0..num_qubits).collect())
183        }
184    }

186    /// The rows for `2^bits` partitions of a `num_qubits` register, with `default_seed` standing in for a `None` seed.
187    ///
188    /// # Panics
189    ///
190    /// As the [`PartitionRows`] constructor each variant names, and if an excluded qubit is out of range.
191    pub fn rows<const W: usize>(
192        &self,
193        num_qubits: usize,
194        bits: u8,
195        default_seed: u64,
196    ) -> PartitionRows<W> {
197        let mask = |qubits: &[usize]| qubit_mask(qubits.iter().copied(), num_qubits);
198        match self {
199            Self::Seeded(seed) => {
200                PartitionRows::from_seed(num_qubits, bits, seed.unwrap_or(default_seed))
201            }
202            Self::Cut(blocks) => PartitionRows::cut(num_qubits, blocks),
203            Self::SeededExcluding { seed, exclude_x, exclude_z } => {
204                PartitionRows::from_seed_excluding(
205                    num_qubits,
206                    bits,
207                    seed.unwrap_or(default_seed),
208                    &mask(exclude_x),
209                    &mask(exclude_z),
210                )
211            }
212            Self::Matrix(rows) => PartitionRows::from_matrix(num_qubits, rows),
213            Self::Parity { axis, order, group } => {
214                assert!(*group > 0, "parity rows need at least one qubit per row");
215                assert_distinct(order, num_qubits);
216                let needed = bits as usize * group;
217                assert!(
218                    needed <= order.len(),
219                    "{bits} parity rows of {group} qubits need {needed} qubits, the order holds {}",
220                    order.len()
221                );
222                let masks = order.chunks(*group).take(bits as usize).map(mask);
223                on_axis(num_qubits, *axis, masks.collect())
224            }
225            Self::Blocks { axis, order } => {
226                assert_distinct(order, num_qubits);
227                let partitions = 1usize << bits;
228                let mut blocks = vec![Vec::new(); partitions];
229                for (i, &q) in order.iter().enumerate() {
230                    blocks[i * partitions / order.len()].push(q);
231                }
232                let cut = PartitionRows::<W>::cut(num_qubits, &blocks);
233                on_axis(num_qubits, *axis, cut.rows().1.to_vec())
234            }
235        }
236    }
237}

239/// Rows reading the `axis` bits of each mask.
240fn on_axis<const W: usize>(
241    num_qubits: usize,
242    axis: PauliAxis,
243    masks: Vec<[u64; W]>,
244) -> PartitionRows<W> {
245    let zero = vec![[0u64; W]; masks.len()];
246    match axis {
247        PauliAxis::X => PartitionRows::from_rows(num_qubits, masks, zero),
248        PauliAxis::Y => PartitionRows::from_rows(num_qubits, masks.clone(), masks),
249        PauliAxis::Z => PartitionRows::from_rows(num_qubits, zero, masks),
250    }
251}

253/// Panics if `order` repeats a qubit or names one outside the register.
254fn assert_distinct(order: &[usize], num_qubits: usize) {
255    let mut seen = vec![false; num_qubits];
256    for &q in order {
257        assert!(q < num_qubits, "qubit {q} is outside the {num_qubits}-qubit register");
258        assert!(!seen[q], "the row order repeats qubit {q}");
259        seen[q] = true;
260    }
261}

263/// The qubits a mask sets, ascending.
264fn mask_qubits<const W: usize>(mask: &[u64; W]) -> Vec<usize> {
265    (0..64 * W).filter(|&q| (mask[q / 64] >> (q % 64)) & 1 == 1).collect()
266}
