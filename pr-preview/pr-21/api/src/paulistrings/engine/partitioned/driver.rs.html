26impl<B> PartitionWork<B> {
27    pub(super) fn take<const W: usize>(local: &mut B, layers: usize, tracing: bool) -> Self
28    where
29        B: PartitionStorage<W>,
30    {
31        Self {
32            local: local.detach(),
33            rows: Vec::with_capacity(if tracing { layers } else { 0 }),
34        }
35    }
36}

38/// One partition's share of `sum` at the agreed bucket count; collective, and run on the partition's own pool for first touch.
39pub(crate) fn scatter_local<const W: usize>(
40    sum: &PauliSum<W>,
41    rows: &PartitionRows<W>,
42    rank: u32,
43    collectives: &dyn Collectives,
44) -> PauliSum<W> {
45    let mut local = sum.filter_partition(rows, rank);
46    let want = desired_bits(local.len(), DEFAULT_TARGET_BUCKET_LEN, DEFAULT_MIN_BUCKETS);
47    let want = collectives.allreduce_max_u8(want);
48    local.coarsen_to(scatter_bits(local.hash().bits(), rows.bits(), want));
49    local
50}

52/// The scatter bits rule of ARCHITECTURE.md §Partitioning, with `pbits = log2(P)` and `want` the all-reduced [`desired_bits`].
53fn scatter_bits(bits: u8, pbits: u8, want: u8) -> u8 {
54    want.max(bits.saturating_sub(pbits)).min(bits)
55}

57/// A driver's own laps since they were last drained.
58#[cfg(feature = "phase-timing")]
59pub(super) struct DriverLaps {
60    scatter_ns: u64,
61    /// Atomic because `gather` takes `&self`.
62    gather_ns: std::sync::atomic::AtomicU64,
63    layers: u64,
64}
65
66#[cfg(feature = "phase-timing")]
67impl DriverLaps {
68    pub(super) fn new(scatter_ns: u64) -> Self {
69        Self {
70            scatter_ns,
71            gather_ns: std::sync::atomic::AtomicU64::new(0),
72            layers: 0,
73        }
74    }
75
76    pub(super) fn lap_gather(&self, ns: u64) {
77        self.gather_ns.fetch_add(ns, Ordering::Relaxed);
78    }
79
80    /// Drain scatter time, gather time and layers driven.
81    pub(super) fn take(&mut self) -> (u64, u64, u64) {
82        (
83            std::mem::take(&mut self.scatter_ns),
84            self.gather_ns.swap(0, Ordering::Relaxed),
85            std::mem::take(&mut self.layers),
86        )
87    }
88}

101impl<const W: usize, T: TruncationPolicy<W> + ?Sized> LayerRun<'_, W, T> {
102    /// Partition `rank`'s [`run_layers`].
103    pub(super) fn run<X: Transport, B: PartitionBackend<W, T>>(
104        &self,
105        rank: usize,
106        work: &mut PartitionWork<B>,
107        transport: &X,
108    ) {
109        let context = PartitionContext {
110            rows: self.rows,
111            rank,
112            size: self.size,
113            tracing: self.tracing,
114        };
115        run_layers(
116            self.circuit,
117            self.policy,
118            self.direction,
119            self.options,
120            context,
121            work,
122            transport,
123        );
124    }
125}

127/// The partitions a driver holds and its records, lent to [`propagate`](Self::propagate) for one call.
128pub(super) struct DriverFrame<'a, const W: usize, B> {
129    /// In rank order: every partition in process, or this rank's one.
130    pub(super) parts: &'a mut [B],
131    pub(super) rows: &'a PartitionRows<W>,
132    pub(super) runtime: &'a PartitionRuntime,
133    pub(super) trace: &'a mut Option<PartitionTrace>,
134    #[cfg(feature = "phase-timing")]
135    pub(super) laps: &'a mut DriverLaps,
136    /// This process's rank in a distributed group, `None` when it holds every partition.
137    pub(super) rank: Option<usize>,
138    /// Partitions in the group.
139    pub(super) size: usize,
140}

142impl<const W: usize, B: PartitionStorage<W>> DriverFrame<'_, W, B> {
143    /// The prologue and epilogue every partitioned driver shares around `run`, which drives the partitions' layers and returns their work in rank order; `agree` runs after the entry line, before any layer.
144    pub(super) fn propagate<T>(
145        self,
146        circuit: &Circuit<W>,
147        policy: &T,
148        direction: Direction,
149        options: PropagateOptions,
150        agree: impl FnOnce(),
151        run: impl FnOnce(Vec<PartitionWork<B>>, &LayerRun<'_, W, T>) -> Vec<PartitionWork<B>>,
152    ) where
153        T: TruncationPolicy<W> + ?Sized,
154        B: PartitionBackend<W, T>,
155    {
156        options.assert_valid();
157        let num_layers = circuit.channels.len();
158        let size = self.size;
159        assert_supports_partitions(policy, size);
160        let terms_in: usize = self.parts.iter().map(PartitionStorage::len).sum();
161        let started = Instant::now();
162        log::info!(
163            target: LOG_TARGET,
164            "propagate: {terms_in} terms through {num_layers} channels ({direction:?}), {}",
165            progress_scope(self.rank, size, Some(&self.runtime.placement_summary())),
166        );
167
168        agree();
169
170        if num_layers > 0 {
171            let tracing = self.trace.is_some();
172            let items: Vec<PartitionWork<B>> = self
173                .parts
174                .iter_mut()
175                .map(|part| PartitionWork::take(part, num_layers, tracing))
176                .collect();
177            let layers = LayerRun {
178                circuit,
179                policy,
180                direction,
181                options,
182                rows: self.rows,
183                size,
184                tracing,
185            };
186            let done = run(items, &layers);
187            assert_eq!(done.len(), self.parts.len(), "one work item per partition");
188            let mut traced = Vec::with_capacity(if tracing { done.len() } else { 0 });
189            for (part, work) in self.parts.iter_mut().zip(done) {
190                *part = work.local;
191                if tracing {
192                    traced.push(work.rows);
193                }
194            }
195            if let Some(trace) = self.trace.as_mut() {
196                assemble(trace, traced, size);
197            }
198            tally!(self.laps.layers, num_layers);
199        }
200
201        log::info!(
202            target: LOG_TARGET,
203            "propagate: {num_layers} layers applied, {terms_in} -> {} terms, {:.3} s, {}",
204            self.parts.iter().map(PartitionStorage::len).sum::<usize>(),
205            started.elapsed().as_secs_f64(),
206            progress_scope(self.rank, size, Some(&self.runtime.placement_summary())),
207        );
208    }
209}

211/// Propagates `sum` through `circuit` on a partitioned engine built from `config`, and gathers the result.
212///
213/// A caller propagating repeatedly should hold a [`PartitionRuntime`](super::runtime::PartitionRuntime) and a [`PartitionedSum`] instead, so pools, split and scratch survive between calls.
214///
215/// # Errors
216///
217/// [`TopologyError`] if `config` cannot be resolved into slots or a pool cannot be built.
218/// Everything else is a panic, as in [`propagate`](crate::propagate) and [`PartitionedSum::propagate_with`].
219pub fn propagate_partitioned<const W: usize, T>(
220    circuit: &Circuit<W>,
221    sum: PauliSum<W>,
222    policy: &T,
223    direction: Direction,
224    config: &PartitionConfig,
225    options: PropagateOptions,
226) -> Result<PauliSum<W>, TopologyError>
227where
228    T: TruncationPolicy<W> + ?Sized,
229{
230    let mut split = PartitionedSum::scatter(sum, config)?;
231    split.propagate_with(circuit, policy, direction, options);
232    Ok(split.into_gathered())
233}
