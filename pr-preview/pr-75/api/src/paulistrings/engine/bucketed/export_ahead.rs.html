1//! Per-bucket counts of the rows the next layer exports, taken by the coset loop as it writes each bucket, so an exchanging layer's export needs no count pass of its own.
2//!
3//! Contract: the counts answer for the sum the coset loop left, under whatever cut the layer pass after it leaves pending, and anything that changes the sum or its buckets in between drops them ([`LayerOctaves::drop_ahead`]).

15/// One bucket's counts ahead: the window counts of its terms that emit a row, and one mark bit per term that does, so the fill visits only those.
16#[derive(Clone, Debug)]
17pub(crate) struct BucketAhead {
18    emitting: BucketOctaves,
19    /// Bit `row % 64` of word `row / 64`.
20    marks: Vec<u64>,
21}
22
23impl Default for BucketAhead {
24    fn default() -> Self {
25        Self { emitting: [0; WINDOW_OCTAVES + 2], marks: Vec::new() }
26    }
27}
28
29impl BucketAhead {
30    /// The rows marked as emitting, ascending.
31    pub(crate) fn marked_rows(&self) -> impl Iterator<Item = usize> + '_ {
32        self.marks.iter().enumerate().flat_map(|(word_index, &word)| {
33            let mut word = word;
34            std::iter::from_fn(move || {
35                (word != 0).then(|| {
36                    let bit = word.trailing_zeros() as usize;
37                    word &= word - 1;
38                    word_index * 64 + bit
39                })
40            })
41        })
42    }
43
44    /// Panics unless the marks fit a bucket of `len` terms and mark as many rows as the counts ahead count.
45    #[cfg(debug_assertions)]
46    pub(crate) fn debug_assert_marks_match(&self, len: usize, bucket: usize) {
47        debug_assert_eq!(
48            self.marks.len(),
49            len.div_ceil(64),
50            "bucket {bucket}: {} mark words for a bucket of {len} terms",
51            self.marks.len()
52        );
53        let marked: u32 = self.marks.iter().map(|word| word.count_ones()).sum();
54        let counted: u32 = self.emitting.iter().sum();
55        debug_assert_eq!(
56            marked, counted,
57            "bucket {bucket}: {marked} rows marked where the counts ahead count {counted}"
58        );
59    }
60}

62/// One remote delta's row-level emitter: a tabulated entry, `dense` when nonzero on every support pattern, or the wide rotation's generator pass.
63pub(crate) enum RowEmitter<'p, const W: usize, C: Coefficient = num_complex::Complex64> {
64    Tabulated {
65        ptm: &'p LocalPtm<W, C>,
66        entry: &'p DeltaEntry<W, C>,
67        dense: bool,
68    },
69    /// The identity pass of a rotation is never remote.
70    Generator {
71        rotation: &'p PreparedRotation<W, C::Real>,
72    },
73}
74
75impl<'p, const W: usize, C: Coefficient> RowEmitter<'p, W, C> {
76    /// The emitter of `remote`, a delta of `prepared`.
77    pub(crate) fn of(prepared: &'p Prepared<W, C>, remote: &RemoteDelta) -> Self {
78        match prepared {
79            Prepared::Local(ptm) => RowEmitter::Tabulated {
80                ptm,
81                entry: &ptm.deltas()[remote.entry],
82                dense: ptm.entry_is_dense(remote.entry),
83            },
84            Prepared::Rotation(rotation) => {
85                debug_assert_eq!(
86                    remote.entry, 1,
87                    "a rotation's only remote entry is the generator pass"
88                );
89                RowEmitter::Generator { rotation }
90            }
91        }
92    }
93
94    pub(crate) fn is_dense(&self) -> bool {
95        matches!(self, RowEmitter::Tabulated { dense: true, .. })
96    }
97
98    pub(crate) fn emit(
99        &self,
100        x: &[u64; W],
101        z: &[u64; W],
102        coeff: C,
103    ) -> Option<([u64; W], [u64; W], C)> {
104        match self {
105            RowEmitter::Tabulated { ptm, entry, .. } => {
106                entry.emit(ptm.support_bits(x, z), x, z, coeff)
107            }
108            RowEmitter::Generator { rotation } => rotation.emit_generator(x, z, coeff),
109        }
110    }
111}

113impl OctaveWindow {
114    /// A coefficient's slot as [`count`](Self::count) bins it, and as the counts ahead bin it, where a NaN, which every cut drops, sits below the window.
115    #[inline(always)]
116    fn slots_of<C: Coefficient>(&self, coeff: C) -> (usize, usize) {
117        let norm_sqr = coeff.norm_sqr();
118        let slot = self.slot(approx_bin(norm_sqr));
119        (slot, slot * usize::from(!norm_sqr.is_nan()))
120    }
121
122    /// Count and mark the terms of `columns` that `emitter` emits for into `ahead`, and count all of them into `counts` if given, both binned under this window and overwritten.
123    pub(crate) fn count_emitting<const W: usize, C: Coefficient>(
124        &self,
125        columns: &BucketColumns<W, C>,
126        emitter: &RowEmitter<'_, W, C>,
127        counts: Option<&mut BucketOctaves>,
128        ahead: &mut BucketAhead,
129    ) {
130        match *emitter {
131            RowEmitter::Tabulated { dense: true, .. } => {
132                self.count_rows(columns, |_, _| true, counts, ahead)
133            }
134            RowEmitter::Tabulated { ptm, entry, .. } => {
135                let amplitude = &entry.amplitude;
136                let emits =
137                    |x: &[u64; W], z: &[u64; W]| amplitude[ptm.support_bits(x, z)] != C::ZERO;
138                self.count_rows(columns, emits, counts, ahead);
139            }
140            RowEmitter::Generator { rotation } => {
141                let generator = &rotation.generator;
142                let emits = |x: &[u64; W], z: &[u64; W]| {
143                    !PauliString::<W> { x: *x, z: *z }.commutes_with(generator)
144                };
145                self.count_rows(columns, emits, counts, ahead);
146            }
147        }
148    }
149
150    /// [`count_emitting`](Self::count_emitting) under one emit test, monomorphized so the loop carries no dispatch.
151    #[inline(always)]
152    fn count_rows<const W: usize, C: Coefficient>(
153        &self,
154        columns: &BucketColumns<W, C>,
155        emits: impl Fn(&[u64; W], &[u64; W]) -> bool,
156        counts: Option<&mut BucketOctaves>,
157        ahead: &mut BucketAhead,
158    ) {
159        let BucketAhead { emitting, marks } = ahead;
160        *emitting = [0; WINDOW_OCTAVES + 2];
161        marks.clear();
162        marks.resize(columns.len().div_ceil(64), 0);
163        let rows = columns.x.iter().zip(&columns.z).zip(&columns.coeff).enumerate();
164        match counts {
165            Some(counts) => {
166                *counts = [0; WINDOW_OCTAVES + 2];
167                for (row, ((x, z), &coeff)) in rows {
168                    let (slot, emitting_slot) = self.slots_of(coeff);
169                    let emits = emits(x, z);
170                    counts[slot] += 1;
171                    emitting[emitting_slot] += u32::from(emits);
172                    marks[row / 64] |= u64::from(emits) << (row % 64);
173                }
174            }
175            None => {
176                for (row, ((x, z), &coeff)) in rows {
177                    let (_, emitting_slot) = self.slots_of(coeff);
178                    let emits = emits(x, z);
179                    emitting[emitting_slot] += u32::from(emits);
180                    marks[row / 64] |= u64::from(emits) << (row % 64);
181                }
182            }
183        }
184    }
185}

187impl LayerOctaves {
188    /// Forget the counts ahead.
189    pub(crate) fn drop_ahead(&mut self) {
190        self.ahead_window = None;
191    }
192
193    /// Whether the last coset loop counted the rows this layer exports.
194    pub(crate) fn has_ahead(&self) -> bool {
195        self.ahead_window.is_some()
196    }
197
198    /// Bucket `b`'s exported rows among the terms the pending cut keeps, from the counts ahead; `None` when they cannot tell, at the one bucket a cut's boundary quota splits.
199    pub(crate) fn rows_ahead(&self, b: usize) -> Option<u32> {
200        let window = self.ahead_window?;
201        let emitting = &self.ahead[b].emitting;
202        let Some(cut) = &self.pending else {
203            return Some(emitting.iter().sum());
204        };
205        // A pending cut was decided from the counts the same coset loop took, under the same window, so its bin is inside it.
206        let slot = window.slot(cut.bin);
207        debug_assert!((1..=WINDOW_OCTAVES).contains(&slot), "the cut's bin is outside the window");
208        let above: u32 = emitting[slot + 1..].iter().sum();
209        match cut.split.quota(b) {
210            0 => Some(above),
211            quota if quota >= self.per_bucket[b][slot] as usize => Some(above + emitting[slot]),
212            _ => None,
213        }
214    }
215
216    /// Bucket `b`'s counts ahead where [`rows_ahead`](Self::rows_ahead) answers: there the pending cut keeps all of `b`'s boundary terms or none, so its filter keeps the same terms asked of the marked rows alone.
217    pub(crate) fn bucket_ahead(&self, b: usize) -> Option<&BucketAhead> {
218        self.rows_ahead(b).map(|_| &self.ahead[b])
219    }
220}
