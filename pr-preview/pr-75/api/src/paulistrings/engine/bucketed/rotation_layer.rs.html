1//! The rotation layer: a two-delta table `{0, g}` applied to each bucket's own columns by a scan, a linear partner order and a merge-join.
2//!
3//! Contracts: the result agrees with the general coset fill to floating-point tolerance, every output passes `keep_term` exactly as there, and on the Clifford arm the identity residues `|a₀| ≤ ε·|a₁|` are dropped where the general path keeps them.

20/// A bucket's `x` and `z` columns, each ORed over its strings, or a superset of them.
21pub(super) type Support<const W: usize> = [[u64; W]; 2];
22
23/// Per bucket, a [`Support`], carried from one rotation layer to the next.
24///
25/// Contract: known only while the sum is exactly as the last layer of the current call left it, give or take removed strings, and that layer was a local rotation layer or run under the call's policy; then every string has passed that policy's `keep_term`, which is a pure function of the term, so a string no layer changes need not pass it again.
26/// Anything else that writes the sum or renumbers its buckets clears it ([`forget`](Self::forget)), and so do the start and end of a call.
27#[derive(Debug)]
28pub(crate) struct BucketSupports<const W: usize> {
29    masks: Vec<Support<W>>,
30    known: bool,
31}
32
33impl<const W: usize> Default for BucketSupports<W> {
34    fn default() -> Self {
35        Self { masks: Vec::new(), known: false }
36    }
37}

39impl<const W: usize> BucketSupports<W> {
40    /// Drop the supports, so the next rotation layer scans and merges every bucket.
41    pub(crate) fn forget(&mut self) {
42        self.known = false;
43    }
44}

46/// A coset loop over `span` whose `fill` also gets its members' supports, and whether they are known; every support is known afterwards.
47pub(super) fn rotation_coset_loop<const W: usize, C: Coefficient>(
48    sum: &mut PauliSum<W, C>,
49    span: &Gf2Span,
50    scratch: &mut LayerScratch<W, C>,
51    octaves: LayerOctaveIo<'_, '_, '_, W, C>,
52    fill: impl Fn(&mut [BucketColumns<W, C>], &mut CosetScratch<W, C>, &mut [Support<W>], bool) + Sync,
53) -> usize {
54    let buckets = sum.num_buckets();
55    let supports = &mut scratch.supports;
56    let known = std::mem::take(&mut supports.known) && supports.masks.len() == buckets;
57    let mut masks = std::mem::take(&mut supports.masks);
58    masks.resize(buckets, [[0; W]; 2]);
59    let slots = DisjointSlots::new(&mut masks);
60    let len = coset_loop(sum, span, scratch, octaves, |members, betas, worker_scratch| {
61        let mut supports = std::mem::take(&mut worker_scratch.member_supports);
62        supports.clear();
63        // SAFETY (both loops): the cosets partition the buckets, so no other task reaches bucket `beta`.
64        supports.extend(betas.iter().map(|&beta| *unsafe { slots.get(beta as usize) }));
65        fill(members, worker_scratch, &mut supports, known);
66        for (support, &beta) in supports.iter().zip(betas) {
67            *unsafe { slots.get(beta as usize) } = *support;
68        }
69        worker_scratch.member_supports = supports;
70    });
71    scratch.supports = BucketSupports { masks, known: true };
72    len
73}

75/// A table shaped like a Pauli rotation about `g`: an identity entry and one generator entry.
76///
77/// On a pattern the generator entry skips, the identity amplitude is exactly one; the patterns it reaches are closed under `⊕ g`, so a partner never lands on a string that keeps its coefficient.
78#[derive(Clone, Debug)]
79pub(super) struct RotationLayer<'p, const W: usize, C: Coefficient = num_complex::Complex64> {
80    ptm: &'p LocalPtm<W, C>,
81    /// `(mask_x, mask_z)` of `g`.
82    mask: ([u64; W], [u64; W]),
83    /// `a₀[s]`, the identity entry's amplitude.
84    identity: [C; LOCAL_DIM],
85    /// `a₁[s]`, the amplitude taking pattern `s` to `s ⊕ g`.
86    partner: [C; LOCAL_DIM],
87    /// Whether every anticommuting pattern has `|a₀| ≤ ε·|a₁|`, so strings move instead of splitting.
88    clifford: bool,
89    order: PartnerOrder<W>,
90    /// [`LocalPtm::support_probe`].
91    probe: [(usize, u32); 2],
92    /// Bit `p` set iff the generator entry reaches probe pattern `p`, whose support pattern is `p & support_mask`.
93    active: u16,
94    support_mask: usize,
95    /// `h(g)`.
96    bucket_delta: u32,
97}
98
99impl<'p, const W: usize, C: Coefficient> RotationLayer<'p, W, C> {
100    /// The rotation shape of `ptm`, or `None` if it has another.
101    pub(super) fn of(ptm: &'p LocalPtm<W, C>) -> Option<Self> {
102        let [identity, generator] = ptm.deltas() else { return None };
103        if identity.local_delta != 0 {
104            return None;
105        }
106        let dim = 1usize << (2 * ptm.k());
107        let delta = generator.local_delta as usize;
108        let mut clifford = true;
109        for s in 0..dim {
110            let (a0, a1) = (identity.amplitude[s], generator.amplitude[s]);
111            if a1 == C::ZERO {
112                if a0 != C::ONE {
113                    return None;
114                }
115            } else {
116                if generator.amplitude[s ^ delta] == C::ZERO {
117                    return None;
118                }
119                // `cos θ` evaluated at a quarter turn is a rounding residue of `sin θ`, not an amplitude.
120                clifford &= a0.norm_sqr() <= f64::EPSILON * f64::EPSILON * a1.norm_sqr();
121            }
122        }
123        let mask = generator.mask();
124        let support_mask = dim - 1;
125        let active = (0..16)
126            .filter(|&p| generator.amplitude[p & support_mask] != C::ZERO)
127            .fold(0u16, |bits, p| bits | 1 << p);
128        Some(Self {
129            ptm,
130            mask,
131            identity: identity.amplitude,
132            partner: generator.amplitude,
133            clifford,
134            order: PartnerOrder::new(&mask.0, &mask.1),
135            probe: ptm.support_probe(),
136            active,
137            support_mask,
138            bucket_delta: generator.bucket_delta,
139        })
140    }
141
142    /// The probe pattern of `(x, z)`, packed like [`LocalPtm::support_bits`] on two probe slots.
143    #[inline(always)]
144    fn pattern(&self, x: &[u64; W], z: &[u64; W]) -> usize {
145        let [(w0, b0), (w1, b1)] = self.probe;
146        let bit = |word: u64, b: u32| ((word >> b) & 1) as usize;
147        bit(x[w0], b0) | bit(z[w0], b0) << 1 | bit(x[w1], b1) << 2 | bit(z[w1], b1) << 3
148    }
149
150    /// Whether strings move instead of splitting, so each input string has at most one output.
151    pub(super) fn is_clifford(&self) -> bool {
152        self.clifford
153    }
154
155    /// Append each anticommuting string's partner `(k ⊕ g, c·a₁)` to `partners`, then scale those strings by `a₀` in place, or on the Clifford arm zero them for the merge to drop; `hits` is scratch.
156    ///
157    /// A first pass reads only the key columns and lists the anticommuting rows, so the coefficients are read and written for the listed rows alone, without a data-dependent branch.
158    /// Unless `known`, `support` becomes the OR of every row; a known support is a superset already, so the pass reads only the column words the anticommute test needs.
159    fn scan(
160        &self,
161        bucket: &mut BucketColumns<W, C>,
162        partners: &mut BucketColumns<W, C>,
163        hits: &mut Vec<u32>,
164        support: &mut Support<W>,
165        known: bool,
166    ) {
167        let BucketColumns { x, z, coeff } = bucket;
168        let (mask_x, mask_z) = &self.mask;
169        let [(w0, b0), (w1, b1)] = self.probe;
170        let bit = |word: u64, b: u32| ((word >> b) & 1) as usize;
171        // An `x` bit anticommutes only under `g`'s `z`, and a `z` bit only under its `x`.
172        let reads_x = mask_z.iter().any(|&w| w != 0);
173        let reads_z = mask_x.iter().any(|&w| w != 0);
174        if !known {
175            *support = [[0; W]; 2];
176        }
177        match (known, reads_x, reads_z) {
178            (true, true, false) => {
179                self.list_hits(x.len(), hits, |row| bit(x[row][w0], b0) | bit(x[row][w1], b1) << 2)
180            }
181            (true, false, true) => self.list_hits(x.len(), hits, |row| {
182                bit(z[row][w0], b0) << 1 | bit(z[row][w1], b1) << 3
183            }),
184            (true, _, _) => self.list_hits(x.len(), hits, |row| self.pattern(&x[row], &z[row])),
185            (false, _, _) => self.list_hits(x.len(), hits, |row| {
186                for w in 0..W {
187                    support[0][w] |= x[row][w];
188                    support[1][w] |= z[row][w];
189                }
190                self.pattern(&x[row], &z[row])
191            }),
192        }
193        let anticommuting = hits.len();
194        if anticommuting == 0 {
195            return;
196        }
197        let start = partners.len();
198        partners.x.reserve(anticommuting);
199        partners.z.reserve(anticommuting);
200        partners.coeff.reserve(anticommuting);
201        // SAFETY: `start` is each column's length, within its allocation.
202        let (out_x, out_z, out_c) = unsafe {
203            (
204                partners.x.as_mut_ptr().add(start),
205                partners.z.as_mut_ptr().add(start),
206                partners.coeff.as_mut_ptr().add(start),
207            )
208        };
209        for (slot, hit) in hits.iter_mut().enumerate() {
210            let row = (*hit >> 4) as usize;
211            let pattern = self.pattern(&x[row], &z[row]);
212            *hit |= pattern as u32;
213            let support = pattern & self.support_mask;
214            let c = coeff[row];
215            // SAFETY: `slot < anticommuting`, reserved past `start` above.
216            unsafe {
217                out_x.add(slot).write(core::array::from_fn(|w| x[row][w] ^ mask_x[w]));
218                out_z.add(slot).write(core::array::from_fn(|w| z[row][w] ^ mask_z[w]));
219                out_c.add(slot).write(c * self.partner[support]);
220            }
221            coeff[row] = if self.clifford { C::ZERO } else { c * self.identity[support] };
222        }
223        // SAFETY: written above.
224        unsafe {
225            partners.x.set_len(start + anticommuting);
226            partners.z.set_len(start + anticommuting);
227            partners.coeff.set_len(start + anticommuting);
228        }
229    }
230
231    /// List into `hits`, as `row << 4`, the rows below `len` whose probe `pattern` the generator reaches; a pattern may leave out the bits that cannot change that.
232    #[inline(always)]
233    fn list_hits(&self, len: usize, hits: &mut Vec<u32>, mut pattern: impl FnMut(usize) -> usize) {
234        assert!(len < 1 << 28, "a hit packs its row into 28 bits");
235        hits.clear();
236        // One slot past the exact bound, for the discarded write.
237        hits.reserve(len + 1);
238        let listed = hits.as_mut_ptr();
239        let mut anticommuting = 0usize;
240        for row in 0..len {
241            let pattern = pattern(row);
242            // SAFETY: `anticommuting ≤ row < len`, within the reserved `len + 1`.
243            unsafe { listed.add(anticommuting).write((row as u32) << 4) };
244            anticommuting += usize::from((self.active >> pattern) & 1 != 0);
245        }
246        // SAFETY: the first `anticommuting` slots were written above.
247        unsafe { hits.set_len(anticommuting) };
248    }
249
250    /// Whether a bucket within `support` may hold a string that anticommutes with `g`, which needs an `x` bit under `g`'s `z` or a `z` bit under its `x`.
251    fn may_touch(&self, support: &Support<W>) -> bool {
252        let (mask_x, mask_z) = &self.mask;
253        (0..W).any(|w| support[0][w] & mask_z[w] | support[1][w] & mask_x[w] != 0)
254    }
255}

257/// A rotation whose generator crosses to a partner, as a receiving bucket applies it: its own strings take the identity half, and the partners arrive from the partner's export.
258// Built once per layer on the stack, so the table variant's size costs nothing.
259#[allow(clippy::large_enum_variant)]
260pub(super) enum ReceivedRotation<'p, const W: usize, C: Coefficient> {
261    Table(RotationLayer<'p, W, C>),
262    /// `order` is `None` when the generator has too many set bits for [`PartnerOrder`], and the partners are sorted instead.
263    Wide {
264        rotation: &'p PreparedRotation<W, C::Real>,
265        order: Option<PartnerOrder<W>>,
266        clifford: bool,
267    },
268}
269
270impl<'p, const W: usize, C: Coefficient> ReceivedRotation<'p, W, C> {
271    pub(super) fn wide(rotation: &'p PreparedRotation<W, C::Real>) -> Self {
272        let (mask_x, mask_z) = rotation.generator_mask();
273        Self::Wide {
274            rotation,
275            order: PartnerOrder::try_new(&mask_x, &mask_z),
276            // As for a table: `cos θ` at a quarter turn is a rounding residue.
277            clifford: (C::ONE * rotation.cos).norm_sqr()
278                <= f64::EPSILON * f64::EPSILON * (C::ONE * rotation.sin).norm_sqr(),
279        }
280    }
281
282    /// The identity half of [`RotationLayer::scan`]: scale each anticommuting string by `a₀`, or on the Clifford arm drop it.
283    fn scale(&self, bucket: &mut BucketColumns<W, C>) {
284        match self {
285            Self::Table(layer) => scale_identity(bucket, layer.clifford, |x, z| {
286                let pattern = layer.ptm.support_bits(x, z);
287                (layer.partner[pattern].is_nonzero(), layer.identity[pattern])
288            }),
289            Self::Wide { rotation, clifford, .. } => {
290                let cos = C::ONE * rotation.cos;
291                scale_identity(bucket, *clifford, |x, z| {
292                    let commutes =
293                        PauliString::<W> { x: *x, z: *z }.commutes_with(&rotation.generator);
294                    (!commutes, if commutes { C::ONE } else { cos })
295                })
296            }
297        }
298    }
299
300    fn partner_order(&self) -> Option<&PartnerOrder<W>> {
301        match self {
302            Self::Table(layer) => Some(&layer.order),
303            Self::Wide { order, .. } => order.as_ref(),
304        }
305    }
306}

308/// Scale `bucket` in place by `identity(x, z) = (anticommutes, a₀)`, or on the Clifford arm drop the anticommuting strings.
309fn scale_identity<const W: usize, C: Coefficient>(
310    bucket: &mut BucketColumns<W, C>,
311    clifford: bool,
312    identity: impl Fn(&[u64; W], &[u64; W]) -> (bool, C),
313) {
314    let BucketColumns { x, z, coeff } = bucket;
315    if clifford {
316        let mut kept = 0usize;
317        for row in 0..coeff.len() {
318            let (key_x, key_z, c) = (x[row], z[row], coeff[row]);
319            x[kept] = key_x;
320            z[kept] = key_z;
321            coeff[kept] = c;
322            kept += usize::from(!identity(&key_x, &key_z).0);
323        }
324        x.truncate(kept);
325        z.truncate(kept);
326        coeff.truncate(kept);
327    } else {
328        for row in 0..coeff.len() {
329            coeff[row] *= identity(&x[row], &z[row]).1;
330        }
331    }
332}

334/// Apply `layer` to bucket `β`: scale it in place and merge in the partners `extra` received for it, which come from the single source bucket `β ⊕ h(g)` in its key order, as [`PartnerOrder`] needs.
335pub(super) fn fill_received_rotation_bucket<const W: usize, C: Coefficient, T, X>(
336    members: &mut [BucketColumns<W, C>],
337    betas: &[u32],
338    layer: &ReceivedRotation<'_, W, C>,
339    policy: &T,
340    scratch: &mut CosetScratch<W, C>,
341    extra: &X,
342) where
343    T: TruncationPolicy<W, C> + ?Sized,
344    X: ExtraRows<W, C>,
345{
346    debug_assert_eq!(members.len(), 1, "a remote generator leaves only the identity delta local");
347    let bucket = &mut members[0];
348    if scratch.runs.is_empty() {
349        scratch.runs.push(GatherRun::default());
350    }
351    if scratch.old.is_empty() {
352        scratch.old.push(BucketColumns::default());
353    }
354    stamp!(stamp);
355    tally!(scratch.stats.rows_gathered, bucket.len());
356    tally!(scratch.stats.rows_id, bucket.len());
357    layer.scale(bucket);
358    let CosetScratch { runs, old, sort, .. } = scratch;
359    let run = &mut runs[0];
360    run.rest.clear();
361    extra.append_into(betas[0], &mut run.rest);
362    lap!(stamp, scratch.stats.gather_ns);
363    tally!(scratch.stats.rows_gathered, run.rest.len());
364    tally!(scratch.stats.rows_sorted, run.rest.len());
365    let partners = match layer.partner_order() {
366        Some(order) => {
367            order.order(&run.rest, &mut run.identity, sort.indices());
368            run.identity.view()
369        }
370        None => {
371            SortKernel::Comparison.sort(&mut run.rest, sort);
372            run.rest.view()
373        }
374    };
375    lap!(stamp, scratch.stats.sort_ns);
376    merge_partners(bucket, partners, policy, &mut old[0], None, false);
377    lap!(stamp, scratch.stats.merge_ns);
378    tally!(scratch.stats.cosets, 1);
379    tally!(scratch.stats.runs, 1);
380}

382/// Apply `layer` to one coset, one bucket or the pair `{β, β ⊕ h(g)}`, each partner run merging into the other member.
383///
384/// `supports[j]` is a superset of member `j`'s support coming in when `known`, and is one coming out either way.
385/// A known support certifies, by [`BucketSupports`]'s contract, that every string already passes `keep_term`: a member it shows commuting with `g` is not read, and a member neither changed nor receiving partners is not merged.
386pub(super) fn fill_rotation_coset<const W: usize, C: Coefficient, T>(
387    members: &mut [BucketColumns<W, C>],
388    layer: &RotationLayer<'_, W, C>,
389    policy: &T,
390    scratch: &mut CosetScratch<W, C>,
391    supports: &mut [Support<W>],
392    known: bool,
393) where
394    T: TruncationPolicy<W, C> + ?Sized,
395{
396    let size = members.len();
397    let floor = policy.coefficient_floor();
398    debug_assert!(size == 1 || size == 2);
399    debug_assert_eq!(supports.len(), size);
400    // The general path's buffers, idle between its cosets: each member's run holds its partners in `rest` and in key order in `identity`, and `old[0]` is the merge fallback.
401    scratch.runs.resize_with(size.max(scratch.runs.len()), GatherRun::default);
402    if scratch.old.is_empty() {
403        scratch.old.push(BucketColumns::default());
404    }
405    stamp!(stamp);
406    let mut changed = [false; 2];
407    // Every scan precedes every merge, so each partner carries its source's unscaled coefficient.
408    for (j, bucket) in members.iter_mut().enumerate() {
409        let partners = &mut scratch.runs[j ^ (size - 1)].rest;
410        partners.clear();
411        scratch.hits[j].clear();
412        if known && !layer.may_touch(&supports[j]) {
413            #[cfg(test)]
414            {
415                scratch.rotation_skips.0 += 1;
416            }
417            continue;
418        }
419        // Counted as the general path's identity rows, whose keys stay put.
420        tally!(scratch.stats.rows_gathered, bucket.len());
421        tally!(scratch.stats.rows_id, bucket.len());
422        layer.scan(bucket, partners, &mut scratch.hits[j], &mut supports[j], known);
423        changed[j] = !scratch.hits[j].is_empty();
424    }
425    lap!(stamp, scratch.stats.gather_ns);
426    let sources = [supports[0], supports[size - 1]];
427    for (j, bucket) in members.iter_mut().enumerate() {
428        let run = &mut scratch.runs[j];
429        if known && !changed[j] && run.rest.len() == 0 {
430            #[cfg(test)]
431            {
432                scratch.rotation_skips.1 += 1;
433            }
434            continue;
435        }
436        if run.rest.len() != 0 {
437            // A partner `k ⊕ g` lies within its source's support and `g`'s.
438            let source = &sources[size - 1 - j];
439            let (mask_x, mask_z) = &layer.mask;
440            supports[j] = [
441                core::array::from_fn(|w| supports[j][0][w] | source[0][w] | mask_x[w]),
442                core::array::from_fn(|w| supports[j][1][w] | source[1][w] | mask_z[w]),
443            ];
444        }
445        tally!(scratch.stats.rows_gathered, run.rest.len());
446        tally!(scratch.stats.rows_sorted, run.rest.len());
447        layer.order.order(&run.rest, &mut run.identity, scratch.sort.indices());
448        lap!(stamp, scratch.stats.sort_ns);
449        fold_hits(bucket, &scratch.hits[j], &mut run.identity, floor);
450        let dirty = known.then_some(scratch.hits[j].as_slice());
451        #[cfg_attr(not(test), allow(unused_variables))]
452        let in_place =
453            merge_partners(bucket, run.identity.view(), policy, &mut scratch.old[0], dirty, true);
454        #[cfg(test)]
455        {
456            scratch.rotation_merges.0 += usize::from(in_place);
457            scratch.rotation_merges.1 += usize::from(!in_place);
458        }
459        lap!(stamp, scratch.stats.merge_ns);
460    }
461    tally!(scratch.stats.cosets, 1);
462    tally!(scratch.stats.runs, size);
463}

465/// Apply `layers` in order to one coset of their joint span, each to the member pairs its generator's coordinate in `coordinates` joins, while the coset is in cache; `supports` and `known` as for [`fill_rotation_coset`].
466#[allow(clippy::too_many_arguments)]
467fn fill_rotation_run<const W: usize, C: Coefficient, T>(
468    members: &mut [BucketColumns<W, C>],
469    layers: &[RotationLayer<'_, W, C>],
470    coordinates: &[u32],
471    policy: &T,
472    scratch: &mut CosetScratch<W, C>,
473    supports: &mut [Support<W>],
474    mut known: bool,
475) where
476    T: TruncationPolicy<W, C> + ?Sized,
477{
478    let mut pair: [BucketColumns<W, C>; 2] = Default::default();
479    for (layer, &coordinate) in layers.iter().zip(coordinates) {
480        for j in 0..members.len() {
481            let partner = j ^ coordinate as usize;
482            if partner == j {
483                let (bucket, support) = (&mut members[j], &mut supports[j..j + 1]);
484                fill_rotation_coset(
485                    core::slice::from_mut(bucket),
486                    layer,
487                    policy,
488                    scratch,
489                    support,
490                    known,
491                );
492            } else if j < partner {
493                std::mem::swap(&mut pair[0], &mut members[j]);
494                std::mem::swap(&mut pair[1], &mut members[partner]);
495                let mut pair_supports = [supports[j], supports[partner]];
496                fill_rotation_coset(&mut pair, layer, policy, scratch, &mut pair_supports, known);
497                [supports[j], supports[partner]] = pair_supports;
498                std::mem::swap(&mut pair[0], &mut members[j]);
499                std::mem::swap(&mut pair[1], &mut members[partner]);
500            }
501        }
502        // After the first layer every support is exact, and every string it scanned passed `keep_term`.
503        known = true;
504    }
505}

507/// Most layers a rotation run joins.
508pub(crate) const MAX_ROTATION_RUN_LAYERS: usize = 8;

510/// Largest span dimension of a host rotation run's bucket deltas, so a coset of `2^r` buckets at the bucket target stays within a core's L2 (research/FINDINGS.md §Consecutive rotation tables run as one coset loop).
511pub(crate) const MAX_ROTATION_RUN_DIMENSION: usize = 3;
512
513/// The bucket deltas of a rotation run being joined, at one bucket count.
514pub(crate) struct RotationRunSpan {
515    deltas: Vec<u32>,
516    bits: u8,
517    max_dimension: usize,
518}
519
520impl RotationRunSpan {
521    /// An empty run at `bits` bucket bits whose span may reach `max_dimension`.
522    pub(crate) fn new(bits: u8, max_dimension: usize) -> Self {
523        Self { deltas: Vec::new(), bits, max_dimension }
524    }
525
526    /// Join `layer` if it is a rotation table that keeps the run's span within its maximum dimension; returns whether it joined.
527    pub(crate) fn join<const W: usize, C: Coefficient>(&mut self, layer: &Prepared<W, C>) -> bool {
528        let Prepared::Local(ptm) = layer else { return false };
529        if RotationLayer::of(ptm).is_none() {
530            return false;
531        }
532        let joined = self.deltas.len();
533        self.deltas.extend(ptm.bucket_deltas());
534        if Gf2Span::new(&self.deltas, self.bits).dimension() > self.max_dimension {
535            self.deltas.truncate(joined);
536            return false;
537        }
538        true
539    }
540}
541
542/// Apply `run`, consecutive rotation tables a [`RotationRunSpan`] joined under the current hash, as one coset loop over their joint span, each coset taking every layer in order while its buckets are in cache.
543///
544/// Each layer moves a string only within its own coset of the joint span, so the cosets stay independent and every output of every layer passes `keep_term` as when the layers run one by one; no layer pass runs between them (`TruncationPolicy::run_length_limit`).
545pub(crate) fn apply_rotation_run<const W: usize, C: Coefficient, T>(
546    sum: &mut PauliSum<W, C>,
547    run: &[Prepared<W, C>],
548    policy: &T,
549    scratch: &mut LayerScratch<W, C>,
550) where
551    T: TruncationPolicy<W, C> + ?Sized,
552{
553    stamp!(stamp);
554    if let Some(cut) = scratch.octaves.pending.take() {
555        cut.apply(sum);
556    }
557    scratch.octaves.counted = false;
558    scratch.octaves.drop_ahead();
559    let layers: Vec<RotationLayer<'_, W, C>> = run
560        .iter()
561        .map(|layer| match layer {
562            Prepared::Local(ptm) => RotationLayer::of(ptm),
563            Prepared::Rotation(_) => None,
564        })
565        .map(|layer| layer.expect("a rotation run holds rotation tables only"))
566        .collect();
567    let deltas: Vec<u32> = layers.iter().map(|layer| layer.bucket_delta).collect();
568    let span = Gf2Span::new(&deltas, sum.hash().bits());
569    let coordinates: Vec<u32> = deltas.iter().map(|&delta| span.coordinate_of(delta)).collect();
570    lap!(stamp, scratch.stats.span_plan_ns);
571    #[cfg(test)]
572    {
573        scratch.rotation_run_layers += run.len();
574    }
575    let octaves = LayerOctaveIo { cut: None, count: None, ahead: None };
576    let len = rotation_coset_loop(
577        sum,
578        &span,
579        scratch,
580        octaves,
581        |members, worker_scratch, supports, known| {
582            fill_rotation_run(
583                members,
584                &layers,
585                &coordinates,
586                policy,
587                worker_scratch,
588                supports,
589                known,
590            )
591        },
592    );
593    lap!(stamp, scratch.stats.coset_loop_ns);
594    sum.set_len_after_layer(len);
595    lap!(stamp, scratch.stats.recount_ns);
596
597    #[cfg(debug_assertions)]
598    sum.assert_invariants();
599}
