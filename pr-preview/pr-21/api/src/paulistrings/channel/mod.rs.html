21/// Fixed-capacity structure-of-arrays buffer that [`Channel::apply`] writes its outputs into.
22///
23/// Capacity is the length of the columns, sized by the caller to at least [`Channel::max_fanout`] per input term.
24pub struct OutputBuffer<'a, const W: usize> {
25    /// X-part column.
26    pub x: &'a mut [[u64; W]],
27    /// Z-part column.
28    pub z: &'a mut [[u64; W]],
29    /// Coefficient column.
30    pub coeff: &'a mut [Complex64],
31    /// Number of terms written so far.
32    pub len: &'a mut usize,
33}
34
35impl<'a, const W: usize> OutputBuffer<'a, W> {
36    /// Append one term.
37    ///
38    /// Panics if the buffer is full; an `apply` body must not push more than its declared `max_fanout`.
39    pub fn push(&mut self, x: [u64; W], z: [u64; W], c: Complex64) {
40        debug_assert!(
41            *self.len < self.x.len(),
42            "OutputBuffer overflow: {} pushes into a buffer of capacity {}",
43            *self.len + 1,
44            self.x.len()
45        );
46        let i = *self.len;
47        self.x[i] = x;
48        self.z[i] = z;
49        self.coeff[i] = c;
50        *self.len = i + 1;
51    }
52
53    /// Reset the length to zero, keeping the storage.
54    pub fn clear(&mut self) {
55        *self.len = 0;
56    }
57}

59/// Pack qubit indices into a [`Channel::support`] bitmask; order and duplicates do not matter.
60pub fn support_mask<const W: usize>(qubits: &[usize]) -> [u64; W] {
61    let mut mask = [0u64; W];
62    for &q in qubits {
63        debug_assert!(q < 64 * W, "qubit {q} out of range for W={W}");
64        mask[q / 64] |= 1u64 << (q % 64);
65    }
66    mask
67}
68
69/// `(word, bit, 1 << bit)` of qubit `q`.
70fn qubit_loc(q: usize) -> (usize, usize, u64) {
71    let word = q / 64;
72    let bit = q % 64;
73    (word, bit, 1u64 << bit)
74}

76/// The packed single-qubit Pauli index `x | (z << 1)` (`I=0, X=1, Z=2, Y=3`) at `(word, bit)`.
77fn read_pauli<const W: usize>(x: &[u64; W], z: &[u64; W], word: usize, bit: usize) -> usize {
78    let x_bit = (x[word] >> bit) & 1;
79    let z_bit = (z[word] >> bit) & 1;
80    (x_bit | (z_bit << 1)) as usize
81}
82
83/// Overwrite the qubit at `(word, bit, mask)` with the packed Pauli index `p`.
84fn write_pauli<const W: usize>(
85    x: &mut [u64; W],
86    z: &mut [u64; W],
87    word: usize,
88    bit: usize,
89    mask: u64,
90    p: usize,
91) {
92    let x_bit = (p & 1) as u64;
93    let z_bit = ((p >> 1) & 1) as u64;
94    x[word] = (x[word] & !mask) | (x_bit << bit);
95    z[word] = (z[word] & !mask) | (z_bit << bit);
96}
97
98/// Set or clear the bit at `(word, mask)` of one bit-plane.
99fn set_bit<const W: usize>(plane: &mut [u64; W], word: usize, mask: u64, value: bool) {
100    if value {
101        plane[word] |= mask;
102    } else {
103        plane[word] &= !mask;
104    }
105}

107/// A gate or noise channel: maps a Pauli string to a small weighted sum of Pauli strings.
108///
109/// Built-ins: [`Clifford1Q`], [`Clifford2Q`], [`PauliRotation`], [`GeneralUnitary1Q`], [`GeneralUnitary2Q`], [`Depolarizing`], [`Dephasing`], [`PauliChannel`], [`Depolarizing2Q`], [`AmplitudeDamping`], [`IdentityChannel`].
110/// A custom channel implements `max_fanout`, `support` and `apply`; the engine derives everything else (ARCHITECTURE.md §Prepared-Channels).
111///
112/// ```
113/// use paulistrings::{Channel, OutputBuffer};
114/// use num_complex::Complex64;
115///
116/// /// Multiplies every coefficient by a complex factor.
117/// struct GlobalPhase {
118///     factor: Complex64,
119/// }
120///
121/// impl<const W: usize> Channel<W> for GlobalPhase {
122///     fn max_fanout(&self) -> usize { 1 }
123///     fn support(&self) -> [u64; W] { [0; W] }
124///     fn apply(
125///         &self,
126///         input_x: &[u64; W],
127///         input_z: &[u64; W],
128///         coeff: Complex64,
129///         out: &mut OutputBuffer<'_, W>,
130///     ) {
131///         out.push(*input_x, *input_z, coeff * self.factor);
132///     }
133/// }
134///
135/// let channel = GlobalPhase {
136///     factor: Complex64::new(0.0, 1.0),
137/// };
138/// let _: Box<dyn Channel<1>> = Box::new(channel);
139/// ```
140pub trait Channel<const W: usize>: Send + Sync {
141    /// Upper bound on the output terms per input term.
142    fn max_fanout(&self) -> usize;
143
144    /// Qubits this channel acts on, as a bitmask built with [`support_mask`].
145    ///
146    /// Outputs must differ from their input only at these qubits.
147    fn support(&self) -> [u64; W];
148
149    /// Short name for the per-layer progress log.
150    ///
151    /// The default is the type name without path or generics, which is unhelpful for tuples, closures and boxes; override it there.
152    fn debug_name(&self) -> &'static str {
153        let full = core::any::type_name::<Self>();
154        let head = match full.find('<') {
155            Some(i) => &full[..i],
156            None => full,
157        };
158        match head.rfind("::") {
159            Some(i) => &head[i + 2..],
160            None => head,
161        }
162    }
163
164    /// Apply the channel to one input term, pushing its outputs to `out`.
165    fn apply(
166        &self,
167        input_x: &[u64; W],
168        input_z: &[u64; W],
169        coeff: Complex64,
170        out: &mut OutputBuffer<'_, W>,
171    );
172
173    /// Apply the channel's adjoint to one input term; run by `Direction::Heisenberg`.
174    ///
175    /// The default calls [`apply`](Self::apply), which is correct only for a self-adjoint channel.
176    fn apply_adjoint(
177        &self,
178        input_x: &[u64; W],
179        input_z: &[u64; W],
180        coeff: Complex64,
181        out: &mut OutputBuffer<'_, W>,
182    ) {
183        self.apply(input_x, input_z, coeff, out);
184    }
185
186    /// Prepare this channel for one engine layer (ARCHITECTURE.md §Prepared-Channels).
187    ///
188    /// The default probes `apply` on the local basis Paulis and supports at most two qubits; `None` makes `propagate` panic.
189    /// It is exact only if the output amplitudes depend on the input solely through its bits at [`support`](Self::support), which debug builds check.
190    fn prepare(&self, hash: &Gf2Hash<W>, adjoint: bool) -> Option<Prepared<W>> {
191        Prepared::derive_local(self, hash, adjoint)
192    }
193}
