1//! The layer loop `run_layers` every driver runs, one partition's share of a call at any partition count.
2//!
3//! Every partition issues the same transport calls in the same order, or the group falls out of step; per layer:
4//! 1. on a layer whose plan has a remote delta, an `exchange` of the proposed bucket bits within its exchange group ([`PartitionPlan::exchange_group`]), whose maximum every member then holds, since partners index an exchange's blocks by the count;
5//! 2. the exchange, only if the plan has a remote delta;
6//! 3. the policy's layer pass, on every layer, if `finalizes_layer` is true;
7//!
8//! and one `allreduce_max_u8` after the last layer, which refines every partition to the group's largest count, so a call returns with one count everywhere.
9//! A local layer issues no call for the count: on the schedule of `rebuckets` a partition moves to its own proposal, so counts differ mid-call.
10//! At `P = 1` there is no group: the loop refines every layer, as `propagate` does.
11//!
12//! Consecutive local layers under a policy without a layer pass may run as one pass ([`Runs`]); local layers issue no transport call, so a partition may take a run whatever its peers do.
13

34/// Layers between two rebuckets of a partition in a group, and the length of the opening ramp that precedes them.
35const REBUCKET_EVERY: usize = 16;
36
37/// Whether a partition of a group rebuckets at the layer at `application_index`, whatever the layer's plan.
38fn rebuckets(application_index: usize) -> bool {
39    application_index < REBUCKET_EVERY || application_index.is_multiple_of(REBUCKET_EVERY)
40}
41
42/// The largest of `bits` over this partition and `group`, one `exchange` among them.
43fn exchange_bits<X: Transport>(transport: &X, group: &[u32], bits: u8) -> u8 {
44    let mut send: Vec<Option<ByteParts>> = (0..transport.size()).map(|_| None).collect();
45    for &member in group {
46        send[member as usize] = Some(ByteParts(vec![vec![bits]]));
47    }
48    let received = transport.exchange(send, &mut Vec::new());
49    received.iter().flatten().map(|payload| payload.0[0][0]).fold(bits, u8::max)
50}

52/// A [`Collectives`] view that counts the policy finalization's calls for the trace and, when `nanos` is given, times them.
53struct CountingCollectives<'a> {
54    inner: &'a dyn Collectives,
55    calls: &'a AtomicU32,
56    nanos: Option<&'a AtomicU64>,
57}
58
59impl CountingCollectives<'_> {
60    fn call<R>(&self, collective: impl FnOnce() -> R) -> R {
61        self.calls.fetch_add(1, Ordering::Relaxed);
62        let start = self.nanos.map(|_| Instant::now());
63        let result = collective();
64        if let (Some(nanos), Some(start)) = (self.nanos, start) {
65            nanos.fetch_add(start.elapsed().as_nanos() as u64, Ordering::Relaxed);
66        }
67        result
68    }
69}
70
71impl crate::collectives::sealed::Sealed for CountingCollectives<'_> {}

73impl Collectives for CountingCollectives<'_> {
74    fn rank(&self) -> u32 {
75        self.inner.rank()
76    }
77    fn size(&self) -> u32 {
78        self.inner.size()
79    }
80    fn allreduce_max_u8(&self, v: u8) -> u8 {
81        self.call(|| self.inner.allreduce_max_u8(v))
82    }
83    fn allreduce_sum_u64(&self, buffer: &mut [u64]) {
84        self.call(|| self.inner.allreduce_sum_u64(buffer))
85    }
86    fn allreduce_sum_f64(&self, buffer: &mut [f64]) {
87        self.call(|| self.inner.allreduce_sum_f64(buffer))
88    }
89    fn barrier(&self) {
90        self.call(|| self.inner.barrier())
91    }
92}

109/// Why a channel's preparation declines, shared by the host panic and the device's `GpuError::Unsupported`.
110pub(crate) const DECLINED_PREPARATION: &str = "support above MAX_LOCAL_SUPPORT (2) qubits is \
111     unsupported except for PauliRotation, and a channel must not write outside its declared support";
112const _: () = assert!(MAX_LOCAL_SUPPORT == 2, "DECLINED_PREPARATION names the value");
113
114/// The channel's preparation, or the unpartitioned engine's hard error naming the partition and layer.
115fn prepare_or_panic<const W: usize, C: Coefficient>(
116    channel: &dyn Channel<W>,
117    hash: &Gf2Hash<W>,
118    adjoint: bool,
119    rank: usize,
120    circuit_index: usize,
121) -> Prepared<W, C> {
122    let prepared = channel.prepare(hash, adjoint).unwrap_or_else(|| {
123        let weight: u32 = channel.support().iter().map(|w| w.count_ones()).sum();
124        panic!(
125            "partition {rank}, layer {circuit_index}: channel preparation declined for a \
126             channel of support {weight}: {DECLINED_PREPARATION}",
127        )
128    });
129    prepared.convert().expect("assert_coefficient_type admitted the circuit")
130}

132/// Panic unless a sum of coefficient type `C` can carry `circuit`: a real type needs [`Circuit::preserves_hermiticity`].
133pub(crate) fn assert_coefficient_type<const W: usize, C: Coefficient>(circuit: &Circuit<W>) {
134    assert!(
135        !C::IS_REAL || circuit.preserves_hermiticity(),
136        "propagate: a {} sum needs every channel to preserve Hermiticity (a real transfer table); \
137         use a complex coefficient type",
138        C::NAME,
139    );
140}

142/// The layers after the current one, prepared on demand under the current hash.
143struct Lookahead<'a, const W: usize> {
144    circuit: &'a Circuit<W>,
145    circuit_index_of: &'a dyn Fn(usize) -> usize,
146    rows: &'a PartitionRows<W>,
147    /// Scratch for [`exchanges`](Self::exchanges).
148    plan: RefCell<PartitionPlan>,
149    adjoint: bool,
150    rank: usize,
151}
152
153impl<const W: usize> Lookahead<'_, W> {
154    /// Whether `prepared` has a remote delta, which depends on the delta alone and so is alike on every partition.
155    fn exchanges<C: Coefficient>(&self, prepared: &Prepared<W, C>) -> bool {
156        let mut plan = self.plan.borrow_mut();
157        plan.rebuild(prepared, self.rows, self.rank as u32);
158        plan.has_remote()
159    }
160
161    /// Hand `accept` the layers from application index `start` on, prepared under `hash`, until it refuses one, one exchanges, or it took `limit`; returns how many it took.
162    fn extend<C: Coefficient>(
163        &self,
164        start: usize,
165        hash: &Gf2Hash<W>,
166        limit: usize,
167        mut accept: impl FnMut(Prepared<W, C>) -> bool,
168    ) -> usize {
169        let end = self.circuit.channels.len().min(start.saturating_add(limit));
170        let mut taken = 0;
171        for application_index in start..end {
172            let index = (self.circuit_index_of)(application_index);
173            let channel = self.circuit.channels[index].as_ref();
174            let prepared = prepare_or_panic(channel, hash, self.adjoint, self.rank, index);
175            if self.exchanges(&prepared) || !accept(prepared) {
176                break;
177            }
178            taken += 1;
179        }
180        taken
181    }
182}

184/// The runs a call takes, consecutive local layers applied in one pass by [`PartitionBackend::apply_run`], and the buffers kept between them.
185struct Runs<const W: usize, C: Coefficient> {
186    /// Runs of one-to-one channels (`engine/bucketed/permutation_run.rs`), tried first.
187    permutations: bool,
188    /// Rotation runs (`engine/bucketed/rotation_layer.rs`).
189    rotations: bool,
190    steps: Vec<PermutationStep<W, C>>,
191    layers: Vec<Prepared<W, C>>,
192}
193
194impl<const W: usize, C: Coefficient> Runs<W, C> {
195    /// Apply the run that starts at application index `start` with `prepared`, and return its end, or `None` to apply layer `start` alone.
196    // Never inlined: `run_layers` inlines the merge kernels, which are sensitive to code motion.
197    #[inline(never)]
198    fn take<T, B>(
199        &mut self,
200        start: usize,
201        prepared: &Prepared<W, C>,
202        local: &mut B,
203        policy: &T,
204        lookahead: &Lookahead<'_, W>,
205        logging: bool,
206    ) -> Option<usize>
207    where
208        T: TruncationPolicy<W, C> + ?Sized,
209        B: PartitionBackend<W, C, T>,
210    {
211        if lookahead.exchanges(prepared) {
212            return None;
213        }
214        if self.permutations {
215            if let Some(first) = PermutationStep::of(prepared) {
216                self.steps.clear();
217                self.steps.push(first);
218                let steps = &mut self.steps;
219                let joined = lookahead.extend(start + 1, local.hash(), usize::MAX, |next| {
220                    PermutationStep::of(&next).map(|step| steps.push(step)).is_some()
221                });
222                if joined > 0 {
223                    local.apply_run(Run::Permutations(&self.steps), policy);
224                    return Some(start + 1 + joined);
225                }
226            }
227        }
228        if !self.rotations || logging {
229            return None;
230        }
231        let mut span = RotationRunSpan::new(local.hash().bits());
232        if !span.join(prepared) {
233            return None;
234        }
235        self.layers.clear();
236        let layers = &mut self.layers;
237        let joined =
238            lookahead.extend(start + 1, local.hash(), MAX_ROTATION_RUN_LAYERS - 1, |next| {
239                span.join(&next) && {
240                    layers.push(next);
241                    true
242                }
243            });
244        if joined == 0 {
245            return None;
246        }
247        self.layers.insert(0, prepared.clone());
248        local.apply_run(Run::Rotations(&self.layers), policy);
249        Some(start + 1 + joined)
250    }
251}

253/// Trace a run of the layers `layers` as one row per layer, its time and phases on the first, into `trace` when given, and log it at `DEBUG`.
254#[cold]
255#[allow(clippy::too_many_arguments)]
256fn record_run<const W: usize, C: Coefficient, B: PartitionStorage<W, C>>(
257    circuit: &Circuit<W>,
258    layers: std::ops::Range<usize>,
259    circuit_index_of: &dyn Fn(usize) -> usize,
260    local: &B,
261    (terms_before, bits_before, nanos): (usize, u8, u64),
262    mut phases: LayerPhases,
263    (rank, size): (usize, usize),
264    trace: Option<&mut Vec<PartitionLayerRow>>,
265) {
266    let first = layers.start;
267    if let Some(trace) = trace {
268        for layer in layers.clone() {
269            let index = circuit_index_of(layer);
270            let leads = layer == first;
271            let row = PartitionLayerRow {
272                bits: local.hash().bits(),
273                remote_deltas: 0,
274                collectives: 0,
275                circuit_index: index as u32,
276                application_index: layer as u32,
277                gate_name: circuit.channels[index].debug_name(),
278                terms_in: if leads { terms_before } else { local.len() },
279                terms_out: local.len(),
280                rows_sent: vec![0; size],
281                bytes_sent: vec![0; size],
282                rows_received: 0,
283                nanos: if leads { nanos } else { 0 },
284                refine_passes: if leads {
285                    u32::from(local.hash().bits().saturating_sub(bits_before))
286                } else {
287                    0
288                },
289                receive_chunks: 0,
290                device: None,
291                phases: if leads { std::mem::take(&mut phases) } else { LayerPhases::default() },
292            };
293            record_layer_row(trace, row);
294        }
295    }
296    log::debug!(
297        target: LOG_TARGET,
298        "layers {}-{}/{} [run]: {terms_before} -> {} terms, {:.1} ms (partition {rank}/{size})",
299        first + 1,
300        layers.end,
301        circuit.channels.len(),
302        local.len(),
303        nanos as f64 * 1e-6,
304    );
305}

307/// One partition's layer loop, run in lock-step with its peers inside its own pool, in the module's collective order.
308pub(crate) fn run_layers<const W: usize, C: Coefficient, T, X, B>(
309    circuit: &Circuit<W>,
310    policy: &T,
311    direction: Direction,
312    options: PropagateOptions,
313    context: PartitionContext<'_, W>,
314    work: &mut PartitionWork<B>,
315    transport: &X,
316) where
317    T: TruncationPolicy<W, C> + ?Sized,
318    X: Transport,
319    B: PartitionBackend<W, C, T>,
320{
321    assert_coefficient_type::<W, C>(circuit);
322    let PartitionContext { rows, rank, size, tracing } = context;
323    let num_layers = circuit.channels.len();
324    let adjoint = matches!(direction, Direction::Heisenberg);
325    let group_size = transport.size();
326    // `finalizes_layer` is a property of the policy type, so skipping the collective call is safe on every partition alike.
327    let finalizes = policy.finalizes_layer();
328    let policy_calls = AtomicU32::new(0);
329    let policy_nanos = AtomicU64::new(0);
330    // At `P = 1` the local answer is the agreed one, so the loop rebuckets every layer exactly as an unpartitioned run would.
331    let solo = group_size == 1;
332    let local = &mut work.local;
333    let mut plan = PartitionPlan::default();
334    let mut next_plan = PartitionPlan::default();
335    // Overlap with the bucket rows only costs balance once the bucket count reaches it, so it is reported there, once, by partition 0.
336    let mut overlap_onset = if rank == 0 { rows.overlap_onset(local.hash()) } else { None };
337
338    let circuit_index_of = |application_index: usize| match direction {
339        Direction::Forward => application_index,
340        Direction::Heisenberg => num_layers - 1 - application_index,
341    };
342    let lookahead = Lookahead {
343        circuit,
344        circuit_index_of: &circuit_index_of,
345        rows,
346        plan: RefCell::default(),
347        adjoint,
348        rank,
349    };
350    let mut runs = Runs::<W, C> {
351        permutations: !finalizes && local.takes_runs(),
352        rotations: !finalizes && local.takes_runs(),
353        steps: Vec::new(),
354        layers: Vec::new(),
355    };
356
357    local.begin_call();
358    let mut application_index = 0;
359    while application_index < num_layers {
360        let circuit_index = circuit_index_of(application_index);
361        let channel: &dyn Channel<W> = circuit.channels[circuit_index].as_ref();
362
363        let debug_enabled = log::log_enabled!(target: LOG_TARGET, log::Level::Debug);
364        let timed_layer = tracing || debug_enabled;
365        let layer_start = timed_layer.then(Instant::now);
366        let terms_before = local.len();
367        let bits_before = local.hash().bits();
368        let mut phases = LayerPhases::default();
369        let mut clock = LayerClock::new(if tracing { layer_start } else { None });
370        #[cfg(feature = "phase-timing")]
371        let stats_before = tracing.then(|| local.phase_snapshot());
372
373        stamp!(stamp);
374        tally!(local.stats().layers, 1);
375        tally!(local.stats().terms_in, terms_before);
376
377        // Alone, a backend that sizes its buckets without the prepared channel settles the count first, so a growing layer prepares once.
378        let refined_early =
379            solo && local.refine_unprepared(options.target_bucket_len, options.min_buckets);
380        if refined_early {
381            lap!(stamp, local.stats().rebucket_ns);
382        }
383        clock.lap(&mut phases.rebucket_ns);
384
385        // Otherwise prepared before the bucket count is settled, since the plan decides whether this layer agrees it; only `bucket_delta` depends on the count, so a refining layer prepares again.
386        let mut prepared = prepare_or_panic(channel, local.hash(), adjoint, rank, circuit_index);
387        plan.rebuild(&prepared, rows, rank as u32);
388        lap!(stamp, local.stats().prepare_ns);
389        clock.lap(&mut phases.prepare_ns);
390
391        let mut collectives = 0u32;
392        if !refined_early && (solo || plan.has_remote() || rebuckets(application_index)) {
393            let mut wanted_bits =
394                local.proposed_bits(&prepared, options.target_bucket_len, options.min_buckets);
395            if !solo && plan.has_remote() {
396                wanted_bits =
397                    exchange_bits(transport, &plan.exchange_group(rank as u32), wanted_bits);
398                collectives += 1;
399            }
400            lap!(stamp, local.stats().collective_ns);
401            clock.lap(&mut phases.collective_ns);
402            if wanted_bits != local.hash().bits() {
403                while local.hash().bits() < wanted_bits {
404                    local.refine();
405                }
406                if wanted_bits < local.hash().bits() {
407                    local.coarsen_to(wanted_bits);
408                }
409                lap!(stamp, local.stats().rebucket_ns);
410                clock.lap(&mut phases.rebucket_ns);
411                prepared = prepare_or_panic(channel, local.hash(), adjoint, rank, circuit_index);
412                plan.rebuild(&prepared, rows, rank as u32);
413                lap!(stamp, local.stats().prepare_ns);
414                clock.lap(&mut phases.prepare_ns);
415            }
416        }
417
418        if overlap_onset.is_some_and(|onset| local.hash().bits() >= onset) {
419            overlap_onset = None;
420            log::warn!(
421                target: LOG_TARGET,
422                "layer {}/{num_layers}: at {} buckets the partition rows overlap the span of the bucket hash's active rows, so partitions and buckets are correlated and some of their pairs stay empty",
423                application_index + 1,
424                local.hash().num_buckets(),
425            );
426        }
427
428        if let Some(end) =
429            runs.take(application_index, &prepared, local, policy, &lookahead, debug_enabled)
430        {
431            timed!(stamp.rearm());
432            clock.lap(&mut phases.coset_loop_ns);
433            // Per layer of the run, as the one-by-one path counts them.
434            tally!(local.stats().layers, end - application_index - 1);
435            tally!(local.stats().terms_in, terms_before * (end - application_index - 1));
436            tally!(local.stats().terms_out, local.len() * (end - application_index));
437            #[cfg(feature = "phase-timing")]
438            {
439                phases.fine = stats_before.map(|before| {
440                    crate::engine::trace::FineLayerPhases::between(&before, &local.phase_snapshot())
441                });
442            }
443            if let Some(start) = layer_start {
444                let before = (terms_before, bits_before, start.elapsed().as_nanos() as u64);
445                let trace = tracing.then_some(&mut work.rows);
446                let layers = application_index..end;
447                let order = &circuit_index_of;
448                record_run(circuit, layers, order, &*local, before, phases, (rank, size), trace);
449            }
450            application_index = end;
451            continue;
452        }
453
454        // The next layer prepared under this layer's hash, so the coset loop can count the rows it exports; a rebucket in between drops the counts.
455        let next_prepared = (!solo && application_index + 1 < num_layers).then(|| {
456            let next_index = circuit_index_of(application_index + 1);
457            let next_channel = circuit.channels[next_index].as_ref();
458            let next = prepare_or_panic(next_channel, local.hash(), adjoint, rank, next_index);
459            next_plan.rebuild(&next, rows, rank as u32);
460            next
461        });
462        let ahead = match (&next_prepared, next_plan.remote.as_slice()) {
463            (Some(next), [remote]) => Some(RowEmitter::of(next, remote)),
464            _ => None,
465        };
466        lap!(stamp, local.stats().prepare_ns);
467        clock.lap(&mut phases.prepare_ns);
468
469        let counts =
470            local.apply_layer(&prepared, &plan, rows, policy, transport, tracing, ahead.as_ref());
471        timed!(stamp.rearm());
472        let mut applied_ns = 0;
473        clock.lap(&mut applied_ns);
474        phases.export_ns = counts.export_ns;
475        phases.exchange_ns = counts.exchange_ns;
476        phases.coset_loop_ns =
477            applied_ns.saturating_sub(counts.export_ns).saturating_sub(counts.exchange_ns);
478
479        if finalizes {
480            policy_calls.store(0, Ordering::Relaxed);
481            policy_nanos.store(0, Ordering::Relaxed);
482            let counting = CountingCollectives {
483                inner: transport,
484                calls: &policy_calls,
485                nanos: tracing.then_some(&policy_nanos),
486            };
487            local.finalize_layer(policy, &counting);
488            collectives += policy_calls.load(Ordering::Relaxed);
489        }
490        lap!(stamp, local.stats().finalize_ns);
491        clock.lap(&mut phases.finalize_ns);
492        let policy_collective_ns = policy_nanos.load(Ordering::Relaxed).min(phases.finalize_ns);
493        phases.finalize_ns -= policy_collective_ns;
494        phases.collective_ns += policy_collective_ns;
495        #[cfg(feature = "phase-timing")]
496        {
497            phases.fine = stats_before.map(|before| {
498                crate::engine::trace::FineLayerPhases::between(&before, &local.phase_snapshot())
499            });
500        }
501        tally!(local.stats().terms_out, local.len());
502
503        // The trace sits behind a hoisted flag and a `#[cold]` callee: this loop inlines the merge kernels, which are sensitive to code motion.
504        let remote_deltas = counts.remote_deltas;
505        let rows_received = counts.rows_received;
506        let elapsed = layer_start.map(|start| start.elapsed());
507        if tracing {
508            let mut row = PartitionLayerRow {
509                bits: local.hash().bits(),
510                remote_deltas: counts.remote_deltas as u32,
511                collectives,
512                circuit_index: circuit_index as u32,
513                application_index: application_index as u32,
514                gate_name: channel.debug_name(),
515                terms_in: terms_before,
516                terms_out: local.len(),
517                rows_sent: counts.rows_sent,
518                bytes_sent: counts.bytes_sent,
519                rows_received: counts.rows_received,
520                nanos: elapsed.unwrap_or_default().as_nanos() as u64,
521                refine_passes: u32::from(local.hash().bits().saturating_sub(bits_before)),
522                receive_chunks: counts.receive_chunks,
523                device: None,
524                phases,
525            };
526            local.trace_layer(&mut row);
527            record_layer_row(&mut work.rows, row);
528        }
529
530        application_index += 1;
531        if debug_enabled {
532            let Some(elapsed) = elapsed else { continue };
533            log::debug!(
534                target: LOG_TARGET,
535                "layer {}/{} [{}]: {} -> {} terms, {:.1} ms (partition {}/{}, {} remote deltas, \
536                 {} rows in)",
537                application_index,
538                num_layers,
539                channel.debug_name(),
540                terms_before,
541                local.len(),
542                elapsed.as_secs_f64() * 1e3,
543                rank,
544                size,
545                remote_deltas,
546                rows_received,
547            );
548        }
549    }
550    if !solo {
551        let agreed = transport.allreduce_max_u8(local.hash().bits());
552        while local.hash().bits() < agreed {
553            local.refine();
554        }
555    }
556    local.settle();
557}
