1//! Runs of one-to-one channels applied as one pass: every string walks the steps that touch its qubits, then the sum is rehashed and each bucket sorted once.
2//!
3//! Contracts: a [`PermutationStep`] maps distinct strings to distinct strings, so a run never sums two strings and agrees with the per-layer path to floating-point tolerance (its coefficients see the same products in the same order); every intermediate string is dropped exactly when the per-layer merge would drop it (an exact zero, or `keep_term` refusing it), with `keep_term` taken as a function of its arguments, so a step that leaves a string unchanged does not ask it again; the run skips the layer passes between its steps, so it runs only within the policy's `TruncationPolicy::run_length_limit`.
4

14/// Steps per segment of a walk, the width of a [`StepSet`].
15const SEGMENT_STEPS: usize = 256;
16
17/// A set of a segment's steps, one bit per step.
18type StepSet = [u64; SEGMENT_STEPS / 64];
19
20/// Below this many terms a run walks, scatters and sorts on the calling thread.
21const PARALLEL_MIN_TERMS: usize = 1 << 13;
22
23/// A rehashed bucket holds `1 / RUN_BUCKET_SLACK_DIVISOR` spare capacity, so the next rotation layer can merge it in place (research/FINDINGS.md §Clifford runs as one pass).
24const RUN_BUCKET_SLACK_DIVISOR: usize = 2;
25
26/// Source-bucket chunks per Rayon thread in a parallel run, each with its own row of destination counts.
27const CHUNKS_PER_THREAD: usize = 4;

29/// A prepared channel as a permutation of strings: support pattern `s` moves to `s ⊕ delta[s]` with its coefficient times `amplitude[s]`, and an exact-zero amplitude drops it.
30#[derive(Clone, Debug)]
31pub(crate) struct PermutationStep<const W: usize, C: Coefficient> {
32    /// Word and bit of support qubit `j`; a one-qubit step repeats qubit 0 and masks its second pattern pair off.
33    word: [usize; MAX_LOCAL_SUPPORT],
34    bit: [u32; MAX_LOCAL_SUPPORT],
35    /// `0b11` for a two-qubit step, `0` for a one-qubit step: the pattern bits of the second qubit.
36    second: usize,
37    /// The support qubits, the first repeated for a one-qubit step.
38    qubit: [usize; MAX_LOCAL_SUPPORT],
39    /// Bit `s` when pattern `s` keeps its key and coefficient exactly.
40    inert: u16,
41    /// Per pattern, bit `j` when support qubit `j` is the identity before the step and not after.
42    grows: [u8; LOCAL_DIM],
43    delta: [u8; LOCAL_DIM],
44    /// `h(lift(delta[s]))` under the hash the channel was prepared for.
45    bucket_delta: [u32; LOCAL_DIM],
46    amplitude: [C; LOCAL_DIM],
47}
48
49impl<const W: usize, C: Coefficient> PermutationStep<W, C> {
50    /// The permutation `prepared` applies, as the per-layer path applies it, or `None` when it can split or merge strings.
51    ///
52    /// A quarter-turn rotation table counts as its Clifford arm, which drops the identity residue.
53    pub(crate) fn of(prepared: &Prepared<W, C>) -> Option<Self> {
54        let Prepared::Local(ptm) = prepared else { return None };
55        let k = ptm.k();
56        if k == 0 {
57            return None;
58        }
59        let dim = 1usize << (2 * k);
60        let deltas = ptm.deltas();
61        // Per pattern, the one entry that emits for it.
62        let mut emitting = [None::<usize>; LOCAL_DIM];
63        match RotationLayer::of(ptm) {
64            Some(layer) if !layer.is_clifford() => return None,
65            Some(_) => {
66                for (s, slot) in emitting.iter_mut().enumerate().take(dim) {
67                    *slot = Some(usize::from(deltas[1].amplitude[s] != C::ZERO));
68                }
69            }
70            None => {
71                for (s, slot) in emitting.iter_mut().enumerate().take(dim) {
72                    let mut entries =
73                        (0..deltas.len()).filter(|&e| deltas[e].amplitude[s] != C::ZERO);
74                    *slot = entries.next();
75                    if entries.next().is_some() {
76                        return None;
77                    }
78                }
79            }
80        }
81        let mut reached = [false; LOCAL_DIM];
82        let mut delta = [0u8; LOCAL_DIM];
83        let mut bucket_delta = [0u32; LOCAL_DIM];
84        let mut amplitude = [C::ZERO; LOCAL_DIM];
85        for s in 0..dim {
86            let Some(entry) = emitting[s] else { continue };
87            let entry = &deltas[entry];
88            let target = s ^ entry.local_delta as usize;
89            if std::mem::replace(&mut reached[target], true) {
90                return None;
91            }
92            delta[s] = entry.local_delta;
93            bucket_delta[s] = entry.bucket_delta;
94            amplitude[s] = entry.amplitude[s];
95        }
96        let qubits: Vec<usize> = ptm.qubits().collect();
97        let last = qubits[k - 1];
98        let mut inert = 0u16;
99        let mut grows = [0u8; LOCAL_DIM];
100        for s in 0..dim {
101            inert |= u16::from(delta[s] == 0 && amplitude[s] == C::ONE) << s;
102            let t = s ^ delta[s] as usize;
103            for j in 0..k {
104                let occupied = |pattern: usize| (pattern >> (2 * j)) & 0b11 != 0;
105                grows[s] |= u8::from(!occupied(s) && occupied(t)) << j;
106            }
107        }
108        let word = [qubits[0] / 64, last / 64];
109        let bit = [(qubits[0] % 64) as u32, (last % 64) as u32];
110        Some(Self {
111            word,
112            bit,
113            second: if k == 2 { 0b11 } else { 0 },
114            qubit: [qubits[0], last],
115            inert,
116            grows,
117            delta,
118            bucket_delta,
119            amplitude,
120        })
121    }
122
123    /// The step as the device walks it: its two support qubits (the first repeated for a one-qubit step), whether the second is one, the inert patterns, and per pattern the local delta and amplitude.
124    #[cfg(feature = "cuda")]
125    pub(crate) fn walk_table(&self) -> StepWalk<'_, C> {
126        StepWalk {
127            qubits: self.qubit,
128            two_qubit: self.second != 0,
129            inert: self.inert,
130            delta: &self.delta,
131            amplitude: &self.amplitude,
132        }
133    }
134
135    /// The support pattern of a key, packed as the prepared table's `local_delta`.
136    #[inline(always)]
137    fn pattern(&self, x: &[u64; W], z: &[u64; W]) -> usize {
138        let [w0, w1] = self.word;
139        let [b0, b1] = self.bit;
140        let first = ((x[w0] >> b0) & 1) | (((z[w0] >> b0) & 1) << 1);
141        let second = ((x[w1] >> b1) & 1) | (((z[w1] >> b1) & 1) << 1);
142        first as usize | ((second as usize & self.second) << 2)
143    }
144
145    /// `key ⊕ lift(delta[s])`.
146    #[inline(always)]
147    fn move_key(&self, x: &mut [u64; W], z: &mut [u64; W], s: usize) {
148        let d = self.delta[s] as u64;
149        let [w0, w1] = self.word;
150        let [b0, b1] = self.bit;
151        x[w0] ^= (d & 1) << b0;
152        z[w0] ^= ((d >> 1) & 1) << b0;
153        x[w1] ^= ((d >> 2) & 1) << b1;
154        z[w1] ^= ((d >> 3) & 1) << b1;
155    }
156}

160pub(crate) struct StepWalk<'s, C> {
161    pub(crate) qubits: [usize; MAX_LOCAL_SUPPORT],
162    pub(crate) two_qubit: bool,
163    pub(crate) inert: u16,
164    pub(crate) delta: &'s [u8; LOCAL_DIM],
165    pub(crate) amplitude: &'s [C; LOCAL_DIM],
166}

168/// The working set of [`apply_permutation_run`], kept across runs for its allocations.
169#[derive(Debug, Default)]
170pub(crate) struct PermutationScratch {
171    /// Per surviving row, in source order, the bucket it moves to.
172    destinations: Vec<u32>,
173    /// Per source chunk, a count and then a write cursor per destination bucket.
174    cursors: Vec<u32>,
175}

177/// Up to [`SEGMENT_STEPS`] consecutive steps of a run, indexed by the qubits they read.
178struct Segment<'s, const W: usize, C: Coefficient> {
179    /// Index of `steps[0]` in the run.
180    first: usize,
181    steps: &'s [PermutationStep<W, C>],
182    /// Per qubit, the steps whose support holds it.
183    by_qubit: Vec<StepSet>,
184    /// The steps that change a string which is the identity on their support.
185    always: StepSet,
186}
187
188impl<'s, const W: usize, C: Coefficient> Segment<'s, W, C> {
189    fn new(first: usize, steps: &'s [PermutationStep<W, C>], num_qubits: usize) -> Self {
190        debug_assert!(steps.len() <= SEGMENT_STEPS);
191        let mut by_qubit = vec![[0u64; SEGMENT_STEPS / 64]; num_qubits];
192        let mut always = [0u64; SEGMENT_STEPS / 64];
193        for (i, step) in steps.iter().enumerate() {
194            for &q in &step.qubit {
195                by_qubit[q][i / 64] |= 1 << (i % 64);
196            }
197            if step.inert & 1 == 0 {
198                always[i / 64] |= 1 << (i % 64);
199            }
200        }
201        Self { first, steps, by_qubit, always }
202    }
203
204    /// Add the steps after `i` that read qubit `q`.
205    #[inline(always)]
206    fn add_after(&self, relevant: &mut StepSet, q: usize, i: usize) {
207        let (word, bit) = (i / 64, i % 64);
208        let mask = &self.by_qubit[q];
209        relevant[word] |= mask[word] & (!0u64 << bit << 1);
210        for w in word + 1..relevant.len() {
211            relevant[w] |= mask[w];
212        }
213    }
214}

216/// Walk one bucket's rows through the run in place, keeping the survivors in order, and record where each moves.
217fn walk_bucket<const W: usize, C: Coefficient, T>(
218    source: u32,
219    bucket: &mut BucketColumns<W, C>,
220    segments: &[Segment<'_, W, C>],
221    policy: &T,
222    destinations: &mut [u32],
223    counts: &mut [u32],
224) where
225    T: TruncationPolicy<W, C> + ?Sized,
226{
227    let survives = |x: &[u64; W], z: &[u64; W], c: C| c.is_nonzero() && policy.keep_term(x, z, c);
228    let BucketColumns { x, z, coeff } = bucket;
229    let len = coeff.len();
230    let mut kept = 0usize;
231    'rows: for row in 0..len {
232        let (mut key_x, mut key_z, mut c, mut destination) = (x[row], z[row], coeff[row], source);
233        // Whether the current string passed `survives`, which the per-layer path asks after every step.
234        let mut tested = false;
235        for segment in segments {
236            let mut relevant = segment.always;
237            for w in 0..W {
238                let mut occupied = key_x[w] | key_z[w];
239                while occupied != 0 {
240                    let q = 64 * w + occupied.trailing_zeros() as usize;
241                    for (slot, &mask) in relevant.iter_mut().zip(&segment.by_qubit[q]) {
242                        *slot |= mask;
243                    }
244                    occupied &= occupied - 1;
245                }
246            }
247            let mut word = 0usize;
248            while word < relevant.len() {
249                let bits = relevant[word];
250                if bits == 0 {
251                    word += 1;
252                    continue;
253                }
254                relevant[word] = bits & (bits - 1);
255                let i = 64 * word + bits.trailing_zeros() as usize;
256                let step = &segment.steps[i];
257                let s = step.pattern(&key_x, &key_z);
258                if (step.inert >> s) & 1 == 1 {
259                    continue;
260                }
261                // A step skipped or inert before this one was asked about the unchanged input.
262                if !tested && segment.first + i > 0 && !survives(&key_x, &key_z, c) {
263                    continue 'rows;
264                }
265                step.move_key(&mut key_x, &mut key_z, s);
266                destination ^= step.bucket_delta[s];
267                c *= step.amplitude[s];
268                if !survives(&key_x, &key_z, c) {
269                    continue 'rows;
270                }
271                tested = true;
272                let grows = step.grows[s];
273                if grows != 0 {
274                    for j in 0..MAX_LOCAL_SUPPORT {
275                        if (grows >> j) & 1 == 1 {
276                            segment.add_after(&mut relevant, step.qubit[j], i);
277                        }
278                    }
279                }
280            }
281        }
282        if !tested && !segments.is_empty() && !survives(&key_x, &key_z, c) {
283            continue;
284        }
285        x[kept] = key_x;
286        z[kept] = key_z;
287        coeff[kept] = c;
288        destinations[kept] = destination;
289        counts[destination as usize] += 1;
290        kept += 1;
291    }
292    x.truncate(kept);
293    z.truncate(kept);
294    coeff.truncate(kept);
295}

297/// Raw write positions into the new buckets' columns, shared by the scatter's tasks.
298struct ScatterTargets<const W: usize, C> {
299    x: Vec<*mut [u64; W]>,
300    z: Vec<*mut [u64; W]>,
301    coeff: Vec<*mut C>,
302}
303
304// SAFETY: each task writes only the slots its own cursors span, disjoint from every other task's, and none reads.
305unsafe impl<const W: usize, C: Send> Sync for ScatterTargets<W, C> {}

307/// Apply `steps` in order to every term of `sum` as one pass, then move each term to its bucket and sort each bucket once.
308pub(crate) fn apply_permutation_run<const W: usize, C: Coefficient, T>(
309    sum: &mut PauliSum<W, C>,
310    steps: &[PermutationStep<W, C>],
311    policy: &T,
312    scratch: &mut PermutationScratch,
313) where
314    T: TruncationPolicy<W, C> + ?Sized,
315{
316    let num_buckets = sum.num_buckets();
317    let num_qubits = sum.num_qubits();
318    let segments: Vec<Segment<'_, W, C>> = steps
319        .chunks(SEGMENT_STEPS)
320        .enumerate()
321        .map(|(j, chunk)| Segment::new(j * SEGMENT_STEPS, chunk, num_qubits))
322        .collect();
323    let segments = segments.as_slice();
324    let parallel = sum.len() >= PARALLEL_MIN_TERMS;
325    let wanted_chunks = if parallel { rayon::current_num_threads() * CHUNKS_PER_THREAD } else { 1 };
326    let buckets_per_chunk = num_buckets.div_ceil(wanted_chunks.clamp(1, num_buckets));
327    let chunks = num_buckets.div_ceil(buckets_per_chunk);
328
329    let PermutationScratch { destinations, cursors } = scratch;
330    destinations.clear();
331    destinations.resize(sum.len(), 0);
332    cursors.clear();
333    cursors.resize(chunks * num_buckets, 0);
334
335    // Each source bucket's destinations start at its row offset before the walk, and its survivors fill them from the front.
336    let mut origins = Vec::with_capacity(num_buckets + 1);
337    origins.push(0usize);
338    for bucket in sum.buckets() {
339        origins.push(origins.last().copied().unwrap_or(0) + bucket.len());
340    }
341
342    {
343        let mut tasks = Vec::with_capacity(chunks);
344        let mut rest_destinations = destinations.as_mut_slice();
345        let mut rest_counts = cursors.as_mut_slice();
346        for (chunk, buckets) in sum.buckets_mut().chunks_mut(buckets_per_chunk).enumerate() {
347            let first = chunk * buckets_per_chunk;
348            let rows = origins[first + buckets.len()] - origins[first];
349            let (mine, later) = std::mem::take(&mut rest_destinations).split_at_mut(rows);
350            rest_destinations = later;
351            let (counts, later) = std::mem::take(&mut rest_counts).split_at_mut(num_buckets);
352            rest_counts = later;
353            tasks.push((first, buckets, mine, counts));
354        }
355        let walk = |(first, buckets, destinations, counts): (
356            usize,
357            &mut [BucketColumns<W, C>],
358            &mut [u32],
359            &mut [u32],
360        )| {
361            let base = origins[first];
362            for (j, bucket) in buckets.iter_mut().enumerate() {
363                let (start, end) = (origins[first + j] - base, origins[first + j + 1] - base);
364                let source = (first + j) as u32;
365                walk_bucket(
366                    source,
367                    bucket,
368                    segments,
369                    policy,
370                    &mut destinations[start..end],
371                    counts,
372                );
373            }
374        };
375        if parallel {
376            tasks.into_par_iter().for_each(walk);
377        } else {
378            tasks.into_iter().for_each(walk);
379        }
380    }
381
382    // Counts to cursors: chunk `c` writes destination `b` after the rows earlier chunks send there.
383    let mut totals = vec![0u32; num_buckets];
384    for row in cursors.chunks_mut(num_buckets) {
385        for (total, cursor) in totals.iter_mut().zip(row.iter_mut()) {
386            let count = *cursor;
387            *cursor = *total;
388            *total += count;
389        }
390    }
391
392    let allocate = |&total: &u32| {
393        BucketColumns::with_capacity(total as usize + total as usize / RUN_BUCKET_SLACK_DIVISOR)
394    };
395    let mut moved: Vec<BucketColumns<W, C>> = if parallel {
396        totals.par_iter().map(allocate).collect()
397    } else {
398        totals.iter().map(allocate).collect()
399    };
400    let targets = ScatterTargets {
401        x: moved.iter_mut().map(|bucket| bucket.x.as_mut_ptr()).collect(),
402        z: moved.iter_mut().map(|bucket| bucket.z.as_mut_ptr()).collect(),
403        coeff: moved.iter_mut().map(|bucket| bucket.coeff.as_mut_ptr()).collect(),
404    };
405
406    {
407        let buckets = sum.buckets();
408        let destinations = destinations.as_slice();
409        let targets = &targets;
410        let scatter = |(chunk, cursors): (usize, &mut [u32])| {
411            let first = chunk * buckets_per_chunk;
412            let last = (first + buckets_per_chunk).min(num_buckets);
413            for source in first..last {
414                let bucket = &buckets[source];
415                let moves = &destinations[origins[source]..origins[source] + bucket.len()];
416                for (row, &destination) in moves.iter().enumerate() {
417                    let destination = destination as usize;
418                    let slot = cursors[destination] as usize;
419                    cursors[destination] += 1;
420                    // SAFETY: `slot` lies in this chunk's span of `destination`'s cursors, below its `totals` capacity and written once.
421                    unsafe {
422                        targets.x[destination].add(slot).write(bucket.x[row]);
423                        targets.z[destination].add(slot).write(bucket.z[row]);
424                        targets.coeff[destination].add(slot).write(bucket.coeff[row]);
425                    }
426                }
427            }
428        };
429        if parallel {
430            cursors.par_chunks_mut(num_buckets).enumerate().for_each(scatter);
431        } else {
432            cursors.chunks_mut(num_buckets).enumerate().for_each(scatter);
433        }
434    }
435    drop(targets);
436
437    let finish = |sort: &mut SortScratch<W, C>,
438                  (bucket, &total): (&mut BucketColumns<W, C>, &u32)| {
439        // SAFETY: the scatter wrote every slot below `total`, the sum of every chunk's count for this bucket.
440        unsafe {
441            bucket.x.set_len(total as usize);
442            bucket.z.set_len(total as usize);
443            bucket.coeff.set_len(total as usize);
444        }
445        SortKernel::Comparison.sort(bucket, sort);
446    };
447    if parallel {
448        moved.par_iter_mut().zip(totals.par_iter()).for_each_init(SortScratch::default, finish);
449    } else {
450        let mut sort = SortScratch::default();
451        moved.iter_mut().zip(totals.iter()).for_each(|item| finish(&mut sort, item));
452    }
453    sum.replace_buckets(moved);
454}
