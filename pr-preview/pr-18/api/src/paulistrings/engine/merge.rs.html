1//! Per-run sort and fused two-stream merge — the bucketed engine's inner kernels.
2//!
3//! [`sort_rows_with_scratch`] and [`sort_rows_radix_with_scratch`] canonicalize one gather run's *rest* stream — the first by comparison, the second by a radix pass over a monotone surrogate; `bucketed.rs` picks between them per layer from the plan's rest-delta count, see [`RADIX_MIN_REST_STREAMS`].
4//! [`merge2_into`] fuses the id/rest two-stream merge with the segmented reduction that restores the `PauliSum` invariant (strictly ascending, no duplicates) inside a destination bucket.
5//! All are called per gather run by `engine::bucketed`; [`SortScratch`] is the worker-persistent scratch the sorts reuse so a steady-state layer allocates nothing.
6//!
7//! Both sorts satisfy one contract and nothing more, pinned by `tests::assert_sort_contract`: the output is **ascending** in lex `(x, z)` with duplicates allowed, and is a permutation of the input `(x, z, c)` triples.
8//! Equal-key order is unspecified (ARCHITECTURE.md §Determinism), so the two kernels are interchangeable to floating-point tolerance and not bitwise.
9

12use crate::truncation::TruncationPolicy;

14/// Worker-persistent scratch for the per-run sorts.
15///
16/// Held across coset tasks (one instance per `CosetScratch`, in turn one per Rayon worker, per `bucketed.rs`'s `LayerScratch`): every buffer retains its high-water capacity across calls, so a run at or below a previously-seen size sorts without allocating.
17/// `perm` serves the comparison kernel; `packed`/`aux` serve the radix kernel; the `tmp_*` triple is the output staging both share.
18#[derive(Clone, Debug, Default)]
19pub(crate) struct SortScratch<const W: usize> {
20    perm: Vec<u32>,
21    /// `(surrogate << 32) | row index` records, in sorted order once the radix kernel has run.
22    /// Empty unless that kernel is selected.
23    packed: Vec<u64>,
24    /// The radix kernel's double buffer for `packed`.
25    aux: Vec<u64>,
26    tmp_x: Vec<[u64; W]>,
27    tmp_z: Vec<[u64; W]>,
28    tmp_c: Vec<Complex64>,
29}
30
31impl<const W: usize> SortScratch<W> {
32    /// Total heap capacity held across this scratch's buffers.
33    /// Exposed only for `bucketed::tests::capacity_stabilizes_across_repeated_layers`.
34    #[cfg(test)]
35    pub(crate) fn total_capacity(&self) -> usize {
36        self.perm.capacity()
37            + self.packed.capacity()
38            + self.aux.capacity()
39            + self.tmp_x.capacity()
40            + self.tmp_z.capacity()
41            + self.tmp_c.capacity()
42    }
43}

45/// Sort `(x, z, c)` columns in place by the key `(x, z)` alone, using `s` as reusable scratch.
46///
47/// Equal-key summation order is not required to be bucket-count- or hash-seed-independent (ARCHITECTURE.md §Determinism), so this sort compares the key alone — cheaper, and with one fewer column to carry through the gather.
48///
49/// The sort is the **stable** `sort_by`, but not for stability — nothing depends on equal-key order any more — it is for *adaptivity*.
50/// A gather run is a concatenation of per-delta streams, each drawn from one sorted source bucket — the identity stream arrives fully sorted, and an XOR-by-constant stream is piecewise sorted (order survives wherever the mask's high bits don't flip) — and Rust's stable driftsort detects and merges those natural ascending runs while the unstable pdqsort does not.
51/// This only matters once there are ≥2 streams to merge: on a single already-ascending run, pdqsort's own presorted-input fast path is just as cheap.
52///
53/// **When those streams are ascending is a partition property, not a given.**
54/// The stream `{v ⊕ d : v ∈ bucket}` is fully ascending exactly when no two of the bucket's keys differ only inside the channel's support — which holds iff the hash separates the support's key-delta space, i.e. the coset dimension `r` is full (4 for a two-qubit channel).
55/// One rank short and each bucket holds two local variants of every off-support pattern, adjacent in key order, and half the deltas invert every such pair: the stream shatters into runs of ~2 and the merge's comparison count rises sharply.
56/// `r` is capped by the bucket bits *and* is a draw from `H`'s rows, so it moves with the term count, the hash seed and `W`; see `research/FINDINGS.md`.
57/// At full rank the merge already sits at the information-theoretic floor for a 15-way merge of sorted runs (`log2(15) + 1 ≈ 4.9` comparisons per row), so headroom here is a matter of configuration, not kernel.
58///
59/// What must still hold — and does, structurally: cosets are write-disjoint, work within one is sequential, and the sort is a deterministic function of its input, so **thread-count determinism and repeat-run determinism at fixed configuration** are unaffected.
60/// A later merge sums whatever order equal keys land in; that sum agrees with any other order to floating-point tolerance, never bit-for-bit across a different order.
61///
62/// Scratch-swap capacity circulation: `s.perm` is filled with the identity permutation `0..len` and reordered by the sort; the caller's columns are then read out through the permutation directly into `s.tmp_*` (one pass, not two), and finally each `tmp_*` is `mem::swap`ped with the caller's `Vec`.
63/// The caller ends up holding the sorted columns; `s` ends up holding the caller's pre-sort columns' storage (cleared next call) as its own scratch capacity — so capacity circulates between the live columns and the scratch instead of either side ever growing past its high-water mark.
64// `#[inline]` is a hint for non-fat-LTO builds only; codegen-inert under the shipping profile.
65#[inline]
66pub(crate) fn sort_rows_with_scratch<const W: usize>(
67    x: &mut Vec<[u64; W]>,
68    z: &mut Vec<[u64; W]>,
69    c: &mut Vec<Complex64>,
70    s: &mut SortScratch<W>,
71) {
72    let len = x.len();
73    debug_assert_eq!(len, z.len());
74    debug_assert_eq!(len, c.len());
75    debug_assert!(len <= u32::MAX as usize);
76    if len < 2 {
77        return;
78    }
79    s.perm.clear();
80    s.perm.extend(0..len as u32);
81    s.perm.sort_by(|&a, &b| {
82        x[a as usize]
83            .cmp(&x[b as usize])
84            .then_with(|| z[a as usize].cmp(&z[b as usize]))
85    });
86    s.tmp_x.clear();
87    s.tmp_x.extend(s.perm.iter().map(|&i| x[i as usize]));
88    s.tmp_z.clear();
89    s.tmp_z.extend(s.perm.iter().map(|&i| z[i as usize]));
90    s.tmp_c.clear();
91    s.tmp_c.extend(s.perm.iter().map(|&i| c[i as usize]));
92    std::mem::swap(x, &mut s.tmp_x);
93    std::mem::swap(z, &mut s.tmp_z);
94    std::mem::swap(c, &mut s.tmp_c);
95}

97/// Minimum number of *rest* delta streams in a gather run before `bucketed.rs` switches that layer to [`sort_rows_radix_with_scratch`].
98///
99/// The two kernels win in opposite regimes and the crossover is steep, so this is deliberately conservative — only a genuinely dense two-qubit PTM (a general SU(4) realizes all 16 bucket deltas, so 15 rest streams) clears it.
100/// A single nearly-sorted stream (the sparse-PTM case) costs about one comparison per row already, so the radix kernel's fixed passes would be pure overhead there.
101/// This is one of two gate arms; see [`RADIX_MAX_REST_ROWS_PER_KEY`] for the other, which separates layers whose streams are pairwise disjoint (a key permutation, where the k-way merge's branch mispredicts often) from those that fan out (predicting well).
102/// Measured crossovers and protocol: `research/FINDINGS.md`.
103pub(crate) const RADIX_MIN_REST_STREAMS: usize = 8;
104
105/// Second arm of the radix gate: the largest `rest_rows_per_key` a layer may have and still be treated as *disjoint*-streamed.
106///
107/// `bucketed::rest_rows_per_key` estimates, from the prepared PTM alone, how many rest rows land on one output key.
108/// Exactly `1.0` means the streams are pairwise disjoint and the k-way merge's branch is a coin flip, which is the regime the radix kernel wins.
109/// Built-ins take only `1.00` (every Clifford) or `3.00`/`14.00` (denser two-qubit gates), so this sits at the midpoint of the one measured gap rather than a tuned curve.
110pub(crate) const RADIX_MAX_REST_ROWS_PER_KEY: f64 = 2.0;
111
112/// Second arm of the radix gate: the minimum rest-stream count for the *disjoint*-streams arm.
113///
114/// A disjoint k-way merge mispredicts about once per two comparisons, and comparisons per row grow as `log2(k) + 1`, so the comparison kernel's cost grows with `k` while the radix kernel's does not.
115/// The measured crossover is at `k = 3`; no built-in channel realizes exactly two rest deltas, so the arm starts there and a two-stream layer keeps the comparison kernel.
116pub(crate) const RADIX_MIN_DISJOINT_STREAMS: usize = 3;
117
118/// Surrogate width the radix kernel sorts on, in [`RADIX_DIGIT_BITS`] digits.
119///
120/// 16 bits is two passes, chosen to resolve duplicate-key *groups* rather than every key: a dense-PTM run's residual tie groups are then ordered by the fixup pass at roughly one full-key comparison per row.
121/// Wider surrogates buy fewer ties at the cost of another whole pass and measured worse across the engine's run sizes.
122const RADIX_SURROGATE_BITS: u32 = 16;
123/// Digit width per radix pass. 256 counters is 1 KiB of stack histogram.
124const RADIX_DIGIT_BITS: u32 = 8;
125const RADIX_BUCKETS: usize = 1 << RADIX_DIGIT_BITS;
126/// Below this many *discriminating* bits in the surrogate window the radix pass cannot separate the run into useful groups, so the comparison kernel runs instead.
127/// Reached by low-weight sums (a `WeightCutoff`-truncated sum whose `x` words are nearly constant) and by narrow key spaces.
128const RADIX_MIN_WINDOW_BITS: u32 = 8;
129
130/// Word `k` of the lex key `(x, z)`: `k < W` selects `x[k]`, `k >= W` selects `z[k - W]`.
131/// Word 0 is the **most** significant, matching the derived `Ord` on `[u64; W]` and hence `PauliString`'s (ARCHITECTURE.md §Data-Model).
132#[inline(always)]
133fn key_word<const W: usize>(x: &[[u64; W]], z: &[[u64; W]], k: usize, i: usize) -> u64 {
134    if k < W {
135        x[i][k]
136    } else {
137        z[i][k - W]
138    }
139}
140
141/// The most significant key word the rows disagree on, its index, and the disagreeing bits within it.
142/// `None` ⟺ every row carries the same key.
143///
144/// One `OR`/`AND` reduction per word, stopping at the first word that disagrees — for the dense-PTM runs this kernel serves that is word 0, so the scan touches only the `x` column.
145fn discriminating_window<const W: usize>(
146    x: &[[u64; W]],
147    z: &[[u64; W]],
148) -> Option<(usize, u32, u64)> {
149    let n = x.len();
150    for k in 0..2 * W {
151        let mut any = 0u64;
152        let mut all = !0u64;
153        for i in 0..n {
154            let v = key_word(x, z, k, i);
155            any |= v;
156            all &= v;
157        }
158        let diff = any & !all;
159        if diff != 0 {
160            // Rows agree on every word before `k` and, within word `k`, on
161            // every bit above the highest set bit of `diff`.
162            return Some((k, 63 - diff.leading_zeros(), diff));
163        }
164    }
165    None
166}

168/// Sort `(x, z, c)` columns in place by the key `(x, z)` alone — radix variant, for gather runs assembled from many delta streams.
169///
170/// Same contract as [`sort_rows_with_scratch`] and freely interchangeable with it (equal-key order differs; see the module doc).
171/// Chosen per layer by `bucketed.rs` when the plan has at least [`RADIX_MIN_REST_STREAMS`] rest streams, and it falls back to the comparison kernel itself whenever its surrogate cannot discriminate.
172///
173/// # Why a surrogate, and why it is order-faithful
174///
175/// The full key is `16·W` bytes — 32 at `W = 2` — so an LSD radix over all of it would be 32 passes.
176/// Instead one pass finds [`discriminating_window`]: the most significant key word `k` the rows actually disagree on, and the highest disagreeing bit `hb` inside it.
177/// Every row then shares the same value on words before `k` and on bits above `hb`, so writing `word_k = H·2^(hb+1) + L` with `H` **constant across the run**,
178///
179/// ```text
180/// surrogate = (word_k >> shift) & (2^NBITS − 1) = L >> shift,
181///     shift = (hb + 1) − NBITS
182/// ```
183///
184/// — the mask erases exactly the constant `H`, and `L >> shift` is monotone in `L`, which is monotone in the key.
185/// So `key₁ < key₂ ⟹ surrogate₁ ≤ surrogate₂`: sorting by the surrogate never puts two rows in the wrong order, it only leaves ties, which the fixup pass resolves on the full key.
186/// A run whose keys are *all equal* needs no work at all and returns early.
187///
188/// # Why this beats the comparison sort where it is selected
189///
190/// Not by doing less work — a dense-PTM run arrives as ~15 ascending blocks and driftsort already merges them near the information-theoretic floor (`log₂ 15 + 1 ≈ 4.9` comparisons per row).
191/// It wins on the cost of that work: each comparison is a dependent indexed load into a 100–400 KiB key column, whereas a radix pass streams 8-byte records sequentially at a fraction of the cost.
192/// Two passes plus the fixup replace those comparisons with about one.
193pub(crate) fn sort_rows_radix_with_scratch<const W: usize>(
194    x: &mut Vec<[u64; W]>,
195    z: &mut Vec<[u64; W]>,
196    c: &mut Vec<Complex64>,
197    s: &mut SortScratch<W>,
198) {
199    let len = x.len();
200    debug_assert_eq!(len, z.len());
201    debug_assert_eq!(len, c.len());
202    if len < 2 {
203        return;
204    }
205    // The record packs the row index into the low 32 bits.
206    if len > u32::MAX as usize {
207        sort_rows_with_scratch(x, z, c, s);
208        return;
209    }
210    let Some((k, hb, diff)) = discriminating_window(x, z) else {
211        // Every row carries the same key, so any order is already ascending.
212        return;
213    };
214    let shift = (hb + 1).saturating_sub(RADIX_SURROGATE_BITS);
215    let mask = (1u64 << RADIX_SURROGATE_BITS) - 1;
216    if ((diff >> shift) & mask).count_ones() < RADIX_MIN_WINDOW_BITS {
217        sort_rows_with_scratch(x, z, c, s);
218        return;
219    }
220
221    let SortScratch {
222        packed,
223        aux,
224        tmp_x,
225        tmp_z,
226        tmp_c,
227        ..
228    } = s;
229    packed.clear();
230    packed.extend((0..len).map(|i| (((key_word(x, z, k, i) >> shift) & mask) << 32) | i as u64));
231    // Exactly `len`, so the pass-to-pass `swap` keeps both buffers that long.
232    // `resize` on the retained capacity writes only when the run grew.
233    aux.resize(len, 0);
234
235    let mut digit = 0u32;
236    while digit * RADIX_DIGIT_BITS < RADIX_SURROGATE_BITS {
237        let sh = 32 + digit * RADIX_DIGIT_BITS;
238        let mut count = [0u32; RADIX_BUCKETS + 1];
239        for &v in packed.iter() {
240            count[(((v >> sh) as usize) & (RADIX_BUCKETS - 1)) + 1] += 1;
241        }
242        // A constant digit contributes no ordering: skip its scatter.
243        // Common on the high digit, whose bits the window shift often leaves fixed.
244        if count[1..].iter().filter(|&&n| n != 0).count() > 1 {
245            for t in 1..=RADIX_BUCKETS {
246                count[t] += count[t - 1];
247            }
248            for &v in packed.iter() {
249                let b = ((v >> sh) as usize) & (RADIX_BUCKETS - 1);
250                aux[count[b] as usize] = v;
251                count[b] += 1;
252            }
253            std::mem::swap(packed, aux);
254        }
255        digit += 1;
256    }
257
258    // Fixup: rows sharing a surrogate are ordered on the full key.
259    // The radix is stable and the index sits in the low bits, so a group arrives in gather order; on the dense-PTM runs this kernel serves, a group is one duplicate key's ~15 rows and this costs ~1 comparison per row.
260    let mut i = 0usize;
261    while i < len {
262        let surrogate = packed[i] >> 32;
263        let mut j = i + 1;
264        while j < len && packed[j] >> 32 == surrogate {
265            j += 1;
266        }
267        if j - i > 1 {
268            packed[i..j].sort_by(|&a, &b| {
269                let (ia, ib) = (a as u32 as usize, b as u32 as usize);
270                x[ia].cmp(&x[ib]).then_with(|| z[ia].cmp(&z[ib]))
271            });
272        }
273        i = j;
274    }
275
276    // Read the columns out through the record order into the staging triple, then swap — the same capacity circulation `sort_rows_with_scratch` does.
277    tmp_x.clear();
278    tmp_x.extend(packed.iter().map(|&v| x[v as u32 as usize]));
279    tmp_z.clear();
280    tmp_z.extend(packed.iter().map(|&v| z[v as u32 as usize]));
281    tmp_c.clear();
282    tmp_c.extend(packed.iter().map(|&v| c[v as u32 as usize]));
283    std::mem::swap(x, tmp_x);
284    std::mem::swap(z, tmp_z);
285    std::mem::swap(c, tmp_c);
286}

288/// Fused two-stream merge + segmented reduction.
289///
290/// `a` is a gather run's identity-delta stream: its keys are untouched source keys, so it inherits the bucket invariant — strictly ascending, no duplicates — and is **never sorted**.
291/// (Under a dense identity plan the key slices are the *source bucket's own columns*, borrowed in place, with only the coefficients gathered; this function cannot tell and need not care.)
292/// `b` is the run's remaining rows, canonicalized by [`sort_rows_with_scratch`] (ascending, duplicates allowed).
293/// The two-pointer walk consumes rows in global key order, seeding a key tie from the `a` row first and then adding the equal-key `b` rows in their sorted order; that order is deterministic for a fixed input but not specified across partitions (ARCHITECTURE.md §Determinism).
294/// Zero-drop and `keep_term` see the fully summed coefficient.
295/// When `a` is empty this degenerates to the plain single-stream segmented reduction, which is the whole story for a channel with no identity delta: everything is gathered into `b`.
296///
297/// **Written as three loops, not one.** The main walk runs only while both streams are live, so its per-row test is the key comparison alone; the two drains then run with no test for the exhausted side at all.
298/// A single combined loop pays two extra conditional branches per output row for the exhausted-side checks, including in the all-Clifford case where `a` is always empty.
299///
300/// Exact-zero rows are consumed like any other (a `θ = π/2` rotation emits `cos·coeff = ±0.0` rows): dropping them *before* the reduction could flip the sign of a zero sum, so the only zero test is on the final accumulator.
301///
302/// Do not restructure this walk into gallop + bulk segment copies: real workloads' id/rest densities make the average id segment one or two rows, so per-segment overhead swamps the per-row compare it would save; see `research/FINDINGS.md`.
303#[allow(clippy::too_many_arguments)]
304pub(crate) fn merge2_into<const W: usize, T: TruncationPolicy<W> + ?Sized>(
305    a_x: &[[u64; W]],
306    a_z: &[[u64; W]],
307    a_c: &[Complex64],
308    b_x: &[[u64; W]],
309    b_z: &[[u64; W]],
310    b_c: &[Complex64],
311    dst_x: &mut Vec<[u64; W]>,
312    dst_z: &mut Vec<[u64; W]>,
313    dst_coeff: &mut Vec<Complex64>,
314    policy: &T,
315) {
316    let zero = Complex64::new(0.0, 0.0);
317    let (an, bn) = (a_c.len(), b_c.len());
318    debug_assert_eq!(an, a_x.len());
319    debug_assert_eq!(an, a_z.len());
320    debug_assert_eq!(bn, b_x.len());
321    debug_assert_eq!(bn, b_z.len());
322    let (mut i, mut j) = (0usize, 0usize);
323    // Main walk: both streams live, so the "which side" test is a pure key comparison with no bounds short-circuit.
324    while i < an && j < bn {
325        // Take the smaller next key; on a tie the `a` row seeds the sum.
326        // After an `a` seed there is no second `a` row for the key (`a` is unique), and after a `b` seed every equal-key `a` row would have compared `<=`, so only `b` rows can extend the segment either way.
327        let take_a = (a_x[i], a_z[i]) <= (b_x[j], b_z[j]);
328        let (key_x, key_z, mut acc) = if take_a {
329            debug_assert!(
330                i == 0 || (a_x[i - 1], a_z[i - 1]) < (a_x[i], a_z[i]),
331                "merge2_into: identity stream must be strictly ascending at {i}",
332            );
333            let t = (a_x[i], a_z[i], a_c[i]);
334            i += 1;
335            t
336        } else {
337            debug_assert!(
338                j == 0 || (b_x[j - 1], b_z[j - 1]) <= (b_x[j], b_z[j]),
339                "merge2_into: rest stream must be sorted at {j}",
340            );
341            let t = (b_x[j], b_z[j], b_c[j]);
342            j += 1;
343            t
344        };
345        while j < bn && b_x[j] == key_x && b_z[j] == key_z {
346            acc += b_c[j];
347            j += 1;
348        }
349        if acc != zero && policy.keep_term(&key_x, &key_z, acc) {
350            dst_x.push(key_x);
351            dst_z.push(key_z);
352            dst_coeff.push(acc);
353        }
354    }
355    // `b` exhausted: every remaining `a` key is unique, so each is its own segment.
356    while i < an {
357        let (key_x, key_z, acc) = (a_x[i], a_z[i], a_c[i]);
358        i += 1;
359        if acc != zero && policy.keep_term(&key_x, &key_z, acc) {
360            dst_x.push(key_x);
361            dst_z.push(key_z);
362            dst_coeff.push(acc);
363        }
364    }
365    // `a` exhausted — which for a channel with no identity delta is the whole call: the plain single-stream segmented reduction, with no `a`-side test in the loop at all.
366    while j < bn {
367        let (key_x, key_z) = (b_x[j], b_z[j]);
368        let mut acc = b_c[j];
369        j += 1;
370        while j < bn && b_x[j] == key_x && b_z[j] == key_z {
371            acc += b_c[j];
372            j += 1;
373        }
374        if acc != zero && policy.keep_term(&key_x, &key_z, acc) {
375            dst_x.push(key_x);
376            dst_z.push(key_z);
377            dst_coeff.push(acc);
378        }
379    }
380}

383mod tests {

389    const TOL: f64 = 1e-12;

391    // ---- the per-run sort kernel's contract ----
392    //
393    // Every kernel `bucketed.rs` may pick for a gather run's rest stream must satisfy exactly this, and nothing more: the output is **ascending** in lex `(x, z)` (duplicates allowed — `merge2_into` reduces them) and is a permutation of the input `(x, z, c)` triples, so a coefficient still travels with its own key.
394    // Equal-key order is explicitly *not* pinned (ARCHITECTURE.md §Determinism), which is why the check is a multiset comparison rather than an element-wise one.
395
396    type SortKernel<const W: usize> =
397        fn(&mut Vec<[u64; W]>, &mut Vec<[u64; W]>, &mut Vec<Complex64>, &mut SortScratch<W>);
398
399    /// Assert the contract above for `kernel` on one run.
400    fn assert_sort_contract<const W: usize>(
401        kernel: SortKernel<W>,
402        x: &[[u64; W]],
403        z: &[[u64; W]],
404        c: &[Complex64],
405        what: &str,
406    ) {
407        let (mut gx, mut gz, mut gc) = (x.to_vec(), z.to_vec(), c.to_vec());
408        let mut scratch = SortScratch::<W>::default();
409        kernel(&mut gx, &mut gz, &mut gc, &mut scratch);
410
411        assert_eq!(gx.len(), x.len(), "{what}: row count changed");
412        assert_eq!(gz.len(), z.len(), "{what}: row count changed");
413        assert_eq!(gc.len(), c.len(), "{what}: row count changed");
414        for i in 1..gx.len() {
415            assert!(
416                (gx[i - 1], gz[i - 1]) <= (gx[i], gz[i]),
417                "{what}: not ascending at row {i}",
418            );
419        }
420        // Multiset of triples, with the coefficient bits as the tiebreak so the comparison is exact and order-insensitive.
421        let key =
422            |(a, b, v): &([u64; W], [u64; W], Complex64)| (*a, *b, v.re.to_bits(), v.im.to_bits());
423        let mut want: Vec<([u64; W], [u64; W], Complex64)> = x
424            .iter()
425            .zip(z)

444    fn xs64(state: &mut u64) -> u64 {
445        *state ^= *state << 13;
446        *state ^= *state >> 7;
447        *state ^= *state << 17;
448        *state

451    /// The shapes a real gather run takes, plus the degenerate ones a kernel that looks at the key *bits* (rather than only comparing keys) can trip over.
452    /// `(label, x, z, c)`.
453    #[allow(clippy::type_complexity)]
454    fn sort_fixtures<const W: usize>(
455        num_qubits: usize,
456    ) -> Vec<(String, Vec<[u64; W]>, Vec<[u64; W]>, Vec<Complex64>)> {
457        let mut out = Vec::new();
458        let mask = |w: usize| crate::test_support::word_mask(num_qubits, w);
459        let mut push = |label: &str, keys: Vec<([u64; W], [u64; W])>, seed: u64| {
460            let mut st = seed | 1;
461            let c: Vec<Complex64> = keys
462                .iter()
463                .map(|_| {
464                    Complex64::new(
465                        (xs64(&mut st) % 17) as f64 - 8.0,
466                        (xs64(&mut st) % 13) as f64 - 6.0,
467                    )
468                })
469                .collect();
470            out.push((
471                format!("{label} W={W} n={}", keys.len()),
472                keys.iter().map(|k| k.0).collect(),
473                keys.iter().map(|k| k.1).collect(),
474                c,
475            ));
476        };
477
478        // Degenerate lengths.
479        for n in [0usize, 1, 2] {
480            let keys: Vec<([u64; W], [u64; W])> = (0..n as u64)
481                .map(|i| {
482                    let mut kx = [0u64; W];
483                    kx[0] = (2 - i) & mask(0);
484                    ([kx[0]; W].map(|v| v & mask(0)), [0u64; W])
485                })
486                .collect();
487            push(&format!("len{n}"), keys, 0x11);
488        }
489
490        // Dense random keys, no duplicates: the sparse-PTM shape.
491        let mut st = 0xC0FF_EE00_1234_5678u64;
492        let mut keys: Vec<([u64; W], [u64; W])> = (0..400)
493            .map(|_| {
494                let mut kx = [0u64; W];
495                let mut kz = [0u64; W];
496                for w in 0..W {
497                    kx[w] = xs64(&mut st) & mask(w);
498                    kz[w] = xs64(&mut st) & mask(w);
499                }
500                (kx, kz)
501            })
502            .collect();
503        push("dense_random", keys.clone(), 0x22);
504        keys.sort_unstable();
505        push("already_sorted", keys.clone(), 0x23);
506        keys.reverse();
507        push("reverse_sorted", keys, 0x24);
508
509        // Heavy duplicates: the dense-PTM shape.
510        // 40 distinct keys, each repeated 15 times, the repeats interleaved as 15 sorted streams.
511        let mut st = 0x5EED_0000_0000_0001u64;
512        let mut distinct: Vec<([u64; W], [u64; W])> = (0..40)
513            .map(|_| {
514                let mut kx = [0u64; W];
515                let mut kz = [0u64; W];
516                for w in 0..W {
517                    kx[w] = xs64(&mut st) & mask(w);
518                    kz[w] = xs64(&mut st) & mask(w);
519                }
520                (kx, kz)
521            })
522            .collect();
523        distinct.sort_unstable();
524        distinct.dedup();
525        let mut dup = Vec::new();
526        for _ in 0..15 {
527            dup.extend(distinct.iter().copied());
528        }
529        push("dup15_streams", dup, 0x25);
530
531        // Every row the same key.
532        push("all_equal", vec![distinct[0]; 64], 0x26);
533
534        // `x` identically zero: all discrimination lives in `z`, so a kernel that keys off the most significant word must walk past `x`.
535        let zero_x: Vec<([u64; W], [u64; W])> = distinct.iter().map(|k| ([0u64; W], k.1)).collect();
536        push("x_all_zero", zero_x, 0x27);
537
538        // Two-bit key space: only bits 0 and 1 of `x[0]` vary, so any fixed-width surrogate window has almost no discriminating power.
539        let thin: Vec<([u64; W], [u64; W])> = (0..200u64)
540            .map(|i| {
541                let mut kx = [0u64; W];
542                kx[0] = (i % 4) & mask(0);
543                (kx, [0u64; W])
544            })
545            .collect();
546        push("thin_window", thin, 0x28);
547
548        // A constant *nonzero* high part above the varying bits — the case where masking a shifted window must not reorder rows.
549        let hi = if num_qubits >= 64 {
550            1u64 << 63
551        } else {
552            1 << (num_qubits - 1)
553        };
554        let biased: Vec<([u64; W], [u64; W])> = (0..300u64)
555            .map(|i| {
556                let mut kx = [0u64; W];
557                kx[0] = (hi | (i * 7)) & mask(0);
558                (kx, [0u64; W])
559            })
560            .collect();
561        push("constant_high_bit", biased, 0x29);
562
563        out
564    }

786    /// Plain single-stream segmented reduction over sorted columns: adjacent equal keys are summed, exact-zero sums are dropped, and `keep_term` sees the summed coefficient.
787    /// Used only to build `merge2_reference`.
788    fn reduce_sorted<const W: usize, T: TruncationPolicy<W> + ?Sized>(
789        sorted_x: &[[u64; W]],
790        sorted_z: &[[u64; W]],
791        sorted_c: &[Complex64],
792        policy: &T,
793    ) -> (Vec<[u64; W]>, Vec<[u64; W]>, Vec<Complex64>) {
794        let zero = Complex64::new(0.0, 0.0);
795        let (mut ox, mut oz, mut oc) = (Vec::new(), Vec::new(), Vec::new());
796        let end = sorted_c.len();
797        let mut i = 0usize;
798        while i < end {
799            let (key_x, key_z) = (sorted_x[i], sorted_z[i]);
800            let mut acc = sorted_c[i];
801            let mut j = i + 1;
802            while j < end && sorted_x[j] == key_x && sorted_z[j] == key_z {
803                acc += sorted_c[j];
804                j += 1;
805            }
806            if acc != zero && policy.keep_term(&key_x, &key_z, acc) {
807                ox.push(key_x);
808                oz.push(key_z);
809                oc.push(acc);
810            }
811            i = j;
812        }
813        (ox, oz, oc)
814    }

816    /// Reference for `merge2_into`: concatenate both streams, sort by key, reduce.
817    /// Coefficients in these tests are small integers, so `f64` addition is exact in any order and the comparison can be `==` even where the two pipelines sum in different orders.
818    #[allow(clippy::type_complexity)]
819    fn merge2_reference<const W: usize, T: TruncationPolicy<W> + ?Sized>(
820        a: (&[[u64; W]], &[[u64; W]], &[Complex64]),
821        b: (&[[u64; W]], &[[u64; W]], &[Complex64]),
822        policy: &T,
823    ) -> (Vec<[u64; W]>, Vec<[u64; W]>, Vec<Complex64>) {
824        let mut rows: Vec<([u64; W], [u64; W], Complex64)> =
825            a.0.iter()
826                .zip(a.1)
827                .zip(a.2)
828                .map(|((&x, &z), &c)| (x, z, c))
829                .chain(b.0.iter().zip(b.1).zip(b.2).map(|((&x, &z), &c)| (x, z, c)))
830                .collect();
831        rows.sort_by_key(|&(x, z, _)| (x, z));
832        let (sx, sz, sc): (Vec<_>, Vec<_>, Vec<_>) =
833            rows.into_iter()
834                .fold((vec![], vec![], vec![]), |(mut x, mut z, mut c), r| {
835                    x.push(r.0);
836                    z.push(r.1);
837                    c.push(r.2);
838                    (x, z, c)
839                });
840        reduce_sorted(&sx, &sz, &sc, policy)
841    }

844        a: (&[[u64; W]], &[[u64; W]], &[Complex64]),
845        b: (&[[u64; W]], &[[u64; W]], &[Complex64]),
846        policy: &T,
847    ) -> (Vec<[u64; W]>, Vec<[u64; W]>, Vec<Complex64>) {

849        let mut oz = vec![];
850        let mut oc = vec![];
851        merge2_into(
852            a.0, a.1, a.2, b.0, b.1, b.2, &mut ox, &mut oz, &mut oc, policy,
853        );
854        (ox, oz, oc)
855    }
