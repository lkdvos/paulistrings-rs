1//! `Gf2Span`, the GF(2) span of a channel's bucket deltas and the coset index algebra over it, and `CosetOrder`, the coset-contiguous bucket order a layer runs in.
2//!
3//! The bucket hash is linear, so output bucket `β′` reads only the input buckets `β′ ⊕ h(D)` for a channel's key deltas `D`.
4//! A coset of the span of `h(D)` is therefore a closed task: its output buckets read only its own input buckets and no other coset writes them, which is why a layer needs no synchronization inside it.
5//! The span rather than `h(D)` itself, since a custom channel's delta set need not be XOR-closed.
6//! Where several deltas share a bucket delta the gather walks them all, so correctness never depends on the draw of the hash rows.
7

12/// Positions from which [`CosetOrder::rebuild`] tabulates on the Rayon pool: the serial table costs tens of ns per position, a fork-join tens of µs.
13pub(crate) const PARALLEL_REBUILD_MIN_POSITIONS: usize = 4096;

15/// Highest set bit of a nonzero word.
16fn highest_bit(v: u32) -> u32 {
17    debug_assert!(v != 0, "highest_bit: zero has no highest set bit");
18    31 - v.leading_zeros()
19}

21/// `pext`: the bits of `value` under `mask`, packed into the low bits.
22fn pext(value: u32, mask: u32) -> u32 {
23    #[cfg(target_arch = "x86_64")]
24    if bmi2::is_fast() {
25        // SAFETY: `is_fast` holds only where the CPU reports BMI2.
26        return unsafe { bmi2::pext(value, mask) };
27    }
28    pext_software(value, mask)
29}
30
31/// `pdep`, the inverse of [`pext`] on `mask`: the low bits of `value` scattered into the set bits of `mask`.
32fn pdep(value: u32, mask: u32) -> u32 {
33    #[cfg(target_arch = "x86_64")]
34    if bmi2::is_fast() {
35        // SAFETY: `is_fast` holds only where the CPU reports BMI2.
36        return unsafe { bmi2::pdep(value, mask) };
37    }
38    pdep_software(value, mask)
39}
40
41fn pext_software(value: u32, mask: u32) -> u32 {
42    let mut out = 0u32;
43    let mut out_bit = 0u32;
44    let mut remaining = mask;
45    while remaining != 0 {
46        let bit = remaining & remaining.wrapping_neg();
47        if value & bit != 0 {
48            out |= 1 << out_bit;
49        }
50        out_bit += 1;
51        remaining ^= bit;
52    }
53    out
54}
55
56fn pdep_software(value: u32, mask: u32) -> u32 {
57    let mut out = 0u32;
58    let mut in_bit = 0u32;
59    let mut remaining = mask;
60    while remaining != 0 {
61        let bit = remaining & remaining.wrapping_neg();
62        if value & (1 << in_bit) != 0 {
63            out |= bit;
64        }
65        in_bit += 1;
66        remaining ^= bit;
67    }
68    out
69}

71/// Hardware `pext`/`pdep`, chosen at run time since the build does not assume BMI2.
72#[cfg(target_arch = "x86_64")]
73mod bmi2 {
74    use std::arch::x86_64::{__cpuid, _pdep_u32, _pext_u32};
75    use std::sync::OnceLock;
76
77    /// Whether the CPU has BMI2 and implements it in hardware.
78    pub(super) fn is_fast() -> bool {
79        static FAST: OnceLock<bool> = OnceLock::new();
80        *FAST.get_or_init(|| std::is_x86_feature_detected!("bmi2") && !is_microcoded())
81    }
82
83    /// Zen 1 and Zen 2 (AMD and Hygon families below 0x19) microcode `pext`/`pdep` at a cost growing with the mask's popcount, slower than the software loop.
84    fn is_microcoded() -> bool {
85        let leaf0 = __cpuid(0);
86        let mut vendor = [0u8; 12];
87        vendor[..4].copy_from_slice(&leaf0.ebx.to_le_bytes());
88        vendor[4..8].copy_from_slice(&leaf0.edx.to_le_bytes());
89        vendor[8..].copy_from_slice(&leaf0.ecx.to_le_bytes());
90        if &vendor != b"AuthenticAMD" && &vendor != b"HygonGenuine" {
91            return false;
92        }
93        let eax = __cpuid(1).eax;
94        let base = (eax >> 8) & 0xF;
95        let family = if base == 0xF { base + ((eax >> 20) & 0xFF) } else { base };
96        family < 0x19
97    }
98
99    #[target_feature(enable = "bmi2")]
100    pub(super) fn pext(value: u32, mask: u32) -> u32 {
101        _pext_u32(value, mask)
102    }
103
104    #[target_feature(enable = "bmi2")]
105    pub(super) fn pdep(value: u32, mask: u32) -> u32 {
106        _pdep_u32(value, mask)
107    }
108}

110/// The span of a channel's bucket deltas, which need not be XOR-closed, and the map between a bucket index and its `(coset, member)` coordinates.
111// `basis` is reduced row echelon, ascending by pivot (its highest set bit), and each pivot bit is set in exactly one vector; bit `j` of a member index selects `basis[j]`.
112#[derive(Clone, Debug)]
113pub(crate) struct Gf2Span {
114    /// Reduced echelon basis, ascending by pivot bit.
115    basis: Vec<u32>,
116    /// OR of the pivot bits — one bit per basis vector.
117    pivot_mask: u32,
118    /// Bucket-index width: indices live in `0..2^bits`.
119    bits: u8,
120    /// The bucket-space bits outside `pivot_mask`: the free bits a representative ranges over.
121    nonpivot_mask: u32,
122}
123
124impl Gf2Span {
125    /// The span of `deltas` in a `bits`-wide index space; panics if `bits > B_MAX_BITS` or a delta lies outside it.
126    pub(crate) fn new(deltas: &[u32], bits: u8) -> Self {
127        assert!(bits <= B_MAX_BITS, "Gf2Span: bits {bits} exceeds B_MAX_BITS {B_MAX_BITS}");
128        let space_mask = if bits == 0 { 0 } else { (1u32 << bits) - 1 };
129
130        let mut basis: Vec<u32> = Vec::new();
131        let mut pivot_mask = 0u32;
132
133        for &delta in deltas {
134            assert!(
135                delta & !space_mask == 0,
136                "Gf2Span: delta {delta} has bits outside the {bits}-bit bucket space"
137            );
138            // Pivot bits are disjoint across basis vectors, so one pass in any order clears them all.
139            let mut reduced = delta;
140            for &vector in &basis {
141                if reduced & (1 << highest_bit(vector)) != 0 {
142                    reduced ^= vector;
143                }
144            }
145            if reduced == 0 {
146                continue;
147            }
148            let pivot = highest_bit(reduced);
149            // Back-substitute so the new pivot is unique to one vector; only vectors with a higher pivot can carry bit `p`.
150            for vector in basis.iter_mut() {
151                if *vector & (1 << pivot) != 0 {
152                    *vector ^= reduced;
153                }
154            }
155            let position = basis.partition_point(|&vector| highest_bit(vector) < pivot);
156            basis.insert(position, reduced);
157            pivot_mask |= 1 << pivot;
158        }
159
160        Self { basis, pivot_mask, bits, nonpivot_mask: space_mask & !pivot_mask }
161    }
162
163    /// Dimension `r` of the span.
164    pub(crate) fn dimension(&self) -> usize {
165        self.basis.len()
166    }
167
168    /// Buckets per coset, `2^r`.
169    pub(crate) fn coset_size(&self) -> usize {
170        1usize << self.basis.len()
171    }
172
173    /// Number of cosets, `2^bits / 2^r`.
174    pub(crate) fn num_cosets(&self) -> usize {
175        (1usize << self.bits) >> self.basis.len()
176    }
177
178    /// Whether `beta` is its coset's representative: the unique member with every pivot bit clear, which is also the coset's minimum.
179    fn is_representative(&self, beta: u32) -> bool {
180        beta & self.pivot_mask == 0
181    }
182
183    /// The representative of `beta`'s coset.
184    // Not `beta & !pivot_mask`: basis vectors carry non-pivot bits too, so masking can leave the coset.
185    pub(crate) fn representative_of(&self, beta: u32) -> u32 {
186        debug_assert!(
187            beta >> self.bits == 0,
188            "Gf2Span::representative_of: beta outside the bucket space"
189        );
190        let mut reduced = beta;
191        for &vector in &self.basis {
192            if reduced & (1 << highest_bit(vector)) != 0 {
193                reduced ^= vector;
194            }
195        }
196        reduced
197    }
198
199    /// The member index of `delta` in the span, read off its pivot bits since the basis is reduced.
200    pub(crate) fn coordinate_of(&self, delta: u32) -> u32 {
201        debug_assert!(
202            self.representative_of(delta) == 0,
203            "Gf2Span::coordinate_of: delta {delta} is not in the span"
204        );
205        pext(delta, self.pivot_mask)
206    }
207
208    /// The position of `representative` among all representatives in ascending order.
209    fn rank_of_representative(&self, representative: u32) -> u32 {
210        debug_assert!(
211            self.is_representative(representative),
212            "Gf2Span::rank_of_representative: {representative} is not a representative"
213        );
214        pext(representative, self.nonpivot_mask)
215    }
216
217    /// The representative of coset `coset`, the `coset`-th smallest, so its members are positions `coset << r ..` of [`Self::permuted_index`].
218    pub(crate) fn representative_at(&self, coset: u32) -> u32 {
219        debug_assert!(
220            (coset as usize) < self.num_cosets(),
221            "Gf2Span::representative_at: coset {coset} out of range"
222        );
223        pdep(coset, self.nonpivot_mask)
224    }
225
226    /// Member `i` of the coset with representative `rep`; bit `j` of `i` selects `basis[j]`.
227    pub(crate) fn member(&self, rep: u32, i: u32) -> u32 {
228        debug_assert!(
229            (i as usize) < self.coset_size(),
230            "Gf2Span::member: index {i} beyond coset size"
231        );
232        let mut v = rep;
233        for (j, &b) in self.basis.iter().enumerate() {
234            if (i >> j) & 1 == 1 {
235                v ^= b;
236            }
237        }
238        v
239    }
240
241    /// `beta` renumbered so coset `c` owns `c << r .. (c + 1) << r`, with the member coordinate in the low `r` bits.
242    pub(crate) fn permuted_index(&self, beta: u32) -> u32 {
243        let representative = self.representative_of(beta);
244        (self.rank_of_representative(representative) << self.basis.len())
245            | self.coordinate_of(beta ^ representative)
246    }
247}

249/// A layer's coset-contiguous bucket order: coset `c` owns positions `c << r .. (c + 1) << r` ([`Gf2Span::permuted_index`]).
250///
251/// Built once per layer and shared by the coset loop, the export pass and the exchange's chunking; empty, i.e. the identity, when `r = 0`.
252#[derive(Clone, Debug, Default)]
253pub(crate) struct CosetOrder {
254    /// `permutation[β]` is bucket `β`'s position.
255    permutation: Vec<u32>,
256    /// `inverse[p]` is the bucket at position `p`; built only on request.
257    inverse: Vec<u32>,
258    positions: u32,
259    dimension: u32,
260}
261
262/// Refill `table` with `f(0..len)`, on the Rayon pool when `parallel`.
263fn tabulate(table: &mut Vec<u32>, len: u32, parallel: bool, f: impl Fn(u32) -> u32 + Sync + Send) {
264    if parallel {
265        (0..len)
266            .into_par_iter()
267            .with_min_len(PARALLEL_REBUILD_MIN_POSITIONS / 8)
268            .map(f)
269            .collect_into_vec(table);
270    } else {
271        table.clear();
272        table.extend((0..len).map(f));
273    }
274}
275
276impl CosetOrder {
277    /// Re-aim the order at `span`, keeping the allocations; the inverse is built only when `with_inverse`.
278    pub(crate) fn rebuild(&mut self, span: &Gf2Span, with_inverse: bool) {
279        self.positions = 1u32 << span.bits;
280        self.dimension = span.dimension() as u32;
281        self.permutation.clear();
282        self.inverse.clear();
283        if self.dimension == 0 {
284            return;
285        }
286        let parallel = self.positions as usize >= PARALLEL_REBUILD_MIN_POSITIONS;
287        tabulate(&mut self.permutation, self.positions, parallel, |beta| span.permuted_index(beta));
288        if with_inverse {
289            // Read off the span rather than scattered from `permutation`, so the table fills in parallel.
290            let r = self.dimension;
291            let member_mask = (1u32 << r) - 1;
292            tabulate(&mut self.inverse, self.positions, parallel, |position| {
293                span.member(span.representative_at(position >> r), position & member_mask)
294            });
295        }
296    }
297
298    /// Bucket `beta`'s position.
299    pub(crate) fn position_of(&self, beta: u32) -> u32 {
300        if self.permutation.is_empty() {
301            beta
302        } else {
303            self.permutation[beta as usize]
304        }
305    }
306
307    /// The bucket at position `position`, the inverse of [`Self::position_of`]; the order must have been built `with_inverse`.
308    pub(crate) fn bucket_at(&self, position: u32) -> u32 {
309        debug_assert!(
310            self.inverse.len() == self.permutation.len(),
311            "CosetOrder::bucket_at: the inverse was not built",
312        );
313        if self.inverse.is_empty() {
314            position
315        } else {
316            self.inverse[position as usize]
317        }
318    }
319
320    /// Positions, i.e. buckets.
321    pub(crate) fn positions(&self) -> usize {
322        self.positions as usize
323    }
324
325    /// `r`, the `log2` of the coset size.
326    pub(crate) fn dimension(&self) -> usize {
327        self.dimension as usize
328    }
329
330    /// Capacity of the permutation, for the footprint tests.
331    #[cfg(test)]
332    pub(crate) fn capacity(&self) -> usize {
333        self.permutation.capacity() + self.inverse.capacity()
334    }
335}
