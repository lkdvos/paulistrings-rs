1//! Transport traits and the exchange wire format.
2//!
3//! A partition talks to the rest of the group only through [`Collectives`] (rank/size, the two reductions, the barrier) and [`Transport`] (the per-layer all-to-all [`Transport::exchange_layer`]); the sum is split across partitions by designated partition rows of the GF(2) hash (ARCHITECTURE.md §Bucketing gives the hash, §Partitioning the split), and the wire unit is one [`ExchangeBlock`] per remote delta, laid out in the receiver's destination-coset order (ARCHITECTURE.md §Engine) so a coset's rows are contiguous and can be sent in chunks while the rest is still in flight.
4//!
5//! **Every partition issues the identical sequence of transport calls, in the same order, on every layer** — a call's `n`-th message pairs with the partner's `n`-th positionally, so empty is sent as `None`, never silence, and every rank stamps its calls with a generation counter so a violation of this invariant panics naming both partitions instead of hanging.
6//!
7//! Payloads are pooled on both sides ([`Payload::byte_parts`]/[`Payload::recv_into`] move each column as a zero-copy byte view), so a steady-state layer neither allocates nor re-zeroes its megabytes.

19/// The order a layer's exchange blocks are laid out in, and the chunks the
20/// bulk transfer is cut into.
21///
22/// # Destination-coset order
23///
24/// The receiver's unit of work is a **coset** of `span(h(D_local))`, and `Gf2Span::perm_index` renumbers the bucket index so that a coset occupies a contiguous run of *positions* (ARCHITECTURE.md §Engine).
25/// Both sides of an exchange can compute that renumbering — the local bucket deltas are a function of the channel and the hash, not of the rank, and the bucket count is agreed by collective before the layer — so the **sender** lays its CSR block out in the receiver's position order instead of its own source-bucket order:
26///
27/// ```text
28/// segment p  holds the rows generated from source bucket  bucket_at(p) ^ bd
29/// ```
30///
31/// and the receiver filling output bucket `β′` reads `segment(position_of(β′))` with no arithmetic of its own.
32/// Reordering costs the sender nothing: it is a permutation of the count and offset arrays, applied before the fill pass walks them.
33///
34/// # Chunks
35///
36/// Because a coset is contiguous in position space, a contiguous *range* of positions is a whole number of cosets, so the block splits into `chunks` pieces that the receiver can consume one at a time, in order, as they arrive.
37/// Chunk `k` covers positions `bound(k)..bound(k + 1)`, always on a coset boundary.
38///
39/// A chunk count above the coset count would produce empty chunks, so it is clamped; `chunks == 1` is the un-pipelined layout.
40#[derive(Clone, Debug, Default, PartialEq, Eq)]
41pub struct ChunkMap {
42    /// `perm[β]` is bucket `β`'s destination position.
43    /// Empty when the permutation is the identity (`r == 0`), which is what a layer whose local deltas are all zero — a rotation with a remote generator, the common case — produces.
44    perm: Vec<u32>,
45    /// `inv[p]` is the bucket at position `p`. Empty with [`Self::perm`].
46    inv: Vec<u32>,
47    /// Positions, i.e. buckets.
48    positions: u32,
49    /// `log2` of the coset size: coset `c` owns positions `c << r .. (c + 1) << r`.
50    r: u32,
51    /// Cosets, `positions >> r`.
52    cosets: u32,
53    /// Chunks the bulk transfer is cut into, `1..=cosets`.
54    chunks: u32,
55}
56
57impl ChunkMap {
58    /// Re-aim the map at `span` over `num_buckets` buckets, cut into at most `chunks` pieces, **keeping the permutation buffers' allocations**.
59    ///
60    /// # Panics
61    ///
62    /// If `num_buckets` is zero or not a power of two, or if `chunks` is zero.
63    pub(crate) fn rebuild(&mut self, span: &Gf2Span, num_buckets: usize, chunks: usize) {
64        assert!(
65            num_buckets.is_power_of_two(),
66            "ChunkMap: {num_buckets} buckets is not a power of two",
67        );
68        assert!(chunks > 0, "ChunkMap: a layer needs at least one chunk");
69        let positions = num_buckets as u32;
70        let r = span.r() as u32;
71        debug_assert!(
72            (1u32 << r) <= positions,
73            "ChunkMap: a coset of {} buckets does not fit in {positions}",
74            1u32 << r,
75        );
76        self.positions = positions;
77        self.r = r;
78        self.cosets = positions >> r;
79        self.chunks = (chunks as u32).min(self.cosets).max(1);
80        self.perm.clear();
81        self.inv.clear();
82        if r == 0 {
83            // `perm_index` compresses over every bit, so it is the identity; the empty vectors say so and save both the build and the indirection.
84            return;
85        }
86        self.perm.reserve(num_buckets);
87        self.inv.resize(num_buckets, 0);
88        for beta in 0..positions {
89            let p = span.perm_index(beta);
90            self.perm.push(p);
91            self.inv[p as usize] = beta;
92        }
93    }
94
95    /// Bucket `beta`'s destination position.
96    #[inline]
97    pub fn position_of(&self, beta: u32) -> u32 {
98        if self.perm.is_empty() {
99            beta
100        } else {
101            self.perm[beta as usize]
102        }
103    }
104
105    /// The bucket at destination position `p` — the inverse of [`position_of`](Self::position_of).
106    #[inline]
107    pub fn bucket_at(&self, p: u32) -> u32 {
108        if self.inv.is_empty() {
109            p
110        } else {
111            self.inv[p as usize]
112        }
113    }
114
115    /// Positions, i.e. buckets.
116    pub fn positions(&self) -> usize {
117        self.positions as usize
118    }
119
120    /// Chunks the bulk transfer is cut into.
121    pub fn chunks(&self) -> usize {
122        self.chunks as usize
123    }
124
125    /// The first position of chunk `k`; `bound(chunks())` is the position count, so chunk `k` is `bound(k)..bound(k + 1)`.
126    ///
127    /// Always a multiple of the coset size, so no coset straddles two chunks.
128    ///
129    /// # Panics
130    ///
131    /// If `k > chunks()`.
132    pub fn bound(&self, k: usize) -> u32 {
133        assert!(k <= self.chunks(), "ChunkMap: chunk {k} is out of range");
134        let c = (k as u64 * u64::from(self.cosets)).div_ceil(u64::from(self.chunks)) as u32;
135        c << self.r
136    }
137
138    /// The chunk position `p` belongs to.
139    ///
140    /// The inverse of [`bound`](Self::bound): `bound(k) <= p < bound(k + 1)`.
141    #[inline]
142    pub fn chunk_of_position(&self, p: u32) -> usize {
143        let c = u64::from(p >> self.r);
144        ((c * u64::from(self.chunks)) / u64::from(self.cosets)) as usize
145    }
146}

148/// Fixed-size prefix describing one [`ExchangeBlock`] on the wire.
149///
150/// `#[repr(C)]` and `Pod`: four `u32`s, 16 bytes, no padding, so it casts to bytes with no copy and reads back from an unaligned buffer.
151#[repr(C)]
152#[derive(Clone, Copy, Debug, Default, PartialEq, Eq, bytemuck::Pod, bytemuck::Zeroable)]
153pub(crate) struct BlockHeader {
154    /// Number of destination positions the block is indexed by — the group's agreed bucket count. `offsets` has `num_buckets + 1` entries.
155    pub num_buckets: u32,
156    /// Total rows in the block: `offsets[num_buckets]`, and the length of each column once the export pass has filled it.
157    pub rows: u32,
158    /// The width `W` the block was built at. Checked on decode: a payload encoded at one width is never silently reinterpreted at another.
159    pub w: u32,
160    /// Which of the layer plan's remote deltas this block carries — the receiver looks it up for the delta's bucket offset `bd[e]` in the [`ExchangeBlock::segment`] rule.
161    pub entry: u32,
162}
163
164/// The rows one remote delta moves from this partition to one partner, in CSR order by the receiver's **destination position** ([`ChunkMap`]).
165///
166/// Columns are structure-of-arrays, matching the bucket storage they are gathered from and scattered into: `x`, `z` and `coeff` are parallel and their first [`rows`](Self::rows) entries are the block.
167/// `offsets` is the CSR index, `num_buckets() + 1` entries, ascending, `offsets[0] == 0` and `offsets[num_buckets] == rows`.
168///
169/// **The columns are grow-only, so they may be longer than `rows`.** A block is reused across layers ([`set_counts`](Self::set_counts)) and the storage a wider layer needed is kept rather than freed and re-faulted, so the row count in the header is the one authority on how much of a column is live: read the block through [`cols`](Self::cols) or [`segment`](Self::segment), never through `x.len()`.
170/// Whatever sits past `rows` is a previous layer's rows, and never travels.
171///
172/// The receiver never scans: for its output bucket `β′` it reads [`segment`](Self::segment)`(map.position_of(β′))` and merges those rows into that bucket — see the module docs for the ordering.
173#[derive(Clone, Debug, Default)]
174pub(crate) struct ExchangeBlock<const W: usize> {
175    /// Wire prefix: source-bucket count, row count, width, remote-delta index.
176    pub header: BlockHeader,
177    /// CSR offsets by destination position, `num_buckets + 1` entries.
178    pub offsets: Vec<u32>,
179    /// X-part column, at least [`rows`](Self::rows) entries.
180    pub x: Vec<[u64; W]>,
181    /// Z-part column, at least [`rows`](Self::rows) entries.
182    pub z: Vec<[u64; W]>,
183    /// Coefficient column, at least [`rows`](Self::rows) entries.
184    pub coeff: Vec<Complex64>,
185}
186
187/// Two blocks are equal when their **live** contents are: the header, the CSR offsets, and the first `rows` entries of each column.
188/// The grow-only tail past `rows` is a previous layer's scratch and is deliberately not compared — a block built through a reused payload must equal the same block built fresh.
189impl<const W: usize> PartialEq for ExchangeBlock<W> {
190    fn eq(&self, other: &Self) -> bool {
191        self.header == other.header && self.offsets == other.offsets && self.cols() == other.cols()
192    }
193}

198#[derive(Clone, Debug, Default, PartialEq)]
199pub(crate) struct PartnerPayload<const W: usize> {
200    /// One block per remote delta, ascending by [`BlockHeader::entry`].
201    pub blocks: Vec<ExchangeBlock<W>>,
202}

204impl<const W: usize> ExchangeBlock<W> {
205    /// Build the CSR skeleton for `counts[p]` rows at each destination position `p` under remote-delta index `entry`, and size the columns.
206    ///
207    /// [`set_counts`](Self::set_counts) on a fresh block: the columns come back `rows` long, ready for the export pass to write by index.
208    ///
209    /// The engine always re-aims a pooled block instead, so this is the tests' constructor.
210    ///
211    /// # Panics
212    ///
213    /// If the counts sum past `u32::MAX` rows.
214    #[cfg(test)]
215    pub fn with_counts(entry: u32, counts: &[u32]) -> Self {
216        let mut block = Self::default();
217        block.set_counts(entry, counts);
218        block
219    }
220
221    /// Re-aim an existing block at `counts[p]` rows per destination position `p` under remote-delta index `entry`, **keeping every allocation**.
222    ///
223    /// The export pass then writes each row by index into the segment the offsets describe.
224    /// The columns are only ever grown, never shrunk or re-zeroed (see the type docs): a steady-state layer re-aims a block it has already used at the same size, which touches nothing but the offsets.
225    /// That is the whole point of holding the payloads across layers — the alternative is faulting in and zeroing the block's megabytes again every layer.
226    ///
227    /// # Panics
228    ///
229    /// If the counts sum past `u32::MAX` rows.
230    pub fn set_counts(&mut self, entry: u32, counts: &[u32]) {
231        self.offsets.clear();
232        self.offsets.reserve(counts.len() + 1);
233        self.offsets.push(0u32);
234        let mut rows = 0u32;
235        for &c in counts {
236            rows = rows
237                .checked_add(c)
238                .expect("exchange block exceeds u32::MAX rows");
239            self.offsets.push(rows);
240        }
241        self.header = BlockHeader {
242            num_buckets: counts.len() as u32,
243            rows,
244            w: W as u32,
245            entry,
246        };
247        self.grow_columns(rows as usize);
248    }
249
250    /// Make every column at least `rows` long, keeping what is there.
251    fn grow_columns(&mut self, rows: usize) {
252        if self.coeff.len() < rows {
253            self.x.resize(rows, [0u64; W]);
254            self.z.resize(rows, [0u64; W]);
255            self.coeff.resize(rows, Complex64::new(0.0, 0.0));
256        }
257    }
258
259    /// The block's live rows: the first [`rows`](Self::rows) entries of the three columns.
260    pub fn cols(&self) -> (&[[u64; W]], &[[u64; W]], &[Complex64]) {
261        let rows = self.rows();
262        (&self.x[..rows], &self.z[..rows], &self.coeff[..rows])
263    }
264
265    /// The rows destined for position `p`, as parallel `x` / `z` / `coeff` slices.
266    ///
267    /// The receiver's rule for its own output bucket `β′` is `segment(map.position_of(β′))` (module docs).
268    /// A position with no rows yields three empty slices.
269    ///
270    /// # Panics
271    ///
272    /// If `p >= num_buckets()`, or if the block's columns are shorter than [`rows`](Self::rows) — which only a hand-built block can be.
273    pub fn segment(&self, p: u32) -> (&[[u64; W]], &[[u64; W]], &[Complex64]) {
274        let lo = self.offsets[p as usize] as usize;
275        let hi = self.offsets[p as usize + 1] as usize;
276        (&self.x[lo..hi], &self.z[lo..hi], &self.coeff[lo..hi])
277    }
278
279    /// The row index each of `map`'s chunk boundaries falls at.
280    ///
281    /// `chunks + 1` ascending entries starting at 0 and ending at [`rows`](Self::rows): chunk `k` carries rows `out[k]..out[k + 1]` of every column.
282    pub fn chunk_rows(&self, map: &ChunkMap) -> Vec<usize> {
283        chunk_rows_of(&self.offsets, map)
284    }
285
286    /// Rows the block carries: `offsets[num_buckets]`, and the length of each column once the export pass has filled it.
287    pub fn rows(&self) -> usize {
288        self.header.rows as usize
289    }
290
291    /// Destination positions the block is indexed by — the group's agreed bucket count, which both sides hold.
292    ///
293    /// The engine reads it only to check a received block against its own count, which is a `debug_assert` (the count is a collective decision, so a mismatch is a driver bug, not a data-dependent outcome).
294    #[cfg(any(test, debug_assertions))]
295    pub fn num_buckets(&self) -> u32 {
296        self.header.num_buckets
297    }
298
299    /// Wire footprint in bytes: the header, the offsets, and the live rows of the three columns ([`Payload::byte_parts`] hands out exactly these bytes).
300    pub fn bytes(&self) -> usize {
301        size_of::<BlockHeader>()
302            + self.offsets.len() * size_of::<u32>()
303            + 2 * self.rows() * W * size_of::<u64>()
304            + self.rows() * size_of::<Complex64>()
305    }
306}

308/// Something a [`Transport`] can move between partitions as bytes.
309///
310/// [`byte_parts`](Self::byte_parts) is zero-copy — borrowed views of the payload's own columns, one part per column — so a sending transport never packs or allocates.
311/// [`recv_into`](Self::recv_into) + [`finish_recv`](Self::finish_recv) is its mirror: the payload sizes its own typed columns from the part lengths the wire declared and hands out mutable byte views *of those columns*, so a transport receives straight into the storage the engine will read, with no decode pass and no second buffer.
312/// That is why a payload is `Default` — a transport pools them across layers, so the steady state allocates nothing.
313///
314/// The in-process transport moves the typed value and calls none of them.
315pub trait Payload: Default + Send + 'static {
316    /// Borrowed byte views of this payload's columns, in wire order.
317    fn byte_parts(&self) -> Vec<&[u8]>;
318
319    /// Reshape this payload for an incoming message whose parts have byte lengths `lens`, and hand out one mutable byte view per part to receive into — the same parts, in the same order, [`byte_parts`](Self::byte_parts) would produce for it.
320    ///
321    /// Reuses whatever storage the payload already holds and only ever grows it.
322    /// The views are of the payload's own typed columns, so filling them is the decode.
323    ///
324    /// # Panics
325    ///
326    /// If `lens` is not a shape this payload can take (a partial block, a part length that is not a whole number of elements).
327    fn recv_into(&mut self, lens: &[usize]) -> Vec<&mut [u8]>;
328
329    /// Check what [`recv_into`](Self::recv_into) could not: the invariants that only hold once the bytes have arrived.
330    ///
331    /// Called by the transport after the receives complete, before the payload is handed to the engine.
332    ///
333    /// # Panics
334    ///
335    /// If the received payload is inconsistent — a block encoded at another width, offsets that disagree with the row count.
336    fn finish_recv(&mut self);
337
338    /// The parts the engine reads **before** it reads a single row, so a two-phase transport must have them in hand before it hands the payload over: for the layer's exchange blocks that is the block headers and the CSR offsets, a few tens of kilobytes against a layer's hundreds of megabytes, and all the engine needs to size a gather run (`ExtraRows::count`).
339    ///
340    /// A prefix of [`byte_parts`](Self::byte_parts) in the same order — the framing still declares every part's length, so the receiver can size the bulk columns from the header alone.
341    /// The default is *every* part, which is what a payload with no interesting internal structure wants: a transport then behaves exactly as a blocking one.
342    fn early_parts(&self) -> Vec<&[u8]> {
343        self.byte_parts()
344    }
345
346    /// The rest of [`byte_parts`](Self::byte_parts), each split into `map`'s chunks: `bulk_parts()[i][k]` is part `i`'s slice for chunk `k`.
347    ///
348    /// A chunk is a contiguous range of the receiver's destination positions ([`ChunkMap`]), so a column's chunk is the byte range of the rows in those positions, which the CSR offsets give exactly.
349    /// Both sides compute it from the same offsets, so the sender's pieces and the receiver's posted receives line up with no further exchange.
350    ///
351    /// The default is no bulk parts at all, the counterpart of [`early_parts`](Self::early_parts)'s default.
352    fn bulk_parts(&self, map: &ChunkMap) -> Vec<Vec<&[u8]>> {
353        let _ = map;
354        Vec::new()
355    }
356
357    /// [`recv_into`](Self::recv_into), returning only the [`early_parts`](Self::early_parts) views.
358    ///
359    /// It still sizes *every* column from `lens` — that is what lets [`bulk_recv_into`](Self::bulk_recv_into) slice them afterwards — but the bulk views are not handed out yet, so the payload can be inspected ([`finish_recv`](Self::finish_recv)) between the two phases with no outstanding borrow.
360    fn early_recv_into(&mut self, lens: &[usize]) -> Vec<&mut [u8]> {
361        self.recv_into(lens)
362    }
363
364    /// The receive-side mirror of [`bulk_parts`](Self::bulk_parts): the same parts, in the same order, split into the same chunks.
365    ///
366    /// Called after the early parts have arrived, so the CSR offsets the split reads are the ones the sender used.
367    fn bulk_recv_into(&mut self, map: &ChunkMap) -> Vec<Vec<&mut [u8]>> {
368        let _ = map;
369        Vec::new()
370    }
371}

373/// What a coset task waits on before it reads a received row.
374///
375/// The partitioned layer hands one of these to its `RecvRows`, which calls [`wait_chunk`](Self::wait_chunk) at the top of `append_into` with the chunk its output bucket belongs to.
376/// A blocking transport's implementation is empty: everything arrived before the body ever ran.
377///
378/// `Sync` because the coset loop calls it from every Rayon worker at once.
379/// An implementation that talks to MPI must therefore serialize itself — one thread in the library at a time is exactly `MPI_THREAD_SERIALIZED`.
380pub trait ChunkWait: Sync {
381    /// Block until every row of chunk `k` has arrived.
382    ///
383    /// Must be safe to call concurrently, repeatedly, and out of order — a coset task knows only its own chunk, and Rayon decides who runs when.
384    fn wait_chunk(&self, k: usize);
385}

394/// Parts per block in the [`PartnerPayload`] encoding: header, offsets, x, z,
395/// coeff.

398/// The row index each of `map`'s chunk boundaries falls at, `chunks + 1` ascending entries — the CSR offsets read at the chunks' destination positions.
399fn chunk_rows_of(offsets: &[u32], map: &ChunkMap) -> Vec<usize> {
400    debug_assert_eq!(
401        offsets.len(),
402        map.positions() + 1,
403        "the chunk map is built for a different bucket count than the block",
404    );
405    (0..=map.chunks())
406        .map(|k| offsets[map.bound(k) as usize] as usize)
407        .collect()
408}

410/// Cut `col` at `bounds` (row indices) and view each piece as bytes.
411fn chunk_slices<'s, T, F>(col: &'s [T], bounds: &[usize], as_bytes: F) -> Vec<&'s [u8]>
412where
413    F: Fn(&'s [T]) -> &'s [u8],
414{
415    bounds
416        .windows(2)
417        .map(|w| as_bytes(&col[w[0]..w[1]]))
418        .collect()
419}
420
421/// [`chunk_slices`] for the receive side: disjoint mutable pieces, in order.
422fn chunk_slices_mut<'s, T, F>(col: &'s mut [T], bounds: &[usize], as_bytes: F) -> Vec<&'s mut [u8]>
423where
424    F: Fn(&'s mut [T]) -> &'s mut [u8],
425{
426    let mut rest = col;
427    let mut out = Vec::with_capacity(bounds.len().saturating_sub(1));
428    let mut at = bounds[0];
429    for &edge in &bounds[1..] {
430        let (head, tail) = rest.split_at_mut(edge - at);
431        out.push(as_bytes(head));
432        rest = tail;
433        at = edge;
434    }
435    out
436}

447impl<const W: usize> Payload for PartnerPayload<W> {
448    fn byte_parts(&self) -> Vec<&[u8]> {
449        let mut parts = Vec::with_capacity(PARTS_PER_BLOCK * self.blocks.len());
450        for block in &self.blocks {
451            let (x, z, coeff) = block.cols();
452            parts.push(bytemuck::bytes_of(&block.header));
453            parts.push(bytemuck::cast_slice(&block.offsets));
454            parts.push(bytemuck::cast_slice(x.as_flattened()));
455            parts.push(bytemuck::cast_slice(z.as_flattened()));
456            parts.push(bytemuck::cast_slice(coeff));
457        }
458        parts
459    }
460
461    /// Size the blocks from the declared part lengths — five parts per block, and each block's shape is readable from the lengths alone: `offsets` gives the bucket count, the `x` column the row count — then hand out the columns as bytes.
462    ///
463    /// The header travels into [`ExchangeBlock::header`] itself, so nothing about a block is known twice; [`finish_recv`](Payload::finish_recv) checks it against the shape sized here once it has arrived.
464    fn recv_into(&mut self, lens: &[usize]) -> Vec<&mut [u8]> {
465        assert_eq!(
466            lens.len() % PARTS_PER_BLOCK,
467            0,
468            "partner payload: {} parts is not a whole number of {PARTS_PER_BLOCK}-part blocks",
469            lens.len(),
470        );
471        let n = lens.len() / PARTS_PER_BLOCK;
472        let key_stride = W * size_of::<u64>();
473        // Grow-only, like the columns: a pooled payload keeps the blocks it held last layer and re-aims them.
474        // Truncate first, grow second: the views handed out below borrow the blocks for as long as the caller holds them, so `self.blocks` cannot be touched again after that.
475        self.blocks.truncate(n);
476        if self.blocks.len() < n {
477            self.blocks.resize_with(n, ExchangeBlock::<W>::default);
478        }
479        let mut parts = Vec::with_capacity(lens.len());
480        for (block, lens) in self
481            .blocks
482            .iter_mut()
483            .zip(lens.chunks_exact(PARTS_PER_BLOCK))
484        {
485            assert_eq!(
486                lens[0],
487                size_of::<BlockHeader>(),
488                "partner payload: block header is {} bytes, expected {}",
489                lens[0],
490                size_of::<BlockHeader>(),
491            );
492            check_stride(lens[1], size_of::<u32>(), "offsets");
493            check_stride(lens[2], key_stride, "x column");
494            check_stride(lens[3], key_stride, "z column");
495            check_stride(lens[4], size_of::<Complex64>(), "coeff column");
496            assert_eq!(
497                lens[2], lens[3],
498                "partner payload: x column is {} bytes and z column {}",
499                lens[2], lens[3],
500            );
501            let rows = lens[2] / key_stride;
502            assert_eq!(
503                lens[4] / size_of::<Complex64>(),
504                rows,
505                "partner payload: coeff column carries {} rows where the keys carry {rows}",
506                lens[4] / size_of::<Complex64>(),
507            );
508            block.offsets.clear();
509            block.offsets.resize(lens[1] / size_of::<u32>(), 0);
510            block.grow_columns(rows);
511            // The row count the wire declared, so `cols` and `segment` see exactly what arrives; the header's own copy is checked against it in `finish_recv`.
512            block.header.rows = rows as u32;
513            let ExchangeBlock {
514                header,
515                offsets,
516                x,
517                z,
518                coeff,
519            } = block;
520            parts.push(bytemuck::bytes_of_mut(header));
521            parts.push(bytemuck::cast_slice_mut(&mut offsets[..]));
522            parts.push(bytemuck::cast_slice_mut(x[..rows].as_flattened_mut()));
523            parts.push(bytemuck::cast_slice_mut(z[..rows].as_flattened_mut()));
524            parts.push(bytemuck::cast_slice_mut(&mut coeff[..rows]));
525        }
526        parts
527    }
528
529    fn finish_recv(&mut self) {
530        for block in &self.blocks {
531            assert_eq!(
532                block.header.w as usize, W,
533                "partner payload: block encoded at width W={} decoded at W={W}",
534                block.header.w,
535            );
536            // The header travelled with the columns, so it could name more rows than the columns the declared part lengths sized.
537            // It never does between ranks running the same build; the check is what keeps a garbled header an assertion rather than an out-of-bounds read.
538            assert!(
539                block.header.rows as usize <= block.coeff.len(),
540                "partner payload: a block header claims {} rows but only {} arrived",
541                block.header.rows,
542                block.coeff.len(),
543            );
544            assert_eq!(
545                block.offsets.len(),
546                block.header.num_buckets as usize + 1,
547                "partner payload: a block indexed by {} buckets arrived with {} offsets",
548                block.header.num_buckets,
549                block.offsets.len(),
550            );
551            assert_eq!(
552                block.offsets.last().copied().unwrap_or(0),
553                block.header.rows,
554                "partner payload: offsets end at {:?}, the columns carry {} rows",
555                block.offsets.last(),
556                block.header.rows,
557            );
558        }
559    }
560
561    /// Parts 0 and 1 of every block — the header and the CSR offsets.
562    /// Those are what `RecvRows::count` reads to size a gather run; the three columns after them are read only inside `append_into`.
563    fn early_parts(&self) -> Vec<&[u8]> {
564        let mut parts = Vec::with_capacity(2 * self.blocks.len());
565        for block in &self.blocks {
566            parts.push(bytemuck::bytes_of(&block.header));
567            parts.push(bytemuck::cast_slice(&block.offsets));
568        }
569        parts
570    }
571
572    /// The three columns of each block, in block order, each cut at the chunks' destination-position boundaries.
573    fn bulk_parts(&self, map: &ChunkMap) -> Vec<Vec<&[u8]>> {
574        let mut parts = Vec::with_capacity(3 * self.blocks.len());
575        for block in &self.blocks {
576            let rows = block.rows();
577            let (x, z, coeff) = (&block.x[..rows], &block.z[..rows], &block.coeff[..rows]);
578            let bounds = block.chunk_rows(map);
579            parts.push(chunk_slices(x, &bounds, |s| {
580                bytemuck::cast_slice(s.as_flattened())
581            }));
582            parts.push(chunk_slices(z, &bounds, |s| {
583                bytemuck::cast_slice(s.as_flattened())
584            }));
585            parts.push(chunk_slices(coeff, &bounds, bytemuck::cast_slice));
586        }
587        parts
588    }
589
590    fn early_recv_into(&mut self, lens: &[usize]) -> Vec<&mut [u8]> {
591        let mut parts = self.recv_into(lens);
592        // `recv_into` sized every column and handed out all five views per block; keep the header and the offsets and drop the columns, which `bulk_recv_into` hands out once their shape is known.
593        let mut early = Vec::with_capacity(2 * parts.len() / PARTS_PER_BLOCK);
594        for (i, view) in parts.drain(..).enumerate() {
595            if i % PARTS_PER_BLOCK < 2 {
596                early.push(view);
597            }
598        }
599        early
600    }
601
602    fn bulk_recv_into(&mut self, map: &ChunkMap) -> Vec<Vec<&mut [u8]>> {
603        let mut parts = Vec::with_capacity(3 * self.blocks.len());
604        for block in &mut self.blocks {
605            let rows = block.header.rows as usize;
606            let bounds = chunk_rows_of(&block.offsets, map);
607            let ExchangeBlock { x, z, coeff, .. } = block;
608            parts.push(chunk_slices_mut(&mut x[..rows], &bounds, |s| {
609                bytemuck::cast_slice_mut(s.as_flattened_mut())
610            }));
611            parts.push(chunk_slices_mut(&mut z[..rows], &bounds, |s| {
612                bytemuck::cast_slice_mut(s.as_flattened_mut())
613            }));
614            parts.push(chunk_slices_mut(
615                &mut coeff[..rows],
616                &bounds,
617                bytemuck::cast_slice_mut,
618            ));
619        }
620        parts
621    }

624/// The collective operations a partition needs outside the exchange itself: its identity in the group, the two reductions a layer's truncation and bookkeeping need, and a barrier.
625///
626/// Both reductions must return **the identical value on every partition** — callers use them to agree on a global decision (a truncation threshold, a term-count total), and a partition that computed a different answer would diverge silently.
627/// Both are exact and order-independent (a maximum, and a wrapping integer sum), so an implementation is free to combine in arrival order.
628///
629/// Every method obeys the collective-order invariant in the module docs: all partitions call them in the same order, the same number of times.
630pub trait Collectives: Send + Sync {
631    /// This partition's index in the group, `0 <= rank < size`.
632    fn rank(&self) -> u32;
633    /// Number of partitions in the group.
634    fn size(&self) -> u32;
635    /// Maximum of `v` over the group. Same value on every partition.
636    fn allreduce_max_u8(&self, v: u8) -> u8;
637    /// Element-wise sum of `buf` over the group, in place. Every partition passes the same length and gets the same values back.
638    /// Sums wrap rather than panic on overflow, so debug and release agree.
639    fn allreduce_sum_u64(&self, buf: &mut [u64]);
640    /// Block until every partition has arrived.
641    fn barrier(&self);
642
643    /// Panic unless every partition passed the same `fingerprint`.
644    ///
645    /// The intended fingerprint is whatever a run's partitions *must* agree on before they can be driven in lock-step — the channel count, the direction, the qubit count, the truncation policy's identity.
646    /// Disagree on any of those and the group deadlocks on the first layer whose collectives no longer pair up; this turns that hang into a message.
647    /// The driver calls it exactly once per propagation, before the first layer.
648    ///
649    /// One collective, and it obeys the collective-order invariant like the rest: every partition calls it, with its own fingerprint.
650    ///
651    /// # Panics
652    ///
653    /// If the fingerprints differ, naming a bit the group disagrees on and how many partitions set it.
654    fn check_consistency(&self, fingerprint: u64) {
655        // 64 counters, one per bit of the fingerprint: "how many partitions have this bit set".
656        // A group that agrees answers 0 or `size` for every bit; a group that disagrees cannot, because the counts pin every partition's value bit by bit.
657        let mut counts = [0u64; 64];
658        for (i, c) in counts.iter_mut().enumerate() {
659            *c = (fingerprint >> i) & 1;
660        }
661        self.allreduce_sum_u64(&mut counts);
662        let size = u64::from(self.size());
663        for (bit, &count) in counts.iter().enumerate() {
664            assert!(
665                count == 0 || count == size,
666                "the partitions disagree about the run: partition {} offered fingerprint \
667                 {fingerprint:#018x}, and {count} of {size} partitions set bit {bit} of theirs. \
668                 Every partition must be driven through the same circuit, in the same direction, \
669                 under the same policy and options — a group that is not stays in step only by \
670                 luck.",
671                self.rank(),
672            );
673        }
674    }
675}

677/// The per-layer all-to-all: each partition hands over what it exports and gets back what its partners exported to it.
678///
679/// Not object-safe ([`exchange_layer`](Self::exchange_layer) is generic over the payload) — deliberate, since the layer code is generic over the transport, so a call monomorphizes into the partition's driving thread with no virtual dispatch on a per-layer path.
680pub trait Transport: Collectives {
681    /// **The** exchange: send `send[q]` to partition `q`, run `body` on what the partners sent here, and give both back.
682    ///
683    /// It is **two-phase**. Everything the coset loop needs to size its gather runs — the block headers and the CSR offsets ([`Payload::early_parts`]) — has arrived before `body` starts; the rows themselves ([`Payload::bulk_parts`]) may still be in flight while it runs, and `body` blocks on the [`ChunkWait`] it is handed before it reads a chunk's rows.
684    /// A transport with nothing to overlap completes the transfer first and hands over a no-op [`ChunkWait`], which makes it a blocking exchange with extra steps — that is what [`InProcessTransport`] is, its "transfer" being a moved pointer.
685    ///
686    /// `send.len()` must be [`size`](Collectives::size) and `send[self.rank()]` must be `None`; the slots handed to `body` have the same length and `None` in the same self slot.
687    /// A partner with nothing to send still participates, with `None` — silence would desynchronize the group (module docs).
688    ///
689    /// `map` is the destination-coset order both sides laid the blocks out in and the chunks the bulk transfer is cut into.
690    /// Every partition passes the same one, for the same reason both sides agree on the bucket count.
691    ///
692    /// `spare` is the caller's **payload pool**, and it is what keeps a steady-state layer from allocating: a transport that has to materialize the received payloads takes them from here rather than building them fresh, and returns any payload from `send` it is finished with.
693    /// The caller returns the results to the pool once the layer has consumed them.
694    /// Pooled payloads are in an unspecified state — a taker reshapes one before it reads anything back — and an empty pool is always legal, so a transport must fall back to [`Default`].
695    /// [`InProcessTransport`] uses neither direction: it *moves* the sender's payload to the receiver, so the pool circulates through the partners instead.
696    ///
697    /// `body` returns whatever the caller needs out of the layer; the received payloads come back with it, for the caller to return to `spare`.
698    fn exchange_layer<P, F, R>(
699        &self,
700        send: Vec<Option<P>>,
701        spare: &mut Vec<P>,
702        map: &ChunkMap,
703        body: F,
704    ) -> (Vec<Option<P>>, R)
705    where
706        P: Payload,
707        F: FnOnce(&[Option<P>], &dyn ChunkWait) -> R;
708
709    /// [`exchange_layer`](Self::exchange_layer) with nothing to overlap: the blocking all-to-all, for a caller that wants the payloads and no more.
710    ///
711    /// The empty [`ChunkMap`] cuts no chunks, so every part travels as an early one — which is what a payload with no interesting internal structure wants, and the gather's is the only one.
712    /// A payload whose `bulk_parts` needs a real map goes through `exchange_layer`.
713    fn exchange<P: Payload>(&self, send: Vec<Option<P>>, spare: &mut Vec<P>) -> Vec<Option<P>> {
714        self.exchange_layer(send, spare, &ChunkMap::default(), |_, _| ())
715            .0
716    }
717
718    /// Collect every partition's `parts` on partition 0.
719    ///
720    /// `Some(v)` on rank 0, where `v[q]` is partition `q`'s parts in the order it passed them (`v[0]` being the caller's own); `None` everywhere else.
721    /// This is the gather that ends a distributed run: each partition hands over its share of the sum as bytes and rank 0 reassembles.
722    ///
723    /// One collective. Unlike the exchange it is deliberately *not* symmetric — every partition talks to rank 0 and to nobody else — so a transport that infers its partner set from the `Some` positions (as the MPI one does) overrides this rather than inheriting it.
724    /// The default body is the honest all-to-all: rank 0 sends nothing and receives from everyone.
725    fn gather_to_root(&self, parts: Vec<&[u8]>) -> Option<Vec<Vec<Vec<u8>>>> {
726        let n = self.size() as usize;
727        let me = self.rank() as usize;
728        let mine = ByteParts(parts.iter().map(|p| p.to_vec()).collect());
729
730        if me != ROOT {
731            let mut send: Vec<Option<ByteParts>> = (0..n).map(|_| None).collect();
732            send[ROOT] = Some(mine);
733            self.exchange(send, &mut Vec::new());
734            return None;
735        }
736
737        let mut recv = self.exchange(
738            (0..n).map(|_| None).collect::<Vec<Option<ByteParts>>>(),
739            &mut Vec::new(),
740        );
741        let mut mine = Some(mine);
742        Some(
743            (0..n)
744                .map(|q| {
745                    let got = if q == ROOT {
746                        mine.take()
747                    } else {
748                        recv[q].take()
749                    };
750                    got.unwrap_or_else(|| {
751                        panic!(
752                            "gather_to_root: partition {q} sent nothing (it must send its \
753                                parts, empty or not)"
754                        )
755                    })
756                    .0
757                })
758                .collect(),
759        )
760    }
761}

767///
768/// [`Transport::gather_to_root`]'s default body moves one of these per partition through [`Transport::exchange`], so a transport gets a gather for free from the exchange it already implements.
769#[derive(Default)]

786    }

833/// `spin_loop` hints a waiting rank issues before it starts yielding instead.
834///
835/// The mechanism is a few hundred nanoseconds between dedicated threads, so the fast path — the partner is already here, or arrives within a couple of microseconds — never leaves this tier.
836const SPINS_BEFORE_YIELD: u32 = 1_000;
837
838/// `yield_now` calls after the spin tier before the waiter starts sleeping.
839///
840/// Covers the ordinary case the spins do not: the partner is a few tens of microseconds behind because its share of the layer was bigger.
841/// A `P = 16` group on a smaller box (the test suite runs one) also makes progress here rather than livelocking.
842const YIELDS_BEFORE_SLEEP: u32 = 100;
843
844/// How long a waiter sleeps per iteration once even yielding has not helped.
845///
846/// `sched_yield` in a loop is not free to the rest of the machine: it re-enters the run queue and takes its fair share of the CPU, which on a partitioned run is a share of the CPUs the partner's *own* workers are trying to finish the layer on.
847/// Past a wait this long the partner is not close, so paying up to one step of extra latency to stay off its cores is the right trade.
848const SLEEP_STEP: Duration = Duration::from_micros(50);
849
850/// Spin iterations between two checks of the departure mask and the deadline while still in the spin tier.
851/// Past it, both are checked every iteration — a yield or a sleep dwarfs two loads.
852///
853/// Both live on lines nobody writes in steady state, but keeping them out of the tight loop leaves the fast path a single load.
854const CHECKS_EVERY: u32 = 64;

856/// How long a rank waits for a partner before declaring it dead.
857///
858/// The backstop, not the mechanism: a partner that panics drops its transport and is reported within `CHECKS_EVERY` spins ([`InProcessTransport::drop`]).
859/// This bound only catches a partner that is neither dead nor arriving — a deadlock elsewhere in the process — so it is generous.
860const WAIT_TIMEOUT: Duration = Duration::from_secs(10);

862/// One rank's contribution to one generation, double-buffered by generation parity (see [`GroupState`] for why two are enough).
863#[derive(Default)]
864struct ValueSlot {
865    /// `(generation << 16) | (kind << 8) | u8 payload`, published last-but-one with `Release`.
866    /// Generation 0 means "never written".
867    tag: AtomicU64,
868    /// Elements of `buf` that belong to this generation.
869    len: AtomicUsize,
870    /// The `allreduce_sum_u64` contribution.
871    ///
872    /// Written only by the owning rank and only while no partner can be reading it, which is what the parity split buys (see [`GroupState`]); the elements are atomics so that a *hypothetical* overlap is a stale read rather than undefined behaviour, and the `UnsafeCell` is there for the resize, which needs `&mut`.
873    /// Allocated on first use and reused at the same length ever after.
874    buf: UnsafeCell<Vec<AtomicU64>>,
875}
876
877/// SAFETY: `buf`'s exclusivity is established by the generation protocol documented on [`GroupState`], not by Rust's borrow checker.
878unsafe impl Sync for ValueSlot {}
879
880impl ValueSlot {
881    /// Store `values` as this generation's contribution.
882    ///
883    /// # Safety
884    ///
885    /// The caller must be the rank that owns this slot, and no partner may be reading it — [`GroupState`]'s parity argument.
886    unsafe fn write_buf(&self, values: &[u64]) {
887        let buf = &mut *self.buf.get();
888        if buf.len() != values.len() {
889            buf.clear();
890            buf.resize_with(values.len(), AtomicU64::default);
891        }
892        for (slot, &v) in buf.iter().zip(values) {
893            slot.store(v, Ordering::Relaxed);
894        }
895        self.len.store(values.len(), Ordering::Relaxed);
896    }
897
898    /// This generation's contribution.
899    ///
900    /// # Safety
901    ///
902    /// The caller must have observed the owner's `Release` of the generation it is reading (so the elements are visible) and must be inside the window in which the owner cannot be writing — [`GroupState`]'s parity argument.
903    unsafe fn read_buf(&self) -> &[AtomicU64] {
904        let buf: &Vec<AtomicU64> = &*self.buf.get();
905        &buf[..]
906    }
907}

909/// One rank's publication point: everything a partner reads to learn where that rank is and what it contributed.
910///
911/// Padded to 128 bytes — two x86 cache lines, the granularity the hardware prefetcher pairs — so `P` ranks polling each other never share a line.
912#[repr(align(128))]
913#[derive(Default)]
914struct RankSlot {
915    /// `(generation << 8) | kind` of the last call this rank published: **monotone**, overwritten every call, both parities.
916    ///
917    /// It is what makes a desynchronized partner a panic instead of a hang: a rank waiting for generation `g` sees a partner that ran *past* `g` here, even though the partner's `values` slot for `g`'s parity never got `g`'s tag.
918    progress: AtomicU64,
919    /// The value published at each generation parity.
920    values: [ValueSlot; 2],
921}
922
923/// The collective state one [`InProcessTransport`] group shares.
924///
925/// Each rank numbers its own calls (generation 1, 2, 3, …) and publishes `(generation, kind, payload)` into its own slot with a `Release` store to `progress`, which every partner spins on with `Acquire`; that pairing is the only synchronization, so the tag, `len` and buffer elements may be loaded `Relaxed` once `progress` is observed.
926/// `values` is double-buffered by generation parity: a rank only overwrites `values[p]` once every partner has observed its previous use, which a rank's own wait for generation `g` before starting `g + 1` guarantees.
927struct GroupState {
928    /// Ranks in the group.
929    size: u32,
930    /// Bit `q` set once rank `q`'s transport has been dropped — it will publish nothing further.
931    /// `P ≤ 64` (`P_MAX_BITS`), so a `u64` mask is ample.
932    departed: AtomicU64,
933    /// One publication point per rank, in rank order.
934    slots: Box<[RankSlot]>,
935}
936
937impl GroupState {
938    fn new(size: u32) -> Self {
939        Self {
940            size,
941            departed: AtomicU64::new(0),
942            slots: (0..size).map(|_| RankSlot::default()).collect(),
943        }
944    }
945
946    /// The slot rank `rank` publishes generation `gen` into.
947    fn value_slot(&self, rank: u32, gen: u64) -> &ValueSlot {
948        &self.slots[rank as usize].values[(gen & 1) as usize]
949    }
950
951    /// Publish generation `gen` of kind `kind` carrying `payload`, storing `tag` then `progress` (see [`GroupState`]); the caller has already written the buffer, if its call carries one.
952    fn publish(&self, rank: u32, gen: u64, kind: CallKind, payload: u8) {
953        let k = kind as u64;
954        self.value_slot(rank, gen).tag.store(
955            (gen << 16) | (k << 8) | u64::from(payload),
956            Ordering::Release,
957        );
958        self.slots[rank as usize]
959            .progress
960            .store((gen << 8) | k, Ordering::Release);
961    }
962
963    /// Wait for rank `src` to publish generation `gen` of kind `kind`, and
964    /// return its tag word (whose low byte is the `u8` payload).
965    ///
966    /// `waiter` is only used to name this rank in a mismatch panic.
967    ///
968    /// # Panics
969    ///
970    /// If `src` published a different call at `gen`, or ran past `gen` without
971    /// publishing it (the collective-order invariant); if `src`'s transport
972    /// has been dropped without publishing `gen` (it panicked); or if `src`
973    /// has neither arrived nor died within [`WAIT_TIMEOUT`].
974    fn wait(&self, waiter: u32, src: u32, gen: u64, kind: CallKind) -> u64 {
975        let slot = &self.slots[src as usize];
976        let value = &slot.values[(gen & 1) as usize];
977        let mut spins: u32 = 0;
978        let mut waiting_since: Option<Instant> = None;
979        loop {
980            let progress = slot.progress.load(Ordering::Acquire);
981            if progress >> 8 >= gen {
982                let tag = value.tag.load(Ordering::Acquire);
983                if tag >> 16 == gen && (tag >> 8) & 0xff == kind as u64 {
984                    return tag;
985                }
986                panic!(
987                    "collective order mismatch: partition {waiter} is at transport call {gen} \
988                     ({}) but partition {src} published {} at call {} — every partition must \
989                     issue the identical sequence of transport calls per layer",
990                    kind.name(),
991                    CallKind::name_of(((tag >> 8) & 0xff) as u8),
992                    tag >> 16,
993                );
994            }
995
996            spins = spins.saturating_add(1);
997            if spins >= SPINS_BEFORE_YIELD || spins.is_multiple_of(CHECKS_EVERY) {
998                if self.departed.load(Ordering::Acquire) & (1u64 << src) != 0
999                    // A rank that published this generation and then left the group is not a dead partner: re-read before condemning it, so the last collective of a call cannot race the partner's return.
1000                    && slot.progress.load(Ordering::Acquire) >> 8 < gen
1001                {
1002                    panic!(
1003                        "partition {src} terminated before completing the {} (it panicked)",
1004                        kind.name(),
1005                    );
1006                }
1007                let since = waiting_since.get_or_insert_with(Instant::now);
1008                if since.elapsed() > WAIT_TIMEOUT {
1009                    panic!(
1010                        "partition {src} terminated before completing the {}: no response in \
1011                         {} s (this partition is at transport call {gen}, that one at {})",
1012                        kind.name(),
1013                        WAIT_TIMEOUT.as_secs(),
1014                        slot.progress.load(Ordering::Acquire) >> 8,
1015                    );
1016                }
1017            }
1018            // Three tiers, cheapest first: burn a few microseconds where the partner is about to arrive, hand the core over where it is a layer's skew behind, and get off the machine entirely where it is further than that.
1019            if spins < SPINS_BEFORE_YIELD {
1020                std::hint::spin_loop();
1021            } else if spins < SPINS_BEFORE_YIELD + YIELDS_BEFORE_SLEEP {
1022                std::thread::yield_now();
1023            } else {
1024                std::thread::sleep(SLEEP_STEP);
1025            }
1026        }
1027    }
1028
1029    /// Wait for every partner of `rank` to publish generation `gen` of kind `kind`, folding their tag words in **rank order** through `fold`.
1030    fn wait_all(&self, rank: u32, gen: u64, kind: CallKind, mut fold: impl FnMut(u32, u64)) {
1031        for src in 0..self.size {
1032            if src != rank {
1033                fold(src, self.wait(rank, src, gen, kind));
1034            }
1035        }
1036    }
1037}

1074/// In-process transport: `P` partitions sharing one `GroupState` for the collectives, and wired as a `P × P` matrix of unbounded `std::sync::mpsc` channels for the exchange.
1075///
1076/// Built as a group by [`group`](Self::group) and moved one per partition thread.
1077/// Deliberately Rayon-free: it is called from the partition's driving thread between layers, never from inside a parallel region — and by *one* thread per rank, since the generation counter is that thread's call index.
1078///
1079/// The three [`Collectives`] operations spin on shared atomics (module docs and `GroupState`); the exchange keeps the channels — it moves a payload, and only on the layers that have one.
1080/// Channels are unbounded, so a send never blocks and the "send everything, then receive in rank order" shape cannot deadlock.
1081/// Its receives block; a partner that died is reported by name rather than waited on forever (its dropped sender disconnects the channel).
1082///
1083/// `size == 1` is a no-op path: no channels exist, no generation is consumed, the exchange returns one `None`, and the reductions return their input.
1084pub struct InProcessTransport {
1085    /// This partition's index.
1086    rank: u32,
1087    /// Partitions in the group.
1088    size: u32,
1089    /// The group's shared collective state, one `Arc` per rank.
1090    /// It outlives every rank's endpoint, which is what lets a partner read a slot's buffer while its owner is on its way out.
1091    state: Arc<GroupState>,
1092    /// This rank's transport-call counter: one increment per call, the generation published to [`GroupState`] and (in debug builds) the stamp on every exchange message.
1093    /// Atomic only because the methods take `&self`; nothing but this rank's driving thread touches it.
1094    gen: AtomicU64,
1095    /// Sender to partition `q`, `None` in the self slot.
1096    outbox: Vec<Option<std::sync::mpsc::Sender<Message>>>,
1097    /// Receiver of what partition `q` sends here, `None` in the self slot.
1098    /// `Mutex` only to make the transport `Sync` — `Receiver` is `Send` but not `Sync`, and nothing here contends for it.
1099    inbox: Vec<Option<std::sync::Mutex<std::sync::mpsc::Receiver<Message>>>>,
1100}
1101
1102/// Leaving the group marks this rank departed, so a partner spinning for a generation this rank will never publish fails fast instead of waiting out its `WAIT_TIMEOUT` backstop.
1103///
1104/// Unconditional rather than `if std::thread::panicking()`: a rank that returns from its partition body with fewer collectives than its partners is exactly as dead to them as one that panicked, and the panic message they raise says so.
1105/// It cannot fire spuriously at the end of a healthy call — `GroupState::wait` re-reads the partner's progress before condemning it, and a rank only leaves after publishing the group's last generation.
1106impl Drop for InProcessTransport {
1107    fn drop(&mut self) {
1108        self.state
1109            .departed
1110            .fetch_or(1u64 << self.rank, Ordering::Release);
1111    }
1112}

1114impl InProcessTransport {
1115    /// Build a group of `size` transports wired to each other, one per partition, in rank order.
1116    ///
1117    /// # Panics
1118    ///
1119    /// If `size` is zero.
1120    pub fn group(size: u32) -> Vec<InProcessTransport> {
1121        assert!(size > 0, "a transport group needs at least one partition");
1122        let n = size as usize;
1123        let mut senders: Vec<Vec<Option<std::sync::mpsc::Sender<Message>>>> =
1124            (0..n).map(|_| (0..n).map(|_| None).collect()).collect();
1125        let mut receivers: Vec<Vec<Option<std::sync::mpsc::Receiver<Message>>>> =
1126            (0..n).map(|_| (0..n).map(|_| None).collect()).collect();
1127        for src in 0..n {
1128            for dst in 0..n {
1129                if src != dst {
1130                    let (tx, rx) = std::sync::mpsc::channel();
1131                    senders[src][dst] = Some(tx);
1132                    receivers[src][dst] = Some(rx);
1133                }
1134            }
1135        }
1136
1137        let state = Arc::new(GroupState::new(size));
1138        (0..n)
1139            .map(|rank| InProcessTransport {
1140                rank: rank as u32,
1141                size,
1142                state: Arc::clone(&state),
1143                gen: AtomicU64::new(0),
1144                outbox: std::mem::take(&mut senders[rank]),
1145                inbox: (0..n)
1146                    .map(|src| receivers[src][rank].take().map(std::sync::Mutex::new))
1147                    .collect(),
1148            })
1149            .collect()
1150    }
1151
1152    /// The generation of the transport call starting now: one more than the
1153    /// last, and never zero (an unwritten slot reads as generation 0).
1154    fn next_gen(&self) -> u64 {
1155        self.gen.fetch_add(1, Ordering::Relaxed) + 1
1156    }
1157
1158    /// Pretend one extra collective was issued, to test the order check.
1159    #[cfg(test)]
1160    fn skip_sequence_for_test(&self) {
1161        self.gen.fetch_add(1, Ordering::Relaxed);
1162    }
1163
1164    /// Send one message to partition `dst`.
1165    fn send_to(&self, dst: usize, seq: u64, body: Box<dyn std::any::Any + Send>, op: &str) {
1166        let tx = self.outbox[dst]
1167            .as_ref()
1168            .expect("a partition has no channel to itself");
1169        if tx.send(Message::new(seq, body)).is_err() {
1170            panic!("partition {dst} terminated before completing the {op} (it panicked)");
1171        }
1172    }
1173
1174    /// Receive this call's message from partition `src`, checking the sequence stamp in debug builds.
1175    fn recv_from(&self, src: usize, seq: u64, op: &str) -> Box<dyn std::any::Any + Send> {
1176        let rx = self.inbox[src]
1177            .as_ref()
1178            .expect("a partition has no channel to itself")
1179            .lock()
1180            .unwrap_or_else(std::sync::PoisonError::into_inner);
1181        match rx.recv() {
1182            Ok(message) => {
1183                #[cfg(debug_assertions)]
1184                assert_eq!(
1185                    message.seq, seq,
1186                    "collective order mismatch: partition {} is at transport call {seq} but \
1187                     partition {src} sent its call {} — every partition must issue the identical \
1188                     sequence of transport calls per layer",
1189                    self.rank, message.seq,
1190                );
1191                let _ = seq;
1192                message.body
1193            }
1194            Err(_) => {
1195                panic!("partition {src} terminated before completing the {op} (it panicked)")
1196            }
1197        }
1198    }

1201impl Collectives for InProcessTransport {
1202    fn rank(&self) -> u32 {
1203        self.rank
1204    }

1207        self.size

1210    /// The maximum rides in the published word's payload byte, so the whole reduction is one store and `P − 1` loads.
1211    /// `max` is order-independent, so every partition returns the same byte however the arrivals interleave.
1212    fn allreduce_max_u8(&self, v: u8) -> u8 {
1213        if self.size == 1 {
1214            return v;
1215        }
1216        let gen = self.next_gen();
1217        self.state.publish(self.rank, gen, CallKind::MaxU8, v);
1218
1219        let mut acc = v;
1220        self.state
1221            .wait_all(self.rank, gen, CallKind::MaxU8, |_, tag| {
1222                acc = acc.max((tag & 0xff) as u8);
1223            });
1224        acc
1225    }
1226
1227    /// Each partition publishes its own contribution once and folds its
1228    /// partners' in place.
1229    ///
1230    /// The local sum runs in rank order, but it would not have to: wrapping `u64` addition is exact, associative and commutative, so every partition ends with the identical bits whatever order it folds in.
1231    fn allreduce_sum_u64(&self, buf: &mut [u64]) {
1232        if self.size == 1 {
1233            return;
1234        }
1235        let gen = self.next_gen();
1236        let slot = self.state.value_slot(self.rank, gen);
1237        // SAFETY: this rank owns the slot, and the generation parity keeps every partner out of it (see `GroupState`). The write must precede the publish below, which is what makes it visible to the partners at all.
1238        unsafe { slot.write_buf(buf) };
1239        self.state.publish(self.rank, gen, CallKind::SumU64, 0);
1240
1241        let n = self.size;
1242        for src in 0..n {
1243            if src == self.rank {
1244                continue;
1245            }
1246            self.state.wait(self.rank, src, gen, CallKind::SumU64);
1247            let theirs = self.state.value_slot(src, gen);
1248            // SAFETY: `wait` returned, so this rank has observed `src`'s `Release` of this generation — its buffer and length are visible — and by the parity argument `src` cannot write the slot again before this rank publishes its next generation.
1249            let values = unsafe { theirs.read_buf() };
1250            assert_eq!(
1251                values.len(),
1252                buf.len(),
1253                "allreduce_sum_u64: partition {src} contributed {} values, this partition {}",
1254                values.len(),
1255                buf.len(),
1256            );
1257            for (a, v) in buf.iter_mut().zip(values) {
1258                *a = a.wrapping_add(v.load(Ordering::Relaxed));
1259            }
1260        }
1261    }
1262
1263    fn barrier(&self) {
1264        if self.size == 1 {
1265            return;
1266        }
1267        let gen = self.next_gen();
1268        self.state.publish(self.rank, gen, CallKind::Barrier, 0);
1269        self.state
1270            .wait_all(self.rank, gen, CallKind::Barrier, |_, _| {});
1271    }
1272}

1274impl Transport for InProcessTransport {
1275    /// A payload is *moved* to its partner rather than copied, so there is nothing for the two-phase shape to overlap: the transfer is complete before `body` runs and its [`ChunkWait`] is a no-op.
1276    /// `map` therefore goes unread, and `spare` is neither drawn from nor added to — a sender's blocks become the receiver's, and the pool circulates through the group rather than through the transport.
1277    fn exchange_layer<P, F, R>(
1278        &self,
1279        send: Vec<Option<P>>,
1280        _spare: &mut Vec<P>,
1281        _map: &ChunkMap,
1282        body: F,
1283    ) -> (Vec<Option<P>>, R)
1284    where
1285        P: Payload,
1286        F: FnOnce(&[Option<P>], &dyn ChunkWait) -> R,
1287    {
1288        let n = self.size as usize;
1289        assert_eq!(
1290            send.len(),
1291            n,
1292            "exchange: send has {} entries, expected one entry per partition ({n})",
1293            send.len(),
1294        );
1295        assert!(
1296            send[self.rank as usize].is_none(),
1297            "exchange: send[{}] is this partition's own slot and must be None",
1298            self.rank,
1299        );
1300        let recv: Vec<Option<P>> = if n == 1 {
1301            vec![None]
1302        } else {
1303            let seq = self.next_gen();
1304            // The exchange consumes a generation like any other call and publishes it before sending, even though it waits on the channels rather than on the slots: that is what lets a partner spinning in a *collective* at the same generation see the kind mismatch and panic, instead of the pair hanging on each other's different call.
1305            self.state.publish(self.rank, seq, CallKind::Exchange, 0);
1306            // Unbounded channels: every send completes before the first receive, so no pair of partitions can block on each other.
1307            for (dst, payload) in send.into_iter().enumerate() {
1308                if dst != self.rank as usize {
1309                    self.send_to(dst, seq, Box::new(payload), "exchange");
1310                }
1311            }
1312            (0..n)
1313                .map(|src| {
1314                    if src == self.rank as usize {
1315                        None
1316                    } else {
1317                        downcast::<Option<P>>(self.recv_from(src, seq, "exchange"), src, "exchange")
1318                    }
1319                })
1320                .collect()
1321        };
1322
1323        let out = body(&recv, &AlreadyHere);
1324        (recv, out)
1325    }
1326}

1671    /// Move `payload` over the wire and back: [`Payload::byte_parts`] on the sender, [`Payload::recv_into`] + [`Payload::finish_recv`] on a fresh receiver, with the bytes copied across the way a transport moves them.
1672    ///
1673    /// That pair is the only encode/decode path the engine has, so it is what the wire-format tests exercise.
1674    fn wire_round_trip<const W: usize>(payload: &PartnerPayload<W>) -> PartnerPayload<W> {
1675        let sent: Vec<Vec<u8>> = payload
1676            .byte_parts()
1677            .into_iter()
1678            .map(<[u8]>::to_vec)
1679            .collect();
1680        let lens: Vec<usize> = sent.iter().map(Vec::len).collect();
1681        let mut back = PartnerPayload::<W>::default();
1682        for (view, bytes) in back.recv_into(&lens).into_iter().zip(&sent) {
1683            view.copy_from_slice(bytes);
1684        }
1685        back.finish_recv();
1686        back
1687    }
