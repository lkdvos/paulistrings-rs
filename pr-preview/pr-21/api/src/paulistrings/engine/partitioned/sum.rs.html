24/// A [`PauliSum`] split across the partitions of a [`PartitionRuntime`], held across calls so a driver scatters and gathers once.
25///
26/// Between calls every partition's sum holds only keys of its own partition and shares one hash and bucket count with its peers ([`PartitionedSum::assert_invariants`]).
27/// `B` is where a partition lives: host memory by default, a CUDA device for `gpu::GpuPartitionedSum`.
28pub struct PartitionedSum<const W: usize, B = HostPartition<W>> {
29    /// In rank order.
30    parts: Vec<B>,
31    rows: PartitionRows<W>,
32    runtime: Arc<PartitionRuntime>,
33    /// `None` unless [`enable_trace`](Self::enable_trace) was called.
34    trace: Option<PartitionTrace>,
35    #[cfg(feature = "phase-timing")]
36    scatter_ns: u64,
37    /// Atomic because [`gather`](Self::gather) takes `&self`.
38    #[cfg(feature = "phase-timing")]
39    gather_ns: std::sync::atomic::AtomicU64,
40    #[cfg(feature = "phase-timing")]
41    layers: u64,
42}

120    /// The host driver's `propagate_with` on any backend.
121    pub(crate) fn propagate_on_backend<T>(
122        &mut self,
123        circuit: &Circuit<W>,
124        policy: &T,
125        direction: Direction,
126        options: PropagateOptions,
127    ) where
128        T: TruncationPolicy<W> + ?Sized,
129        B: PartitionBackend<W, T>,
130    {
131        let n = circuit.channels.len();
132        let size = self.num_partitions();
133        assert_supports_partitions(policy, size);
134        let terms_in = self.len();
135        let started = Instant::now();
136        log::info!(
137            target: LOG_TARGET,
138            "propagate_partitioned: {terms_in} terms through {n} channels ({direction:?}) \
139             on {size} partitions [{}]",
140            self.runtime.placement_summary(),
141        );
142
143        if n > 0 {
144            let tracing = self.trace.is_some();
145            let items: Vec<PartitionWork<B>> = self
146                .parts
147                .iter_mut()
148                .map(|part| PartitionWork::take(part, n, tracing))
149                .collect();
150
151            let runtime = Arc::clone(&self.runtime);
152            let rows = &self.rows;
153            let done = runtime.map_partitions(items, |rank, mut work, transport| {
154                let context = PartitionContext {
155                    rows,
156                    rank,
157                    size,
158                    tracing,
159                };
160                run_layers(
161                    circuit, policy, direction, options, context, &mut work, transport,
162                );
163                work
164            });
165            let mut traced = Vec::with_capacity(if tracing { size } else { 0 });
166            for (rank, work) in done.into_iter().enumerate() {
167                self.parts[rank] = work.local;
168                if tracing {
169                    traced.push(work.rows);
170                }
171            }
172            if let Some(trace) = self.trace.as_mut() {
173                assemble(trace, traced);
174            }
175            #[cfg(feature = "phase-timing")]
176            {
177                self.layers += n as u64;
178            }
179        }
180
181        log::info!(
182            target: LOG_TARGET,
183            "propagate_partitioned: {n} layers applied, {terms_in} -> {} terms, {:.3} s",
184            self.len(),
185            started.elapsed().as_secs_f64(),
186        );
187    }
188}

190impl<const W: usize, B: PartitionStorage<W>> PartitionedSum<W, B> {
191    /// Terms in the whole sum, summed over partitions.
192    pub fn len(&self) -> usize {
193        self.parts.iter().map(|p| p.len()).sum()
194    }
195
196    /// Whether every partition is empty.
197    pub fn is_empty(&self) -> bool {
198        self.parts.iter().all(|p| p.len() == 0)
199    }
200
201    /// The bucket bits every partition currently holds.
202    pub fn bits(&self) -> u8 {
203        self.parts[0].hash().bits()
204    }
205
206    /// Qubits the sum is over.
207    pub fn num_qubits(&self) -> usize {
208        self.rows.num_qubits()
209    }

212impl<const W: usize> PartitionedSum<W> {
213    /// Splits `sum` across the partitions `config` places, by seeded rows ([`PartitionRowPolicy::Seeded`] at `config.partition_row_seed`).
214    ///
215    /// # Errors
216    ///
217    /// [`TopologyError`] if `config` cannot be resolved or a pool cannot be built.
218    ///
219    /// # Panics
220    ///
221    /// In debug builds, if the partition rows are not independent of the sum's hash rows.
222    pub fn scatter(sum: PauliSum<W>, config: &PartitionConfig) -> Result<Self, TopologyError> {
223        let options = ScatterOptions {
224            runtime: PartitionRuntime::new(config)?,
225            rows: ScatterRows::Policy(PartitionRowPolicy::Seeded(config.partition_row_seed)),
226        };
227        Ok(Self::scatter_with(sum, options))
228    }
229
230    /// [`scatter`](Self::scatter) onto a caller-built runtime, split by the rows `options` names; at `P = 1` the scatter changes nothing.
231    ///
232    /// # Panics
233    ///
234    /// If the rows do not name the runtime's partition count, or as [`PartitionRows::cut`].
235    /// In debug builds, if the rows are not independent of the sum's hash rows, which costs load balance but not correctness.
236    pub fn scatter_with(sum: PauliSum<W>, options: ScatterOptions<W>) -> Self {
237        let ScatterOptions { runtime, rows } = options;
238        let rows = rows.resolve(&sum, runtime.partition_bits());
239        let size = runtime.num_partitions();
240        rows.assert_splits(sum.hash(), sum.num_qubits(), size);
241
242        let started = Instant::now();
243        let locals = {
244            let sum = &sum;
245            let rows = &rows;
246            runtime.map_partitions((0..size).collect(), |rank, _, transport| {
247                scatter_local(sum, rows, rank as u32, transport)
248            })
249        };
250        log::info!(
251            target: LOG_TARGET,
252            "scatter: {} terms over {size} partitions, {} bucket bits, {:.3} s",
253            sum.len(),
254            locals[0].hash().bits(),
255            started.elapsed().as_secs_f64(),
256        );
257
258        Self {
259            parts: locals.into_iter().map(HostPartition::new).collect(),
260            rows,
261            runtime,
262            trace: None,
263            #[cfg(feature = "phase-timing")]
264            scatter_ns: started.elapsed().as_nanos() as u64,
265            #[cfg(feature = "phase-timing")]
266            gather_ns: std::sync::atomic::AtomicU64::new(0),
267            #[cfg(feature = "phase-timing")]
268            layers: 0,
269        }
270    }
271
272    /// Propagates through `circuit` under `policy`, in place; see [`propagate_with`](Self::propagate_with).
273    pub fn propagate<T>(&mut self, circuit: &Circuit<W>, policy: &T, direction: Direction)
274    where
275        T: TruncationPolicy<W> + ?Sized,
276    {
277        self.propagate_with(circuit, policy, direction, PropagateOptions::default())
278    }
279
280    /// Propagates through `circuit` under `policy` with explicit [`PropagateOptions`], as [`propagate_with`](crate::propagate_with) does unpartitioned.
281    ///
282    /// The per-layer `DEBUG` log line comes once per partition, tagged `partition r/P`, from that partition's own thread.
283    ///
284    /// # Panics
285    ///
286    /// If a channel's [`Channel::prepare`](crate::Channel::prepare) declines, as the unpartitioned engine does.
287    /// Before the first layer, above one partition, unless [`policy.supports_partitioned()`](TruncationPolicy::supports_partitioned) — exact [`TopN`](crate::TopN) is not yet supported partitioned.
288    /// A panic in any partition is re-raised on the calling thread.
289    pub fn propagate_with<T>(
290        &mut self,
291        circuit: &Circuit<W>,
292        policy: &T,
293        direction: Direction,
294        options: PropagateOptions,
295    ) where
296        T: TruncationPolicy<W> + ?Sized,
297    {
298        self.propagate_on_backend(circuit, policy, direction, options);
299    }

301    /// Merges the partitions back into one sum on the calling thread, leaving `self` intact.
302    pub fn gather(&self) -> PauliSum<W> {
303        #[cfg(feature = "phase-timing")]
304        let started = Instant::now();
305        let out = PauliSum::merge_partitions(self.parts.iter().map(|p| p.sum.clone()).collect());
306        #[cfg(feature = "phase-timing")]
307        self.gather_ns.fetch_add(
308            started.elapsed().as_nanos() as u64,
309            std::sync::atomic::Ordering::Relaxed,
310        );
311        out
312    }
313
314    /// [`gather`](Self::gather) by value — no clone of the parts.
315    pub fn into_gathered(self) -> PauliSum<W> {
316        PauliSum::merge_partitions(self.parts.into_iter().map(|p| p.sum).collect())
317    }

323    /// If `r` is not a partition of this sum.
324    pub fn partition(&self, r: usize) -> &PauliSum<W> {

326    }

328    /// `⟨ψ|O|ψ⟩` in a uniform single-qubit product state, the partitions' own values added in rank order.
329    pub fn expectation_product_state(&self, state: ProductState) -> Complex64 {
330        self.parts
331            .iter()
332            .map(|part| part.sum.expectation_product_state(state))
333            .sum()
334    }

336    /// [`PauliSum::anticommute_histogram`] of the whole sum, the partitions' histograms added in rank order.
337    pub fn anticommute_histogram(&self, sites: &[usize], axis: RotationAxis) -> Vec<f64> {
338        let mut histogram = vec![0.0f64; sites.len() + 1];
339        for part in &self.parts {
340            for (h, v) in histogram
341                .iter_mut()
342                .zip(part.sum.anticommute_histogram(sites, axis))
343            {
344                *h += v;
345            }
346        }
347        histogram
348    }

350    /// [`PauliSum::rotated_overlap`] of the whole sum; gathers first unless the rows avoid the flipped coordinates of `sites`.
351    pub fn rotated_overlap(&self, sites: &[usize], delta: f64, axis: RotationAxis) -> f64 {
352        if self.rows.keeps_flip_classes(sites, axis) {
353            self.parts
354                .iter()
355                .map(|part| part.sum.rotated_overlap(sites, delta, axis))
356                .sum()
357        } else {
358            self.gather().rotated_overlap(sites, delta, axis)
359        }
360    }

362    /// Drain the per-phase timing counters, zeroing them.
363    #[cfg(feature = "phase-timing")]
364    pub fn take_stats(&mut self) -> PartitionPhaseStats {
365        PartitionPhaseStats {
366            per_partition: self
367                .parts
368                .iter_mut()
369                .map(|part| part.state.layer.take_stats())
370                .collect(),
371            scatter_ns: std::mem::take(&mut self.scatter_ns),
372            gather_ns: self.gather_ns.swap(0, std::sync::atomic::Ordering::Relaxed),
373            layers: std::mem::take(&mut self.layers),
374        }
375    }

377    /// Checks every invariant a partitioned sum holds between calls, in `O(terms)`; release builds skip the per-partition structural check.
378    ///
379    /// # Panics
380    ///
381    /// If a partition's sum is internally inconsistent, holds a key of another partition, or disagrees with partition 0 about the hash rows, the bucket count or the qubit count.
382    pub fn assert_invariants(&self) {
383        assert_eq!(
384            self.parts.len(),
385            self.rows.num_partitions(),
386            "partition count disagrees with the rows",
387        );
388        let head = &self.parts[0].sum;
389        for (rank, local) in self.parts.iter().map(|p| &p.sum).enumerate() {
390            #[cfg(any(test, debug_assertions))]
391            local.assert_invariants();
392            assert_eq!(
393                local.num_qubits(),
394                head.num_qubits(),
395                "partition {rank}: qubit count differs from partition 0",
396            );
397            assert!(
398                local.hash().same_rows_as(head.hash()),
399                "partition {rank}: hash rows differ from partition 0",
400            );
401            assert_eq!(
402                local.hash().bits(),
403                head.hash().bits(),
404                "partition {rank}: bucket bits differ from partition 0",
405            );
406            let held = local.partition_rank_of_all(&self.rows);
407            assert!(
408                held.is_none() || held == Some(rank as u32),
409                "partition {rank} holds keys of partition {held:?}",
410            );
411        }
412    }

415/// The partitioned engine's phase breakdown, drained by [`PartitionedSum::take_stats`].
416///
417/// The per-partition wall-clock fields are concurrent, not additive; `exchange_ns` absorbs the wait for a partner, so the load imbalance shows there.
418#[cfg(feature = "phase-timing")]
419#[derive(Clone, Debug, Default, PartialEq, Eq)]
420pub struct PartitionPhaseStats {
421    /// One breakdown per partition, in rank order.
422    pub per_partition: Vec<PhaseStats>,
423    /// Wall time of the scatter that built this sum.
424    pub scatter_ns: u64,
425    /// Wall time summed over [`PartitionedSum::gather`] calls.
426    pub gather_ns: u64,
427    /// Layers driven since the counters were last drained.
428    pub layers: u64,
429}
