1//! Per-layer classification of a prepared channel's deltas into local (`part(d) = 0`) and remote ones (ARCHITECTURE.md §Partitioning).
2
3use crate::channel::prepared::Prepared;
4#[cfg(any(test, feature = "test-utils"))]
5use crate::circuit::Circuit;
6#[cfg(any(test, feature = "test-utils"))]
7use crate::pauli_sum::hash::Gf2Hash;
8use crate::pauli_sum::hash::PartitionRows;
9
10/// One delta of a prepared channel that crosses a partition boundary.
11#[derive(Clone, Debug, PartialEq, Eq)]
12pub(crate) struct RemoteDelta {
13    /// Index into `ptm.deltas()`; for a rotation, `0` is the identity pass and `1` the generator pass.
14    pub entry: usize,
15    /// `rank ^ part(d)`, never `rank` itself.
16    pub partner: u32,
17    /// `h(d)`.
18    pub bucket_delta: u32,
19}
20
21/// How one prepared channel's deltas split under a partitioning, seen from one partition.
22#[derive(Clone, Debug, Default)]
23pub(crate) struct PartitionPlan {
24    /// Per entry (numbered as [`RemoteDelta::entry`]), whether its partition delta is zero.
25    pub local_entries: Vec<bool>,
26    /// The local deltas' bucket deltas, ascending, deduplicated and always containing `0`.
27    pub local_bucket_deltas: Vec<u32>,
28    /// Ascending by [`RemoteDelta::entry`].
29    pub remote: Vec<RemoteDelta>,
30    /// Non-identity realized deltas, local and remote together, so the sort-kernel gate sees the channel's whole fanout.
31    pub rest_streams_total: usize,
32}
33
34impl PartitionPlan {
35    /// Classify `prepared`'s deltas under `rows`, as seen from partition `rank`.
36    #[cfg(any(test, feature = "test-utils"))]
37    pub(crate) fn new<const W: usize>(
38        prepared: &Prepared<W>,
39        rows: &PartitionRows<W>,
40        rank: u32,
41    ) -> Self {
42        let mut plan = Self::default();
43        plan.rebuild(prepared, rows, rank);
44        plan
45    }
46
47    /// [`new`](Self::new) in place, keeping the allocations across layers.
48    pub(crate) fn rebuild<const W: usize>(
49        &mut self,
50        prepared: &Prepared<W>,
51        rows: &PartitionRows<W>,
52        rank: u32,
53    ) {
54        debug_assert!(
55            rows.num_partitions() > rank as usize,
56            "PartitionPlan: rank {rank} outside {} partitions",
57            rows.num_partitions(),
58        );
59        self.local_entries.clear();
60        self.local_bucket_deltas.clear();
61        self.local_bucket_deltas.push(0);
62        self.remote.clear();
63
64        let mut classify =
65            |entry: usize, mask_x: &[u64; W], mask_z: &[u64; W], bucket_delta: u32| {
66                let partition_delta = rows.partition_of(mask_x, mask_z);
67                self.local_entries.push(partition_delta == 0);
68                if partition_delta == 0 {
69                    self.local_bucket_deltas.push(bucket_delta);
70                } else {
71                    self.remote.push(RemoteDelta {
72                        entry,
73                        partner: rank ^ partition_delta,
74                        bucket_delta,
75                    });
76                }
77            };
78        match prepared {
79            Prepared::Local(ptm) => {
80                for (entry, delta) in ptm.deltas().iter().enumerate() {
81                    let (mask_x, mask_z) = delta.mask();
82                    classify(entry, &mask_x, &mask_z, delta.bucket_delta);
83                }
84            }
85            Prepared::Rotation(rotation) => {
86                let (generator_x, generator_z) = rotation.generator_mask();
87                classify(0, &[0u64; W], &[0u64; W], rotation.bucket_delta_identity);
88                classify(
89                    1,
90                    &generator_x,
91                    &generator_z,
92                    rotation.bucket_delta_generator,
93                );
94            }
95        }
96
97        // `engine::bucketed`'s `rest_streams`, over the whole realized delta set (ARCHITECTURE.md §Bucketing).
98        self.rest_streams_total = match prepared {
99            Prepared::Local(ptm) => {
100                let has_identity = ptm.deltas().first().is_some_and(|d| d.local_delta == 0);
101                debug_assert!(
102                    !has_identity || self.local_entries[0],
103                    "the identity delta must be local: part(0) = 0",
104                );
105                ptm.deltas().len() - has_identity as usize
106            }
107            Prepared::Rotation(_) => {
108                debug_assert!(self.local_entries[0], "the identity pass must be local");
109                1
110            }
111        };
112
113        self.local_bucket_deltas.sort_unstable();
114        self.local_bucket_deltas.dedup();
115    }
116
117    /// Whether this layer moves anything across a partition boundary.
118    pub(crate) fn has_remote(&self) -> bool {
119        !self.remote.is_empty()
120    }
121
122    /// The remote deltas destined for partition `q`, ascending by entry.
123    pub(crate) fn remote_for_partner(&self, q: u32) -> impl Iterator<Item = &RemoteDelta> {
124        self.remote.iter().filter(move |r| r.partner == q)
125    }
126}

128/// Per layer in application order, `(local, remote)` delta counts of `circuit`'s channels prepared against `hash` under `rows`.
129///
130/// # Panics
131///
132/// If any channel declines [`Channel::prepare`](crate::Channel::prepare).
133#[cfg(any(test, feature = "test-utils"))]
134pub fn count_remote_deltas<const W: usize>(
135    circuit: &Circuit<W>,
136    hash: &Gf2Hash<W>,
137    rows: &PartitionRows<W>,
138    adjoint: bool,
139) -> Vec<(usize, usize)> {
140    let n = circuit.channels.len();
141    (0..n)
142        .map(|k| if adjoint { n - 1 - k } else { k })
143        .map(|index| {
144            let prepared = circuit.channels[index]
145                .prepare(hash, adjoint)
146                .unwrap_or_else(|| {
147                    panic!("count_remote_deltas: layer {index} declined Channel::prepare")
148                });
149            let plan = PartitionPlan::new(&prepared, rows, 0);
150            let local = plan.local_entries.iter().filter(|b| **b).count();
151            (local, plan.remote.len())
152        })
153        .collect()
154}
