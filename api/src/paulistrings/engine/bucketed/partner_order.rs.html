1//! [`PartnerOrder`], which puts a rotation's partner run in key order in linear time.
2
3use crate::channel::prepared::MAX_LOCAL_SUPPORT;
4use crate::pauli_sum::storage::BucketColumns;
5
6/// Set bits a two-qubit key delta can have.
7const MAX_LEVELS: usize = 2 * MAX_LOCAL_SUPPORT;
8
9/// The key order of `k ⊕ g` over a run of distinct `k` in key order.
10///
11/// Two keys first differ at the same bit before and after the XOR, and their order flips exactly when `g` sets that bit.
12/// So within each group of keys sharing every bit above `g`'s most significant set bit, the halves on either side of that bit swap, and so on recursively down `g`'s set bits; below the last one the source order stands.
13/// That is a fixed block permutation, applied by [`order`](Self::order) from each adjacent pair's first differing bit, with no comparison sort.
14#[derive(Clone, Copy, Debug)]
15pub(super) struct PartnerOrder<const W: usize> {
16    /// `g`'s set bits as significance ranks in the lex key `(x, z)`, 0 the most significant, ascending.
17    levels: [u32; MAX_LEVELS],
18    depth: usize,
19}
20
21impl<const W: usize> PartnerOrder<W> {
22    /// The order for the XOR mask `g = (mask_x, mask_z)`, which has at most [`MAX_LEVELS`] set bits.
23    pub(super) fn new(mask_x: &[u64; W], mask_z: &[u64; W]) -> Self {
24        let mut levels = [0u32; MAX_LEVELS];
25        let mut depth = 0;
26        for word in 0..2 * W {
27            let mut live = if word < W { mask_x[word] } else { mask_z[word - W] };
28            while live != 0 {
29                let rank = live.leading_zeros();
30                assert!(
31                    depth < MAX_LEVELS,
32                    "PartnerOrder: a key delta has at most {MAX_LEVELS} set bits"
33                );
34                levels[depth] = 64 * word as u32 + rank;
35                depth += 1;
36                live &= !(1u64 << (63 - rank));
37            }
38        }
39        Self { levels, depth }
40    }
41
42    /// Write `run`, whose keys are `k ⊕ g` for distinct `k` in key order, into `out` in key order; `divergence` is scratch.
43    pub(super) fn order(
44        &self,
45        run: &BucketColumns<W>,
46        out: &mut BucketColumns<W>,
47        divergence: &mut Vec<u32>,
48    ) {
49        out.clear();
50        let len = run.len();
51        out.x.reserve(len);
52        out.z.reserve(len);
53        out.coeff.reserve(len);
54        divergence.clear();
55        divergence.push(0);
56        divergence.extend((1..len).map(|row| first_difference(run, row - 1, row)));
57        let mut placer = Placer { run, out, divergence, levels: &self.levels[..self.depth] };
58        placer.place(0, len, 0);
59    }
60}
61
62/// The significance rank of the first bit at which rows `a` and `b` differ.
63fn first_difference<const W: usize>(run: &BucketColumns<W>, a: usize, b: usize) -> u32 {
64    for w in 0..W {
65        let diff = run.x[a][w] ^ run.x[b][w];
66        if diff != 0 {
67            return 64 * w as u32 + diff.leading_zeros();
68        }
69    }
70    for w in 0..W {
71        let diff = run.z[a][w] ^ run.z[b][w];
72        if diff != 0 {
73            return 64 * (W + w) as u32 + diff.leading_zeros();
74        }
75    }
76    unreachable!("PartnerOrder: the run's keys must be distinct")
77}

79/// One [`PartnerOrder::order`] call; `divergence[i]` is the first differing bit of rows `i - 1` and `i`.
80struct Placer<'a, const W: usize> {
81    run: &'a BucketColumns<W>,
82    out: &'a mut BucketColumns<W>,
83    divergence: &'a [u32],
84    levels: &'a [u32],
85}
86
87impl<const W: usize> Placer<'_, W> {
88    fn copy(&mut self, low: usize, high: usize) {
89        self.out.x.extend_from_slice(&self.run.x[low..high]);
90        self.out.z.extend_from_slice(&self.run.z[low..high]);
91        self.out.coeff.extend_from_slice(&self.run.coeff[low..high]);
92    }
93
94    /// Rows `low..high`, which share every bit above `levels[level - 1]` and that bit, in key order.
95    fn place(&mut self, low: usize, high: usize, level: usize) {
96        if level == self.levels.len() || high - low < 2 {
97            self.copy(low, high);
98            return;
99        }
100        let bit = self.levels[level];
101        // Groups share every bit above `bit`; one split by it has its XORed-set half first, which swaps.
102        // Runs of groups without a split pass to the next level whole, since its groups only refine them.
103        let (mut pending, mut group_start, mut row) = (low, low, low + 1);
104        while row < high {
105            let d = self.divergence[row];
106            if d < bit {
107                group_start = row;
108            } else if d == bit {
109                let split = row;
110                let mut end = split + 1;
111                while end < high && self.divergence[end] > bit {
112                    end += 1;
113                }
114                debug_assert!(end == high || self.divergence[end] < bit);
115                self.place(pending, group_start, level + 1);
116                self.place(split, end, level + 1);
117                self.place(group_start, split, level + 1);
118                pending = end;
119                group_start = end;
120                row = end;
121            }
122            row += 1;
123        }
124        self.place(pending, high, level + 1);
125    }
126}
