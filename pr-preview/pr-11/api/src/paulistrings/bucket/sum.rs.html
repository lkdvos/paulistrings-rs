16/// Exposed as a constant rather than hidden so a caller who needs a different partition can build their own [`Gf2Hash`].

19/// Bucket bits for a sum of `len` terms: the smallest `b` with `len <= target << b`, clamped below by the parallelism floor.
20/// Used to size the partition once at ingestion and at the start of `propagate`. [`PauliSum::rebucket`] tracks it afterwards, but only upward.
21pub fn desired_bits(len: usize, target: usize, min_buckets: usize) -> u8 {
22    debug_assert!(target > 0);
23    let worth_splitting = len >= min_buckets.saturating_mul(MIN_TERMS_PER_TASK);
24    let mut floor = 0u8;
25    if worth_splitting {
26        while (1usize << floor) < min_buckets && floor < super::hash::B_MAX_BITS {
27            floor += 1;
28        }
29    }
30    let mut b = floor;
31    while b < super::hash::B_MAX_BITS && len > target.saturating_mul(1usize << b) {
32        b += 1;
33    }
34    b
35}

37/// Minimum terms per bucket for the parallelism floor to apply.
38/// The floor in [`PauliSum::rebucket`] exists to give Rayon enough independent tasks, but a task carrying almost nothing is pure overhead; below `min_buckets × MIN_TERMS_PER_TASK` total terms we would rather have few buckets and let the small-`n` fallback handle it. See ARCHITECTURE.md §Bucket-Policy for the sweep that set this value.
39///
40/// This gate is not only about parallelism: the bucket count caps the engine's coset dimension, and the per-run sort's comparison count reaches its floor only at full delta rank — so it also controls how much sort work a dense-PTM layer pays. Do not lower or drop the gate without checking a sparse-PTM layer too; the two regimes pull in opposite directions (see `research/FINDINGS.md`).
41pub const MIN_TERMS_PER_TASK: usize = 64;

43/// Default target terms per bucket.
44/// Chosen so a bucket plus its gather scratch stays L2-resident on the reference host. See ARCHITECTURE.md §Bucket-Policy for the sweep that set this value, and [`MIN_TERMS_PER_TASK`] for the dense-PTM caveat.
45pub const DEFAULT_TARGET_BUCKET_LEN: usize = 1024;

90/// Split one input bucket into its "low" (kept in place) and "high" (new bucket at `b + old_nb`) halves under a hash that has just gained `new_bit` as its top bit.
91/// Shared by [`PauliSum::refine`]'s serial and parallel branches. `b` is the old bucket index, used only by the debug-only low-bits invariant check.
92fn refine_bucket<const W: usize>(
93    cols: &mut BucketCols<W>,
94    up: &mut BucketCols<W>,
95    hash: &Gf2Hash<W>,
96    new_bit: u8,
97    b: u32,
98) {
99    let _ = b; // referenced only inside the `cfg(debug_assertions)` block below
100    let n = cols.len();
101    let mut keep = 0usize;
102    for i in 0..n {
103        let bit = hash.row_parity(&cols.x[i], &cols.z[i], new_bit);
104        #[cfg(debug_assertions)]
105        {
106            let full = hash.bucket_of(&cols.x[i], &cols.z[i]);
107            debug_assert_eq!(
108                full & ((1u32 << new_bit) - 1),
109                b,
110                "refine: low bits must be preserved",
111            );
112        }
113        if bit == 1 {
114            up.push(cols.x[i], cols.z[i], cols.coeff[i]);
115        } else {
116            // Compact in place: `keep <= i` always, so this never overwrites an unread slot.
117            cols.x[keep] = cols.x[i];
118            cols.z[keep] = cols.z[i];
119            cols.coeff[keep] = cols.coeff[i];
120            keep += 1;
121        }
122    }
123    cols.x.truncate(keep);
124    cols.z.truncate(keep);
125    cols.coeff.truncate(keep);
126}

128/// Merge two sorted runs. No coefficient combining: keys are globally unique.
129fn merge_two<const W: usize>(a: &BucketCols<W>, b: &BucketCols<W>) -> BucketCols<W> {
130    let mut out = BucketCols::<W>::new();
131    let total = a.len() + b.len();
132    out.x.reserve_exact(total);
133    out.z.reserve_exact(total);
134    out.coeff.reserve_exact(total);
135    let (mut i, mut j) = (0usize, 0usize);
136    while i < a.len() && j < b.len() {
137        if (&a.x[i], &a.z[i]) <= (&b.x[j], &b.z[j]) {
138            out.push(a.x[i], a.z[i], a.coeff[i]);
139            i += 1;
140        } else {
141            out.push(b.x[j], b.z[j], b.coeff[j]);
142            j += 1;
143        }
144    }
145    while i < a.len() {
146        out.push(a.x[i], a.z[i], a.coeff[i]);
147        i += 1;
148    }
149    while j < b.len() {
150        out.push(b.x[j], b.z[j], b.coeff[j]);
151        j += 1;
152    }
153    out
154}

156/// Merge two sorted runs, summing equal keys and dropping exact-zero sums.
157/// The counterpart of [`merge_two`] for operands that may share keys: within a partition, equal keys are always in the same bucket pair, so a two-pointer pass over one bucket of each operand sees every collision there is.
158fn merge_two_adding<const W: usize>(a: &BucketCols<W>, b: &BucketCols<W>) -> BucketCols<W> {
159    let mut out = BucketCols::<W>::new();
160    let total = a.len() + b.len();
161    out.x.reserve_exact(total);
162    out.z.reserve_exact(total);
163    out.coeff.reserve_exact(total);
164    let zero = Complex64::new(0.0, 0.0);
165    let (mut i, mut j) = (0usize, 0usize);
166    while i < a.len() && j < b.len() {
167        match (&a.x[i], &a.z[i]).cmp(&(&b.x[j], &b.z[j])) {
168            std::cmp::Ordering::Less => {
169                out.push(a.x[i], a.z[i], a.coeff[i]);
170                i += 1;
171            }
172            std::cmp::Ordering::Greater => {
173                out.push(b.x[j], b.z[j], b.coeff[j]);
174                j += 1;
175            }
176            std::cmp::Ordering::Equal => {
177                let c = a.coeff[i] + b.coeff[j];
178                if c != zero {
179                    out.push(a.x[i], a.z[i], c);
180                }
181                i += 1;
182                j += 1;
183            }
184        }
185    }
186    while i < a.len() {
187        out.push(a.x[i], a.z[i], a.coeff[i]);
188        i += 1;
189    }
190    while j < b.len() {
191        out.push(b.x[j], b.z[j], b.coeff[j]);
192        j += 1;
193    }
194    out
195}

197/// Merge `B` sorted runs into one, by `log2(B)` rounds of pairwise merges.
198/// Faster than a `BinaryHeap`-based `B`-way merge, whose pops need `log2(B)` comparisons against keys scattered across `B` runs and which is inherently sequential. The tree does the same `O(n log B)` comparisons but reads two sequential streams at a time, and every pair within a round is independent, so the rounds parallelize — sequential bandwidth traded for `log B` passes over the payload instead of one.
199fn merge_runs<const W: usize>(mut runs: Vec<BucketCols<W>>) -> BucketCols<W> {
200    if runs.is_empty() {
201        return BucketCols::new();
202    }
203    while runs.len() > 1 {
204        runs = runs
205            .par_chunks(2)
206            .map(|pair| match pair {
207                [a, b] => merge_two(a, b),
208                [a] => a.clone(),
209                _ => unreachable!("par_chunks(2) yields 1 or 2 elements"),
210            })
211            .collect();
212    }
213    runs.pop().expect("non-empty by the check above")
214}

216/// Copy the terms of one bucket whose partition rank is `rank`.
217/// Shared by `PauliSum::filter_partition`'s serial and parallel branches. A subsequence of a strictly ascending run is strictly ascending, so the output needs no sort.
218/// Plain pushes, no `reserve`: the surviving count is not known without a second pass, and a speculative reservation would inflate peak RSS across all `P` parts for no measured gain (see `research/FINDINGS.md`).
219fn filter_bucket<const W: usize>(
220    cols: &BucketCols<W>,
221    rows: &PartitionRows<W>,
222    rank: u32,
223) -> BucketCols<W> {
224    let mut out = BucketCols::<W>::new();
225    for i in 0..cols.len() {
226        if rows.partition_of(&cols.x[i], &cols.z[i]) == rank {
227            out.push(cols.x[i], cols.z[i], cols.coeff[i]);
228        }
229    }
230    out
231}

233/// Merge `k` disjoint sorted runs into one, in `ceil(log2 k)` pairwise rounds — the gather half of the partition scatter/gather.
234/// The same tree as [`merge_runs`], but serial and clone-free: the caller is already parallel over buckets and `k = P <= 16`, so a nested Rayon split would be pure overhead. Runs come from distinct partitions of one sum, so no key can appear twice; debug builds check that by re-verifying the strict ascent of the result.
235fn merge_disjoint_runs<const W: usize>(mut runs: Vec<BucketCols<W>>) -> BucketCols<W> {
236    while runs.len() > 1 {
237        let mut next: Vec<BucketCols<W>> = Vec::with_capacity(runs.len().div_ceil(2));
238        let mut it = runs.into_iter();
239        while let Some(a) = it.next() {
240            match it.next() {
241                Some(b) => next.push(merge_two(&a, &b)),
242                None => next.push(a),
243            }
244        }
245        runs = next;
246    }
247    let out = runs.pop().unwrap_or_default();
248    #[cfg(debug_assertions)]
249    {
250        for i in 1..out.len() {
251            debug_assert!(
252                (&out.x[i - 1], &out.z[i - 1]) < (&out.x[i], &out.z[i]),
253                "PauliSum::merge_partitions: duplicate key across partitions",
254            );
255        }
256    }
257    out
258}

260/// Weighted sum of Pauli operators, stored as structure-of-arrays columns partitioned by a GF(2)-linear hash.
261///
262/// # Canonical order
263///
264/// Terms are ordered by bucket index ascending, then lexicographic `(x, z)` within a bucket — the order [`Self::iter`] and [`Self::to_arrays`] produce. This is a public promise, not an implementation detail.
265///
266/// # Invariant
267///
268/// Every term lies in `buckets[hash.bucket_of(term)]`, and each bucket is sorted by the lexicographic `(x, z)` key with no duplicate keys. Because `h` is a function, equal keys always share a bucket, so per-bucket dedup implies global dedup and no global sort is ever needed.
269///
270/// # Partition scatter/gather
271///
272/// The crate-internal `filter_partition` / `merge_partitions` pair are the scatter and gather primitives of the partitioned engine (ARCHITECTURE.md §Partitioning). Neither direction sorts or combines coefficients, so a round trip is bitwise.
273///
274/// [`propagate`]: crate::propagate
275#[derive(Clone, Debug)]
276pub struct PauliSum<const W: usize> {
277    buckets: Vec<BucketCols<W>>,
278    hash: Gf2Hash<W>,
279    num_qubits: usize,
280    len: usize,
281}

284    /// Partition a globally key-sorted stream of terms. `O(n)`: one hash evaluation and one scatter per term, and each bucket comes out sorted for free since order within it is inherited from the input.
285    /// The caller owes the sortedness: `x`, `z`, `coeff` must be parallel columns ascending in `(x, z)` with no duplicate keys, or the per-bucket sort invariant silently breaks.
286    pub(crate) fn from_key_sorted(
287        x: &[[u64; W]],
288        z: &[[u64; W]],
289        coeff: &[Complex64],
290        hash: Gf2Hash<W>,
291        num_qubits: usize,
292    ) -> Self {
293        let n = coeff.len();
294        let nb = hash.num_buckets();
295
296        // Hashing is the expensive part, so it runs in parallel; the counts come from the resulting indices rather than a second hashing pass.
297        // The scatter below stays sequential: buckets are separate allocations, so a parallel scatter would need every thread to write into every bucket.
298        let idx: Vec<u32> = (0..n)
299            .into_par_iter()
300            .map(|i| hash.bucket_of(&x[i], &z[i]))
301            .collect();
302        let mut counts: Vec<usize> = vec![0; nb];
303        for &b in idx.iter() {
304            counts[b as usize] += 1;
305        }
306
307        let mut buckets: Vec<BucketCols<W>> = Vec::with_capacity(nb);
308        for &c in counts.iter() {
309            let mut cols = BucketCols::<W>::new();
310            cols.x.reserve_exact(c);
311            cols.z.reserve_exact(c);
312            cols.coeff.reserve_exact(c);
313            buckets.push(cols);
314        }
315
316        for i in 0..n {
317            buckets[idx[i] as usize].push(x[i], z[i], coeff[i]);
318        }
319
320        Self {
321            buckets,
322            hash,
323            num_qubits,
324            len: n,
325        }
326    }

368    /// Repartition under `hash`, keeping every term. Flattens to a globally key-sorted stream and rescatters.
369    /// Prefer [`Self::refine`] / [`Self::coarsen`] when only the bucket count changes and the hash rows are the same; those are `O(n)` and never merge.

385    /// A copy of `self` partitioned exactly as `target` partitions.
386    /// Three cases, cheapest first: identical partition is a clone; same hash rows at a different bucket count is a clone plus `O(n)` refine/coarsen; different rows falls back to [`Self::with_hash`]'s `O(n log B)` flatten.
387    pub(crate) fn align_to(&self, target: &Gf2Hash<W>) -> Self {
388        if !self.hash.same_rows_as(target) {
389            return self.clone().with_hash(target.clone());
390        }
391        let mut out = self.clone();
392        while out.hash.bits() < target.bits() {
393            out.refine();
394        }
395        while out.hash.bits() > target.bits() {
396            out.coarsen();
397        }
398        out
399    }

401    /// Expectation value `⟨ψ|O|ψ⟩` in a uniform single-qubit product state.
402    /// For each [`ProductState`] there is exactly one single-qubit Pauli with expectation `1`; a term contributes its full coefficient iff every factor is `I` or that Pauli — a masked scan over the key columns, run as a per-bucket parallel reduction.
403    /// The uniform states are the special case of [`ProductBasis`] with sign `+1` everywhere, so this is a thin wrapper over [`Self::expectation_product_basis`].
404    /// Returns `Complex64` rather than `f64` because `self` need not be Hermitian; take `.re` when it is.
405    ///
406    /// # Summation order
407    ///
408    /// Partial sums are combined in bucket order, which is deterministic given the partition; two partitions of the same terms can differ in the last bits, since `f64` addition is not associative.
409    pub fn expectation_product_state(&self, state: ProductState) -> Complex64 {
410        self.expectation_product_basis(&ProductBasis::<W>::uniform(state))
411    }

413    /// Expectation value `⟨ψ|O|ψ⟩` in an arbitrary single-qubit product state.
414    /// `basis` gives each qubit its own axis and sign; see [`ProductBasis`] for the per-word match condition and the sign rule this evaluates. A term contributes its coefficient (negated when an odd number of support sites are `-1` eigenstates) iff every non-identity factor equals that qubit's axis exactly. Cost is one pass over the key columns as a per-bucket parallel reduction, with no expansion over basis states.
415    /// Returns `Complex64` rather than `f64` because `self` need not be Hermitian; take `.re` when it is.
416    ///
417    /// # Summation order
418    ///
419    /// As in [`Self::expectation_product_state`] — partials are combined in bucket order, so two partitions of the same terms can differ in the last bits.
420    pub fn expectation_product_basis(&self, basis: &ProductBasis<W>) -> Complex64 {
421        self.buckets
422            .par_iter()
423            .map(|cols| {
424                let mut acc = Complex64::new(0.0, 0.0);
425                for i in 0..cols.len() {
426                    // `mismatch` stays zero iff every word's non-identity sites carry exactly the local axis Pauli; `sign_bits` counts the `-1` eigenstates inside the support.
427                    let mut mismatch = 0u64;
428                    let mut sign_bits = 0u32;
429                    for w in 0..W {
430                        let x = cols.x[i][w];
431                        let z = cols.z[i][w];
432                        let sup = x | z;
433                        mismatch |= (x ^ (sup & basis.ax_x[w])) | (z ^ (sup & basis.ax_z[w]));
434                        sign_bits += (sup & basis.neg[w]).count_ones();
435                    }
436                    if mismatch == 0 {
437                        if sign_bits & 1 == 0 {
438                            acc += cols.coeff[i];
439                        } else {
440                            acc -= cols.coeff[i];
441                        }
442                    }
443                }
444                acc
445            })
446            .collect::<Vec<_>>()
447            .into_iter()
448            .fold(Complex64::new(0.0, 0.0), |a, b| a + b)
449    }

451    /// Expectation value `⟨ψ|O|ψ⟩` in a stabilizer state.
452    /// `⟨ψ|P|ψ⟩` is `±1` when `±P` lies in the state's stabilizer group and `0` otherwise, so this is a filter with a sign, exactly like [`Self::expectation_product_basis`], but the admissible state widens to any stabilizer state (Bell, GHZ, cluster, a Clifford circuit's output). See [`StabilizerState`] for the membership test and the sign bookkeeping.
453    /// Cost is `O(terms · n²/64)` word operations after the state's one-time `O(n³/64)` reduction — `n` times more work per term than the product-state scan, so prefer [`Self::expectation_product_basis`] for states that factorize.
454    /// Returns `Complex64` rather than `f64` because `self` need not be Hermitian; take `.re` when it is.
455    ///
456    /// # Summation order
457    ///
458    /// As in [`Self::expectation_product_state`] — partials are combined in bucket order, so two partitions of the same terms can differ in the last bits.
459    ///
460    /// # Panics
461    ///
462    /// Panics if `state.num_qubits()` differs from [`Self::num_qubits`].
463    pub fn expectation_stabilizer(&self, state: &StabilizerState<W>) -> Complex64 {
464        assert_eq!(
465            self.num_qubits,
466            state.num_qubits(),
467            "PauliSum::expectation_stabilizer: num_qubits mismatch ({} vs {})",
468            self.num_qubits,
469            state.num_qubits(),
470        );
471        self.buckets
472            .par_iter()
473            .map(|cols| {
474                let mut acc = Complex64::new(0.0, 0.0);
475                for i in 0..cols.len() {
476                    let key = PauliString::<W> {
477                        x: cols.x[i],
478                        z: cols.z[i],
479                    };
480                    match state.sign_of(&key) {
481                        None => {}
482                        Some(false) => acc += cols.coeff[i],
483                        Some(true) => acc -= cols.coeff[i],
484                    }
485                }
486                acc
487            })
488            .collect::<Vec<_>>()
489            .into_iter()
490            .fold(Complex64::new(0.0, 0.0), |a, b| a + b)
491    }

544    /// Double the bucket count, splitting each bucket in two.
545    /// One `Gf2Hash::row_parity` evaluation per term against just the new high bit — `O(n)` total — since only whether the new bit is set decides which half a term lands in; both halves inherit the source bucket's order, so nothing is re-sorted.
546    /// Bucket pairs are independent, so above [`MIN_TERMS_PER_TASK`] × [`DEFAULT_MIN_BUCKETS`] total terms the per-bucket work runs across Rayon; below it the sequential loop avoids per-task overhead.
547    pub fn refine(&mut self) {
548        let old_nb = self.buckets.len();
549        self.hash.refine();
550        let new_bit = self.hash.bits() - 1;
551        let hash = &self.hash;
552
553        // Take the old buckets out so the upper halves can reuse their storage.
554        let mut old = std::mem::take(&mut self.buckets);
555        let mut upper: Vec<BucketCols<W>> = (0..old_nb).map(|_| BucketCols::new()).collect();
556
557        if self.len < DEFAULT_MIN_BUCKETS * MIN_TERMS_PER_TASK {
558            for (b, (cols, up)) in old.iter_mut().zip(upper.iter_mut()).enumerate() {
559                refine_bucket(cols, up, hash, new_bit, b as u32);
560            }
561        } else {
562            old.par_iter_mut()
563                .zip(upper.par_iter_mut())
564                .enumerate()
565                .for_each(|(b, (cols, up))| {
566                    refine_bucket(cols, up, hash, new_bit, b as u32);
567                });
568        }
569
570        old.extend(upper);
571        self.buckets = old;
572    }

576    pub fn coarsen(&mut self) {

618    /// The scatter half of the partition scatter/gather (ARCHITECTURE.md §Partitioning). 620    pub(crate) fn filter_partition(&self, rows: &PartitionRows<W>, rank: u32) -> Self {

650    /// Gather partition-disjoint sums that share a partition back into one.
651    /// The inverse of [`Self::filter_partition`] (ARCHITECTURE.md §Partitioning). Bucket `b` of the result is a `P`-way merge of bucket `b` of each input — sorted runs in, one sorted run out, no coefficient arithmetic, so a `filter_partition`/`merge_partitions` round trip is bitwise. Debug builds assert the key sets are disjoint.
652    /// Same threshold and rationale as [`Self::refine`]. A single input is returned unchanged.
653    ///
654    /// # Panics
655    ///
656    /// Panics if `parts` is empty, or if the inputs disagree on their hash rows or bucket count — align them with [`Self::coarsen_to`] or [`Self::align_to`] first.
657    pub(crate) fn merge_partitions(parts: Vec<Self>) -> Self {
658        let first = parts
659            .first()
660            .expect("PauliSum::merge_partitions: no inputs to merge");
661        let hash = first.hash.clone();
662        let num_qubits = first.num_qubits;
663        for p in parts.iter().skip(1) {
664            assert!(
665                p.hash.same_rows_as(&hash),
666                "PauliSum::merge_partitions: hash mismatch (seed or num_qubits differs)",
667            );
668            assert_eq!(
669                p.hash.bits(),
670                hash.bits(),
671                "PauliSum::merge_partitions: bucket count mismatch",
672            );
673        }
674
675        let nb = hash.num_buckets();
676        let k = parts.len();
677        let len: usize = parts.iter().map(|p| p.len).sum();
678
679        // Transpose parts×buckets into buckets×parts, moving the columns so the merge below never copies a run it does not have to.
680        let mut runs: Vec<Vec<BucketCols<W>>> = (0..nb).map(|_| Vec::with_capacity(k)).collect();
681        for p in parts {
682            for (b, cols) in p.buckets.into_iter().enumerate() {
683                runs[b].push(cols);
684            }
685        }
686
687        let buckets: Vec<BucketCols<W>> = if len < DEFAULT_MIN_BUCKETS * MIN_TERMS_PER_TASK {
688            runs.into_iter().map(merge_disjoint_runs).collect()
689        } else {
690            runs.into_par_iter().map(merge_disjoint_runs).collect()
691        };
692
693        Self {
694            buckets,
695            hash,
696            num_qubits,
697            len,
698        }
699    }

800    /// Bring the bucket count up to what [`desired_bits`] would choose for the current length — but never down.
801    ///
802    /// # Grow-only policy
803    ///
804    /// A sum's bucket count is monotone non-decreasing over its lifetime: this clamps the target to `self.hash.bits()`, so `rebucket` only ever refines, never coarsens. Term counts oscillate every layer (fanout grows them, truncation cuts them back), so tracking `desired_bits` exactly in both directions would refine and coarsen on alternate layers near a power-of-two boundary, each an `O(n · bits)` serial pass — see `research/FINDINGS.md`.
805    /// Keeping the larger partition is otherwise free: the cost of not coarsening back down is three empty `Vec` headers per surplus bucket, not a term-proportional cost. A caller that wants to shrink a sum can still do so explicitly via [`Self::with_hash`] or [`Self::coarsen`].
806    /// Also keeps at least `min_buckets` buckets once there is enough work to spread, so the bucket-parallel decomposition has slack to load-balance.
807    pub fn rebucket(&mut self, target: usize, min_buckets: usize) {
808        debug_assert!(target > 0);
809        let want = desired_bits(self.len, target, min_buckets).max(self.hash.bits());
810        while self.hash.bits() < want {
811            self.refine();
812        }
813    }

826    /// This is not globally sorted — a bucket is a hash class, and the classes interleave arbitrarily in key order — but it is a total, deterministic order fixed by the partition, and the one [`Self::to_arrays`] concatenates in.
827    pub fn iter(&self) -> impl Iterator<Item = (&[u64; W], &[u64; W], Complex64)> + '_ {

832                .zip(cols.coeff.iter())

844            x.extend_from_slice(&cols.x);

851    /// Coefficient of the term with key `(x, z)`, or `None` if absent.
852    /// `O(b·W + log m)`: one hash evaluation to find the bucket, then a binary search of that bucket's `m` terms — one bucket is the whole search space, which is the lookup that per-bucket dedup buys.
853    pub fn get(&self, x: &[u64; W], z: &[u64; W]) -> Option<Complex64> {
854        let cols = &self.buckets[self.hash.bucket_of(x, z) as usize];
855        let mut lo = 0usize;
856        let mut hi = cols.len();
857        while lo < hi {
858            let mid = lo + (hi - lo) / 2;
859            match (&cols.x[mid], &cols.z[mid]).cmp(&(x, z)) {
860                std::cmp::Ordering::Less => lo = mid + 1,
861                std::cmp::Ordering::Greater => hi = mid,
862                std::cmp::Ordering::Equal => return Some(cols.coeff[mid]),
863            }
864        }
865        None
866    }

886    /// Keep only the terms for which `f(x, z, coeff)` is `true`.
887    /// Order-preserving and in place, so the per-bucket sort and no-duplicates invariant both survive; a term never changes bucket, so the hash invariant does too. `f` may run on several threads at once, hence the `Sync` bound.
888    pub fn retain(&mut self, f: impl Fn(&[u64; W], &[u64; W], Complex64) -> bool + Sync) {
889        self.buckets.par_iter_mut().for_each(|cols| {
890            let n = cols.len();
891            let mut w = 0usize;
892            for r in 0..n {
893                if f(&cols.x[r], &cols.z[r], cols.coeff[r]) {
894                    if w != r {
895                        cols.x[w] = cols.x[r];
896                        cols.z[w] = cols.z[r];
897                        cols.coeff[w] = cols.coeff[r];
898                    }
899                    w += 1;
900                }
901            }
902            cols.x.truncate(w);
903            cols.z.truncate(w);
904            cols.coeff.truncate(w);
905        });
906        self.recount();
907    }

909    /// Hilbert-Schmidt overlap `tr(self† · other) / 2ⁿ`, i.e. `Σ conj(aᵢ)·bᵢ` over the keys the two sums share.
910    /// Equal keys always land in the same bucket under a shared hash, so this is `B` independent two-pointer merges, one per bucket, and no term is ever compared across buckets.
911    ///
912    /// # Summation order
913    ///
914    /// Each bucket's partial is accumulated in key order, and the partials then combined in ascending bucket index; different partitions of the same operands agree to within rounding, not bit for bit.
915    ///
916    /// # Panics
917    ///
918    /// Panics unless the two sums share a partition: same hash rows and the same bucket count. Combining sums under different partitions is [`Self::add`]'s job; overlap does not realign. Under the grow-only [`Self::rebucket`] policy, two sums can have equal `len()` but different bucket counts, so align them with [`Self::with_hash`] first if that is a possibility.
919    pub fn overlap(&self, other: &Self) -> Complex64 {
920        assert!(
921            self.hash.same_rows_as(&other.hash),
922            "PauliSum::overlap: hash mismatch (seed or num_qubits differs)",
923        );
924        assert_eq!(
925            self.hash.bits(),
926            other.hash.bits(),
927            "PauliSum::overlap: bucket count mismatch",
928        );
929        self.buckets
930            .par_iter()
931            .zip(other.buckets.par_iter())
932            .map(|(a, b)| {
933                let mut acc = Complex64::new(0.0, 0.0);
934                let (mut i, mut j) = (0usize, 0usize);
935                while i < a.len() && j < b.len() {
936                    match (&a.x[i], &a.z[i]).cmp(&(&b.x[j], &b.z[j])) {
937                        std::cmp::Ordering::Less => i += 1,
938                        std::cmp::Ordering::Greater => j += 1,
939                        std::cmp::Ordering::Equal => {
940                            acc += a.coeff[i].conj() * b.coeff[j];
941                            i += 1;
942                            j += 1;
943                        }
944                    }
945                }
946                acc
947            })
948            .collect::<Vec<_>>()
949            .into_iter()
950            .fold(Complex64::new(0.0, 0.0), |a, b| a + b)
951    }

953    /// Sum of two bucketed sums.
954    /// The left partition wins: the result is partitioned exactly as `self` is, and `other` is realigned onto that partition first. `self` and `other` are both left untouched.
955    /// Once aligned, equal keys sit in the same bucket index on both sides, so this is `B` independent two-pointer merges — no global sort, no cross-bucket comparison. Terms whose coefficients sum to exactly `0+0i` are dropped.
956    ///
957    /// # Summation order
958    ///
959    /// Each surviving coefficient is a single `self + other` addition, bit-identical to what the flat merge would produce. Only derived quantities that accumulate across terms (e.g. [`Self::overlap`]) see the bucket-order effect.
960    ///
961    /// # Panics
962    ///
963    /// Panics if the two sums disagree about `num_qubits`.
964    pub fn add(&self, other: &Self) -> Self {
965        assert_eq!(
966            self.num_qubits, other.num_qubits,
967            "PauliSum::add: num_qubits mismatch ({} vs {})",
968            self.num_qubits, other.num_qubits,
969        );
970        let rhs = other.align_to(&self.hash);
971        let buckets: Vec<BucketCols<W>> = self
972            .buckets
973            .par_iter()
974            .zip(rhs.buckets.par_iter())
975            .map(|(a, b)| merge_two_adding(a, b))
976            .collect();
977        let len = buckets.iter().map(|c| c.len()).sum();
978        Self {
979            buckets,
980            hash: self.hash.clone(),
981            num_qubits: self.num_qubits,
982            len,
983        }
984    }

1080    /// Low-weight keys — the physically relevant regime, and the one where a badly chosen hash would collapse into one bucket.
1081    fn rand_low_weight_sum<const W: usize>(
1082        n: usize,
1083        num_qubits: usize,
1084        weight: usize,
1085        seed: u64,
1086    ) -> PauliSum<W> {
1087        let mut rng = Xs64::new(seed);
1088        let mut acc = BuildAccumulator::<W>::with_capacity(num_qubits, n);
1089        for _ in 0..n {
1090            let mut p = PauliString::<W> {
1091                x: [0u64; W],
1092                z: [0u64; W],
1093            };
1094            for _ in 0..weight {
1095                let q = (rng.next_u64() as usize) % num_qubits;
1096                let bit = 1u64 << (q % 64);
1097                match rng.next_u64() % 3 {
1098                    0 => p.x[q / 64] |= bit,
1099                    1 => p.z[q / 64] |= bit,
1100                    _ => {
1101                        p.x[q / 64] |= bit;
1102                        p.z[q / 64] |= bit;
1103                    }
1104                }
1105            }
1106            let re = (rng.next_u64() as i64 as f64) / (i64::MAX as f64);
1107            acc.add_term(p, Phase::ONE, Complex64::new(re, 0.0));
1108        }
1109        acc.finalize()
1110    }

1732    /// Reference overlap: two-pointer over globally key-sorted triples — the accumulation order a single-bucket sum uses.
1733    fn flat_overlap<const W: usize>(a: &PauliSum<W>, b: &PauliSum<W>) -> Complex64 {
1734        let ta = sorted_triples(a);
1735        let tb = sorted_triples(b);
1736        let mut acc = Complex64::new(0.0, 0.0);
1737        let (mut i, mut j) = (0usize, 0usize);
1738        while i < ta.len() && j < tb.len() {
1739            match (ta[i].0, ta[i].1).cmp(&(tb[j].0, tb[j].1)) {
1740                std::cmp::Ordering::Less => i += 1,
1741                std::cmp::Ordering::Greater => j += 1,
1742                std::cmp::Ordering::Equal => {
1743                    acc += ta[i].2.conj() * tb[j].2;
1744                    i += 1;
1745                    j += 1;
1746                }
1747            }
1748        }
1749        acc
1750    }

2203    //   |0⟩ = Z+:  ⟨Z⟩ = +1,  ⟨X⟩ = ⟨Y⟩ = 0

2212    /// Per-qubit label string → [`ProductBasis`], in the alphabet the Python binding accepts (qiskit `Statevector.from_label`): `0`/`1` = Z±, `+`/`-` = X±, `r`/`l` = Y±. Character `i` addresses qubit `i`.

2214    fn basis_from_labels<const W: usize>(labels: &str) -> ProductBasis<W> {
2215        ProductBasis::<W>::from_axes(labels.chars().map(|ch| match ch {
2216            '0' => (PauliAxis::Z, false),
2217            '1' => (PauliAxis::Z, true),
2218            '+' => (PauliAxis::X, false),
2219            '-' => (PauliAxis::X, true),
2220            'r' => (PauliAxis::Y, false),
2221            'l' => (PauliAxis::Y, true),
2222            other => panic!("unexpected label {other:?}"),
2223        }))
2224    }

2226    /// Differential oracle: `⟨ψ|O|ψ⟩` evaluated one qubit at a time straight from the Bloch table above, sharing no code with the masked scan.
2227    fn naive_labelled_expectation<const W: usize>(sum: &PauliSum<W>, labels: &str) -> Complex64 {
2228        let mut total = Complex64::new(0.0, 0.0);
2229        for (x, z, c) in sum.iter() {
2230            let mut factor = 1.0f64;
2231            for (q, label) in labels.chars().enumerate() {
2232                let bx = (x[q / 64] >> (q % 64)) & 1 == 1;
2233                let bz = (z[q / 64] >> (q % 64)) & 1 == 1;
2234                factor *= match (bx, bz, label) {
2235                    (false, false, _) => 1.0, // identity factor: no constraint
2236                    (true, false, '+') => 1.0,
2237                    (true, false, '-') => -1.0,
2238                    (false, true, '0') => 1.0,
2239                    (false, true, '1') => -1.0,
2240                    (true, true, 'r') => 1.0,
2241                    (true, true, 'l') => -1.0,
2242                    _ => 0.0, // off-axis Pauli: zero overlap
2243                };
2244            }
2245            total += c * factor;
2246        }
2247        total
2248    }

2903    /// Scatter `sum` (rehashed to `bits`) into `1 << pbits` partitions, check each part is a well-formed single-rank sum on the same partition, and gather it back — which must reproduce the input bit for bit, in the same canonical order.
2904    fn check_split_merge<const W: usize>(
2905        sum: &PauliSum<W>,
2906        num_qubits: usize,
2907        bits: u8,
2908        pbits: u8,
2909        what: &str,
2910    ) {
2911        let s = sum
2912            .clone()
2913            .with_hash(Gf2Hash::<W>::new(num_qubits, bits, 0xBEEF));
2914        let rows = PartitionRows::<W>::from_seed(num_qubits, pbits, 0x5EED);
2915        let p = rows.num_partitions();
2916
2917        let mut parts: Vec<PauliSum<W>> = Vec::with_capacity(p);
2918        let mut total = 0usize;
2919        for r in 0..p as u32 {
2920            let part = s.filter_partition(&rows, r);
2921            part.assert_invariants();
2922            assert_eq!(
2923                part.num_buckets(),
2924                s.num_buckets(),
2925                "{what}: part {r} bucket count",
2926            );
2927            assert!(
2928                part.hash().same_rows_as(s.hash()),
2929                "{what}: part {r} hash rows",
2930            );
2931            assert_eq!(
2932                part.hash().bits(),
2933                s.hash().bits(),
2934                "{what}: part {r} hash bits",
2935            );
2936            assert_eq!(
2937                part.num_qubits(),
2938                s.num_qubits(),
2939                "{what}: part {r} num_qubits",
2940            );
2941            if !part.is_empty() {
2942                assert_eq!(
2943                    part.partition_rank_of_all(&rows),
2944                    Some(r),
2945                    "{what}: part {r} is not single-rank",
2946                );
2947            }
2948            total += part.len();
2949            parts.push(part);
2950        }
2951        assert_eq!(total, s.len(), "{what}: partition lengths must add up");
2952
2953        let back = PauliSum::<W>::merge_partitions(parts);
2954        back.assert_invariants();
2955        assert_eq!(
2956            back.num_buckets(),
2957            s.num_buckets(),
2958            "{what}: merged bucket count",
2959        );
2960        assert_same_columns(&back, &s, what);
2961    }
