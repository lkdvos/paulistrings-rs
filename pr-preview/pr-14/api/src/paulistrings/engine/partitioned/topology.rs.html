1//! CPU sets, NUMA node discovery, thread/memory pinning, and the pinned Rayon pool one partition runs on.
2//! See ARCHITECTURE.md §Partitioning.
3//!
4//! A partition owns one Rayon pool whose workers are pinned to one NUMA domain's CPUs and whose allocations are bound to that domain's memory, so work stealing stays inside a socket while first-touch of a partition's columns lands on the socket that reads them.
5//! Discovery is sysfs + `libc` (`sched_{get,set}affinity`, `sched_getcpu`, `set_mempolicy`); there is no hwloc or libnuma dependency.
6//! Every entry point compiles on non-Linux targets, where it reports one node covering `available_parallelism` CPUs and pins nothing.
7
8use std::fmt;
9use std::io;

11/// `log` target for topology events (pinning failures, overlapping explicit
12/// sets). Separate from the engine's progress target so a consumer can filter
13/// placement diagnostics on their own.
14const LOG_TARGET: &str = "paulistrings::partitioned";
15
16/// A set of logical CPU indices, kept sorted and deduplicated.
17///
18/// The tuple field is public so callers can build a set literally; the
19/// constructors ([`CpuSet::parse`], [`CpuSet::intersect`]) and every consumer
20/// in this module normalize, so an unsorted literal is tolerated rather than
21/// relied upon.
22#[derive(Clone, Debug, PartialEq, Eq)]
23pub struct CpuSet(
24    /// The CPU indices, ascending and unique.
25    pub Vec<usize>,
26);
27
28impl CpuSet {
29    /// Parses a Linux cpulist, the `"0-7,16-23"` form written by sysfs.
30    ///
31    /// Surrounding and per-token whitespace is trimmed and empty tokens (a
32    /// trailing comma, a trailing newline) are skipped.
33    ///
34    /// # Errors
35    ///
36    /// [`TopologyError::EmptySet`] if the list names no CPU at all, and
37    /// [`TopologyError::InvalidCpuList`] if a token is not a decimal index or
38    /// an ascending `lo-hi` range.
39    pub fn parse(s: &str) -> Result<Self, TopologyError> {
40        let invalid = || TopologyError::InvalidCpuList(s.trim().to_string());
41        let index = |token: &str| token.trim().parse::<usize>().map_err(|_| invalid());
42
43        let mut cpus = Vec::new();
44        for token in s.split(',') {
45            let token = token.trim();
46            if token.is_empty() {
47                continue;
48            }
49            match token.split_once('-') {
50                None => cpus.push(index(token)?),
51                Some((lo, hi)) => {
52                    let (lo, hi) = (index(lo)?, index(hi)?);
53                    if hi < lo {
54                        return Err(invalid());
55                    }
56                    cpus.extend(lo..=hi);
57                }
58            }
59        }
60        if cpus.is_empty() {
61            return Err(TopologyError::EmptySet);
62        }
63        cpus.sort_unstable();
64        cpus.dedup();
65        Ok(Self(cpus))
66    }
67
68    /// The number of distinct CPUs in the set.
69    #[must_use]
70    pub fn len(&self) -> usize {
71        self.normalized().len()
72    }
73
74    /// Whether the set names no CPU.
75    #[must_use]
76    pub fn is_empty(&self) -> bool {
77        self.0.is_empty()
78    }
79
80    /// The CPUs present in both sets.
81    #[must_use]
82    pub fn intersect(&self, other: &Self) -> Self {
83        let other = other.normalized();
84        Self(
85            self.normalized()
86                .into_iter()
87                .filter(|cpu| other.binary_search(cpu).is_ok())
88                .collect(),
89        )
90    }
91
92    /// Whether every CPU of `self` is also in `other`.
93    fn is_subset_of(&self, other: &Self) -> bool {
94        let other = other.normalized();
95        self.0.iter().all(|cpu| other.binary_search(cpu).is_ok())
96    }
97
98    /// A sorted, deduplicated copy of the indices.
99    fn normalized(&self) -> Vec<usize> {
100        let mut cpus = self.0.clone();
101        cpus.sort_unstable();
102        cpus.dedup();
103        cpus
104    }
105
106    /// The union of two sets.
107    fn union(&self, other: &Self) -> Self {
108        let mut cpus = self.normalized();
109        cpus.extend(other.normalized());
110        cpus.sort_unstable();
111        cpus.dedup();
112        Self(cpus)
113    }
114}

116impl fmt::Display for CpuSet {
117    /// Writes the sysfs cpulist form, compressing runs into `lo-hi` ranges.
118    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
119        let cpus = self.normalized();
120        let mut first = true;
121        let mut i = 0;
122        while i < cpus.len() {
123            let mut j = i;
124            while j + 1 < cpus.len() && cpus[j + 1] == cpus[j] + 1 {
125                j += 1;
126            }
127            if !first {
128                write!(f, ",")?;
129            }
130            first = false;
131            if i == j {
132                write!(f, "{}", cpus[i])?;
133            } else {
134                write!(f, "{}-{}", cpus[i], cpus[j])?;
135            }
136            i = j + 1;
137        }
138        Ok(())
139    }
140}

142/// The CPUs this process may run on: the current affinity mask.
143///
144/// Falls back to `0..available_parallelism` when the mask cannot be read and
145/// on non-Linux targets.
146#[must_use]
147pub(crate) fn allowed_cpus() -> CpuSet {
148    #[cfg(target_os = "linux")]
149    {
150        // SAFETY: `set` is a valid, correctly sized `cpu_set_t`; pid 0 is the
151        // calling thread.
152        unsafe {
153            let mut set: libc::cpu_set_t = std::mem::zeroed();
154            if libc::sched_getaffinity(0, std::mem::size_of::<libc::cpu_set_t>(), &mut set) == 0 {
155                let cpus: Vec<usize> = (0..libc::CPU_SETSIZE as usize)
156                    .filter(|&cpu| libc::CPU_ISSET(cpu, &set))
157                    .collect();
158                if !cpus.is_empty() {
159                    return CpuSet(cpus);
160                }
161            }
162        }
163    }
164    CpuSet((0..available_parallelism()).collect())
165}

167/// The NUMA nodes visible in the current affinity mask, ascending by node id.
168///
169/// Each node's CPU set is intersected with the process's affinity mask, and
170/// nodes left empty by that intersection are dropped. When sysfs exposes no
171/// usable node — a kernel without NUMA, a sandbox without `/sys`, a non-Linux
172/// target — the whole affinity mask is reported as node 0.
173#[must_use]
174pub fn numa_nodes() -> Vec<(usize, CpuSet)> {
175    let allowed = allowed_cpus();
176    let mut nodes = sysfs_numa_nodes(&allowed);
177    nodes.sort_unstable_by_key(|(id, _)| *id);
178    if nodes.is_empty() {
179        vec![(0, allowed)]
180    } else {
181        nodes
182    }
183}
184
185#[cfg(target_os = "linux")]
186fn sysfs_numa_nodes(allowed: &CpuSet) -> Vec<(usize, CpuSet)> {
187    const ROOT: &str = "/sys/devices/system/node";
188    let Ok(entries) = std::fs::read_dir(ROOT) else {
189        return Vec::new();
190    };
191    let mut nodes = Vec::new();
192    for entry in entries.flatten() {
193        let name = entry.file_name();
194        let Some(id) = name.to_str().and_then(|n| n.strip_prefix("node")) else {
195            continue;
196        };
197        let Ok(id) = id.parse::<usize>() else {
198            continue;
199        };
200        let Ok(list) = std::fs::read_to_string(entry.path().join("cpulist")) else {
201            continue;
202        };
203        let Ok(cpus) = CpuSet::parse(&list) else {
204            continue;
205        };
206        let cpus = cpus.intersect(allowed);
207        if !cpus.is_empty() {
208            nodes.push((id, cpus));
209        }
210    }
211    nodes
212}

214#[cfg(not(target_os = "linux"))]
215fn sysfs_numa_nodes(_allowed: &CpuSet) -> Vec<(usize, CpuSet)> {
216    Vec::new()

219/// Pins the calling thread to `set`.
220///
221/// A no-op returning `Ok(())` on non-Linux targets.
222///
223/// # Errors
224///
225/// [`io::ErrorKind::InvalidInput`] if the set is empty or names a CPU at or
226/// beyond the kernel's `CPU_SETSIZE`; otherwise the `sched_setaffinity` error.
227pub(crate) fn pin_current_thread(set: &CpuSet) -> io::Result<()> {
228    #[cfg(target_os = "linux")]
229    {
230        if set.is_empty() {
231            return Err(io::Error::new(
232                io::ErrorKind::InvalidInput,
233                "cannot pin a thread to an empty CPU set",
234            ));
235        }
236        // SAFETY: `mask` is a valid, zeroed `cpu_set_t`; every index is
237        // checked against `CPU_SETSIZE` before `CPU_SET` touches it.
238        unsafe {
239            let mut mask: libc::cpu_set_t = std::mem::zeroed();
240            for &cpu in &set.0 {
241                if cpu >= libc::CPU_SETSIZE as usize {
242                    return Err(io::Error::new(
243                        io::ErrorKind::InvalidInput,
244                        format!("CPU {cpu} is beyond CPU_SETSIZE"),
245                    ));
246                }
247                libc::CPU_SET(cpu, &mut mask);
248            }
249            if libc::sched_setaffinity(0, std::mem::size_of::<libc::cpu_set_t>(), &mask) != 0 {
250                return Err(io::Error::last_os_error());
251            }
252        }
253    }
254    #[cfg(not(target_os = "linux"))]
255    let _ = set;
256    Ok(())
257}

259/// Binds the calling thread's allocations to one NUMA node, or restores the default policy.
260///
261/// `Some(node)` installs `MPOL_BIND` on that node alone, so pages this thread first-touches come from its own domain; `None` restores `MPOL_DEFAULT`.
262/// A no-op returning `Ok(())` on non-Linux targets.
263///
264/// # Errors
265///
266/// The `set_mempolicy` error — notably [`io::ErrorKind::Unsupported`] on a
267/// kernel built without NUMA support.
268pub(crate) fn bind_current_thread_memory(node: Option<usize>) -> io::Result<()> {
269    #[cfg(target_os = "linux")]
270    {
271        // `libc` exposes no `set_mempolicy` wrapper, so go through `syscall`.
272        const BITS: usize = 8 * std::mem::size_of::<libc::c_ulong>();
273        let rc = match node {
274            Some(node) => {
275                let words = node / BITS + 1;
276                let mut mask = vec![0 as libc::c_ulong; words];
277                mask[node / BITS] = 1 << (node % BITS);
278                // SAFETY: `mask` outlives the call and holds `words` words,
279                // which is exactly the `maxnode` bits the kernel reads.
280                unsafe {
281                    libc::syscall(
282                        libc::SYS_set_mempolicy,
283                        libc::MPOL_BIND,
284                        mask.as_ptr(),
285                        (words * BITS) as libc::c_ulong,
286                    )
287                }
288            }
289            // SAFETY: `MPOL_DEFAULT` takes a null mask of zero length.
290            None => unsafe {
291                libc::syscall(
292                    libc::SYS_set_mempolicy,
293                    libc::MPOL_DEFAULT,
294                    std::ptr::null::<libc::c_ulong>(),
295                    0 as libc::c_ulong,
296                )
297            },
298        };
299        if rc != 0 {
300            return Err(io::Error::last_os_error());
301        }
302    }
303    #[cfg(not(target_os = "linux"))]
304    let _ = node;
305    Ok(())
306}

308/// The CPU the calling thread is running on right now, if the platform can say. `None` on non-Linux targets.
309///
310/// Only [`build_pool`]'s own test asks; the engine pins and then trusts the kernel.
311#[cfg(test)]
312fn current_cpu() -> Option<usize> {
313    #[cfg(target_os = "linux")]
314    {
315        // SAFETY: `sched_getcpu` takes no arguments and cannot fail beyond
316        // returning a negative value.
317        let cpu = unsafe { libc::sched_getcpu() };
318        usize::try_from(cpu).ok()
319    }
320    #[cfg(not(target_os = "linux"))]
321    {
322        None
323    }
324}

326/// One resolved partition: where its pool's workers run and how many there
327/// are.
328#[derive(Clone, Debug)]
329pub struct PartitionSlot {
330    /// CPUs to pin the pool's workers to, or `None` to leave them unpinned.
331    pub cpus: Option<CpuSet>,
332    /// NUMA node to bind the workers' allocations to, when the slot sits in
333    /// exactly one node.
334    pub node: Option<usize>,
335    /// Worker count for the slot's Rayon pool.
336    pub threads: usize,
337}
338
339/// Builds the Rayon pool for one partition, pinning each worker to the slot's CPUs and (when `bind_memory`) binding its allocations to the slot's node.
340///
341/// Pinning happens on the worker itself, before it enters Rayon's main loop, so the pool's own stacks and per-worker allocations are first-touched locally.
342/// A worker whose pinning call fails logs a warning and runs unpinned: a placement failure degrades locality, it does not stop the run.
343///
344/// # Errors
345///
346/// [`TopologyError::Io`] wrapping the spawn or pool-build failure.
347pub(crate) fn build_pool(
348    slot: &PartitionSlot,
349    bind_memory: bool,
350    name: &str,
351) -> Result<rayon::ThreadPool, TopologyError> {
352    let cpus = slot.cpus.clone();
353    let node = slot.node;
354    let pool_name = name.to_string();
355    rayon::ThreadPoolBuilder::new()
356        .num_threads(slot.threads)
357        .thread_name(move |index| format!("{pool_name}-{index}"))
358        .spawn_handler(move |thread| {
359            let cpus = cpus.clone();
360            let mut builder = std::thread::Builder::new();
361            if let Some(name) = thread.name() {
362                builder = builder.name(name.to_string());
363            }
364            if let Some(size) = thread.stack_size() {
365                builder = builder.stack_size(size);
366            }
367            builder.spawn(move || {
368                if let Some(cpus) = &cpus {
369                    if let Err(err) = pin_current_thread(cpus) {
370                        log::warn!(target: LOG_TARGET, "failed to pin worker to {cpus}: {err}");
371                    }
372                }
373                if bind_memory {
374                    if let Err(err) = bind_current_thread_memory(node) {
375                        log::warn!(target: LOG_TARGET, "failed to bind worker memory to {node:?}: {err}");
376                    }
377                }
378                thread.run();
379            })?;
380            Ok(())
381        })
382        .build()
383        .map_err(|err| TopologyError::Io(io::Error::other(err.to_string())))
384}

386/// How partitions map onto the machine.
387#[derive(Clone, Debug, PartialEq, Eq)]
388pub enum Placement {
389    /// One partition per NUMA node in the affinity mask.
390    ///
391    /// The partition count is the node count rounded **down** to a power of two (at least one); when that is fewer than the node count, adjacent nodes are merged into a partition rather than dropped, so every allowed CPU stays in play.
392    /// Each partition takes as many threads as it has CPUs.
393    Auto {
394        /// Upper bound on the partition count, itself rounded down to a power
395        /// of two. `None` for no bound.
396        max_partitions: Option<usize>,
397    },
398    /// One partition per listed CPU set, exactly as given.
399    ///
400    /// Sets may overlap (the resolve logs a warning and proceeds), but each must be non-empty and a subset of the process's affinity mask.
401    Explicit(
402        /// The CPU sets, one per partition; the count must be a power of two.
403        Vec<CpuSet>,
404    ),
405    /// Partitions with no pinning at all: the shape of a partitioned run
406    /// without its placement, for tests and CI.
407    Unpinned {
408        /// Number of partitions; must be a power of two.
409        partitions: usize,
410        /// Threads per partition, or `None` to split `available_parallelism`
411        /// evenly (at least one thread each).
412        threads_per_partition: Option<usize>,
413    },
414}
415
416/// Placement plus the knobs the partitioned engine reads alongside it.
417#[derive(Clone, Debug, PartialEq, Eq)]
418pub struct PartitionConfig {
419    /// How partitions map onto the machine.
420    pub placement: Placement,
421    /// Whether each pool's workers bind their allocations to the partition's
422    /// NUMA node (`MPOL_BIND`, through `set_mempolicy`).
423    ///
424    /// The Python surface spells this `pin_memory=`, and the probe's JSON
425    /// sidecar follows the Python name; everything on the Rust side is
426    /// `bind_memory`, because it is a memory policy and not a thread affinity.
427    pub bind_memory: bool,
428    /// Seed selecting which GF(2) hash rows designate the partition, or
429    /// `None` to take the engine's default choice.
430    pub partition_row_seed: Option<u64>,
431}
432
433impl Default for PartitionConfig {
434    /// One partition per NUMA node, memory bound to it, default partition
435    /// rows.
436    fn default() -> Self {
437        Self {
438            placement: Placement::Auto {
439                max_partitions: None,
440            },
441            bind_memory: true,
442            partition_row_seed: None,
443        }
444    }
445}
446
447impl PartitionConfig {
448    /// Resolves the placement against the machine into one slot per partition.
449    ///
450    /// The slot count is always a power of two, which is what makes a partition index a fixed set of GF(2) hash rows.
451    ///
452    /// # Errors
453    ///
454    /// [`TopologyError::NotPowerOfTwo`] if an explicit or unpinned partition
455    /// count is not a power of two, [`TopologyError::EmptySet`] if an explicit
456    /// set is empty, and [`TopologyError::CpuNotAllowed`] if an explicit set
457    /// names a CPU outside the process's affinity mask.
458    pub fn resolve(&self) -> Result<Vec<PartitionSlot>, TopologyError> {
459        match &self.placement {
460            Placement::Auto { max_partitions } => Ok(resolve_auto(*max_partitions)),
461            Placement::Explicit(sets) => resolve_explicit(sets),
462            Placement::Unpinned {
463                partitions,
464                threads_per_partition,
465            } => resolve_unpinned(*partitions, *threads_per_partition),
466        }
467    }
468}
469
470fn resolve_auto(max_partitions: Option<usize>) -> Vec<PartitionSlot> {
471    let nodes = numa_nodes();
472    let mut count = prev_power_of_two(nodes.len());
473    if let Some(max) = max_partitions {
474        count = prev_power_of_two(count.min(max));
475    }
476
477    // Merge adjacent nodes when the count was rounded down, so no allowed CPU is left out of the run: the first `nodes.len() % count` partitions take one node more than the rest.
478    let (base, remainder) = (nodes.len() / count, nodes.len() % count);
479    let mut slots = Vec::with_capacity(count);
480    let mut rest = nodes.as_slice();
481    for partition in 0..count {
482        let take = base + usize::from(partition < remainder);
483        let (group, tail) = rest.split_at(take);
484        rest = tail;
485        let cpus = group
486            .iter()
487            .fold(CpuSet(Vec::new()), |acc, (_, set)| acc.union(set));
488        slots.push(PartitionSlot {
489            node: (group.len() == 1).then(|| group[0].0),
490            threads: cpus.len(),
491            cpus: Some(cpus),
492        });
493    }
494    slots
495}
496
497fn resolve_explicit(sets: &[CpuSet]) -> Result<Vec<PartitionSlot>, TopologyError> {
498    if !sets.len().is_power_of_two() {
499        return Err(TopologyError::NotPowerOfTwo(sets.len()));
500    }
501    let allowed = allowed_cpus();
502    let nodes = numa_nodes();
503
504    let mut slots = Vec::with_capacity(sets.len());
505    for set in sets {
506        if set.is_empty() {
507            return Err(TopologyError::EmptySet);
508        }
509        let cpus = CpuSet(set.normalized());
510        if let Some(&cpu) = cpus.0.iter().find(|cpu| !allowed.0.contains(cpu)) {
511            return Err(TopologyError::CpuNotAllowed(cpu));
512        }
513        let mut containing = nodes
514            .iter()
515            .filter(|(_, node)| cpus.is_subset_of(node))
516            .map(|(id, _)| *id);
517        let node = match (containing.next(), containing.next()) {
518            (Some(id), None) => Some(id),
519            _ => None,
520        };
521        slots.push(PartitionSlot {
522            node,
523            threads: cpus.len(),
524            cpus: Some(cpus),
525        });
526    }
527
528    for (i, slot) in slots.iter().enumerate() {
529        for other in slots.iter().skip(i + 1) {
530            let (a, b) = (slot.cpus.as_ref(), other.cpus.as_ref());
531            if let (Some(a), Some(b)) = (a, b) {
532                let shared = a.intersect(b);
533                if !shared.is_empty() {
534                    log::warn!(
535                        target: LOG_TARGET,
536                        "explicit partitions {a} and {b} overlap on {shared}; \
537                         their pools will contend for those CPUs",
538                    );
539                }
540            }
541        }
542    }
543    Ok(slots)
544}
545
546fn resolve_unpinned(
547    partitions: usize,
548    threads_per_partition: Option<usize>,
549) -> Result<Vec<PartitionSlot>, TopologyError> {
550    if !partitions.is_power_of_two() {
551        return Err(TopologyError::NotPowerOfTwo(partitions));
552    }
553    let threads = threads_per_partition
554        .unwrap_or(available_parallelism() / partitions)
555        .max(1);
556    Ok(vec![
557        PartitionSlot {
558            cpus: None,
559            node: None,
560            threads,
561        };
562        partitions
563    ])
564}

566/// The largest power of two `<= n`, and 1 for `n == 0`.
567fn prev_power_of_two(n: usize) -> usize {
568    if n <= 1 {
569        1
570    } else {
571        1 << (usize::BITS - 1 - n.leading_zeros())
572    }
573}
574
575fn available_parallelism() -> usize {
576    std::thread::available_parallelism().map_or(1, std::num::NonZeroUsize::get)
577}

579/// What can go wrong discovering topology or resolving a [`PartitionConfig`].
580#[derive(Debug)]
581pub enum TopologyError {
582    /// A cpulist string was not the `"0-7,16-23"` form.
583    InvalidCpuList(
584        /// The offending list, trimmed.
585        String,
586    ),
587    /// A partition count was not a power of two.
588    NotPowerOfTwo(
589        /// The rejected count.
590        usize,
591    ),
592    /// A CPU set was empty where a partition needs at least one CPU.
593    EmptySet,
594    /// A syscall or sysfs read failed, or a pool could not be built.
595    Io(
596        /// The underlying error.
597        io::Error,
598    ),
599    /// An explicit set named a CPU outside the process's affinity mask.
600    CpuNotAllowed(
601        /// The CPU index that is outside the process's affinity mask.
602        usize,
603    ),
604}
605
606impl fmt::Display for TopologyError {
607    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
608        match self {
609            Self::InvalidCpuList(list) => {
610                write!(f, "{list:?} is not a cpulist of the form \"0-7,16-23\"")
611            }
612            Self::NotPowerOfTwo(count) => write!(
613                f,
614                "partition count {count} is not a power of two; a partition index is a fixed \
615                 set of GF(2) hash rows",
616            ),
617            Self::EmptySet => write!(f, "a partition needs a non-empty CPU set"),
618            Self::Io(err) => write!(f, "topology I/O failed: {err}"),
619            Self::CpuNotAllowed(cpu) => {
620                write!(f, "CPU {cpu} is outside this process's affinity mask",)
621            }
622        }
623    }
624}

626impl std::error::Error for TopologyError {
627    fn source(&self) -> Option<&(dyn std::error::Error + 'static)> {
628        match self {
629            Self::Io(err) => Some(err),
630            _ => None,
