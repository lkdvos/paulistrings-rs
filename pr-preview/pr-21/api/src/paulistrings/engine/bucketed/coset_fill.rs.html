16/// One coset task's working set: the swapped-out input columns and the per-output-member gather runs.
17#[derive(Clone, Debug, Default)]
18pub(in crate::engine) struct CosetScratch<const W: usize> {
19    /// The coset's input columns, swapped with the live bucket slots so the layer runs in place.
20    pub(super) old: Vec<BucketColumns<W>>,
21    /// Per-output-member gather runs.
22    pub(super) runs: Vec<GatherRun<W>>,
23    /// The per-run sort's scratch.
24    pub(super) sort: SortScratch<W>,
25    /// This slot's busy-time counters.
26    #[cfg(feature = "phase-timing")]
27    pub(super) stats: CosetStats,
28}

30/// One output member's gather run, split into the identity stream and the rest.
31#[derive(Clone, Debug, Default)]
32pub(in crate::engine) struct GatherRun<const W: usize> {
33    /// The identity-delta stream, already sorted; under a dense identity only `coeff` is filled.
34    pub(super) identity: BucketColumns<W>,
35    /// Every other delta's rows, unsorted until the per-run sort.
36    pub(super) rest: BucketColumns<W>,
37}
38
39/// `value != ZERO`, without the `||` short-circuit — one flag, no branch.
40pub(super) fn nonzero(value: Complex64) -> bool {
41    (value.re != 0.0) | (value.im != 0.0)
42}
43
44impl<const W: usize> GatherRun<W> {
45    pub(super) fn reset(
46        &mut self,
47        id_key_capacity: usize,
48        id_coeff_capacity: usize,
49        rest_capacity: usize,
50    ) {
51        self.identity.clear();
52        self.rest.clear();
53        // One slot past the exact capacity: `append_if` writes a discarded row at `len` before deciding not to publish it.
54        self.identity.x.reserve(id_key_capacity + 1);
55        self.identity.z.reserve(id_key_capacity + 1);
56        self.identity.coeff.reserve(id_coeff_capacity + 1);
57        self.rest.x.reserve(rest_capacity + 1);
58        self.rest.z.reserve(rest_capacity + 1);
59        self.rest.coeff.reserve(rest_capacity + 1);
60    }
61
62    /// Write one row at `len` and publish it iff `keep`.
63    ///
64    /// # Safety
65    ///
66    /// `len < capacity` on each of the three columns, which [`GatherRun::reset`] guarantees.
67    unsafe fn append_if(
68        columns: &mut BucketColumns<W>,
69        keep: bool,
70        x: [u64; W],
71        z: [u64; W],
72        c: Complex64,
73    ) {
74        let BucketColumns {
75            x: xs,
76            z: zs,
77            coeff: cs,
78        } = columns;
79        let len = xs.len();
80        debug_assert!(len < xs.capacity());
81        debug_assert!(len < zs.capacity());
82        debug_assert!(len < cs.capacity());
83        unsafe {
84            xs.as_mut_ptr().add(len).write(x);
85            zs.as_mut_ptr().add(len).write(z);
86            cs.as_mut_ptr().add(len).write(c);
87            let published = len + keep as usize;
88            xs.set_len(published);
89            zs.set_len(published);
90            cs.set_len(published);
91        }
92    }
93
94    /// Branchless filtered append to the rest stream.
95    pub(super) fn push_if(&mut self, keep: bool, x: [u64; W], z: [u64; W], c: Complex64) {
96        // SAFETY: `reset` reserved `rest_capacity + 1` on all three columns, and `rest_capacity` bounds every row the plan's deltas can emit into this run.
97        unsafe { Self::append_if(&mut self.rest, keep, x, z, c) }
98    }
99
100    /// Unconditional append to the rest stream.
101    pub(super) fn push_row(&mut self, x: [u64; W], z: [u64; W], c: Complex64) {
102        self.push_if(true, x, z, c);
103    }
104
105    /// Branchless filtered append to the identity stream.
106    pub(super) fn push_id_if(&mut self, keep: bool, x: [u64; W], z: [u64; W], c: Complex64) {
107        // SAFETY: as `push_if`, with `id_key_capacity` / `id_coeff_capacity` bounding the identity stream's rows.
108        unsafe { Self::append_if(&mut self.identity, keep, x, z, c) }
109    }
110
111    /// Unchecked append to the identity coefficient column, the only one a dense identity materializes.
112    pub(super) fn push_id_coeff(&mut self, c: Complex64) {
113        let coeff = &mut self.identity.coeff;
114        let len = coeff.len();
115        debug_assert!(len < coeff.capacity());
116        // SAFETY: `reset` reserved `id_coeff_capacity + 1`, and `id_coeff_capacity` is the source bucket's length, exactly the number of calls made.
117        unsafe {
118            coeff.as_mut_ptr().add(len).write(c);
119            coeff.set_len(len + 1);
120        }
121    }
122
123    #[cfg(any(test, feature = "phase-timing"))]
124    pub(super) fn len(&self) -> usize {
125        self.identity.len() + self.rest.len()
126    }
127}

132/// Gather, sort and merge one coset's `2^r` bucket columns in place; `chunk_base` and `inverse_permutation` name each member's original bucket for `extra`.
133pub(in crate::engine) fn fill_coset<const W: usize, T, X>(
134    chunk: &mut [BucketColumns<W>],
135    plan: &DeltaPlan<'_, W>,
136    policy: &T,
137    scratch: &mut CosetScratch<W>,
138    extra: &X,
139    chunk_base: usize,
140    inverse_permutation: &[u32],
141) where
142    T: TruncationPolicy<W> + ?Sized,
143    X: ExtraRows<W>,
144{
145    let members = chunk.len();
146    let beta_of = |j: usize| -> u32 {
147        if inverse_permutation.is_empty() {
148            (chunk_base + j) as u32
149        } else {
150            inverse_permutation[chunk_base + j]
151        }
152    };
153    #[cfg(feature = "phase-timing")]
154    let CosetScratch {
155        old,
156        runs,
157        sort,
158        stats,
159    } = scratch;
160    #[cfg(not(feature = "phase-timing"))]
161    let CosetScratch { old, runs, sort } = scratch;
162    #[cfg(feature = "phase-timing")]
163    let mut stamp = Stamp::now();
164    old.resize_with(members, BucketColumns::default);
165    runs.resize_with(members, GatherRun::default);
166
167    for (slot, columns) in chunk.iter_mut().zip(old.iter_mut()) {
168        std::mem::swap(slot, columns);
169        slot.clear();
170    }
171    #[cfg(feature = "phase-timing")]
172    stamp.lap(&mut stats.swap_ns);
173
174    // Exact per-run capacity per stream; it is also the bound `append_if`'s safety relies on.
175    for (j, run) in runs.iter_mut().enumerate() {
176        let (id_key_capacity, id_coeff_capacity, mut rest_capacity): (usize, usize, usize) =
177            match plan {
178                DeltaPlan::Local {
179                    coordinates,
180                    has_identity,
181                    dense_identity,
182                    ..
183                } => {
184                    let mut id = 0usize;
185                    let mut rest = 0usize;
186                    for (entry, &coordinate) in coordinates.iter().enumerate() {
187                        let len = old[j ^ coordinate as usize].len();
188                        if *has_identity && entry == 0 {
189                            id += len;
190                        } else {
191                            rest += len;
192                        }
193                    }
194                    (if *dense_identity { 0 } else { id }, id, rest)
195                }
196                DeltaPlan::Rotation {
197                    coordinate_identity,
198                    generator_coordinate,
199                    generator_local,
200                    ..
201                } => (
202                    0,
203                    old[j ^ *coordinate_identity as usize].len(),
204                    if *generator_local {
205                        old[j ^ *generator_coordinate as usize].len()
206                    } else {
207                        0
208                    },
209                ),
210            };
211        if X::NEEDS_BETA {
212            rest_capacity += extra.count(beta_of(j));
213        }
214        run.reset(id_key_capacity, id_coeff_capacity, rest_capacity);
215    }
216    #[cfg(feature = "phase-timing")]
217    stamp.lap(&mut stats.size_ns);
218
219    match plan {
220        DeltaPlan::Local {
221            ptm,
222            coordinates,
223            has_identity,
224            dense_identity,
225            ..
226        } => {
227            if members >= 1 << GATHER_OUTPUT_MAJOR_MIN_R {
228                gather_local_output_major(
229                    old,
230                    runs,
231                    ptm,
232                    coordinates,
233                    *has_identity,
234                    *dense_identity,
235                );
236            } else {
237                gather_local_input_major(
238                    old,
239                    runs,
240                    ptm,
241                    coordinates,
242                    *has_identity,
243                    *dense_identity,
244                );
245            }
246        }
247        DeltaPlan::Rotation {
248            rotation,
249            coordinate_identity,
250            generator_coordinate,
251            generator_local,
252        } => {
253            // Every term emits one id row, kept even when `cos == 0` (the signed-zero contract on `merge_identity_and_rest`), so the merge borrows the source keys.
254            for (i, source) in old.iter().enumerate() {
255                for row in 0..source.len() {
256                    let key = PauliString::<W> {
257                        x: source.x[row],
258                        z: source.z[row],
259                    };
260                    if key.commutes_with(&rotation.generator) {
261                        runs[i ^ *coordinate_identity as usize].push_id_coeff(source.coeff[row]);
262                    } else {
263                        runs[i ^ *coordinate_identity as usize]
264                            .push_id_coeff(source.coeff[row] * rotation.cos);
265                        if *generator_local {
266                            let mut product = key;
267                            let phase = product.mul_assign(&rotation.generator);
268                            let total = Phase::I + phase;
269                            runs[i ^ *generator_coordinate as usize].push_row(
270                                product.x,
271                                product.z,
272                                total.apply(source.coeff[row]) * rotation.sin,
273                            );
274                        }
275                    }
276                }
277            }
278        }
279    }
280    if X::NEEDS_BETA {
281        for (j, run) in runs.iter_mut().enumerate() {
282            let rest = &mut run.rest;
283            extra.append_into(beta_of(j), &mut rest.x, &mut rest.z, &mut rest.coeff);
284        }
285    }
286    #[cfg(feature = "phase-timing")]
287    stamp.lap(&mut stats.gather_ns);
288
289    let sort_kernel = match plan {
290        DeltaPlan::Local { sort_kernel, .. } => *sort_kernel,
291        DeltaPlan::Rotation { .. } => SortKernel::Comparison,
292    };
293    for (j, (run, destination)) in runs.iter_mut().zip(chunk.iter_mut()).enumerate() {
294        #[cfg(feature = "phase-timing")]
295        {
296            stats.rows_gathered += run.len() as u64;
297            stats.rows_sorted += run.rest.len() as u64;
298        }
299        sort_kernel.sort(&mut run.rest, sort);
300        #[cfg(feature = "phase-timing")]
301        stamp.lap(&mut stats.sort_ns);
302        let (a_x, a_z): (&[[u64; W]], &[[u64; W]]) = match plan {
303            DeltaPlan::Local {
304                dense_identity: true,
305                ..
306            } => {
307                let source = &old[j];
308                debug_assert_eq!(source.len(), run.identity.len());
309                #[cfg(feature = "phase-timing")]
310                {
311                    stats.rows_id += source.len() as u64;
312                }
313                (&source.x, &source.z)
314            }
315            DeltaPlan::Local { .. } => (&run.identity.x, &run.identity.z),
316            DeltaPlan::Rotation {
317                coordinate_identity,
318                ..
319            } => {
320                let source = &old[j ^ *coordinate_identity as usize];
321                debug_assert_eq!(source.len(), run.identity.len());
322                #[cfg(feature = "phase-timing")]
323                {
324                    stats.rows_id += source.len() as u64;
325                }
326                (&source.x, &source.z)
327            }
328        };
329        let identity = ColumnsRef {
330            x: a_x,
331            z: a_z,
332            coeff: &run.identity.coeff,
333        };
334        merge_identity_and_rest::<W, T>(identity, run.rest.view(), destination, policy);
335        #[cfg(feature = "phase-timing")]
336        stamp.lap(&mut stats.merge_ns);
337    }
338
339    for columns in old.iter_mut() {
340        columns.clear();
341    }
342    #[cfg(feature = "phase-timing")]
343    {
344        stamp.lap(&mut stats.clear_ns);
345        stats.cosets += 1;
346        stats.runs += members as u64;
347    }
348}

350/// Coset dimension at or above which the gather switches to output-major, trading re-reads for fewer open write streams.
351/// Both orders gather the same rows, so the choice is performance only; a dense two-qubit gate takes output-major at `r = 4`.
352const GATHER_OUTPUT_MAJOR_MIN_R: u8 = 3;
353
354/// Input-major gather for a `Local` plan: each term is loaded once and scattered by `member(i) ⊕ δ = member(i ⊕ coord(δ))`.
355// Not a branch on zero amplitudes: research/FINDINGS.md §Branch misprediction: merge loop split and branchless gather filter
356pub(super) fn gather_local_input_major<const W: usize>(
357    old: &[BucketColumns<W>],
358    runs: &mut [GatherRun<W>],
359    ptm: &LocalPtm<W>,
360    coordinates: &[u32],
361    has_identity: bool,
362    dense_identity: bool,
363) {
364    let rest_start = has_identity as usize;
365    for (i, source) in old.iter().enumerate() {
366        for row in 0..source.len() {
367            let pattern = ptm.support_bits(&source.x[row], &source.z[row]);
368            if has_identity {
369                let amplitude = ptm.deltas()[0].amplitude[pattern];
370                if dense_identity {
371                    debug_assert!(amplitude != ZERO);
372                    runs[i].push_id_coeff(source.coeff[row] * amplitude);
373                } else {
374                    runs[i].push_id_if(
375                        nonzero(amplitude),
376                        source.x[row],
377                        source.z[row],
378                        source.coeff[row] * amplitude,
379                    );
380                }
381            }
382            for (entry, delta) in ptm.deltas().iter().enumerate().skip(rest_start) {
383                let amplitude = delta.amplitude[pattern];
384                let mut key_x = source.x[row];
385                let mut key_z = source.z[row];
386                for w in 0..W {
387                    key_x[w] ^= delta.mask_x[w];
388                    key_z[w] ^= delta.mask_z[w];
389                }
390                runs[i ^ coordinates[entry] as usize].push_if(
391                    nonzero(amplitude),
392                    key_x,
393                    key_z,
394                    source.coeff[row] * amplitude,
395                );
396            }
397        }
398    }
399}

401/// Output-major gather for a `Local` plan: each output member streams one input bucket per delta.
402pub(super) fn gather_local_output_major<const W: usize>(
403    old: &[BucketColumns<W>],
404    runs: &mut [GatherRun<W>],
405    ptm: &LocalPtm<W>,
406    coordinates: &[u32],
407    has_identity: bool,
408    dense_identity: bool,
409) {
410    let rest_start = has_identity as usize;
411    for (j, run) in runs.iter_mut().enumerate() {
412        if has_identity {
413            let delta = &ptm.deltas()[0];
414            let source = &old[j];
415            for row in 0..source.len() {
416                let pattern = ptm.support_bits(&source.x[row], &source.z[row]);
417                let amplitude = delta.amplitude[pattern];
418                if dense_identity {
419                    debug_assert!(amplitude != ZERO);
420                    run.push_id_coeff(source.coeff[row] * amplitude);
421                } else {
422                    run.push_id_if(
423                        nonzero(amplitude),
424                        source.x[row],
425                        source.z[row],
426                        source.coeff[row] * amplitude,
427                    );
428                }
429            }
430        }
431        for (entry, delta) in ptm.deltas().iter().enumerate().skip(rest_start) {
432            let source = &old[j ^ coordinates[entry] as usize];
433            for row in 0..source.len() {
434                let pattern = ptm.support_bits(&source.x[row], &source.z[row]);
435                let amplitude = delta.amplitude[pattern];
436                let mut key_x = source.x[row];
437                let mut key_z = source.z[row];
438                for w in 0..W {
439                    key_x[w] ^= delta.mask_x[w];
440                    key_z[w] ^= delta.mask_z[w];
441                }
442                run.push_if(
443                    nonzero(amplitude),
444                    key_x,
445                    key_z,
446                    source.coeff[row] * amplitude,
447                );
448            }
449        }
450    }
