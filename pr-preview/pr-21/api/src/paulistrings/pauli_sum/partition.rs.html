9/// Copy the terms of one bucket whose partition rank is `rank`.
10// Not reserved up front: research/FINDINGS.md §Reserving a safe upper bound in the merge.
11fn filter_bucket<const W: usize>(
12    columns: &BucketColumns<W>,
13    rows: &PartitionRows<W>,
14    rank: u32,
15) -> BucketColumns<W> {
16    let mut out = BucketColumns::<W>::new();
17    for i in 0..columns.len() {
18        if rows.partition_of(&columns.x[i], &columns.z[i]) == rank {
19            out.push(columns.x[i], columns.z[i], columns.coeff[i]);
20        }
21    }
22    out
23}

25/// Merge key-disjoint sorted runs into one by serial pairwise rounds; the caller is already parallel over buckets.
26fn merge_disjoint_runs<const W: usize>(mut runs: Vec<BucketColumns<W>>) -> BucketColumns<W> {
27    while runs.len() > 1 {
28        let mut next: Vec<BucketColumns<W>> = Vec::with_capacity(runs.len().div_ceil(2));
29        let mut remaining = runs.into_iter();
30        while let Some(a) = remaining.next() {
31            match remaining.next() {
32                Some(b) => next.push(merge_two(&a, &b)),
33                None => next.push(a),
34            }
35        }
36        runs = next;
37    }
38    let out = runs.pop().unwrap_or_default();
39    #[cfg(debug_assertions)]
40    {
41        for i in 1..out.len() {
42            debug_assert!(
43                (&out.x[i - 1], &out.z[i - 1]) < (&out.x[i], &out.z[i]),
44                "PauliSum::merge_partitions: duplicate key across partitions",
45            );
46        }
47    }
48    out
49}

51impl<const W: usize> PauliSum<W> {
52    /// Coarsen until the bucket count is `1 << bits`; panics if that would grow it.
53    pub(crate) fn coarsen_to(&mut self, bits: u8) {
54        assert!(
55            bits <= self.hash.bits(),
56            "PauliSum::coarsen_to: {bits} exceeds the current {} bucket bits (use refine to grow)",
57            self.hash.bits(),
58        );
59        while self.hash.bits() > bits {
60            self.coarsen();
61        }
62    }

64    /// The terms `rows` assigns to partition `rank`, under the same hash and bucket count.
65    pub(crate) fn filter_partition(&self, rows: &PartitionRows<W>, rank: u32) -> Self {
66        debug_assert_eq!(
67            self.num_qubits,
68            rows.num_qubits(),
69            "PauliSum::filter_partition: num_qubits mismatch",
70        );
71        debug_assert!(
72            (rank as usize) < rows.num_partitions(),
73            "PauliSum::filter_partition: rank {rank} out of range",
74        );
75        let buckets: Vec<BucketColumns<W>> = if self.len < PARALLEL_MIN_TERMS {
76            self.buckets
77                .iter()
78                .map(|columns| filter_bucket(columns, rows, rank))
79                .collect()
80        } else {
81            self.buckets
82                .par_iter()
83                .map(|columns| filter_bucket(columns, rows, rank))
84                .collect()
85        };
86        let len = buckets.iter().map(|c| c.len()).sum();
87        Self {
88            buckets,
89            hash: self.hash.clone(),
90            num_qubits: self.num_qubits,
91            len,
92        }
93    }

95    /// Gather key-disjoint sums sharing hash rows and bucket count back into one; panics on an empty or misaligned input.
96    pub(crate) fn merge_partitions(parts: Vec<Self>) -> Self {
97        let first = parts
98            .first()
99            .expect("PauliSum::merge_partitions: no inputs to merge");
100        let hash = first.hash.clone();
101        let num_qubits = first.num_qubits;
102        for p in parts.iter().skip(1) {
103            assert!(
104                p.hash.same_rows_as(&hash),
105                "PauliSum::merge_partitions: hash mismatch (seed or num_qubits differs)",
106            );
107            assert_eq!(
108                p.hash.bits(),
109                hash.bits(),
110                "PauliSum::merge_partitions: bucket count mismatch",
111            );
112        }
113
114        let num_buckets = hash.num_buckets();
115        let num_parts = parts.len();
116        let len: usize = parts.iter().map(|p| p.len).sum();
117
118        let mut runs: Vec<Vec<BucketColumns<W>>> = (0..num_buckets)
119            .map(|_| Vec::with_capacity(num_parts))
120            .collect();
121        for p in parts {
122            for (b, columns) in p.buckets.into_iter().enumerate() {
123                runs[b].push(columns);
124            }
125        }
126
127        let buckets: Vec<BucketColumns<W>> = if len < PARALLEL_MIN_TERMS {
128            runs.into_iter().map(merge_disjoint_runs).collect()
129        } else {
130            runs.into_par_iter().map(merge_disjoint_runs).collect()
131        };
132
133        Self {
134            buckets,
135            hash,
136            num_qubits,
137            len,
138        }
139    }

141    /// Rebuild a sum from the columns [`Self::to_arrays`] produced, cut back into buckets by `lens`; nothing is sorted.
142    pub(crate) fn from_bucket_columns(
143        lens: &[usize],
144        x: Vec<[u64; W]>,
145        z: Vec<[u64; W]>,
146        coeff: Vec<Complex64>,
147        hash: Gf2Hash<W>,
148        num_qubits: usize,
149    ) -> Self {
150        assert_eq!(
151            lens.len(),
152            hash.num_buckets(),
153            "PauliSum::from_bucket_columns: {} bucket lengths for a hash with {} buckets",
154            lens.len(),
155            hash.num_buckets(),
156        );
157        assert!(
158            x.len() == coeff.len() && z.len() == coeff.len(),
159            "PauliSum::from_bucket_columns: columns are not parallel ({}, {}, {})",
160            x.len(),
161            z.len(),
162            coeff.len(),
163        );
164        let len: usize = lens.iter().sum();
165        assert_eq!(
166            len,
167            coeff.len(),
168            "PauliSum::from_bucket_columns: bucket lengths sum to {len}, columns hold {}",
169            coeff.len(),
170        );
171
172        let mut x = x.into_iter();
173        let mut z = z.into_iter();
174        let mut coeff = coeff.into_iter();
175        let buckets = lens
176            .iter()
177            .map(|&n| BucketColumns {
178                x: x.by_ref().take(n).collect(),
179                z: z.by_ref().take(n).collect(),
180                coeff: coeff.by_ref().take(n).collect(),
181            })
182            .collect();
183
184        Self {
185            buckets,
186            hash,
187            num_qubits,
188            len,
189        }
190    }

192    /// Wrap per-bucket columns that already satisfy the invariant, one entry per bucket of `hash`.
193    #[cfg(feature = "cuda")]
194    pub(crate) fn from_buckets(
195        buckets: Vec<BucketColumns<W>>,
196        hash: Gf2Hash<W>,
197        num_qubits: usize,
198    ) -> Self {
199        assert_eq!(
200            buckets.len(),
201            hash.num_buckets(),
202            "PauliSum::from_buckets: {} buckets for a hash with {} buckets",
203            buckets.len(),
204            hash.num_buckets(),
205        );
206        let len = buckets.iter().map(BucketColumns::len).sum();
207        Self {
208            buckets,
209            hash,
210            num_qubits,
211            len,
212        }
213    }

215    /// `Some(r)` if every term lies in partition `r`, `None` on an empty or a mixed sum.
216    pub(crate) fn partition_rank_of_all(&self, rows: &PartitionRows<W>) -> Option<u32> {
217        let mut seen: Option<u32> = None;
218        for (x, z, _) in self.iter() {
219            let r = rows.partition_of(x, z);
220            match seen {
221                None => seen = Some(r),
222                Some(s) if s == r => {}
223                Some(_) => return None,
224            }
225        }
226        seen
227    }
