1//! Operator products of a replicated generator with a partitioned operand, as collectives over a [`Transport`]: [`liouvillian_share`], [`product_share`] and [`anticommutator_share`].
2//!
3//! Partition rows are GF(2)-linear, so a generator term `h` takes an operand term from partition `p` to `p ⊕ R·h`, and rank `r` computes `Σ_δ H_δ · O_{r ⊕ δ}` with `H_δ` the generator terms of partition delta `δ`.
4//! Partner `r ⊕ δ` sends rank `r` the operand terms `H_δ` can reach, only the anticommuting ones for a Liouvillian, so operands cross the wire and products never do.

21/// The generator's terms keyed by partition delta, ascending.
22type ByDelta<const W: usize, C> = BTreeMap<u32, Vec<Entry<W, C>>>;

24/// Every rank must hold the same generator; folded over its keys and coefficient bytes in canonical order.
25fn generator_fingerprint<const W: usize, C: Coefficient>(generator: &PauliSum<W, C>) -> u64 {
26    let mut folded: u64 = 0xcbf2_9ce4_8422_2325;
27    let mut mix = |word: u64| folded = (folded ^ word).wrapping_mul(0x0000_0100_0000_01b3);
28    mix(generator.len() as u64);
29    for (x, z, c) in generator.iter() {
30        x.iter().chain(z).for_each(|&w| mix(w));
31        bytemuck::bytes_of(&c).chunks(8).for_each(|chunk| {
32            let mut word = [0u8; 8];
33            word[..chunk.len()].copy_from_slice(chunk);
34            mix(u64::from_le_bytes(word));
35        });
36    }
37    folded
38}

40/// The terms of bucket `b` that some term of `generator` anticommutes with, or every term without an `index`, in order.
41fn reachable_in_bucket<const W: usize, C: Coefficient>(
42    share: &PauliSum<W, C>,
43    b: usize,
44    index: Option<&SupportIndex>,
45    generator: &[Entry<W, C>],
46    candidates: &mut Vec<u64>,
47) -> (Vec<[u64; W]>, Vec<[u64; W]>, Vec<C>) {
48    let (bucket_x, bucket_z, bucket_coeff) = share.bucket(b);
49    let Some(index) = index else {
50        return (bucket_x.to_vec(), bucket_z.to_vec(), bucket_coeff.to_vec());
51    };
52    let mut reached = vec![0u64; bucket_x.len().div_ceil(64)];
53    for h in generator {
54        let key = PauliString { x: h.x, z: h.z };
55        let support: [u64; W] = std::array::from_fn(|w| h.x[w] | h.z[w]);
56        let weight: u32 = support.iter().map(|w| w.count_ones()).sum();
57        if weight <= SCAN_ABOVE_WEIGHT {
58            index.candidates(b, &support, candidates);
59        } else {
60            candidates.clear();
61            candidates.resize(reached.len(), u64::MAX);
62        }
63        for (w, &word) in candidates.iter().enumerate() {
64            let mut bits = word;
65            while bits != 0 {
66                let i = w * 64 + bits.trailing_zeros() as usize;
67                bits &= bits - 1;
68                if i < bucket_x.len() {
69                    let term = PauliString { x: bucket_x[i], z: bucket_z[i] };
70                    reached[w] |= (!key.commutes_with(&term) as u64) << (i % 64);
71                }
72            }
73        }
74    }
75    let kept = (0..bucket_x.len()).filter(|&i| reached[i / 64] >> (i % 64) & 1 == 1);
76    let mut out = (Vec::new(), Vec::new(), Vec::new());
77    for i in kept {
78        out.0.push(bucket_x[i]);
79        out.1.push(bucket_z[i]);
80        out.2.push(bucket_coeff[i]);
81    }
82    out
83}

85/// The terms of `share` that some term of `generator` anticommutes with, or every term without an `index`, in the gather's wire form, keeping every bucket's order.
86fn encode_reachable<const W: usize, C: Coefficient>(
87    share: &PauliSum<W, C>,
88    index: Option<&SupportIndex>,
89    generator: &[Entry<W, C>],
90) -> ByteParts {
91    let buckets: Vec<_> = (0..share.num_buckets())
92        .into_par_iter()
93        .map_init(Vec::new, |candidates, b| {
94            reachable_in_bucket(share, b, index, generator, candidates)
95        })
96        .collect();
97    let lens: Vec<u64> = buckets.iter().map(|(_, _, c)| c.len() as u64).collect();
98    let terms: usize = buckets.iter().map(|(_, _, c)| c.len()).sum();
99    let (mut x, mut z, mut coeff) =
100        (Vec::with_capacity(terms), Vec::with_capacity(terms), Vec::with_capacity(terms));
101    for (bx, bz, bc) in buckets {
102        x.extend(bx);
103        z.extend(bz);
104        coeff.extend(bc);
105    }
106    ByteParts(vec![
107        bytemuck::cast_slice(&lens).to_vec(),
108        bytemuck::cast_slice(x.as_flattened()).to_vec(),
109        bytemuck::cast_slice(z.as_flattened()).to_vec(),
110        bytemuck::cast_slice(&coeff).to_vec(),
111    ])
112}

114/// `dest ← alpha · pairing(generator, operand) + beta · dest` on this rank, where `share` is this rank's share of the operand under `rows`, `dest` holds only this rank's keys and `generator` is the same on every rank. **Collective.**
115pub(crate) fn pair_share_into<const W: usize, C, X, P>(
116    dest: &mut PauliSum<W, C>,
117    generator: &PauliSum<W, C>,
118    share: &PauliSum<W, C>,
119    rows: &PartitionRows<W>,
120    transport: &X,
121    (pairing, weights): (Pairing, Weights<C>),
122    policy: &P,
123) where
124    C: Coefficient,
125    X: Transport,
126    P: TruncationPolicy<W, C> + ?Sized,
127{
128    if pairing == Pairing::Product {
129        assert!(
130            !C::IS_REAL,
131            "product_share: a product is not Hermitian, so a real coefficient type ({}) cannot hold it; cast to a complex type first",
132            C::NAME,
133        );
134    }
135    let num_qubits = share.num_qubits();
136    assert!(
137        generator.num_qubits() == num_qubits && dest.num_qubits() == num_qubits,
138        "{pairing:?} of a partitioned operand: num_qubits mismatch (generator {}, share {num_qubits}, destination {})",
139        generator.num_qubits(),
140        dest.num_qubits(),
141    );
142    let size = transport.size() as usize;
143    rows.assert_splits(num_qubits, 1usize << group_bits(transport.size()));
144    assert_supports_partitions(policy, size);
145    transport.check_consistency(
146        identity_fingerprint(rows, share.hash()) ^ generator_fingerprint(generator).rotate_left(7),
147    );
148
149    let bits = transport.allreduce_max_u8(share.hash().bits());
150    let refined;
151    let share = if share.hash().bits() < bits {
152        let mut copy = share.clone();
153        while copy.hash().bits() < bits {
154            copy.refine();
155        }
156        refined = copy;
157        &refined
158    } else {
159        share
160    };
161
162    let mut by_delta: ByDelta<W, C> = BTreeMap::new();
163    for (x, z, c) in generator.iter() {
164        let c = c * weights.alpha;
165        by_delta.entry(rows.partition_of(x, z)).or_default().push(Entry { x: *x, z: *z, c });
166    }
167    let remote: Vec<u32> = by_delta.keys().copied().filter(|&delta| delta != 0).collect();
168
169    let rank = transport.rank();
170    // Built once: the sends filter by it and the local job reads by it.
171    let mut index = (pairing == Pairing::Liouvillian).then(|| SupportIndex::new(share));
172    // Every rank holds every delta, so each sends to exactly the partners it receives from, an empty payload included.
173    let mut send: Vec<Option<ByteParts>> = (0..size).map(|_| None).collect();
174    for &delta in &remote {
175        send[(rank ^ delta) as usize] =
176            Some(encode_reachable(share, index.as_ref(), &by_delta[&delta]));
177    }
178    let mut received = transport.exchange(send, &mut Vec::new());
179    let slices: Vec<(u32, PauliSum<W, C>)> = remote
180        .iter()
181        .map(|&delta| {
182            let from = (rank ^ delta) as usize;
183            let payload = received[from]
184                .take()
185                .unwrap_or_else(|| panic!("{pairing:?}: rank {from} sent nothing"));
186            (delta, decode_rank(&payload.0, share.hash().clone(), num_qubits, from))
187        })
188        .collect();
189
190    let mut jobs = Vec::with_capacity(slices.len() + 1);
191    if let Some(local) = by_delta.get(&0) {
192        jobs.push(Job::with_index(local.iter().copied(), share, index.take(), true));
193    }
194    for (delta, slice) in &slices {
195        jobs.push(Job::new(by_delta[delta].iter().copied(), slice, pairing, true));
196    }
197    dest.align_destination(share.hash(), weights.beta);
198    accumulate_into(dest, weights.beta, &jobs, pairing, &|x, z, c| policy.keep_term(x, z, c));
199    drop(jobs);
200    drop(slices);
201    if policy.finalizes_layer() {
202        policy.finalize_layer_partitioned(dest, transport);
203    }
204    dest.rebucket(DEFAULT_TARGET_BUCKET_LEN, DEFAULT_MIN_BUCKETS);
205}

207/// This rank's share of `pairing(generator, operand)` as a fresh sum. **Collective.**
208fn pair_share<const W: usize, C, X, P>(
209    generator: &PauliSum<W, C>,
210    share: &PauliSum<W, C>,
211    rows: &PartitionRows<W>,
212    transport: &X,
213    pairing: Pairing,
214    policy: &P,
215) -> PauliSum<W, C>
216where
217    C: Coefficient,
218    X: Transport,
219    P: TruncationPolicy<W, C> + ?Sized,
220{
221    let mut out = PauliSum::empty_with_hash(share.num_qubits(), share.hash().clone());
222    pair_share_into(
223        &mut out,
224        generator,
225        share,
226        rows,
227        transport,
228        (pairing, Weights::fresh()),
229        policy,
230    );
231    out
232}

234/// This rank's share of the Liouvillian `i[generator, operand]`, truncated by `policy` as one layer, where `share` is this rank's share of the operand under `rows`. **Collective.**
235///
236/// Every rank passes the same `generator`, `rows` and `policy`; partner `r ⊕ δ` receives only the terms of `share` that anticommute with a generator term of partition delta `δ`, so a generator whose terms mostly keep their partition exchanges a halo rather than the share.
237/// Every rank's share must carry the same bucket hash rows; bucket counts may differ and are raised to the group's largest.
238///
239/// # Panics
240///
241/// If the ranks disagree about `generator`, `rows` or the bucket hash rows, if `rows` does not name one partition per rank, if the qubit counts differ, or if `policy` has a layer pass with no partitioned form.
242pub fn liouvillian_share<const W: usize, C, X, P>(
243    generator: &PauliSum<W, C>,
244    share: &PauliSum<W, C>,
245    rows: &PartitionRows<W>,
246    transport: &X,
247    policy: &P,
248) -> PauliSum<W, C>
249where
250    C: Coefficient,
251    X: Transport,
252    P: TruncationPolicy<W, C> + ?Sized,
253{
254    pair_share(generator, share, rows, transport, Pairing::Liouvillian, policy)
255}
