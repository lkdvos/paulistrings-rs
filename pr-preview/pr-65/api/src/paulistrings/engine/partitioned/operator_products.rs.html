1//! Operator products of a replicated generator with a partitioned operand, as collectives over a [`Transport`]: [`liouvillian_share`], [`product_share`] and [`anticommutator_share`].
2//!
3//! Partition rows are GF(2)-linear, so a generator term `h` takes an operand term from partition `p` to `p ⊕ R·h`, and rank `r` computes `Σ_δ H_δ · O_{r ⊕ δ}` with `H_δ` the generator terms of partition delta `δ`.
4//! Partner `r ⊕ δ` sends rank `r` the operand terms `H_δ` can reach, only the anticommuting ones for a Liouvillian, so operands cross the wire and products never do.

23/// The generator's terms keyed by partition delta, ascending.
24type ByDelta<const W: usize, C> = BTreeMap<u32, Vec<Entry<W, C>>>;

26/// Every rank must hold the same generator; folded over its keys and coefficient bytes in canonical order.
27fn generator_fingerprint<const W: usize, C: Coefficient>(generator: &PauliSum<W, C>) -> u64 {
28    let mut folded: u64 = 0xcbf2_9ce4_8422_2325;
29    let mut mix = |word: u64| folded = (folded ^ word).wrapping_mul(0x0000_0100_0000_01b3);
30    mix(generator.len() as u64);
31    for (x, z, c) in generator.iter() {
32        x.iter().chain(z).for_each(|&w| mix(w));
33        bytemuck::bytes_of(&c).chunks(8).for_each(|chunk| {
34            let mut word = [0u8; 8];
35            word[..chunk.len()].copy_from_slice(chunk);
36            mix(u64::from_le_bytes(word));
37        });
38    }
39    folded
40}

42/// The terms of bucket `b` that some term of `generator` anticommutes with, or every term without an `index`, in order.
43fn reachable_in_bucket<const W: usize, C: Coefficient>(
44    share: &PauliSum<W, C>,
45    b: usize,
46    index: Option<&SupportIndex>,
47    generator: &[Entry<W, C>],
48    candidates: &mut Vec<u64>,
49) -> (Vec<[u64; W]>, Vec<[u64; W]>, Vec<C>) {
50    let (bucket_x, bucket_z, bucket_coeff) = share.bucket(b);
51    let Some(index) = index else {
52        return (bucket_x.to_vec(), bucket_z.to_vec(), bucket_coeff.to_vec());
53    };
54    let mut reached = vec![0u64; bucket_x.len().div_ceil(64)];
55    for h in generator {
56        let key = PauliString { x: h.x, z: h.z };
57        let support: [u64; W] = std::array::from_fn(|w| h.x[w] | h.z[w]);
58        let weight: u32 = support.iter().map(|w| w.count_ones()).sum();
59        if weight <= SCAN_ABOVE_WEIGHT {
60            index.candidates(b, &support, candidates);
61        } else {
62            candidates.clear();
63            candidates.resize(reached.len(), u64::MAX);
64        }
65        for (w, &word) in candidates.iter().enumerate() {
66            let mut bits = word;
67            while bits != 0 {
68                let i = w * 64 + bits.trailing_zeros() as usize;
69                bits &= bits - 1;
70                if i < bucket_x.len() {
71                    let term = PauliString { x: bucket_x[i], z: bucket_z[i] };
72                    reached[w] |= (!key.commutes_with(&term) as u64) << (i % 64);
73                }
74            }
75        }
76    }
77    let kept = (0..bucket_x.len()).filter(|&i| reached[i / 64] >> (i % 64) & 1 == 1);
78    let mut out = (Vec::new(), Vec::new(), Vec::new());
79    for i in kept {
80        out.0.push(bucket_x[i]);
81        out.1.push(bucket_z[i]);
82        out.2.push(bucket_coeff[i]);
83    }
84    out
85}

87/// The terms of `share` that some term of `generator` anticommutes with, or every term without an `index`, in the gather's wire form, keeping every bucket's order.
88fn encode_reachable<const W: usize, C: Coefficient>(
89    share: &PauliSum<W, C>,
90    index: Option<&SupportIndex>,
91    generator: &[Entry<W, C>],
92) -> ByteParts {
93    let buckets: Vec<_> = (0..share.num_buckets())
94        .into_par_iter()
95        .map_init(Vec::new, |candidates, b| {
96            reachable_in_bucket(share, b, index, generator, candidates)
97        })
98        .collect();
99    let lens: Vec<u64> = buckets.iter().map(|(_, _, c)| c.len() as u64).collect();
100    let terms: usize = buckets.iter().map(|(_, _, c)| c.len()).sum();
101    let (mut x, mut z, mut coeff) =
102        (Vec::with_capacity(terms), Vec::with_capacity(terms), Vec::with_capacity(terms));
103    for (bx, bz, bc) in buckets {
104        x.extend(bx);
105        z.extend(bz);
106        coeff.extend(bc);
107    }
108    ByteParts(vec![
109        bytemuck::cast_slice(&lens).to_vec(),
110        bytemuck::cast_slice(x.as_flattened()).to_vec(),
111        bytemuck::cast_slice(z.as_flattened()).to_vec(),
112        bytemuck::cast_slice(&coeff).to_vec(),
113    ])
114}

116/// `dest ← alpha · pairing(generator, operand) + beta · dest` on this rank, where `share` is this rank's share of the operand under `rows`, `dest` holds only this rank's keys and `generator` is the same on every rank, returning the call as one trace row. **Collective.**
117pub(crate) fn pair_share_into<const W: usize, C, X, P>(
118    dest: &mut PauliSum<W, C>,
119    generator: &PauliSum<W, C>,
120    share: &PauliSum<W, C>,
121    rows: &PartitionRows<W>,
122    transport: &X,
123    (pairing, weights): (Pairing, Weights<C>),
124    policy: &P,
125) -> PartitionLayerRow
126where
127    C: Coefficient,
128    X: Transport,
129    P: TruncationPolicy<W, C> + ?Sized,
130{
131    let started = Instant::now();
132    let mut clock = LayerClock::new(Some(started));
133    let mut phases = LayerPhases::default();
134    if pairing == Pairing::Product {
135        assert!(
136            !C::IS_REAL,
137            "product_share: a product is not Hermitian, so a real coefficient type ({}) cannot hold it; cast to a complex type first",
138            C::NAME,
139        );
140    }
141    let num_qubits = share.num_qubits();
142    assert!(
143        generator.num_qubits() == num_qubits && dest.num_qubits() == num_qubits,
144        "{pairing:?} of a partitioned operand: num_qubits mismatch (generator {}, share {num_qubits}, destination {})",
145        generator.num_qubits(),
146        dest.num_qubits(),
147    );
148    let size = transport.size() as usize;
149    rows.assert_splits(num_qubits, 1usize << group_bits(transport.size()));
150    assert_supports_partitions(policy, size);
151    transport.check_consistency(
152        identity_fingerprint(rows, share.hash()) ^ generator_fingerprint(generator).rotate_left(7),
153    );
154
155    let bits = transport.allreduce_max_u8(share.hash().bits());
156    clock.lap(&mut phases.collective_ns);
157    let refined;
158    let share = if share.hash().bits() < bits {
159        let mut copy = share.clone();
160        while copy.hash().bits() < bits {
161            copy.refine();
162        }
163        refined = copy;
164        &refined
165    } else {
166        share
167    };
168    clock.lap(&mut phases.rebucket_ns);
169
170    let mut by_delta: ByDelta<W, C> = BTreeMap::new();
171    for (x, z, c) in generator.iter() {
172        let c = c * weights.alpha;
173        by_delta.entry(rows.partition_of(x, z)).or_default().push(Entry { x: *x, z: *z, c });
174    }
175    let remote: Vec<u32> = by_delta.keys().copied().filter(|&delta| delta != 0).collect();
176
177    let rank = transport.rank();
178    // Built once: the sends filter by it and the local job reads by it.
179    let mut index = (pairing == Pairing::Liouvillian).then(|| SupportIndex::new(share));
180    clock.lap(&mut phases.prepare_ns);
181    // Every rank holds every delta, so each sends to exactly the partners it receives from, an empty payload included.
182    let mut send: Vec<Option<ByteParts>> = (0..size).map(|_| None).collect();
183    for &delta in &remote {
184        send[(rank ^ delta) as usize] =
185            Some(encode_reachable(share, index.as_ref(), &by_delta[&delta]));
186    }
187    let (mut rows_sent, mut bytes_sent) = (vec![0u64; size], vec![0u64; size]);
188    for (to, payload) in send.iter().enumerate() {
189        if let Some(ByteParts(parts)) = payload {
190            rows_sent[to] = bytemuck::cast_slice::<u8, u64>(&parts[0]).iter().sum();
191            bytes_sent[to] = parts.iter().map(|part| part.len() as u64).sum();
192        }
193    }
194    clock.lap(&mut phases.export_ns);
195    let mut received = transport.exchange(send, &mut Vec::new());
196    let slices: Vec<(u32, PauliSum<W, C>)> = remote
197        .iter()
198        .map(|&delta| {
199            let from = (rank ^ delta) as usize;
200            let payload = received[from]
201                .take()
202                .unwrap_or_else(|| panic!("{pairing:?}: rank {from} sent nothing"));
203            (delta, decode_rank(&payload.0, share.hash().clone(), num_qubits, from))
204        })
205        .collect();
206    clock.lap(&mut phases.exchange_ns);
207    let rows_received = slices.iter().map(|(_, slice)| slice.len() as u64).sum();
208
209    let mut jobs = Vec::with_capacity(slices.len() + 1);
210    if let Some(local) = by_delta.get(&0) {
211        jobs.push(Job::with_index(local.iter().copied(), share, index.take(), true));
212    }
213    for (delta, slice) in &slices {
214        jobs.push(Job::new(by_delta[delta].iter().copied(), slice, pairing, true));
215    }
216    dest.align_destination(share.hash(), weights.beta);
217    accumulate_into(dest, weights.beta, &jobs, pairing, &|x, z, c| policy.keep_term(x, z, c));
218    drop(jobs);
219    drop(slices);
220    clock.lap(&mut phases.coset_loop_ns);
221    if policy.finalizes_layer() {
222        policy.finalize_layer_partitioned(dest, transport);
223    }
224    clock.lap(&mut phases.finalize_ns);
225    dest.rebucket(DEFAULT_TARGET_BUCKET_LEN, DEFAULT_MIN_BUCKETS);
226    clock.lap(&mut phases.rebucket_ns);
227    PartitionLayerRow {
228        bits,
229        remote_deltas: remote.len() as u32,
230        collectives: 2 + u32::from(policy.finalizes_layer()),
231        gate_name: pairing.name(),
232        terms_in: share.len(),
233        terms_out: dest.len(),
234        rows_sent,
235        bytes_sent,
236        rows_received,
237        nanos: started.elapsed().as_nanos() as u64,
238        phases,
239        ..PartitionLayerRow::default()
240    }
241}

243/// This rank's share of `pairing(generator, operand)` as a fresh sum. **Collective.**
244fn pair_share<const W: usize, C, X, P>(
245    generator: &PauliSum<W, C>,
246    share: &PauliSum<W, C>,
247    rows: &PartitionRows<W>,
248    transport: &X,
249    pairing: Pairing,
250    policy: &P,
251) -> PauliSum<W, C>
252where
253    C: Coefficient,
254    X: Transport,
255    P: TruncationPolicy<W, C> + ?Sized,
256{
257    let mut out = PauliSum::empty_with_hash(share.num_qubits(), share.hash().clone());
258    let _ = pair_share_into(
259        &mut out,
260        generator,
261        share,
262        rows,
263        transport,
264        (pairing, Weights::fresh()),
265        policy,
266    );
267    out
268}

270/// This rank's share of the Liouvillian `i[generator, operand]`, truncated by `policy` as one layer, where `share` is this rank's share of the operand under `rows`. **Collective.**
271///
272/// Every rank passes the same `generator`, `rows` and `policy`; partner `r ⊕ δ` receives only the terms of `share` that anticommute with a generator term of partition delta `δ`, so a generator whose terms mostly keep their partition exchanges a halo rather than the share.
273/// Every rank's share must carry the same bucket hash rows; bucket counts may differ and are raised to the group's largest.
274///
275/// # Panics
276///
277/// If the ranks disagree about `generator`, `rows` or the bucket hash rows, if `rows` does not name one partition per rank, if the qubit counts differ, or if `policy` has a layer pass with no partitioned form.
278pub fn liouvillian_share<const W: usize, C, X, P>(
279    generator: &PauliSum<W, C>,
280    share: &PauliSum<W, C>,
281    rows: &PartitionRows<W>,
282    transport: &X,
283    policy: &P,
284) -> PauliSum<W, C>
285where
286    C: Coefficient,
287    X: Transport,
288    P: TruncationPolicy<W, C> + ?Sized,
289{
290    pair_share(generator, share, rows, transport, Pairing::Liouvillian, policy)
291}
