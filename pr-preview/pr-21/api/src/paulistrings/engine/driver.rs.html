20/// Layers between two bucket-count agreements, and the length of the opening ramp that precedes them (ARCHITECTURE.md §Partitioning).
21pub(crate) const BITS_AGREE_EVERY: usize = 16;
22
23/// Whether the layer at `application_index` of a call agrees the bucket count; a pure function of it, so every partition answers alike.
24fn agrees_bucket_bits(application_index: usize) -> bool {
25    application_index < BITS_AGREE_EVERY || application_index.is_multiple_of(BITS_AGREE_EVERY)
26}

28/// A [`Collectives`] view that counts the policy finalization's calls for the trace.
29struct CountingCollectives<'a> {
30    inner: &'a dyn Collectives,
31    calls: &'a AtomicU32,
32}
33
34impl crate::collectives::sealed::Sealed for CountingCollectives<'_> {}
35
36impl Collectives for CountingCollectives<'_> {
37    fn rank(&self) -> u32 {
38        self.inner.rank()
39    }
40    fn size(&self) -> u32 {
41        self.inner.size()
42    }
43    fn allreduce_max_u8(&self, v: u8) -> u8 {
44        self.calls.fetch_add(1, Ordering::Relaxed);
45        self.inner.allreduce_max_u8(v)
46    }
47    fn allreduce_sum_u64(&self, buffer: &mut [u64]) {
48        self.calls.fetch_add(1, Ordering::Relaxed);
49        self.inner.allreduce_sum_u64(buffer)
50    }
51    fn allreduce_sum_f64(&self, buffer: &mut [f64]) {
52        self.calls.fetch_add(1, Ordering::Relaxed);
53        self.inner.allreduce_sum_f64(buffer)
54    }
55    fn barrier(&self) {
56        self.calls.fetch_add(1, Ordering::Relaxed);
57        self.inner.barrier()
58    }
59}

61/// One partition's payload, moved into its thread for the duration of a call and handed back.

68/// What a partition knows about itself while it walks the layers.

76/// Why a channel's preparation declines, shared by the host panic and the device's `GpuError::Unsupported`.
77pub(crate) const DECLINED_PREPARATION: &str = "support above MAX_LOCAL_SUPPORT (2) qubits is \
78     unsupported except for PauliRotation, and a channel must not write outside its declared support";
79const _: () = assert!(
80    MAX_LOCAL_SUPPORT == 2,
81    "DECLINED_PREPARATION names the value"
82);
83
84/// The channel's preparation, or the unpartitioned engine's hard error naming the partition and layer.
85fn prepare_or_panic<const W: usize>(
86    channel: &dyn Channel<W>,
87    hash: &Gf2Hash<W>,
88    adjoint: bool,
89    rank: usize,
90    circuit_index: usize,
91) -> crate::channel::prepared::Prepared<W> {
92    channel.prepare(hash, adjoint).unwrap_or_else(|| {
93        let weight: u32 = channel.support().iter().map(|w| w.count_ones()).sum();
94        panic!(
95            "partition {rank}, layer {circuit_index}: channel preparation declined for a \
96             channel of support {weight}: {DECLINED_PREPARATION}",
97        )
98    })
99}

101/// One partition's layer loop, run in lock-step with its peers inside its own pool (ARCHITECTURE.md §Partitioning).
102pub(crate) fn run_layers<const W: usize, T, X, B>(
103    circuit: &Circuit<W>,
104    policy: &T,
105    direction: Direction,
106    options: PropagateOptions,
107    context: PartitionContext<'_, W>,
108    work: &mut PartitionWork<B>,
109    transport: &X,
110) where
111    T: TruncationPolicy<W> + ?Sized,
112    X: Transport,
113    B: PartitionBackend<W, T>,
114{
115    let PartitionContext {
116        rows,
117        rank,
118        size,
119        tracing,
120    } = context;
121    let num_layers = circuit.channels.len();
122    let adjoint = matches!(direction, Direction::Heisenberg);
123    let group_size = transport.size();
124    // `finalizes_layer` is a property of the policy type, so skipping the collective call is safe on every partition alike.
125    let finalizes = policy.finalizes_layer();
126    let policy_calls = AtomicU32::new(0);
127    // At `P = 1` the local answer is the agreed one, so the loop rebuckets every layer exactly as an unpartitioned run would.
128    let solo = group_size == 1;
129    let local = &mut work.local;
130    let mut plan = PartitionPlan::default();
131
132    for application_index in 0..num_layers {
133        let circuit_index = match direction {
134            Direction::Forward => application_index,
135            Direction::Heisenberg => num_layers - 1 - application_index,
136        };
137        let channel: &dyn Channel<W> = circuit.channels[circuit_index].as_ref();
138
139        let debug_enabled = log::log_enabled!(target: LOG_TARGET, log::Level::Debug);
140        let timed_layer = tracing || debug_enabled;
141        let layer_start = timed_layer.then(Instant::now);
142        let terms_before = local.len();
143        let bits_before = local.hash().bits();
144
145        stamp!(stamp);
146        tally!(local.stats().layers, 1);
147        tally!(local.stats().terms_in, terms_before);
148
149        // Alone, a backend that sizes its buckets without the prepared channel settles the count first, so a growing layer prepares once.
150        let refined_early =
151            solo && local.refine_unprepared(options.target_bucket_len, options.min_buckets);
152        if refined_early {
153            lap!(stamp, local.stats().rebucket_ns);
154        }
155
156        // Otherwise prepared before the bucket count is settled, since the plan decides whether this layer agrees it; only `bucket_delta` depends on the count, so a refining layer prepares again.
157        let mut prepared = prepare_or_panic(channel, local.hash(), adjoint, rank, circuit_index);
158        plan.rebuild(&prepared, rows, rank as u32);
159        lap!(stamp, local.stats().prepare_ns);
160
161        let mut collectives = 0u32;
162        if !refined_early && (solo || plan.has_remote() || agrees_bucket_bits(application_index)) {
163            let mut wanted_bits =
164                local.proposed_bits(&prepared, options.target_bucket_len, options.min_buckets);
165            if !solo {
166                wanted_bits = transport.allreduce_max_u8(wanted_bits);
167                collectives += 1;
168            }
169            lap!(stamp, local.stats().collective_ns);
170            if wanted_bits > local.hash().bits() {
171                while local.hash().bits() < wanted_bits {
172                    local.refine();
173                }
174                lap!(stamp, local.stats().rebucket_ns);
175                prepared = prepare_or_panic(channel, local.hash(), adjoint, rank, circuit_index);
176                plan.rebuild(&prepared, rows, rank as u32);
177                lap!(stamp, local.stats().prepare_ns);
178            }
179        }
180
181        let counts = local.apply_layer(&prepared, &plan, rows, policy, transport);
182        timed!(stamp.rearm());
183
184        if finalizes {
185            policy_calls.store(0, Ordering::Relaxed);
186            let counting = CountingCollectives {
187                inner: transport,
188                calls: &policy_calls,
189            };
190            local.finalize_layer(policy, &counting);
191            collectives += policy_calls.load(Ordering::Relaxed);
192        }
193        lap!(stamp, local.stats().finalize_ns);
194        tally!(local.stats().terms_out, local.len());
195
196        // The trace sits behind a hoisted flag and a `#[cold]` callee: this loop inlines the merge kernels, which are sensitive to code motion.
197        let remote_deltas = counts.remote_deltas;
198        let rows_received = counts.rows_received;
199        let elapsed = layer_start.map(|start| start.elapsed());
200        if tracing {
201            let mut row = PartitionLayerRow {
202                bits: local.hash().bits(),
203                remote_deltas: counts.remote_deltas as u32,
204                collectives,
205                circuit_index: circuit_index as u32,
206                application_index: application_index as u32,
207                gate_name: channel.debug_name(),
208                terms_in: terms_before,
209                terms_out: local.len(),
210                rows_sent: counts.rows_sent,
211                bytes_sent: counts.bytes_sent,
212                rows_received: counts.rows_received,
213                nanos: elapsed.unwrap_or_default().as_nanos() as u64,
214                refine_passes: u32::from(local.hash().bits() - bits_before),
215                receive_chunks: counts.receive_chunks,
216                device: None,
217            };
218            local.trace_layer(&mut row);
219            record_layer_row(&mut work.rows, row);
220        }
221
222        if debug_enabled {
223            let Some(elapsed) = elapsed else { continue };
224            log::debug!(
225                target: LOG_TARGET,
226                "layer {}/{} [{}]: {} -> {} terms, {:.1} ms (partition {}/{}, {} remote deltas, \
227                 {} rows in)",
228                application_index + 1,
229                num_layers,
230                channel.debug_name(),
231                terms_before,
232                local.len(),
233                elapsed.as_secs_f64() * 1e3,
234                rank,
235                size,
236                remote_deltas,
237                rows_received,
238            );
239        }
240    }
