1//! Per-layer prepared form of a channel. See ARCHITECTURE.md §Prepared-Channels.
2//!
3//! Prepares a channel once per layer into a table, reducing the inner loop to one lookup on ≤ 4 extracted bits, one XOR with a precomputed mask, and one complex multiply — instead of redoing the channel's setup (e.g. `theta.cos()/sin()`) per term through a vtable call.
4//! The prepared form also carries, for each key delta `d`, the bucket delta `δ = H·d` the engine needs to know which buckets to read.
5
6use num_complex::Complex64;
7
8use super::{Channel, OutputBuffer};
9use crate::bucket::hash::Gf2Hash;
10use crate::pauli_string::PauliString;
11use crate::phase::Phase;

13/// Largest support size handled by [`Prepared::Local`].
14///
15/// The dense local Pauli-transfer matrix is `4^k × 4^k`, so `k = 2` is 4 KB of `Complex64` per layer; `k = 3` would be 64 KB, too large to be worth building per layer.
16pub const MAX_LOCAL_SUPPORT: usize = 2;
17
18/// `4^MAX_LOCAL_SUPPORT` — the number of local Pauli basis elements.
19pub const LOCAL_DIM: usize = 16;
20
21const ZERO: Complex64 = Complex64::new(0.0, 0.0);
22
23/// One key delta, with the amplitude it carries for each input support pattern.
24#[derive(Clone, Debug)]
25pub struct DeltaEntry<const W: usize> {
26    /// `δ = H·d`. Output bucket `β'` reads input bucket `β' ^ bucket_delta` for this delta.
27    /// Several entries may share a `bucket_delta` (a hash collision); it costs a wasted read, never correctness.
28    pub bucket_delta: u32,
29    /// The delta in local support coordinates: bit `2j` is the x-bit of support qubit `j`, bit `2j+1` its z-bit.
30    /// The canonical ordering key, since unlike `bucket_delta` it does not depend on the bucket count.
31    pub local_delta: u8,
32    /// `d` lifted to a full-width XOR mask.
33    pub mask_x: [u64; W],
34    /// `d` lifted to a full-width XOR mask.
35    pub mask_z: [u64; W],
36    /// `amp[s]` is the amplitude taking input support pattern `s` to
37    /// `s ^ local_delta`. Exactly zero means "no output for this `s`".
38    pub amp: [Complex64; LOCAL_DIM],
39}
40
41impl<const W: usize> DeltaEntry<W> {
42    /// The output row this entry emits for a term with support pattern `s`, or `None` when the amplitude is exactly zero.
43    ///
44    /// The row-level form of the engine's gather inner loop (`engine::bucketed::gather_local_input_major`): same lookup, same mask XOR, same multiply, in the same order, so the partitioned export pass produces bitwise the rows a local gather would have.
45    #[inline]
46    pub(crate) fn emit(
47        &self,
48        s: usize,
49        x: &[u64; W],
50        z: &[u64; W],
51        c: Complex64,
52    ) -> Option<([u64; W], [u64; W], Complex64)> {
53        let a = self.amp[s];
54        if a == ZERO {
55            return None;
56        }
57        let mut kx = *x;
58        let mut kz = *z;
59        for w in 0..W {
60            kx[w] ^= self.mask_x[w];
61            kz[w] ^= self.mask_z[w];
62        }
63        Some((kx, kz, c * a))
64    }
65
66    /// The entry's key delta as a full-width XOR mask pair, `(mask_x, mask_z)`.
67    ///
68    /// What a partitioning reads to classify the entry: `part(mask)` is the partition delta, `h(mask)` the (already stored) [`bucket_delta`](Self::bucket_delta).
69    #[inline]
70    pub(crate) fn mask(&self) -> ([u64; W], [u64; W]) {
71        (self.mask_x, self.mask_z)
72    }
73}

75/// A channel with support on at most [`MAX_LOCAL_SUPPORT`] qubits, as a dense
76/// local Pauli-transfer matrix grouped by bucket delta.
77#[derive(Clone, Debug)]
78pub struct LocalPtm<const W: usize> {
79    /// Support qubits, ascending. Only the first `k` are meaningful.
80    qubits: [u32; MAX_LOCAL_SUPPORT],
81    /// Number of support qubits, `0 ≤ k ≤ MAX_LOCAL_SUPPORT`.
82    k: u8,
83    /// The delta set, **ascending by `local_delta`** — the canonical construction order (index 0 is the identity entry when present). Does not define summation order; see ARCHITECTURE.md §Determinism.
84    deltas: Vec<DeltaEntry<W>>,
85}
86
87impl<const W: usize> LocalPtm<W> {
88    /// Number of support qubits.
89    #[inline]
90    pub fn k(&self) -> usize {
91        self.k as usize
92    }
93
94    /// Support qubits, ascending.
95    #[inline]
96    pub fn qubits(&self) -> &[u32] {
97        &self.qubits[..self.k as usize]
98    }
99
100    /// The delta set, ascending by `local_delta` — the canonical construction
101    /// order (see the field doc).
102    #[inline]
103    pub fn deltas(&self) -> &[DeltaEntry<W>] {
104        &self.deltas
105    }
106
107    /// Number of distinct key deltas, i.e. `|D|`.
108    #[inline]
109    pub fn num_deltas(&self) -> usize {
110        self.deltas.len()
111    }
112
113    /// The distinct bucket deltas `h(D)`, ascending. Length is
114    /// `2^rank(H|_D)` — the number of input buckets each output bucket reads.
115    pub fn bucket_deltas(&self) -> Vec<u32> {
116        let mut v: Vec<u32> = self.deltas.iter().map(|d| d.bucket_delta).collect();
117        v.sort_unstable();
118        v.dedup();
119        v
120    }
121
122    /// Extract the local support pattern `s` of a key.
123    ///
124    /// Bit `2j` is the x-bit of support qubit `j`, bit `2j+1` its z-bit — the
125    /// same packing `Clifford2Q` uses (`idx = x0 | z0<<1 | x1<<2 | z1<<3`).
126    #[inline]
127    pub fn support_bits(&self, x: &[u64; W], z: &[u64; W]) -> usize {
128        let mut s = 0usize;
129        for j in 0..self.k as usize {
130            let q = self.qubits[j] as usize;
131            let w = q / 64;
132            let b = q % 64;
133            s |= (((x[w] >> b) & 1) as usize) << (2 * j);
134            s |= (((z[w] >> b) & 1) as usize) << (2 * j + 1);
135        }
136        s
137    }
138
139    /// `true` if this channel leaves every key bitwise unchanged, so a layer is
140    /// an in-place coefficient rescale: no gather, no sort, no merge.
141    ///
142    /// Covers `IdentityChannel`, `Depolarizing`, `Dephasing` and
143    /// `Clifford1Q::{x, y, z}`.
144    pub fn is_key_preserving(&self) -> bool {
145        self.deltas.len() == 1 && self.deltas[0].local_delta == 0
146    }
147
148    /// A copy keeping only the entries with `keep[e] == true`, in order, with the same support qubits and `k`.
149    ///
150    /// Used at prepare time to split a partitioned layer's table into a local-only one, so the gather loops keep running over a plain `deltas()` slice with no per-entry predicate in the inner loop.
151    /// The result is still ascending by `local_delta`, a subsequence of an ascending sequence.
152    ///
153    /// # Panics
154    ///
155    /// Panics in debug builds if `keep` is not one flag per entry.
156    pub(crate) fn retain_entries(&self, keep: &[bool]) -> LocalPtm<W> {
157        debug_assert_eq!(
158            keep.len(),
159            self.deltas.len(),
160            "retain_entries: one flag per delta entry",
161        );
162        LocalPtm {
163            qubits: self.qubits,
164            k: self.k,
165            deltas: self
166                .deltas
167                .iter()
168                .zip(keep)
169                .filter(|(_, &k)| k)
170                .map(|(d, _)| d.clone())
171                .collect(),
172        }
173    }
174
175    /// Lift a local delta to full-width XOR masks.
176    fn lift(&self, local_delta: u8) -> ([u64; W], [u64; W]) {
177        let mut mx = [0u64; W];
178        let mut mz = [0u64; W];
179        for j in 0..self.k as usize {
180            let q = self.qubits[j] as usize;
181            let w = q / 64;
182            let bit = 1u64 << (q % 64);
183            if (local_delta >> (2 * j)) & 1 == 1 {
184                mx[w] |= bit;
185            }
186            if (local_delta >> (2 * j + 1)) & 1 == 1 {
187                mz[w] |= bit;
188            }
189        }
190        (mx, mz)
191    }
192}

194/// A rotation with support wider than [`MAX_LOCAL_SUPPORT`].
195///
196/// The delta set is `{0, P}` for any generator weight, so only two buckets are ever read — but the amplitude's `i^k` phase depends on `2w` support bits, which stops being tabulable, so amplitudes are computed per term instead, with `cos`/`sin` hoisted out of the loop.
197#[derive(Clone, Debug)]
198pub struct RotationPrep<const W: usize> {
199    /// The generator `P`.
200    pub gen: PauliString<W>,
201    /// `cos(θ)`, hoisted.
202    pub cos: f64,
203    /// `sin(θ)`, hoisted.
204    pub sin: f64,
205    /// `H·0 = 0`: the bucket delta for the identity output.
206    pub bucket_delta_identity: u32,
207    /// `H·P`: the bucket delta for the `v ⊕ P` output.
208    pub bucket_delta_gen: u32,
209}
210
211impl<const W: usize> RotationPrep<W> {
212    /// The generator-pass row for a term, or `None` if it commutes with the generator (in which case the rotation leaves the term alone and only the identity pass emits).
213    ///
214    /// Copied verbatim from the `DeltaPlan::Rotation` arm of `engine::bucketed::fill_coset` so it stays bitwise-equal to the gather: `mul_assign` returns the `i^k` of the Pauli product, the leading `i` of `i · Q · P` folds in as `Phase::I + phase`, applied to the coefficient before the hoisted `sin` multiplies last.
215    ///
216    /// The identity pass has no emitter: every term contributes one identity row unconditionally, so there is nothing to decide per row.
217    #[inline]
218    pub(crate) fn emit_gen(
219        &self,
220        x: &[u64; W],
221        z: &[u64; W],
222        c: Complex64,
223    ) -> Option<([u64; W], [u64; W], Complex64)> {
224        let v = PauliString::<W> { x: *x, z: *z };
225        if v.commutes_with(&self.gen) {
226            return None;
227        }
228        let mut prod = v;
229        let phase = prod.mul_assign(&self.gen);
230        let total = Phase::I + phase;
231        Some((prod.x, prod.z, total.apply(c) * self.sin))
232    }
233
234    /// The generator-pass key delta as a full-width XOR mask pair — i.e. the
235    /// generator itself, since the delta set is `{0, P}`.
236    pub(crate) fn gen_mask(&self) -> ([u64; W], [u64; W]) {
237        (self.gen.x, self.gen.z)
238    }
239}

241/// A channel prepared for one layer of the bucketed engine.
242#[derive(Clone, Debug)]
243pub enum Prepared<const W: usize> {
244    /// Amplitudes depend only on ≤ 4 support bits, so they are tabulated.
245    /// Covers `Clifford1Q`/`Clifford2Q`, all noise channels,
246    /// `GeneralUnitary1Q`/`2Q`, and `PauliRotation` at generator weight ≤ 2.
247    Local(LocalPtm<W>),
248    /// `PauliRotation` at generator weight > 2.
249    Rotation(RotationPrep<W>),
250}
251
252impl<const W: usize> Prepared<W> {
253    /// Bucket deltas this channel can produce, i.e. `h(D)`.
254    ///
255    /// Output bucket `β'` reads input buckets `β' ^ δ` for each `δ` here. The
256    /// length is `2^rank(H|_D)` — 1, 2, 4 or 16 for the built-ins.
257    pub fn bucket_deltas(&self) -> Vec<u32> {
258        match self {
259            Prepared::Local(p) => p.bucket_deltas(),
260            Prepared::Rotation(r) => {
261                if r.bucket_delta_gen == r.bucket_delta_identity {
262                    vec![r.bucket_delta_identity]
263                } else {
264                    vec![r.bucket_delta_identity, r.bucket_delta_gen]
265                }
266            }
267        }
268    }
269
270    /// Derive the prepared form of a bounded-support channel by probing its own [`Channel::apply`].
271    ///
272    /// Returns `None` when the channel's support is wider than [`MAX_LOCAL_SUPPORT`], or when it writes outside its declared support; in both cases `propagate` panics rather than proceeding with an unsound preparation (see ARCHITECTURE.md §Prepared-Channels).
273    ///
274    /// # The soundness precondition
275    ///
276    /// Exact iff the channel honours the bounded-support contract: the output amplitude may depend on the input only through its support bits.
277    /// Probing cannot fully verify that — a channel reading qubit 5 while declaring support `[0]` would produce a table wrong for inputs never tried — so debug builds re-derive with an all-ones background outside the support and assert the two tables agree, and a property test checks every built-in against `apply` on randomized full-width inputs.
278    pub fn derive_local<C>(channel: &C, hash: &Gf2Hash<W>, adjoint: bool) -> Option<Self>
279    where
280        C: Channel<W> + ?Sized,
281    {
282        let mask = channel.support();
283        // Popcount first, and bail before materializing anything, so a wide
284        // support never pays for qubit extraction it will just discard.
285        let k: usize = mask.iter().map(|w| w.count_ones() as usize).sum();
286        if k > MAX_LOCAL_SUPPORT {
287            return None;
288        }
289
290        // Extract qubit indices ascending via per-word `trailing_zeros`. A
291        // bitmask is already a set, so this is automatically sorted and
292        // duplicate-free.
293        let mut qubits = [0u32; MAX_LOCAL_SUPPORT];
294        let mut n = 0usize;
295        for (w, &word) in mask.iter().enumerate() {
296            let mut live = word;
297            while live != 0 {
298                let bit = live.trailing_zeros();
299                qubits[n] = (64 * w) as u32 + bit;
300                n += 1;
301                live &= live - 1;
302            }
303        }
304        debug_assert_eq!(n, k);
305
306        let mut ptm = LocalPtm {
307            qubits,
308            k: k as u8,
309            deltas: Vec::new(),
310        };
311
312        let table = probe_table(channel, &ptm, adjoint, false)?;
313
314        #[cfg(debug_assertions)]
315        {
316            // Same table, but with every non-support bit set. A channel that
317            // reads outside its support will disagree.
318            let shadow = probe_table(channel, &ptm, adjoint, true);
319            debug_assert!(
320                shadow.as_ref() == Some(&table),
321                "Channel::apply depends on bits outside its declared support; \
322                 the bounded-support contract of Prepared::derive_local is violated",
323            );
324        }
325
326        // Collect the deltas, ascending by local delta -- the canonical,
327        // bucket-count-independent order that determinism relies on.
328        let dim = 1usize << (2 * k);
329        let mut has_delta = [false; LOCAL_DIM];
330        for s in 0..dim {
331            for t in 0..dim {
332                if table[s][t] != ZERO {
333                    has_delta[s ^ t] = true;
334                }
335            }
336        }
337
338        for d in 0..dim {
339            if !has_delta[d] {
340                continue;
341            }
342            let mut amp = [ZERO; LOCAL_DIM];
343            for s in 0..dim {
344                amp[s] = table[s][s ^ d];
345            }
346            let (mask_x, mask_z) = ptm.lift(d as u8);
347            // `d` ascends over this loop, so `deltas` ends up sorted by
348            // `local_delta` with no explicit sort.
349            ptm.deltas.push(DeltaEntry {
350                bucket_delta: hash.bucket_of(&mask_x, &mask_z),
351                local_delta: d as u8,
352                mask_x,
353                mask_z,
354                amp,
355            });
356        }
357
358        Some(Prepared::Local(ptm))
359    }
360}

362/// Probe `channel.apply` on every local basis Pauli and read off `amp[s][t]`.
363///
364/// With `background`, every non-support bit of the probe is set — used in debug builds to detect a channel that reads outside its support.
365fn probe_table<const W: usize, C>(
366    channel: &C,
367    ptm: &LocalPtm<W>,
368    adjoint: bool,
369    background: bool,
370) -> Option<[[Complex64; LOCAL_DIM]; LOCAL_DIM]>
371where
372    C: Channel<W> + ?Sized,
373{
374    let k = ptm.k as usize;
375    let dim = 1usize << (2 * k);
376    let fanout = channel.max_fanout().max(1);
377
378    let mut table = [[ZERO; LOCAL_DIM]; LOCAL_DIM];
379
380    // Bits belonging to the support, so they can be excluded from a background.
381    let (sup_x, sup_z) = {
382        let mut mx = [0u64; W];
383        let mut mz = [0u64; W];
384        for j in 0..k {
385            let q = ptm.qubits[j] as usize;
386            let bit = 1u64 << (q % 64);
387            mx[q / 64] |= bit;
388            mz[q / 64] |= bit;
389        }
390        (mx, mz)
391    };
392
393    let mut buf_x = vec![[0u64; W]; fanout];
394    let mut buf_z = vec![[0u64; W]; fanout];
395    let mut buf_c = vec![ZERO; fanout];
396
397    for (s, row) in table.iter_mut().enumerate().take(dim) {
398        let (mut in_x, mut in_z) = ptm.lift(s as u8);
399        if background {
400            for w in 0..W {
401                in_x[w] |= !sup_x[w];
402                in_z[w] |= !sup_z[w];
403            }
404        }
405
406        let mut len = 0usize;
407        {
408            let mut out = OutputBuffer::<W> {
409                x: &mut buf_x,
410                z: &mut buf_z,
411                coeff: &mut buf_c,
412                len: &mut len,
413            };
414            if adjoint {
415                channel.apply_adjoint(&in_x, &in_z, Complex64::new(1.0, 0.0), &mut out);
416            } else {
417                channel.apply(&in_x, &in_z, Complex64::new(1.0, 0.0), &mut out);
418            }
419        }
420
421        for i in 0..len {
422            // The output must differ from the input only inside the support,
423            // otherwise this channel cannot be expressed as a local PTM.
424            for w in 0..W {
425                if (buf_x[i][w] ^ in_x[w]) & !sup_x[w] != 0
426                    || (buf_z[i][w] ^ in_z[w]) & !sup_z[w] != 0
427                {
428                    return None;
429                }
430            }
431            let t = ptm.support_bits(&buf_x[i], &buf_z[i]);
432            // Several outputs can share a `t` only if the channel emits the same
433            // Pauli twice; sum rather than overwrite so the table stays faithful.
434            row[t] += buf_c[i];
435        }
436    }
437
438    Some(table)
439}

454    type Term<const W: usize> = ([u64; W], [u64; W], Complex64);

456    /// Outputs of `apply` / `apply_adjoint`, with exact zeros dropped and equal
457    /// keys summed — the same normalization the merge phase performs.
458    fn via_apply<const W: usize, C: Channel<W> + ?Sized>(
459        ch: &C,
460        adjoint: bool,
461        x: &[u64; W],
462        z: &[u64; W],
463        coeff: Complex64,
464    ) -> Vec<Term<W>> {
465        let f = ch.max_fanout().max(1);
466        let mut bx = vec![[0u64; W]; f];
467        let mut bz = vec![[0u64; W]; f];
468        let mut bc = vec![ZERO; f];
469        let mut len = 0usize;
470        {
471            let mut out = OutputBuffer::<W> {
472                x: &mut bx,
473                z: &mut bz,
474                coeff: &mut bc,
475                len: &mut len,
476            };
477            if adjoint {
478                ch.apply_adjoint(x, z, coeff, &mut out);
479            } else {
480                ch.apply(x, z, coeff, &mut out);
481            }
482        }
483        normalize((0..len).map(|i| (bx[i], bz[i], bc[i])).collect())
484    }

486    /// The same outputs, reconstructed from the prepared table.
487    fn via_prepared<const W: usize>(
488        prep: &Prepared<W>,
489        x: &[u64; W],
490        z: &[u64; W],
491        coeff: Complex64,
492    ) -> Vec<Term<W>> {
493        let mut out: Vec<Term<W>> = Vec::new();
494        match prep {
495            Prepared::Local(p) => {
496                let s = p.support_bits(x, z);
497                for m in p.deltas() {
498                    let a = m.amp[s];
499                    if a == ZERO {
500                        continue;
501                    }
502                    let mut ox = *x;
503                    let mut oz = *z;
504                    for w in 0..W {
505                        ox[w] ^= m.mask_x[w];
506                        oz[w] ^= m.mask_z[w];
507                    }
508                    out.push((ox, oz, coeff * a));
509                }
510            }
511            Prepared::Rotation(r) => {
512                let input = PauliString::<W> { x: *x, z: *z };
513                if input.commutes_with(&r.gen) {
514                    out.push((*x, *z, coeff));
515                } else {
516                    out.push((*x, *z, coeff * r.cos));
517                    let mut prod = input;
518                    let phase = prod.mul_assign(&r.gen);
519                    let total = crate::phase::Phase::I + phase;
520                    out.push((prod.x, prod.z, total.apply(coeff) * r.sin));
521                }
522            }
523        }
524        normalize(out)
525    }

527    fn normalize<const W: usize>(mut v: Vec<Term<W>>) -> Vec<Term<W>> {
528        v.sort_by(|a, b| (a.0, a.1).cmp(&(b.0, b.1)));
529        let mut out: Vec<Term<W>> = Vec::new();
530        for (x, z, c) in v {
531            match out.last_mut() {
532                Some(last) if last.0 == x && last.1 == z => last.2 += c,
533                _ => out.push((x, z, c)),
534            }
535        }
536        out.retain(|t| t.2 != ZERO);
537        out
538    }

540    fn assert_terms_eq<const W: usize>(a: &[Term<W>], b: &[Term<W>], what: &str) {
541        assert_eq!(a.len(), b.len(), "{what}: term count {a:?} vs {b:?}");
542        for (p, q) in a.iter().zip(b.iter()) {
543            assert_eq!(p.0, q.0, "{what}: x key");
544            assert_eq!(p.1, q.1, "{what}: z key");
545            assert!((p.2 - q.2).norm() < TOL, "{what}: coeff {} vs {}", p.2, q.2);
546        }
547    }

549    /// The derived table must reproduce `apply` on randomized full-width inputs, not just the basis probes prepare-time derivation reads — this catches a channel reading outside its declared support.
550    fn check_agrees_on_random_inputs<const W: usize, C: Channel<W>>(
551        ch: &C,
552        num_qubits: usize,
553        label: &str,
554    ) {
555        let hash = Gf2Hash::<W>::new(num_qubits, 10, 0xD00D);
556        for &adjoint in &[false, true] {
557            let prep = ch
558                .prepare(&hash, adjoint)
559                .unwrap_or_else(|| panic!("{label}: prepare returned None"));
560            let mut rng = Xs64::new(0x5EED ^ (adjoint as u64));
561            for _ in 0..400 {
562                let mut x = [0u64; W];
563                let mut z = [0u64; W];
564                for w in 0..W {
565                    x[w] = rng.next_u64();
566                    z[w] = rng.next_u64();
567                }

671    // `DeltaEntry::emit` / `RotationPrep::emit_gen` are the row-level form the partitioned export pass uses in place of the engine's gather loops; these check them against `Channel::apply` itself, not against the gather.
672
673    /// The same outputs, reconstructed one entry at a time through the emitters.
674    /// Mirrors `via_prepared`, but routes every row through [`DeltaEntry::emit`] / [`RotationPrep::emit_gen`].
675    fn via_emit<const W: usize>(
676        prep: &Prepared<W>,
677        x: &[u64; W],
678        z: &[u64; W],
679        coeff: Complex64,
680    ) -> Vec<Term<W>> {
681        let mut out: Vec<Term<W>> = Vec::new();
682        match prep {
683            Prepared::Local(p) => {
684                let s = p.support_bits(x, z);
685                for m in p.deltas() {
686                    if let Some(row) = m.emit(s, x, z, coeff) {
687                        out.push(row);
688                    }
689                }
690            }
691            Prepared::Rotation(r) => {
692                // Entry 0, the identity pass, has no `DeltaEntry`: every term
693                // emits one such row, full coefficient when it commutes and
694                // `cos`-scaled when it does not.
695                let input = PauliString::<W> { x: *x, z: *z };
696                let commutes = input.commutes_with(&r.gen);
697                out.push((*x, *z, if commutes { coeff } else { coeff * r.cos }));
698                let gen_row = r.emit_gen(x, z, coeff);
699                assert_eq!(
700                    gen_row.is_none(),
701                    commutes,
702                    "emit_gen must be None exactly when the term commutes",
703                );
704                if let Some(row) = gen_row {
705                    out.push(row);
706                }
707            }
708        }
709        normalize(out)
710    }

712    fn check_emit_agrees_on_random_inputs<const W: usize, C: Channel<W>>(
713        ch: &C,
714        num_qubits: usize,
715        label: &str,
716    ) {
717        let hash = Gf2Hash::<W>::new(num_qubits, 10, 0xD00D);
718        for &adjoint in &[false, true] {
719            let prep = ch
720                .prepare(&hash, adjoint)
721                .unwrap_or_else(|| panic!("{label}: prepare returned None"));
722            let mut rng = Xs64::new(0x5EED ^ (adjoint as u64));
723            for _ in 0..400 {
724                let mut x = [0u64; W];
725                let mut z = [0u64; W];
726                for w in 0..W {
727                    x[w] = rng.next_u64();
728                    z[w] = rng.next_u64();
729                }
730                let coeff = Complex64::new(
731                    (rng.next_u64() as i64 as f64) / (i64::MAX as f64),
732                    (rng.next_u64() as i64 as f64) / (i64::MAX as f64),
733                );
734                let direct = via_apply(ch, adjoint, &x, &z, coeff);
735                let emitted = via_emit(&prep, &x, &z, coeff);
736                assert_terms_eq(&direct, &emitted, &format!("{label} adjoint={adjoint}"));
737            }
738        }
739    }
