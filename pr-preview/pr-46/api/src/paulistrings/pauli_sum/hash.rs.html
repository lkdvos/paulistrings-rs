3use crate::pauli_string::word_mask;
4use crate::rng::{mix64, Rng, SPLITMIX_GAMMA};

6/// Maximum number of bucket bits; all rows are drawn up front so the active hash is a prefix of one fixed matrix.
7pub const B_MAX_BITS: u8 = 22;
8
9/// Maximum number of partition bits, i.e. `P ≤ 64` partitions.
10pub const P_MAX_BITS: u8 = 6;
11
12/// Salt keeping [`PartitionRows::from_seed`]'s rows independent of [`Gf2Hash::new`]'s under the same seed.
13const PARTITION_ROW_SALT: u64 = 0xD1B5_4A32_D192_ED03;
14
15/// Salts keeping the sparse and projection families' coordinate draws independent of each other and of the dense rows.
16const SPARSE_SALT: u64 = 0x2545_F491_4F6C_DD1D;
17const PROJECTION_SALT: u64 = 0x94D0_49BB_1331_11EB;
18
19/// How [`Gf2Hash`]'s rows are drawn.
20///
21/// Every family is GF(2)-linear, so every family partitions correctly, and full rank: its first `min(B_MAX_BITS, 2 · num_qubits)` rows are linearly independent, so every active bucket is reachable.
22/// They differ in how evenly they spread a sum over the buckets.
23#[derive(Clone, Copy, Debug, Default, PartialEq, Eq, Hash)]
24pub enum HashFamily {
25    /// Every live coordinate set with probability 1/2, a universal hash.
26    #[default]
27    Dense,
28    /// Exactly `weight` live coordinates per row, drawn without replacement.
29    Sparse {
30        /// Coordinates set per row; [`HashFamily::check`] says which weights reach full rank.
31        weight: u16,
32    },
33    /// One live coordinate per row, distinct across the first `2 · num_qubits` rows and cycling after: a coordinate projection.
34    Projection,
35}
36
37impl HashFamily {
38    /// `Err` with the reason unless this family's rows can be full rank over `num_qubits` qubits.
39    ///
40    /// Weight-`w` rows over `m = 2 · num_qubits` coordinates span the whole space for odd `w < m`, the even-weight subspace of dimension `m - 1` for even `w < m`, and one vector for `w = m`; the hash needs `min(B_MAX_BITS, m)` independent rows.
41    pub fn check(self, num_qubits: usize) -> Result<(), String> {
42        let HashFamily::Sparse { weight } = self else {
43            return Ok(());
44        };
45        let coordinates = 2 * num_qubits;
46        let weight = weight as usize;
47        if num_qubits == 0 {
48            return Ok(());
49        }
50        if !(1..=coordinates).contains(&weight) {
51            return Err(format!("HashFamily::Sparse: weight {weight} must lie in 1..={coordinates} for {num_qubits} qubits"));
52        }
53        let reachable = if weight == coordinates {
54            1
55        } else if weight % 2 == 1 {
56            coordinates
57        } else {
58            coordinates - 1
59        };
60        let needed = coordinates.min(B_MAX_BITS as usize);
61        if reachable < needed {
62            return Err(format!(
63                "HashFamily::Sparse: weight-{weight} rows over {num_qubits} qubits span only {reachable} of the {needed} independent rows a full-rank hash needs"
64            ));
65        }
66        Ok(())
67    }
68}

70/// Word `word` of the `half` (0 = x, 1 = z) of row `row`, draw `attempt`.
71// Not a GF(2)-linear generator such as xorshift: research/FINDINGS.md §`Gf2Hash` rows are splitmix64, not xorshift successors.
72fn row_word(seed: u64, row: usize, attempt: u32, word: usize, half: u64) -> u64 {
73    debug_assert!(row < 1 << 16 && word < 1 << 15);
74    let position = ((attempt as u64) << 32) | ((row as u64) << 16) | ((word as u64) << 1) | half;
75    mix64(seed.wrapping_add(SPLITMIX_GAMMA.wrapping_mul(position.wrapping_add(1))))
76}

78/// A GF(2) matrix over the key space `(x, z)`, each row masked to the live qubit columns; `Gf2Hash` and `PartitionRows` are its two uses.
79#[derive(Clone, Debug, PartialEq, Eq)]
80pub(crate) struct Gf2Matrix<const W: usize> {
81    rows_x: Vec<[u64; W]>,
82    rows_z: Vec<[u64; W]>,
83    num_qubits: usize,
84}
85
86impl<const W: usize> Gf2Matrix<W> {
87    /// `num_rows` rows drawn from `seed` with the `exclude` coordinates cleared, an all-zero row redrawn unless `num_qubits == 0`.
88    pub(crate) fn draw(
89        num_qubits: usize,
90        num_rows: usize,
91        seed: u64,
92        exclude_x: &[u64; W],
93        exclude_z: &[u64; W],
94    ) -> Self {
95        let mut rows_x: Vec<[u64; W]> = Vec::with_capacity(num_rows);
96        let mut rows_z: Vec<[u64; W]> = Vec::with_capacity(num_rows);
97        let has_live_columns = num_qubits > 0;
98        for row in 0..num_rows {
99            let mut attempt = 0u32;
100            let (row_x, row_z) = loop {
101                let mut row_x = [0u64; W];
102                let mut row_z = [0u64; W];
103                let mut any = false;
104                for w in 0..W {
105                    let mask = word_mask(num_qubits, w);
106                    row_x[w] = row_word(seed, row, attempt, w, 0) & mask & !exclude_x[w];
107                    row_z[w] = row_word(seed, row, attempt, w, 1) & mask & !exclude_z[w];
108                    any |= (row_x[w] | row_z[w]) != 0;
109                }
110                if any || !has_live_columns {
111                    break (row_x, row_z);
112                }
113                attempt += 1;
114            };
115            rows_x.push(row_x);
116            rows_z.push(row_z);
117        }
118        Self { rows_x, rows_z, num_qubits }
119    }
120
121    /// `num_rows` rows of `family` drawn from `seed`, the first `min(num_rows, 2 · num_qubits)` linearly independent and every later one nonzero.
122    /// A candidate failing that is redrawn, so the rows depend on `(num_qubits, seed, family)` alone, not on `W`.
123    ///
124    /// # Panics
125    ///
126    /// If [`HashFamily::check`] rejects `family` for `num_qubits`.
127    pub(crate) fn draw_family(
128        num_qubits: usize,
129        num_rows: usize,
130        seed: u64,
131        family: HashFamily,
132    ) -> Self {
133        if let Err(reason) = family.check(num_qubits) {
134            panic!("{reason}");
135        }
136        let coordinates = 2 * num_qubits;
137        if num_qubits == 0 {
138            return Self::from_rows(0, vec![[0; W]; num_rows], vec![[0; W]; num_rows]);
139        }
140        let projection_order = matches!(family, HashFamily::Projection).then(|| {
141            partial_shuffle(coordinates, coordinates, &mut Rng::from_key(&[PROJECTION_SALT, seed]))
142        });
143        let candidate = |row: usize, attempt: u32| -> KeyRow<W> {
144            match family {
145                HashFamily::Dense => {
146                    let mut key = ([0u64; W], [0u64; W]);
147                    for w in 0..W {
148                        let mask = word_mask(num_qubits, w);
149                        key.0[w] = row_word(seed, row, attempt, w, 0) & mask;
150                        key.1[w] = row_word(seed, row, attempt, w, 1) & mask;
151                    }
152                    key
153                }
154                HashFamily::Sparse { weight } => {
155                    let mut stream =
156                        Rng::from_key(&[SPARSE_SALT, seed, row as u64, attempt as u64]);
157                    coordinate_row(
158                        num_qubits,
159                        &partial_shuffle(coordinates, weight as usize, &mut stream),
160                    )
161                }
162                HashFamily::Projection => {
163                    let order =
164                        projection_order.as_ref().expect("drawn above for the projection family");
165                    coordinate_row(num_qubits, &[order[row % coordinates]])
166                }
167            }
168        };
169        let independent = coordinates.min(num_rows);
170        let mut basis = Gf2Basis::<W>::default();
171        let mut rows_x = Vec::with_capacity(num_rows);
172        let mut rows_z = Vec::with_capacity(num_rows);
173        for row in 0..num_rows {
174            let mut attempt = 0u32;
175            let key = loop {
176                let key = candidate(row, attempt);
177                let accepted =
178                    if row < independent { basis.insert(&key) } else { key != ([0; W], [0; W]) };
179                if accepted {
180                    break key;
181                }
182                attempt += 1;
183                assert!(
184                    attempt < 1 << 16,
185                    "{family:?}: no acceptable row {row} in 65536 draws over {num_qubits} qubits"
186                );
187            };
188            rows_x.push(key.0);
189            rows_z.push(key.1);
190        }
191        Self::from_rows(num_qubits, rows_x, rows_z)
192    }

194    /// Explicit rows, masked to the live columns.
195    pub(crate) fn from_rows(
196        num_qubits: usize,
197        mut rows_x: Vec<[u64; W]>,
198        mut rows_z: Vec<[u64; W]>,
199    ) -> Self {
200        assert_eq!(rows_x.len(), rows_z.len(), "Gf2Matrix: row count mismatch");
201        for (row_x, row_z) in rows_x.iter_mut().zip(rows_z.iter_mut()) {
202            for w in 0..W {
203                let mask = word_mask(num_qubits, w);
204                row_x[w] &= mask;
205                row_z[w] &= mask;
206            }
207        }
208        Self { rows_x, rows_z, num_qubits }
209    }
210
211    pub(crate) fn num_rows(&self) -> usize {
212        self.rows_x.len()
213    }
214
215    pub(crate) fn num_qubits(&self) -> usize {
216        self.num_qubits
217    }
218
219    pub(crate) fn row(&self, i: usize) -> KeyRow<W> {
220        (self.rows_x[i], self.rows_z[i])
221    }
222
223    pub(crate) fn rows(&self) -> (&[[u64; W]], &[[u64; W]]) {
224        (&self.rows_x, &self.rows_z)
225    }

227    /// Bit `row` of `M·v`, as `0` or `1`.
228    pub(crate) fn row_parity(&self, x: &[u64; W], z: &[u64; W], row: usize) -> u32 {
229        let row_x = &self.rows_x[row];
230        let row_z = &self.rows_z[row];
231        let mut folded: u64 = 0;
232        for w in 0..W {
233            folded ^= (x[w] & row_x[w]) ^ (z[w] & row_z[w]);
234        }
235        folded.count_ones() & 1
236    }
237
238    /// The first `rows` bits of `M·v`, row `i` at bit `i`.
239    pub(crate) fn apply(&self, x: &[u64; W], z: &[u64; W], rows: usize) -> u32 {
240        let mut index: u32 = 0;
241        for i in 0..rows {
242            index |= self.row_parity(x, z, i) << i;
243        }
244        index
245    }
246
247    /// Whether this matrix's first `rows` rows and `other`'s first `other_rows` are jointly GF(2)-independent.
248    pub(crate) fn independent_with(&self, rows: usize, other: &Self, other_rows: usize) -> bool {
249        let stacked: Vec<KeyRow<W>> =
250            (0..rows).map(|i| self.row(i)).chain((0..other_rows).map(|i| other.row(i))).collect();
251        gf2_rank_wide(&stacked) == rows + other_rows
252    }

255/// The GF(2)-linear hash `h(v) = H·v` from a Pauli key `v = (x, z)` to a bucket index, for an `H` drawn from a seed by a [`HashFamily`], dense random by default.
256///
257/// Linearity gives `h(v ⊕ d) = h(v) ⊕ h(d)`, so a channel's output buckets are predictable from its input buckets.
258/// The active rows are a prefix of one fixed matrix, so a refine splits every bucket in two by one row parity with the within-bucket order inherited, and a coarsen merges bucket pairs; neither needs a sort to keep the [`PauliSum`](crate::PauliSum) invariant.
259#[derive(Clone, Debug)]
260pub struct Gf2Hash<const W: usize> {
261    /// All [`B_MAX_BITS`] rows of `H`.
262    matrix: Gf2Matrix<W>,
263    /// Active prefix length: `B = 1 << bits` buckets.
264    bits: u8,
265    /// Seed the rows were generated from.
266    seed: u64,
267    /// How the rows were drawn.
268    family: HashFamily,
269}
270
271impl<const W: usize> Gf2Hash<W> {
272    /// Build a hash over `num_qubits` qubits with `bits` active bucket bits.
273    ///
274    /// The rows are a function of `(num_qubits, seed)` alone, independent of `W`.
275    ///
276    /// # Panics
277    ///
278    /// Panics if `bits > B_MAX_BITS`, or in debug builds if `num_qubits > 64 · W`.
279    pub fn new(num_qubits: usize, bits: u8, seed: u64) -> Self {
280        Self::with_family(num_qubits, bits, seed, HashFamily::Dense)
281    }
282
283    /// As [`Gf2Hash::new`], with the rows drawn by `family`.
284    ///
285    /// # Panics
286    ///
287    /// As [`Gf2Hash::new`], and if [`HashFamily::check`] rejects `family` for `num_qubits`.
288    pub fn with_family(num_qubits: usize, bits: u8, seed: u64, family: HashFamily) -> Self {
289        assert!(bits <= B_MAX_BITS, "Gf2Hash: bits {bits} exceeds B_MAX_BITS {B_MAX_BITS}",);
290        debug_assert!(num_qubits <= 64 * W);
291
292        Self {
293            matrix: Gf2Matrix::draw_family(num_qubits, B_MAX_BITS as usize, seed, family),
294            bits,
295            seed,
296            family,
297        }
298    }
299
300    /// Number of active bucket bits.
301    pub fn bits(&self) -> u8 {
302        self.bits
303    }
304
305    /// Number of buckets, `1 << bits()`.
306    pub fn num_buckets(&self) -> usize {
307        1usize << self.bits
308    }
309
310    /// The seed the rows were generated from.
311    pub fn seed(&self) -> u64 {
312        self.seed
313    }
314
315    /// The family the rows were drawn by.
316    pub fn family(&self) -> HashFamily {
317        self.family
318    }
319
320    /// The qubit count the rows were masked against.
321    pub fn num_qubits(&self) -> usize {
322        self.matrix.num_qubits()
323    }
324
325    /// `h(v)` for a key given as separate `x` and `z` words.
326    pub fn bucket_of(&self, x: &[u64; W], z: &[u64; W]) -> u32 {
327        self.matrix.apply(x, z, self.bits as usize)
328    }
329
330    /// Bit `row` of `H·v`, as `0` or `1`.
331    pub(super) fn row_parity(&self, x: &[u64; W], z: &[u64; W], row: u8) -> u32 {
332        self.matrix.row_parity(x, z, row as usize)
333    }
334
335    /// Double the bucket count: `B → 2B`.
336    ///
337    /// # Panics
338    ///
339    /// Panics if already at `B_MAX_BITS`.
340    pub fn refine(&mut self) {
341        assert!(self.bits < B_MAX_BITS, "Gf2Hash::refine: already at B_MAX_BITS {B_MAX_BITS}",);
342        self.bits += 1;
343    }
344
345    /// Halve the bucket count: `B → B/2`, merging buckets `b` and `b + B/2`.
346    ///
347    /// # Panics
348    ///
349    /// Panics if already at a single bucket.
350    pub fn coarsen(&mut self) {
351        assert!(self.bits > 0, "Gf2Hash::coarsen: already at a single bucket");
352        self.bits -= 1;
353    }
354
355    /// `true` if `other` was generated with the same rows, so sums partitioned by the two can be combined (after matching `bits`).
356    pub(crate) fn same_rows_as(&self, other: &Self) -> bool {
357        self.seed == other.seed
358            && self.family == other.family
359            && self.num_qubits() == other.num_qubits()
360    }
361
362    /// Row `i < B_MAX_BITS` of `H`, which may lie beyond the active prefix.
363    #[cfg(any(test, feature = "cuda"))]
364    pub(crate) fn row(&self, i: usize) -> ([u64; W], [u64; W]) {
365        self.matrix.row(i)
366    }
367}

369/// The `p` GF(2) rows `part(v) = P·v` that split a sum across `2^p` partitions: a partition holds exactly the terms with `part(v)` equal to its rank.
370///
371/// A matrix separate from [`Gf2Hash`]'s and fixed for a run, since a row that moved as the bucket hash refines would change a term's owner mid-run.
372/// Dependence on the bucket hash's rows costs load balance, never correctness.
373#[derive(Clone, Debug, PartialEq, Eq)]
374pub struct PartitionRows<const W: usize> {
375    /// One row per partition bit: `P = 1 << rows` partitions.
376    matrix: Gf2Matrix<W>,
377}
378
379impl<const W: usize> PartitionRows<W> {
380    /// Panics unless the rows split a sum over `num_qubits` qubits into `partitions` partitions; debug builds also check them independent of `hash`.
381    pub(crate) fn assert_splits(&self, hash: &Gf2Hash<W>, num_qubits: usize, partitions: usize) {
382        assert_eq!(
383            self.num_partitions(),
384            partitions,
385            "partition rows name {} partitions for a split into {partitions}",
386            self.num_partitions(),
387        );
388        assert_eq!(
389            self.num_qubits(),
390            num_qubits,
391            "partition rows are for {} qubits, the sum for {num_qubits}",
392            self.num_qubits(),
393        );
394        debug_assert!(
395            self.is_independent_of(hash),
396            "partition rows are dependent on the bucket hash rows — the split will correlate \
397             with the bucket partition and load-balance badly",
398        );
399    }
400
401    /// Draw `bits` partition rows from `seed`, unrelated to [`Gf2Hash::new`]'s rows under the same seed.
402    ///
403    /// # Panics
404    ///
405    /// Panics if `bits > P_MAX_BITS`, or in debug builds if `num_qubits > 64 · W`.
406    pub fn from_seed(num_qubits: usize, bits: u8, seed: u64) -> Self {
407        Self::from_seed_excluding(num_qubits, bits, seed, &[0; W], &[0; W])
408    }
409
410    /// [`Self::from_seed`] with the coordinates in `(exclude_x, exclude_z)` cleared from every row, so keys differing only there share a partition.
411    ///
412    /// # Panics
413    ///
414    /// As [`Self::from_seed`], and if `bits > 0` while the exclusion covers every live column.
415    pub fn from_seed_excluding(
416        num_qubits: usize,
417        bits: u8,
418        seed: u64,
419        exclude_x: &[u64; W],
420        exclude_z: &[u64; W],
421    ) -> Self {
422        assert!(bits <= P_MAX_BITS, "PartitionRows: bits {bits} exceeds P_MAX_BITS {P_MAX_BITS}",);
423        debug_assert!(num_qubits <= 64 * W);
424
425        let keeps_a_column = (0..W).any(|w| {
426            let mask = word_mask(num_qubits, w);
427            (mask & !exclude_x[w]) | (mask & !exclude_z[w]) != 0
428        });
429        assert!(
430            bits == 0 || num_qubits == 0 || keeps_a_column,
431            "PartitionRows::from_seed_excluding: the exclusion covers every column",
432        );
433        Self {
434            matrix: Gf2Matrix::draw(
435                num_qubits,
436                bits as usize,
437                mix64(seed) ^ PARTITION_ROW_SALT,
438                exclude_x,
439                exclude_z,
440            ),
441        }
442    }

444    /// Build from explicit rows, masked to the live qubit columns.
445    ///
446    /// # Panics
447    ///
448    /// Panics if `rows_x` and `rows_z` differ in length, if there are more than [`P_MAX_BITS`] rows, or if any row masks to all-zero while `num_qubits > 0`.
449    pub(crate) fn from_rows(
450        num_qubits: usize,
451        rows_x: Vec<[u64; W]>,
452        rows_z: Vec<[u64; W]>,
453    ) -> Self {
454        assert_eq!(
455            rows_x.len(),
456            rows_z.len(),
457            "PartitionRows::from_rows: row count mismatch, {} x-rows vs {} z-rows",
458            rows_x.len(),
459            rows_z.len(),
460        );
461        assert!(
462            rows_x.len() <= P_MAX_BITS as usize,
463            "PartitionRows: bits {} exceeds P_MAX_BITS {P_MAX_BITS}",
464            rows_x.len(),
465        );
466        debug_assert!(num_qubits <= 64 * W);
467
468        let matrix = Gf2Matrix::from_rows(num_qubits, rows_x, rows_z);
469        for i in 0..matrix.num_rows() {
470            let (row_x, row_z) = matrix.row(i);
471            assert!(
472                num_qubits == 0 || row_x.iter().chain(&row_z).any(|&w| w != 0),
473                "PartitionRows::from_rows: row {i} masks to zero",
474            );
475        }
476        Self { matrix }
477    }

479    /// Rows that label a qubit cut: `log2(blocks.len())` z-only rows giving block `b` the label `b`.
480    ///
481    /// A term's label is the XOR of the labels of the blocks it has odd z-weight in, so an x-only generator is local under any cut and a `ZZ(i, j)` bond is remote exactly when it crosses blocks.
482    /// Blocks must be disjoint; qubits in no block behave as if in block 0.
483    ///
484    /// # Panics
485    ///
486    /// Panics if `blocks.len()` is not a power of two or exceeds `2^P_MAX_BITS`, if a qubit is `>= num_qubits` or appears in two blocks, or if some row would be all-zero.
487    pub fn cut(num_qubits: usize, blocks: &[Vec<usize>]) -> Self {
488        assert!(
489            blocks.len().is_power_of_two(),
490            "PartitionRows::cut: block count {} is not a power of two",
491            blocks.len(),
492        );
493        let bits = blocks.len().trailing_zeros() as u8;
494        assert!(bits <= P_MAX_BITS, "PartitionRows: bits {bits} exceeds P_MAX_BITS {P_MAX_BITS}",);
495        debug_assert!(num_qubits <= 64 * W);
496
497        let mut seen = vec![false; num_qubits];
498        let mut rows_z = vec![[0u64; W]; bits as usize];
499        for (b, qubits) in blocks.iter().enumerate() {
500            for &q in qubits {
501                assert!(
502                    q < num_qubits,
503                    "PartitionRows::cut: qubit {q} in block {b} is outside 0..{num_qubits}",
504                );
505                assert!(
506                    !seen[q],
507                    "PartitionRows::cut: blocks must be disjoint, qubit {q} appears twice",
508                );
509                seen[q] = true;
510                for (i, row) in rows_z.iter_mut().enumerate() {
511                    if (b >> i) & 1 == 1 {
512                        row[q / 64] |= 1u64 << (q % 64);
513                    }
514                }
515            }
516        }
517        for (i, row) in rows_z.iter().enumerate() {
518            assert!(
519                row.iter().any(|w| *w != 0),
520                "PartitionRows::cut: row {i} is empty — no qubit lies in a block \
521                 whose label has bit {i} set",
522            );
523        }
524
525        Self::from_rows(num_qubits, vec![[0u64; W]; bits as usize], rows_z)
526    }

531    }

556    }

558    /// `true` if no row reads a coordinate in `(mask_x, mask_z)`, i.e. keys differing only there always share a partition.
559    pub fn avoids(&self, mask_x: &[u64; W], mask_z: &[u64; W]) -> bool {
560        let (rows_x, rows_z) = self.matrix.rows();
561        rows_x.iter().zip(rows_z).all(|(row_x, row_z)| {
562            (0..W).all(|w| row_x[w] & mask_x[w] == 0 && row_z[w] & mask_z[w] == 0)
563        })
564    }
565
566    /// `true` if the partition rows and `hash`'s active rows are jointly GF(2)-independent.
567    pub fn is_independent_of(&self, hash: &Gf2Hash<W>) -> bool {
568        self.matrix.independent_with(self.matrix.num_rows(), &hash.matrix, hash.bits() as usize)
569    }
570}

572/// One row of the key space as `(x-masks, z-masks)`.
573type KeyRow<const W: usize> = ([u64; W], [u64; W]);
574
575/// GF(2) rank of rows over the key space, by Gaussian elimination.
576fn gf2_rank_wide<const W: usize>(rows: &[KeyRow<W>]) -> usize {
577    let mut pivots: Vec<(usize, KeyRow<W>)> = Vec::with_capacity(rows.len());
578    for row in rows {
579        let mut reduced = *row;
580        'reduce: loop {
581            let Some(lead) = leading_column(&reduced) else {
582                break;
583            };
584            for (pivot_column, pivot_row) in pivots.iter() {
585                if *pivot_column == lead {
586                    for w in 0..W {
587                        reduced.0[w] ^= pivot_row.0[w];
588                        reduced.1[w] ^= pivot_row.1[w];
589                    }
590                    continue 'reduce;
591                }
592            }
593            pivots.push((lead, reduced));
594            break;
595        }
596    }
597    pivots.len()
598}
599
600/// Index of the highest set column of a key-space row, or `None` if it is zero.
601fn leading_column<const W: usize>(row: &KeyRow<W>) -> Option<usize> {
602    for w in (0..W).rev() {
603        if row.1[w] != 0 {
604            return Some(W * 64 + w * 64 + (63 - row.1[w].leading_zeros() as usize));
605        }
606    }
607    for w in (0..W).rev() {
608        if row.0[w] != 0 {
609            return Some(w * 64 + (63 - row.0[w].leading_zeros() as usize));
610        }
611    }
612    None
613}

618/// The first `count` entries of a uniformly random permutation of `0..len`, by a partial Fisher–Yates shuffle.
619fn partial_shuffle(len: usize, count: usize, stream: &mut Rng) -> Vec<usize> {
620    let mut order: Vec<usize> = (0..len).collect();
621    for i in 0..count {
622        let j = i + (stream.next_u64() % (len - i) as u64) as usize;
623        order.swap(i, j);
624    }
625    order.truncate(count);
626    order
627}

629/// The row setting each of `coordinates`: `c` is the x-bit of qubit `c` below `num_qubits` and the z-bit of qubit `c - num_qubits` above.
630fn coordinate_row<const W: usize>(num_qubits: usize, coordinates: &[usize]) -> KeyRow<W> {
631    let mut key = ([0u64; W], [0u64; W]);
632    for &c in coordinates {
633        let (half, q) = if c < num_qubits { (&mut key.0, c) } else { (&mut key.1, c - num_qubits) };
634        half[q / 64] |= 1u64 << (q % 64);
635    }
636    key
637}

639/// An incrementally grown row-echelon basis of key-space rows.
640struct Gf2Basis<const W: usize> {
641    pivots: Vec<(usize, KeyRow<W>)>,
642}
643
644impl<const W: usize> Default for Gf2Basis<W> {
645    fn default() -> Self {
646        Self { pivots: Vec::new() }
647    }
648}
649
650impl<const W: usize> Gf2Basis<W> {
651    /// Add `row` and return `true` if it is independent of the rows so far; leave the basis unchanged and return `false` otherwise.
652    fn insert(&mut self, row: &KeyRow<W>) -> bool {
653        let mut reduced = *row;
654        while let Some(lead) = leading_column(&reduced) {
655            match self.pivots.iter().find(|(column, _)| *column == lead) {
656                Some((_, pivot)) => {
657                    for w in 0..W {
658                        reduced.0[w] ^= pivot.0[w];
659                        reduced.1[w] ^= pivot.1[w];
660                    }
661                }
662                None => {
663                    self.pivots.push((lead, reduced));
664                    return true;
665                }
666            }
667        }
668        false
669    }
670}
