2//! See ARCHITECTURE.md §Partitioning.

14/// The policy-independent half of a partition: what the layer loop reads and reshapes between layers.
15pub(crate) trait PartitionStorage<const W: usize>: Send + Sized {
16    /// Terms this partition holds.
17    fn len(&self) -> usize;
18    /// The hash and bucket count every partition of the group shares.
19    fn hash(&self) -> &Gf2Hash<W>;
20    /// Add one bucket bit, as [`PauliSum::refine`].
21    fn refine(&mut self);
22    /// The bucket bits this partition proposes to the group's agreement for the layer `prep` prepares: the host formula, which a backend may raise but never lower.
23    fn proposed_bits(
24        &self,
25        prep: &Prepared<W>,
26        target_bucket_len: usize,
27        min_buckets: usize,
28    ) -> u8 {
29        let _ = prep;
30        crate::bucket::sum::desired_bits(self.len(), target_bucket_len, min_buckets)
31            .max(self.hash().bits())
32    }
33    /// Move the partition out, leaving a valid empty partition under the same hash behind.
34    ///
35    /// The placeholder keeps a driver self-consistent (same rows, same hash, same bucket count on every partition) if the partition panics and the work is never handed back.
36    fn detach(&mut self) -> Self;
37    /// The phase counters the loop laps into.
38    #[cfg(feature = "phase-timing")]
39    fn stats(&mut self) -> &mut PhaseStats;
40}

42/// The layer itself under policy type `T`.
43///
44/// Everything collective stays in `run_layers`; an implementor must issue exactly the transport calls the host layer issues, in the same order, or the group falls out of step.
45pub(crate) trait PartitionBackend<const W: usize, T: ?Sized>: PartitionStorage<W> {
46    /// One layer's export, exchange and merge, as [`apply_layer_partitioned_with_plan`].
47    fn apply_layer<X: Transport>(
48        &mut self,
49        prep: &Prepared<W>,
50        plan: &PartitionPlan,
51        rows: &PartitionRows<W>,
52        policy: &T,
53        transport: &X,
54    ) -> LayerExchangeCounts;
55    /// The policy's collective layer pass, as [`PartitionedTruncation::finalize_layer_partitioned`].
56    fn finalize_layer(&mut self, policy: &T, coll: &dyn Collectives);
57}

59/// A partition held in host memory: its sum and its layer and export scratch, retained across calls.
60///
61/// Nominally `pub` only so it can be `DistributedSum`'s default backend; the module is crate-private, so it cannot be named outside the crate.
62#[derive(Debug)]
63pub struct HostPartition<const W: usize> {
64    pub(crate) sum: PauliSum<W>,
65    pub(super) state: PartitionState<W>,
66}
67
68impl<const W: usize> HostPartition<W> {
69    /// `sum` with fresh scratch.
70    pub(crate) fn new(sum: PauliSum<W>) -> Self {
71        Self {
72            sum,
73            state: PartitionState::default(),
74        }
75    }
76}
77
78impl<const W: usize> PartitionStorage<W> for HostPartition<W> {
79    #[inline]
80    fn len(&self) -> usize {
81        self.sum.len()
82    }
83
84    #[inline]
85    fn hash(&self) -> &Gf2Hash<W> {
86        self.sum.hash()
87    }
88
89    #[inline]
90    fn refine(&mut self) {
91        self.sum.refine();
92    }
93
94    fn detach(&mut self) -> Self {
95        let placeholder = PauliSum::empty_with_hash(self.sum.num_qubits(), self.sum.hash().clone());
96        Self {
97            sum: std::mem::replace(&mut self.sum, placeholder),
98            state: std::mem::take(&mut self.state),
99        }
100    }
101
102    #[cfg(feature = "phase-timing")]
103    #[inline]
104    fn stats(&mut self) -> &mut PhaseStats {
105        &mut self.state.layer.stats
106    }
107}
108
109impl<const W: usize, T> PartitionBackend<W, T> for HostPartition<W>
110where
111    T: PartitionedTruncation<W> + ?Sized,
112{
113    #[inline]
114    fn apply_layer<X: Transport>(
115        &mut self,
116        prep: &Prepared<W>,
117        plan: &PartitionPlan,
118        rows: &PartitionRows<W>,
119        policy: &T,
120        transport: &X,
121    ) -> LayerExchangeCounts {
122        apply_layer_partitioned_with_plan(
123            &mut self.sum,
124            prep,
125            plan,
126            rows,
127            policy,
128            &mut self.state,
129            transport,
130        )
131    }
132
133    #[inline]
134    fn finalize_layer(&mut self, policy: &T, coll: &dyn Collectives) {
135        policy.finalize_layer_partitioned(&mut self.sum, coll);
136    }
137}

139#[cfg(test)]
140mod tests {
141    use super::*;
142    use crate::test_support::{assert_same_terms, rand_sum};
143
144    fn detach_leaves_an_empty_partition_under_the_same_hash<const W: usize>(num_qubits: usize) {
145        let mut sum = rand_sum::<W>(3_000, num_qubits, 0xDE7AC4 + W as u64);
146        sum.refine();
147        sum.refine();
148        let want = sum.clone();
149        let mut part = HostPartition::new(sum);
150
151        let moved = part.detach();
152
153        assert_eq!(part.len(), 0);
154        assert!(part.sum.is_empty());
155        assert_eq!(part.hash().bits(), want.hash().bits());
156        assert!(part.hash().same_rows_as(want.hash()));
157        assert_eq!(part.sum.num_qubits(), want.num_qubits());
158        assert_eq!(part.sum.num_buckets(), want.num_buckets());
159        part.sum.assert_invariants();
160
161        assert_eq!(moved.len(), want.len());
162        assert_same_terms(&moved.sum, &want, "detached partition");
163        moved.sum.assert_invariants();
164    }
165
166    #[test]
167    fn detach_leaves_an_empty_partition_under_the_same_hash_w1() {
168        detach_leaves_an_empty_partition_under_the_same_hash::<1>(40);
169    }
170
171    #[test]
172    fn detach_leaves_an_empty_partition_under_the_same_hash_w2() {
173        detach_leaves_an_empty_partition_under_the_same_hash::<2>(100);
174    }
