1//! The export pass: a layer's rows that belong to another partition, gathered into one [`ExchangeBlock`] per remote delta.
2//!
3//! A row of remote delta `d` from local bucket `β` belongs to partition `rank ⊕ part(d)`, bucket `β ⊕ h(d)`, so the sender fills segment `p` from its bucket `bucket_at(p) ⊕ h(d)` and the block arrives in the receiver's coset order.
4//! A partner gets one block per remote delta to it, in ascending plan order and empty ones included, since the receiver walks them in lockstep with its own plan.
5//! Both passes skip the terms a pending `ApproxTopN` cut drops, which the coset loop applies after, and where the last coset loop counted this layer's rows ahead ([`LayerOctaves::bucket_ahead`]) pass 1 reads those counts and the fill visits only the marked rows.
6

20/// Fill work ([`fill_work`]) below which a block's fill stays on one thread.
21const FILL_PARALLEL_MIN_WORK: usize = 16384;
22
23/// A marked row's fill cost in scanned input terms: a marked fill reaches each row at a cache miss (§Marked export fills split by rows).
24const MARKED_ROW_WORK: usize = 32;
25
26/// Reusable scratch for [`export_layer`], so a steady-state layer allocates nothing.
27#[derive(Debug, Default)]
28pub(crate) struct ExportScratch<const W: usize, C: Coefficient = num_complex::Complex64> {
29    /// Pass-1 counts, bucket-major: `counts[β * num_remote + i]`.
30    counts: Vec<u32>,
31    /// One block's counts by destination position.
32    block_counts: Vec<u32>,
33    /// `source_of[position] = order.bucket_at(position) ^ bucket_delta`.
34    source_of: Vec<u32>,
35    /// Pass-1 fill work in scanned input terms ([`fill_work`]), laid out like `counts`.
36    work: Vec<usize>,
37    /// One block's prefix sums of `work`, by destination position.
38    work_prefix: Vec<usize>,
39    /// Payloads not in flight, columns intact; outgoing and incoming payloads are both drawn from here.
40    pub(crate) pool: Vec<PartnerPayload<W, C>>,
41    /// Test hook: the buckets whose pass 1 read the counts ahead under a pending cut, summed over layers.
42    #[cfg(any(test, feature = "test-utils"))]
43    pub(crate) buckets_ahead_under_cut: u64,
44}

46/// This partition's outgoing payloads for one layer, one slot per partner rank, from `local` as its pending cut in `octaves` leaves it.
47///
48/// A partner's blocks are one per remote delta destined for it in ascending order, empty ones included, since the receiver indexes blocks positionally.
49pub(crate) fn export_layer<const W: usize, C: Coefficient>(
50    local: &PauliSum<W, C>,
51    octaves: &LayerOctaves,
52    prepared: &Prepared<W, C>,
53    plan: &PartitionPlan,
54    size: u32,
55    order: &CosetOrder,
56    scratch: &mut ExportScratch<W, C>,
57) -> (Vec<Option<PartnerPayload<W, C>>>, LayerExchangeCounts) {
58    export_layer_with(local, octaves, prepared, plan, size, order, scratch, FILL_PARALLEL_MIN_WORK)
59}

61/// [`export_layer`] with the fill's leaf size in fill work ([`fill_work`]).
62#[allow(clippy::too_many_arguments)]
63fn export_layer_with<const W: usize, C: Coefficient>(
64    local: &PauliSum<W, C>,
65    octaves: &LayerOctaves,
66    prepared: &Prepared<W, C>,
67    plan: &PartitionPlan,
68    size: u32,
69    order: &CosetOrder,
70    scratch: &mut ExportScratch<W, C>,
71    min_work: usize,
72) -> (Vec<Option<PartnerPayload<W, C>>>, LayerExchangeCounts) {
73    let num_remote = plan.remote.len();
74    let num_buckets = local.num_buckets();
75    let cut = octaves.pending();
76    let ahead = Some(octaves).filter(|octaves| num_remote == 1 && octaves.has_ahead());
77    let mut send: Vec<Option<PartnerPayload<W, C>>> = (0..size).map(|_| None).collect();
78    let mut counts = LayerExchangeCounts::none(size);
79    if num_remote == 0 {
80        return (send, counts);
81    }
82
83    let emitters: Vec<RowEmitter<'_, W, C>> =
84        plan.remote.iter().map(|remote| RowEmitter::of(prepared, remote)).collect();
85    let dense: Vec<bool> = emitters.iter().map(RowEmitter::is_dense).collect();
86    let all_dense = dense.iter().all(|&is_dense| is_dense);
87
88    // Pass 1: bucket-major counts and fill work, `counts[β * num_remote + i]`, so each bucket owns one disjoint chunk.
89    scratch.counts.clear();
90    scratch.counts.resize(num_buckets * num_remote, 0);
91    scratch.work.clear();
92    scratch.work.resize(num_buckets * num_remote, 0);
93    let slots =
94        scratch.counts.par_chunks_mut(num_remote).zip(scratch.work.par_chunks_mut(num_remote));
95    slots.enumerate().for_each(|(bucket, (slot, work))| {
96        let len = local.bucket_len(bucket);
97        let counted = |slot: &mut [u32]| {
98            let terms = local.bucket(bucket);
99            match cut {
100                None if all_dense => slot.fill(terms.2.len() as u32),
101                None => count_bucket(terms, prepared, plan, &dense, |_, _, _| true, slot),
102                Some(cut) => count_bucket(terms, prepared, plan, &dense, cut.filter(bucket), slot),
103            }
104        };
105        match ahead.and_then(|octaves| octaves.rows_ahead(bucket)) {
106            Some(rows) => {
107                slot[0] = rows;
108                work[0] = fill_work(len, rows, true);
109                #[cfg(debug_assertions)]
110                {
111                    let mut recounted = [0u32];
112                    counted(&mut recounted);
113                    debug_assert_eq!(
114                        rows, recounted[0],
115                        "bucket {bucket}: the counts ahead disagree"
116                    );
117                }
118            }
119            None => {
120                counted(slot);
121                for (work, &rows) in work.iter_mut().zip(slot.iter()) {
122                    *work = fill_work(len, rows, false);
123                }
124            }
125        }
126    });
127    #[cfg(any(test, feature = "test-utils"))]
128    if let (Some(octaves), Some(_)) = (ahead, cut) {
129        scratch.buckets_ahead_under_cut +=
130            (0..num_buckets).filter(|&bucket| octaves.rows_ahead(bucket).is_some()).count() as u64;
131    }
132
133    // Pass 2: one block per remote delta, written by index into pooled storage that is neither allocated nor zeroed here.
134    debug_assert_eq!(
135        order.positions(),
136        num_buckets,
137        "the coset order is built for a different bucket count than the sum has",
138    );
139    scratch.block_counts.clear();
140    scratch.block_counts.resize(num_buckets, 0);
141    scratch.source_of.clear();
142    scratch.source_of.resize(num_buckets, 0);
143    scratch.work_prefix.clear();
144    scratch.work_prefix.resize(num_buckets + 1, 0);
145    let mut blocks_used = vec![0usize; size as usize];
146    for (i, remote) in plan.remote.iter().enumerate() {
147        let mut work = 0usize;
148        for position in 0..num_buckets {
149            let source = order.bucket_at(position as u32) ^ remote.bucket_delta;
150            let index = source as usize * num_remote + i;
151            scratch.source_of[position] = source;
152            scratch.block_counts[position] = scratch.counts[index];
153            work += scratch.work[index];
154            scratch.work_prefix[position + 1] = work;
155        }
156        let partner = remote.partner as usize;
157        let payload = send[partner].get_or_insert_with(|| scratch.pool.pop().unwrap_or_default());
158        let block_index = plan.sent_block_index(i);
159        blocks_used[partner] = block_index + 1;
160        if payload.blocks.len() <= block_index {
161            payload.blocks.resize_with(block_index + 1, ExchangeBlock::<W, C>::default);
162        }
163        let block = &mut payload.blocks[block_index];
164        block.set_counts(remote.entry as u32, &scratch.block_counts);
165        let rows = block.rows();
166        {
167            let ExchangeBlock { ref offsets, ref mut x, ref mut z, ref mut coeff, .. } = *block;
168            let columns =
169                ColumnsMut { x: &mut x[..rows], z: &mut z[..rows], coeff: &mut coeff[..rows] };
170            let fill = Fill {
171                local,
172                cut,
173                ahead,
174                emitter: &emitters[i],
175                offsets,
176                source_of: &scratch.source_of,
177                work_prefix: &scratch.work_prefix,
178                min_work,
179            };
180            fill.range(0, num_buckets, columns);
181        }
182        counts.rows_sent[partner] += rows as u64;
183        counts.bytes_sent[partner] += block.bytes() as u64;
184    }
185    // A pooled payload may carry more blocks than this layer fills; the extras must not travel.
186    for (partner, payload) in send.iter_mut().enumerate() {
187        if let Some(payload) = payload {
188            payload.blocks.truncate(blocks_used[partner]);
189        }
190    }
191
192    (send, counts)
193}

195/// Count one source bucket's exported rows among the terms `keep` keeps, one slot per remote delta.
196fn count_bucket<const W: usize, C: Coefficient>(
197    (bucket_x, bucket_z, bucket_coeff): (&[[u64; W]], &[[u64; W]], &[C]),
198    prepared: &Prepared<W, C>,
199    plan: &PartitionPlan,
200    dense: &[bool],
201    mut keep: impl FnMut(&[u64; W], &[u64; W], C) -> bool,
202    out: &mut [u32],
203) {
204    let len = bucket_coeff.len();
205    match prepared {
206        Prepared::Local(ptm) => {
207            let all_dense = dense.iter().all(|&is_dense| is_dense);
208            out.fill(0);
209            let mut kept = 0u32;
210            for term in 0..len {
211                let survives = keep(&bucket_x[term], &bucket_z[term], bucket_coeff[term]);
212                kept += u32::from(survives);
213                if all_dense {
214                    continue;
215                }
216                let support = ptm.support_bits(&bucket_x[term], &bucket_z[term]);
217                for (i, remote) in plan.remote.iter().enumerate() {
218                    let emits = ptm.deltas()[remote.entry].amplitude[support] != C::ZERO;
219                    out[i] += u32::from(survives & !dense[i] & emits);
220                }
221            }
222            for (slot, &is_dense) in out.iter_mut().zip(dense) {
223                if is_dense {
224                    *slot = kept;
225                }
226            }
227        }
228        Prepared::Rotation(rotation) => {
229            debug_assert_eq!(out.len(), 1, "a rotation has at most one remote delta");
230            let mut anticommuting = 0u32;
231            for term in 0..len {
232                let survives = keep(&bucket_x[term], &bucket_z[term], bucket_coeff[term]);
233                let string = PauliString::<W> { x: bucket_x[term], z: bucket_z[term] };
234                anticommuting += u32::from(survives & !string.commutes_with(&rotation.generator));
235            }
236            out[0] = anticommuting;
237        }
238    }
239}

241/// One block's pass-2 inputs, shared by every task of its fill.
242struct Fill<'a, 'p, const W: usize, C: Coefficient> {
243    local: &'a PauliSum<W, C>,
244    cut: Option<&'a BoundaryCut>,
245    /// The counts ahead, when they are for this layer's one remote delta.
246    ahead: Option<&'a LayerOctaves>,
247    emitter: &'a RowEmitter<'p, W, C>,
248    offsets: &'a [u32],
249    source_of: &'a [u32],
250    work_prefix: &'a [usize],
251    /// Fill work below which a range stays on one thread.
252    min_work: usize,
253}
254
255impl<const W: usize, C: Coefficient> Fill<'_, '_, W, C> {
256    /// Fill destination positions `start..end`; the halves of a split are disjoint, so the contents are independent of the thread count.
257    ///
258    /// The split balances fill work ([`fill_work`]) rather than emitted rows, since bucket bits can decide the emit test for whole buckets and leave most of the scan where few rows land.
259    fn range(&self, start: usize, end: usize, columns: ColumnsMut<'_, W, C>) {
260        let offsets = self.offsets;
261        debug_assert_eq!(columns.len(), (offsets[end] - offsets[start]) as usize);
262        if end - start > 1 && self.work_prefix[end] - self.work_prefix[start] > self.min_work {
263            let middle = split_at_half_work(self.work_prefix, start, end);
264            let (head, tail) = columns.split_at((offsets[middle] - offsets[start]) as usize);
265            rayon::join(|| self.range(start, middle, head), || self.range(middle, end, tail));
266            return;
267        }
268        let ColumnsMut { x, z, coeff } = columns;
269        let mut written = 0usize;
270        for position in start..end {
271            if offsets[position] == offsets[position + 1] {
272                continue;
273            }
274            let source = self.source_of[position] as usize;
275            let out = ColumnsMut {
276                x: &mut x[written..],
277                z: &mut z[written..],
278                coeff: &mut coeff[written..],
279            };
280            let all = 0..self.local.bucket_len(source);
281            let marked = self.ahead.and_then(|octaves| octaves.bucket_ahead(source));
282            #[cfg(debug_assertions)]
283            if let Some(marked) = marked {
284                marked.debug_assert_marks_match(all.len(), source);
285            }
286            written += match (self.cut, marked) {
287                (None, None) => self.fill_rows(source, all, |_, _, _| true, out),
288                (Some(cut), None) => self.fill_rows(source, all, cut.filter(source), out),
289                (None, Some(marked)) => {
290                    self.fill_rows(source, marked.marked_rows(), |_, _, _| true, out)
291                }
292                (Some(cut), Some(marked)) => {
293                    self.fill_rows(source, marked.marked_rows(), cut.filter(source), out)
294                }
295            };
296        }
297        debug_assert_eq!(
298            written,
299            x.len(),
300            "export: pass 2 filled {written} rows where pass 1 counted {}",
301            x.len(),
302        );
303    }
304
305    /// Write the rows bucket `source`'s terms at `rows` emit, among those `keep` keeps, to the head of `out`; returns how many; `keep` is asked of `rows` in order, so `rows` may skip only terms that move no boundary quota.
306    fn fill_rows(
307        &self,
308        source: usize,
309        rows: impl Iterator<Item = usize>,
310        mut keep: impl FnMut(&[u64; W], &[u64; W], C) -> bool,
311        out: ColumnsMut<'_, W, C>,
312    ) -> usize {
313        let ColumnsMut { x, z, coeff } = out;
314        let (bucket_x, bucket_z, bucket_coeff) = self.local.bucket(source);
315        let mut written = 0usize;
316        for term in rows {
317            let survives = keep(&bucket_x[term], &bucket_z[term], bucket_coeff[term]);
318            if let (true, Some((out_x, out_z, out_coeff))) =
319                (survives, self.emitter.emit(&bucket_x[term], &bucket_z[term], bucket_coeff[term]))
320            {
321                x[written] = out_x;
322                z[written] = out_z;
323                coeff[written] = out_coeff;
324                written += 1;
325            }
326        }
327        written
328    }
329}

331/// A source bucket's fill work in scanned input terms: nothing without rows, else its marked rows plus mark words, or its whole scan.
332fn fill_work(len: usize, rows: u32, marked: bool) -> usize {
333    match (rows, marked) {
334        (0, _) => 0,
335        (rows, true) => rows as usize * MARKED_ROW_WORK + len / 64,
336        (_, false) => len,
337    }
338}

340/// The position in `start + 1..end` where the scan work of `start..end`, read from its prefix sums, first reaches half.
341fn split_at_half_work(work_prefix: &[usize], start: usize, end: usize) -> usize {
342    let target = work_prefix[start] + (work_prefix[end] - work_prefix[start]) / 2;
343    let middle = start + 1 + work_prefix[start + 1..end].partition_point(|&work| work < target);
344    middle.min(end - 1)
345}

347/// Every exported row belongs to the partner it is addressed to.
348#[cfg(debug_assertions)]
349pub(super) fn debug_assert_exported_partitions<const W: usize, C: Coefficient>(
350    send: &[Option<PartnerPayload<W, C>>],
351    rows: &crate::pauli_sum::hash::PartitionRows<W>,
352) {
353    for (partner, payload) in send.iter().enumerate() {
354        let Some(payload) = payload else { continue };
355        for block in &payload.blocks {
356            for i in 0..block.rows() {
357                let owner = rows.partition_of(&block.x[i], &block.z[i]);
358                debug_assert_eq!(
359                    owner, partner as u32,
360                    "export: row {i} of the block for entry {} is addressed to partition {partner} \
361                     but belongs to partition {owner}",
362                    block.header.entry,
363                );
364            }
365        }
366    }
