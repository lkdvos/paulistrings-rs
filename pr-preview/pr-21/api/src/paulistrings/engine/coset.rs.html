1//! `Gf2Span`, the GF(2) span of a channel's bucket deltas and the coset index algebra over it, and `CosetOrder`, the coset-contiguous bucket order a layer runs in (ARCHITECTURE.md §Engine).

3use crate::pauli_sum::hash::B_MAX_BITS;

5/// Highest set bit of a nonzero word.
6fn highest_bit(v: u32) -> u32 {
7    debug_assert!(v != 0, "highest_bit: zero has no highest set bit");
8    31 - v.leading_zeros()
9}
10
11/// Software `pext`, since BMI2 is not assumed and this runs off any inner loop.
12fn pext(value: u32, mask: u32) -> u32 {
13    let mut out = 0u32;
14    let mut out_bit = 0u32;
15    let mut remaining = mask;
16    while remaining != 0 {
17        let bit = remaining & remaining.wrapping_neg();
18        if value & bit != 0 {
19            out |= 1 << out_bit;
20        }
21        out_bit += 1;
22        remaining ^= bit;
23    }
24    out
25}

27/// The span of a channel's bucket deltas, which need not be XOR-closed, and the map between a bucket index and its `(coset, member)` coordinates.
28// `basis` is reduced row echelon, ascending by pivot (its highest set bit), and each pivot bit is set in exactly one vector; bit `j` of a member index selects `basis[j]`.
29#[derive(Clone, Debug)]
30pub(crate) struct Gf2Span {
31    /// Reduced echelon basis, ascending by pivot bit.
32    basis: Vec<u32>,
33    /// OR of the pivot bits — one bit per basis vector.
34    pivot_mask: u32,
35    /// Bucket-index width: indices live in `0..2^bits`.
36    bits: u8,
37    /// The bucket-space bits outside `pivot_mask`: the free bits a representative ranges over.
38    nonpivot_mask: u32,
39}
40
41impl Gf2Span {
42    /// The span of `deltas` in a `bits`-wide index space; panics if `bits > B_MAX_BITS` or a delta lies outside it.
43    pub(crate) fn new(deltas: &[u32], bits: u8) -> Self {
44        assert!(
45            bits <= B_MAX_BITS,
46            "Gf2Span: bits {bits} exceeds B_MAX_BITS {B_MAX_BITS}"
47        );
48        let space_mask = if bits == 0 { 0 } else { (1u32 << bits) - 1 };
49
50        let mut basis: Vec<u32> = Vec::new();
51        let mut pivot_mask = 0u32;
52
53        for &delta in deltas {
54            assert!(
55                delta & !space_mask == 0,
56                "Gf2Span: delta {delta} has bits outside the {bits}-bit bucket space"
57            );
58            // Pivot bits are disjoint across basis vectors, so one pass in any order clears them all.
59            let mut reduced = delta;
60            for &vector in &basis {
61                if reduced & (1 << highest_bit(vector)) != 0 {
62                    reduced ^= vector;
63                }
64            }
65            if reduced == 0 {
66                continue;
67            }
68            let pivot = highest_bit(reduced);
69            // Back-substitute so the new pivot is unique to one vector; only vectors with a higher pivot can carry bit `p`.
70            for vector in basis.iter_mut() {
71                if *vector & (1 << pivot) != 0 {
72                    *vector ^= reduced;
73                }
74            }
75            let position = basis.partition_point(|&vector| highest_bit(vector) < pivot);
76            basis.insert(position, reduced);
77            pivot_mask |= 1 << pivot;
78        }
79
80        Self {
81            basis,
82            pivot_mask,
83            bits,
84            nonpivot_mask: space_mask & !pivot_mask,
85        }
86    }
87
88    /// Dimension `r` of the span.
89    pub(crate) fn dimension(&self) -> usize {
90        self.basis.len()
91    }
92
93    /// Buckets per coset, `2^r`.
94    pub(crate) fn coset_size(&self) -> usize {
95        1usize << self.basis.len()
96    }
97
98    /// Number of cosets, `2^bits / 2^r`.
99    pub(crate) fn num_cosets(&self) -> usize {
100        (1usize << self.bits) >> self.basis.len()
101    }
102
103    /// Whether `beta` is its coset's representative: the unique member with every pivot bit clear, which is also the coset's minimum.
104    fn is_representative(&self, beta: u32) -> bool {
105        beta & self.pivot_mask == 0
106    }
107
108    /// The representative of `beta`'s coset.
109    // Not `beta & !pivot_mask`: basis vectors carry non-pivot bits too, so masking can leave the coset.
110    pub(crate) fn representative_of(&self, beta: u32) -> u32 {
111        debug_assert!(
112            beta >> self.bits == 0,
113            "Gf2Span::representative_of: beta outside the bucket space"
114        );
115        let mut reduced = beta;
116        for &vector in &self.basis {
117            if reduced & (1 << highest_bit(vector)) != 0 {
118                reduced ^= vector;
119            }
120        }
121        reduced
122    }
123
124    /// The member index of `delta` in the span, read off its pivot bits since the basis is reduced.
125    pub(crate) fn coordinate_of(&self, delta: u32) -> u32 {
126        debug_assert!(
127            self.representative_of(delta) == 0,
128            "Gf2Span::coordinate_of: delta {delta} is not in the span"
129        );
130        pext(delta, self.pivot_mask)
131    }
132
133    /// The position of `representative` among all representatives in ascending order.
134    fn rank_of_representative(&self, representative: u32) -> u32 {
135        debug_assert!(
136            self.is_representative(representative),
137            "Gf2Span::rank_of_representative: {representative} is not a representative"
138        );
139        pext(representative, self.nonpivot_mask)
140    }
141
142    /// `beta` renumbered so coset `c` owns `c << r .. (c + 1) << r`, with the member coordinate in the low `r` bits.
143    pub(crate) fn permuted_index(&self, beta: u32) -> u32 {
144        let representative = self.representative_of(beta);
145        (self.rank_of_representative(representative) << self.basis.len())
146            | self.coordinate_of(beta ^ representative)
147    }
148}

150/// A layer's coset-contiguous bucket order: coset `c` owns positions `c << r .. (c + 1) << r` ([`Gf2Span::permuted_index`]).
151///
152/// Built once per layer and shared by the coset loop, the export pass and the exchange's chunking; empty, i.e. the identity, when `r = 0`.
153#[derive(Clone, Debug, Default)]
154pub(crate) struct CosetOrder {
155    /// `permutation[β]` is bucket `β`'s position.
156    permutation: Vec<u32>,
157    /// `inverse[p]` is the bucket at position `p`; built only on request.
158    inverse: Vec<u32>,
159    positions: u32,
160    dimension: u32,
161}
162
163impl CosetOrder {
164    /// Re-aim the order at `span`, keeping the allocations; the inverse is built only when `with_inverse`.
165    pub(crate) fn rebuild(&mut self, span: &Gf2Span, with_inverse: bool) {
166        self.positions = 1u32 << span.bits;
167        self.dimension = span.dimension() as u32;
168        self.permutation.clear();
169        self.inverse.clear();
170        if self.dimension == 0 {
171            return;
172        }
173        self.permutation
174            .extend((0..self.positions).map(|beta| span.permuted_index(beta)));
175        if with_inverse {
176            self.inverse.resize(self.permutation.len(), 0);
177            for (beta, &position) in self.permutation.iter().enumerate() {
178                self.inverse[position as usize] = beta as u32;
179            }
180        }
181    }
182
183    /// Whether every bucket is its own position (`r = 0`).
184    pub(crate) fn is_identity(&self) -> bool {
185        self.dimension == 0
186    }
187
188    /// Bucket `beta`'s position.
189    pub(crate) fn position_of(&self, beta: u32) -> u32 {
190        if self.permutation.is_empty() {
191            beta
192        } else {
193            self.permutation[beta as usize]
194        }
195    }
196
197    /// The bucket at position `position`, the inverse of [`Self::position_of`]; the order must have been built `with_inverse`.
198    pub(crate) fn bucket_at(&self, position: u32) -> u32 {
199        debug_assert!(
200            self.inverse.len() == self.permutation.len(),
201            "CosetOrder::bucket_at: the inverse was not built",
202        );
203        if self.inverse.is_empty() {
204            position
205        } else {
206            self.inverse[position as usize]
207        }
208    }
209
210    /// Positions, i.e. buckets.
211    pub(crate) fn positions(&self) -> usize {
212        self.positions as usize
213    }
214
215    /// `r`, the `log2` of the coset size.
216    pub(crate) fn dimension(&self) -> usize {
217        self.dimension as usize
218    }
219
220    /// Capacity of the permutation, for the footprint tests.
221    #[cfg(test)]
222    pub(crate) fn capacity(&self) -> usize {
223        self.permutation.capacity() + self.inverse.capacity()
224    }
225}
