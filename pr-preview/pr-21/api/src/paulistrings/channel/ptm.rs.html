13const ZERO: Complex64 = Complex64::new(0.0, 0.0);
14const ONE: Complex64 = Complex64::new(1.0, 0.0);

16/// A one- or two-qubit gate or noise channel, stored as its Pauli-transfer matrix.
17///
18/// Built by the named constructors: the Cliffords [`hadamard`](Self::hadamard), [`phase_gate`](Self::phase_gate), [`pauli_x`](Self::pauli_x), [`pauli_y`](Self::pauli_y), [`pauli_z`](Self::pauli_z), [`cnot`](Self::cnot), [`cz`](Self::cz), [`swap`](Self::swap);
19/// the general unitaries [`unitary_1q`](Self::unitary_1q), [`unitary_2q`](Self::unitary_2q);
20/// and the noise channels [`depolarizing`](Self::depolarizing), [`dephasing`](Self::dephasing), [`pauli_channel`](Self::pauli_channel), [`depolarizing_2q`](Self::depolarizing_2q), [`amplitude_damping`](Self::amplitude_damping).
21///
22/// On the support, Paulis are indexed `x | (z << 1)` per qubit (`I = 0, X = 1, Z = 2, Y = 3`), the constructor's first qubit in the low two bits.
23/// [`Channel::apply`] maps input Pauli `s` to `Σ_t ptm[s][t] · P_t`, and [`Channel::apply_adjoint`] reads the transpose.
24/// [`Channel::debug_name`] is the constructor's name (`"hadamard"`, `"cnot"`, …), with `_adjoint` appended after [`adjoint`](Self::adjoint).
25#[derive(Clone, Debug)]
26pub struct PtmChannel {
27    /// Support qubits in constructor order; only the first `num_support_qubits` are meaningful.
28    qubits: [u32; 2],
29    num_support_qubits: u8,
30    kind: PtmKind,
31    /// Whether this is the [`adjoint`](Self::adjoint) of the constructed channel; [`Channel::debug_name`] appends `_adjoint`.
32    adjoint: bool,
33    storage: PtmStorage,
34}
35
36/// The constructor a [`PtmChannel`] came from, which [`Channel::debug_name`] reports.
37#[derive(Clone, Copy, Debug, PartialEq, Eq)]
38enum PtmKind {
39    Hadamard,
40    PhaseGate,
41    PauliX,
42    PauliY,
43    PauliZ,
44    Cnot,
45    Cz,
46    Swap,
47    Unitary1q,
48    Unitary2q,
49    Depolarizing,
50    Dephasing,
51    PauliChannel,
52    Depolarizing2q,
53    AmplitudeDamping,
54}
55
56impl PtmKind {
57    /// The constructor's name, with `_adjoint` appended for the adjoint.
58    fn name(self, adjoint: bool) -> &'static str {
59        let [plain, adjoint_name] = match self {
60            Self::Hadamard => ["hadamard", "hadamard_adjoint"],
61            Self::PhaseGate => ["phase_gate", "phase_gate_adjoint"],
62            Self::PauliX => ["pauli_x", "pauli_x_adjoint"],
63            Self::PauliY => ["pauli_y", "pauli_y_adjoint"],
64            Self::PauliZ => ["pauli_z", "pauli_z_adjoint"],
65            Self::Cnot => ["cnot", "cnot_adjoint"],
66            Self::Cz => ["cz", "cz_adjoint"],
67            Self::Swap => ["swap", "swap_adjoint"],
68            Self::Unitary1q => ["unitary_1q", "unitary_1q_adjoint"],
69            Self::Unitary2q => ["unitary_2q", "unitary_2q_adjoint"],
70            Self::Depolarizing => ["depolarizing", "depolarizing_adjoint"],
71            Self::Dephasing => ["dephasing", "dephasing_adjoint"],
72            Self::PauliChannel => ["pauli_channel", "pauli_channel_adjoint"],
73            Self::Depolarizing2q => ["depolarizing_2q", "depolarizing_2q_adjoint"],
74            Self::AmplitudeDamping => ["amplitude_damping", "amplitude_damping_adjoint"],
75        };
76        if adjoint {
77            adjoint_name
78        } else {
79            plain
80        }
81    }
82}

84/// The nonzero PTM entries, in the cheapest form the constructor knows to be exact.
85#[derive(Clone, Debug)]
86enum PtmStorage {
87    /// A Clifford: input `s` goes to `image[s]` with coefficient `-1` where bit `s` of `negated` is set, `+1` otherwise.
88    SignedPermutation {
89        image: [u8; LOCAL_DIM],
90        negated: u16,
91    },
92    /// A Pauli-diagonal channel: input `s` keeps its key and is scaled by `scale[s]`.
93    Diagonal { scale: Box<[f64]> },
94    /// Everything else, ascending by `(input, output)`.
95    Sparse { entries: Box<[SparseEntry]> },
96}
97
98/// One nonzero PTM entry: `amplitude` takes local Pauli `input` to `output`.
99#[derive(Clone, Copy, Debug)]
100struct SparseEntry {
101    input: u8,
102    output: u8,
103    amplitude: Complex64,
104}
105
106impl PtmChannel {
107    /// A channel on `qubits` (one or two, distinct) from its storage.
108    fn new(qubits: &[usize], storage: PtmStorage, kind: PtmKind) -> Self {
109        debug_assert!(qubits.len() == 1 || qubits.len() == 2);
110        let narrow = |q: usize| u32::try_from(q).expect("PtmChannel: qubit index exceeds u32");
111        let mut padded = [narrow(qubits[0]); 2];
112        for (slot, &q) in padded.iter_mut().zip(qubits) {
113            *slot = narrow(q);
114        }
115        Self {
116            qubits: padded,
117            num_support_qubits: qubits.len() as u8,
118            kind,
119            adjoint: false,
120            storage,
121        }
122    }
123
124    /// The support qubits, in constructor order.
125    pub fn qubits(&self) -> impl ExactSizeIterator<Item = usize> + '_ {
126        self.qubits[..self.num_support_qubits as usize]
127            .iter()
128            .map(|&q| q as usize)
129    }
130
131    /// The adjoint channel, whose PTM is this one's transpose.
132    pub fn adjoint(&self) -> Self {
133        let storage = match &self.storage {
134            PtmStorage::SignedPermutation { image, negated } => {
135                let mut inverse = [0u8; LOCAL_DIM];
136                let mut inverse_negated = 0u16;
137                for (s, &t) in image[..self.dimension()].iter().enumerate() {
138                    let t = t as usize;
139                    inverse[t] = s as u8;
140                    inverse_negated |= ((negated >> s) & 1) << t;
141                }
142                PtmStorage::SignedPermutation {
143                    image: inverse,
144                    negated: inverse_negated,
145                }
146            }
147            PtmStorage::Diagonal { scale } => PtmStorage::Diagonal {
148                scale: scale.clone(),
149            },
150            PtmStorage::Sparse { entries } => {
151                let mut transposed: Vec<((u8, u8), Complex64)> = entries
152                    .iter()
153                    .map(|entry| ((entry.output, entry.input), entry.amplitude))
154                    .collect();
155                transposed.sort_by_key(|&(position, _)| position);
156                sparse_storage(transposed)
157            }
158        };
159        Self {
160            storage,
161            adjoint: !self.adjoint,
162            ..*self
163        }
164    }
165
166    /// `4^k` for `k` support qubits.
167    fn dimension(&self) -> usize {
168        1 << (2 * self.num_support_qubits)
169    }
170
171    /// The local Pauli index of the input's restriction to the support.
172    fn local_index<const W: usize>(&self, x: &[u64; W], z: &[u64; W]) -> usize {
173        self.qubits()
174            .enumerate()
175            .map(|(j, q)| {
176                let (word, bit, _) = qubit_loc(q);
177                read_pauli(x, z, word, bit) << (2 * j)
178            })
179            .fold(0, |index, part| index | part)
180    }
181
182    /// The input with its support overwritten by local Pauli index `t`.
183    fn with_local_index<const W: usize>(
184        &self,
185        x: &[u64; W],
186        z: &[u64; W],
187        t: usize,
188    ) -> ([u64; W], [u64; W]) {
189        let mut new_x = *x;
190        let mut new_z = *z;
191        for (j, q) in self.qubits().enumerate() {
192            let (word, bit, mask) = qubit_loc(q);
193            write_pauli(&mut new_x, &mut new_z, word, bit, mask, (t >> (2 * j)) & 3);
194        }
195        (new_x, new_z)
196    }
197
198    /// Body of `apply` (`transpose == false`) and `apply_adjoint`.
199    fn apply_rows<const W: usize>(
200        &self,
201        transpose: bool,
202        x: &[u64; W],
203        z: &[u64; W],
204        coeff: Complex64,
205        out: &mut OutputBuffer<W>,
206    ) {
207        let s = self.local_index(x, z);
208        match &self.storage {
209            PtmStorage::SignedPermutation { image, negated } => {
210                let (source, t) = if transpose {
211                    let source = image[..self.dimension()]
212                        .iter()
213                        .position(|&image| image as usize == s)
214                        .expect("a signed permutation is a bijection");
215                    (source, source)
216                } else {
217                    (s, image[s] as usize)
218                };
219                let (new_x, new_z) = self.with_local_index(x, z, t);
220                let signed = if (negated >> source) & 1 == 1 {
221                    -coeff
222                } else {
223                    coeff
224                };
225                out.push(new_x, new_z, signed);
226            }
227            PtmStorage::Diagonal { scale } => out.push(*x, *z, coeff * scale[s]),
228            PtmStorage::Sparse { entries } => {
229                for entry in entries.iter() {
230                    let (from, to) = if transpose {
231                        (entry.output, entry.input)
232                    } else {
233                        (entry.input, entry.output)
234                    };
235                    if from as usize == s {
236                        let (new_x, new_z) = self.with_local_index(x, z, to as usize);
237                        out.push(new_x, new_z, coeff * entry.amplitude);
238                    }
239                }
240            }
241        }
242    }
243
244    /// Calls `visit(s, t, amplitude)` for every stored entry.
245    fn for_each_entry(&self, mut visit: impl FnMut(usize, usize, Complex64)) {
246        match &self.storage {
247            PtmStorage::SignedPermutation { image, negated } => {
248                for (s, &t) in image[..self.dimension()].iter().enumerate() {
249                    let amplitude = if (negated >> s) & 1 == 1 { -ONE } else { ONE };
250                    visit(s, t as usize, amplitude);
251                }
252            }
253            PtmStorage::Diagonal { scale } => {
254                for (s, &value) in scale.iter().enumerate() {
255                    visit(s, s, Complex64::new(value, 0.0));
256                }
257            }
258            PtmStorage::Sparse { entries } => {
259                for entry in entries.iter() {
260                    visit(entry.input as usize, entry.output as usize, entry.amplitude);
261                }
262            }
263        }
264    }
265
266    /// The dense `table[s][t]` (transposed when `adjoint`) in ascending-qubit local indices, the order [`Prepared`] uses.
267    fn transfer_matrix(&self, adjoint: bool) -> [[Complex64; LOCAL_DIM]; LOCAL_DIM] {
268        let reversed = self.num_support_qubits == 2 && self.qubits[0] > self.qubits[1];
269        let reorder = |index: usize| {
270            if reversed {
271                ((index & 3) << 2) | (index >> 2)
272            } else {
273                index
274            }
275        };
276        let mut table = [[ZERO; LOCAL_DIM]; LOCAL_DIM];
277        self.for_each_entry(|s, t, amplitude| {
278            let (row, column) = if adjoint { (t, s) } else { (s, t) };
279            table[reorder(row)][reorder(column)] += amplitude;
280        });
281        table
282    }
283}

285/// Sparse storage from `(position, amplitude)` pairs ascending by position, exact zeros dropped.
286fn sparse_storage(entries: Vec<((u8, u8), Complex64)>) -> PtmStorage {
287    PtmStorage::Sparse {
288        entries: entries
289            .into_iter()
290            .filter(|&(_, amplitude)| amplitude != ZERO)
291            .map(|((input, output), amplitude)| SparseEntry {
292                input,
293                output,
294                amplitude,
295            })
296            .collect(),
297    }
298}

300/// Panics unless the two support qubits differ.
301fn assert_distinct(constructor: &str, q0: usize, q1: usize) {
302    assert!(
303        q0 != q1,
304        "PtmChannel::{constructor}: the two qubits must differ (both are {q0})"
305    );
306}
307
308/// Panics unless `value` lies in `[0, 1]`.
309fn assert_probability(constructor: &str, name: &str, value: f64) {
310    assert!(
311        (0.0..=1.0).contains(&value),
312        "PtmChannel::{constructor}: {name} must lie in [0, 1] (got {value})"
313    );
314}

316impl<const W: usize> Channel<W> for PtmChannel {
317    fn support(&self) -> [u64; W] {
318        let mut qubits = [0usize; 2];
319        for (slot, q) in qubits.iter_mut().zip(self.qubits()) {
320            *slot = q;
321        }
322        support_mask(&qubits[..self.num_support_qubits as usize])
323    }
324
325    fn debug_name(&self) -> &'static str {
326        self.kind.name(self.adjoint)
327    }
328
329    fn apply(
330        &self,
331        input_x: &[u64; W],
332        input_z: &[u64; W],
333        coeff: Complex64,
334        out: &mut OutputBuffer<W>,
335    ) {
336        self.apply_rows(false, input_x, input_z, coeff, out);
337    }
338
339    fn apply_adjoint(
340        &self,
341        input_x: &[u64; W],
342        input_z: &[u64; W],
343        coeff: Complex64,
344        out: &mut OutputBuffer<W>,
345    ) {
346        self.apply_rows(true, input_x, input_z, coeff, out);
347    }
348
349    /// Tabulated straight from the stored PTM, without probing `apply`.
350    fn prepare_for_layer(
351        &self,
352        hash: &Gf2Hash<W>,
353        adjoint: bool,
354        _: PrepareToken,
355    ) -> Option<Prepared<W>> {
356        Some(Prepared::from_transfer_matrix(
357            &self.support(),
358            &self.transfer_matrix(adjoint),
359            hash,
360        ))
361    }
362}
