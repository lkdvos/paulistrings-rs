1//! [`PartitionRuntime`]: the placement resolved once, the pinned pools built
2//! once, and the scoped fan-out every partitioned call runs inside.
3//!
4//! A partitioned run has two kinds of thread.
5//! Each partition has one **driving** thread that walks the layers, issues the collectives and calls into the engine; partition 0's driving thread is the *calling* thread, and partitions `1..P` get a scoped thread each.
6//! Inside a partition, the layer itself runs on that partition's own pinned Rayon pool — the driving thread enters it with `ThreadPool::install`, so every allocation a layer makes is first-touched by a worker in the partition's own NUMA domain (ARCHITECTURE.md §Parallelism for why the layer needs no synchronization of its own).
7//! That `install` is also why the MPI transport requires `MPI_THREAD_SERIALIZED` rather than `FUNNELED`: a distributed rank's collectives are issued from a pool worker, not from the process's main thread.
8//!
9//! There is no synchronization between partitions other than the transport: every partition issues the identical sequence of transport calls per layer (see [`transport`](super::transport)'s collective-order invariant), so the group stays in step without a barrier.
10

23/// The resolved placement and its pools, shared by every [`PartitionedSum`](super::PartitionedSum) that runs on it.
24///
25/// Built once — pools are not cheap — and held behind an [`Arc`], so a driver stepping an observable through many circuits, or several sums propagated in turn, pays for the pinned pools once.
26/// Cloning the `Arc` is the intended way to share it.
27///
28/// # Examples
29///
30/// ```
31/// use paulistrings::engine::partitioned::{PartitionConfig, PartitionRuntime, Placement};
32///
33/// let config = PartitionConfig {
34///     placement: Placement::Unpinned { partitions: 2, threads_per_partition: Some(1) },
35///     bind_memory: false,
36///     partition_row_seed: None,
37/// };
38/// let runtime = PartitionRuntime::new(&config).expect("topology resolves");
39/// assert_eq!(runtime.num_partitions(), 2);
40/// assert_eq!(runtime.slots().len(), 2);
41/// ```
42pub struct PartitionRuntime {
43    /// One slot per partition, in rank order.
44    slots: Vec<PartitionSlot>,
45    /// One pinned pool per partition, in rank order.
46    pools: Vec<rayon::ThreadPool>,
47    /// Whether driving threads and pool workers bind their allocations to the slot's NUMA node.
48    bind_memory: bool,
49}
50
51impl PartitionRuntime {
52    /// Resolves `config` against the machine and builds one pinned pool per partition.
53    ///
54    /// # Errors
55    ///
56    /// Whatever [`PartitionConfig::resolve`] reports (a partition count that is not a power of two, an empty or out-of-mask explicit CPU set), or [`TopologyError::Io`] if a pool cannot be built.
57    pub fn new(config: &PartitionConfig) -> Result<Arc<Self>, TopologyError> {
58        Self::with_threads_per_partition(config, None)
59    }
60
61    /// [`new`](Self::new) with the pool width chosen by the caller instead of by the placement.
62    ///
63    /// `Some(t)` gives every partition a `t`-worker pool (at least one), leaving the CPU set each pool is pinned to exactly as [`PartitionConfig::resolve`] derived it.
64    /// `None` keeps the resolved widths ([`Placement::Auto`](super::Placement::Auto) and [`Explicit`](super::Placement::Explicit) size a pool by the number of CPUs in its set).
65    ///
66    /// This is the knob a measurement harness needs to hold the *total* thread count fixed across partition counts — `T` threads unpartitioned against `P` pools of `T / P` on the same CPUs — without shrinking the CPU masks and changing what is being compared.
67    /// In production, prefer [`new`](Self::new): a pool narrower than its CPU set leaves cores idle, and a wider one oversubscribes them.
68    ///
69    /// # Errors
70    ///
71    /// Whatever [`PartitionConfig::resolve`] reports (a partition count that is not a power of two, an empty or out-of-mask explicit CPU set), or [`TopologyError::Io`] if a pool cannot be built.
72    pub fn with_threads_per_partition(
73        config: &PartitionConfig,
74        threads_per_partition: Option<usize>,
75    ) -> Result<Arc<Self>, TopologyError> {
76        let mut slots = config.resolve()?;
77        if let Some(threads) = threads_per_partition {
78            for slot in &mut slots {
79                slot.threads = threads.max(1);
80            }
81        }
82        let mut pools = Vec::with_capacity(slots.len());
83        for (rank, slot) in slots.iter().enumerate() {
84            pools.push(build_pool(
85                slot,
86                config.bind_memory,
87                &format!("ps-part{rank}"),
88            )?);
89        }
90        Ok(Arc::new(Self {
91            slots,
92            pools,
93            bind_memory: config.bind_memory,
94        }))
95    }
96
97    /// Partitions in the group — always a power of two.
98    pub fn num_partitions(&self) -> usize {
99        self.slots.len()
100    }
101
102    /// The resolved slots, in rank order.
103    pub fn slots(&self) -> &[PartitionSlot] {
104        &self.slots
105    }
106
107    /// `log2` of the partition count: the number of GF(2) rows a
108    /// [`PartitionRows`](crate::PartitionRows) needs to name a partition.
109    pub(crate) fn partition_bits(&self) -> u8 {
110        debug_assert!(self.slots.len().is_power_of_two());
111        self.slots.len().trailing_zeros() as u8
112    }
113
114    /// A one-line summary of the placement, for the entry log line.
115    pub(crate) fn placement_summary(&self) -> String {
116        let mut s = String::new();
117        for (rank, slot) in self.slots.iter().enumerate() {
118            if rank > 0 {
119                s.push_str(", ");
120            }
121            match &slot.cpus {
122                Some(cpus) => s.push_str(&format!("{rank}:{cpus}x{}", slot.threads)),
123                None => s.push_str(&format!("{rank}:unpinned x{}", slot.threads)),
124            }
125        }
126        s
127    }
128
129    /// Runs `f` on partition 0's pool and returns its result.
130    ///
131    /// The distributed shape of [`map_partitions`](Self::map_partitions): a process that *is* one partition has nothing to fan out to, but its work still belongs on the pinned pool, so that every allocation is first-touched by a worker inside the process's own domain.
132    /// As with `map_partitions`, the calling thread is not re-affinitized — the work runs on pool 0's workers, which `build_pool` pinned.
133    ///
134    /// `rayon::ThreadPool::install` blocks the caller and runs `f` on a pool worker, so `f` (and any MPI call inside it) is not on the process's main thread — that is why the MPI transport documents `MPI_THREAD_SERIALIZED` rather than `FUNNELED`.
135    pub(crate) fn install<R: Send>(&self, f: impl FnOnce() -> R + Send) -> R {
136        self.pools[0].install(f)
137    }
138
139    /// Runs `f(rank, item, transport)` once per partition, concurrently, and returns the results in rank order.
140    ///
141    /// `items` carries one value per partition — the partition's local sum and scratch — moved in and handed back, so nothing is shared between partitions but the transport.
142    /// Partition 0 runs on the calling thread; partitions `1..P` on scoped threads that pin themselves to their slot first.
143    /// Every partition's body runs inside its own pool (`ThreadPool::install`), so a `rayon` call inside `f` lands on the partition's own workers.
144    ///
145    /// # Panics
146    ///
147    /// A panic in any partition is propagated to the caller with its original payload, after the scope has joined the others.
148    /// It cannot hang the group: the transport group is *moved* into the partitions, so a dying partition drops its endpoints and its partners' next collective reports a dead partner rather than blocking (see [`transport`](super::transport)).
149    pub(crate) fn map_partitions<I, O, F>(&self, items: Vec<I>, f: F) -> Vec<O>
150    where
151        I: Send,
152        O: Send,
153        F: Fn(usize, I, &InProcessTransport) -> O + Send + Sync,
154    {
155        let size = self.num_partitions();
156        assert_eq!(items.len(), size, "map_partitions: one item per partition");
157        // A fresh group per call, rather than one held in the runtime: each partition *owns* its endpoint for the duration, which is what turns a partner's panic into a panic (dropped senders) instead of a hang.
158        // It also means an aborted call cannot leave the group's collective counter out of step for the next one.
159        let transports = InProcessTransport::group(size as u32);
160        let f = &f;
161
162        let mut items = items.into_iter();
163        let mut transports = transports.into_iter();
164        let item0 = items.next().expect("at least one partition");
165        let transport0 = transports.next().expect("at least one transport");
166
167        std::thread::scope(|scope| {
168            let handles: Vec<_> = items
169                .zip(transports)
170                .enumerate()
171                .map(|(i, (item, transport))| {
172                    let rank = i + 1;
173                    let slot = &self.slots[rank];
174                    let pool = &self.pools[rank];
175                    let bind_memory = self.bind_memory;
176                    scope.spawn(move || {
177                        place_current_thread(slot, bind_memory);
178                        pool.install(move || f(rank, item, &transport))
179                    })
180                })
181                .collect();
182
183            // Partition 0 on the calling thread — deliberately *not* pinned: the caller's thread is not ours to re-affinitize, and the work itself runs on pool 0's workers, which `build_pool` pinned.
184            let mut out = Vec::with_capacity(size);
185            out.push(self.pools[0].install(|| f(0, item0, &transport0)));
186            for handle in handles {
187                match handle.join() {
188                    Ok(o) => out.push(o),
189                    // Re-raise the partition's own payload, so the caller sees the original message rather than a join error.
190                    Err(payload) => std::panic::resume_unwind(payload),
191                }
192            }
193            out
194        })
195    }
196}

198/// Pins a partition's driving thread to its slot, warning (not failing) when the platform declines — the same degradation policy `build_pool` applies to pool workers.
199fn place_current_thread(slot: &PartitionSlot, bind_memory: bool) {
200    if let Some(cpus) = &slot.cpus {
201        if let Err(err) = pin_current_thread(cpus) {
202            log::warn!(target: LOG_TARGET, "failed to pin partition driver to {cpus}: {err}");
203        }
204    }
205    if bind_memory {
206        if let Err(err) = bind_current_thread_memory(slot.node) {
207            log::warn!(
208                target: LOG_TARGET,
209                "failed to bind partition driver memory to {:?}: {err}",
210                slot.node,
211            );
212        }
213    }
214}

242    /// The pool-width override replaces the resolved worker count on every slot and leaves the CPU sets alone — the probe's "total threads split over P partitions" knob.
243    #[test]
244    fn with_threads_per_partition_overrides_the_resolved_width() {
245        let cpus = allowed_cpus();
246        let placement = Placement::Explicit(vec![cpus.clone(), cpus.clone()]);
247        let config = PartitionConfig {
248            placement,
249            bind_memory: false,
250            partition_row_seed: None,
251        };
252
253        let resolved = PartitionRuntime::new(&config).expect("resolve");
254        for slot in resolved.slots() {
255            assert_eq!(slot.threads, cpus.len());
256        }
257
258        let narrowed = PartitionRuntime::with_threads_per_partition(&config, Some(1))
259            .expect("resolve with an explicit width");
260        for slot in narrowed.slots() {
261            assert_eq!(slot.threads, 1);
262            assert_eq!(slot.cpus.as_ref(), Some(&cpus), "the CPU set is untouched");
263        }
264    }

266    #[test]
267    fn map_partitions_returns_results_in_rank_order() {
268        let runtime = PartitionRuntime::new(&config(4)).expect("resolve");
269        let got = runtime.map_partitions(vec![10usize, 20, 30, 40], |rank, item, transport| {
270            assert_eq!(transport.rank() as usize, rank);
271            assert_eq!(transport.size(), 4);
272            // Inside the partition's own pool.
273            assert!(rayon::current_thread_index().is_some());
274            (rank, item)
275        });
276        assert_eq!(got, vec![(0, 10), (1, 20), (2, 30), (3, 40)]);

288    /// A partition that panics while its partners are inside a collective must surface as a panic, not a deadlock.
289    /// (The test itself would hang on failure; that is the assertion.)
290    ///
291    /// *Which* panic surfaces is a race, so the assertion is bare `should_panic` with no expected text: a partner can notice the dead partition first and report that instead, or die itself before another partner reads its channel.
292    /// Every one of those outcomes is the behaviour under test: the group terminates instead of blocking forever.
293    #[test]
294    #[should_panic]
295    fn a_panicking_partition_does_not_hang_the_group() {
296        let runtime = PartitionRuntime::new(&config(4)).expect("resolve");
297        runtime.map_partitions(vec![0usize; 4], |rank, _, transport| {
298            if rank == 2 {
299                panic!("rank 2 fell over");
300            }
301            // The partners keep collecting, so they are blocked on the dead partition until its endpoint drops.
302            for _ in 0..4 {
303                transport.allreduce_max_u8(rank as u8);
304            }
305        });
306    }
