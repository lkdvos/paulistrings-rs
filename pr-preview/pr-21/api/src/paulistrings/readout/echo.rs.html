1//! Operator Loschmidt-echo read-outs on a propagated operator `A`: the exact rotated overlap `2⁻ⁿ Tr(A V† A V)` and the anticommutation histogram behind its diagonal approximation.
2//! `V = ⊗_{q ∈ sites} exp(-iδ G_q)` with `G_q` the single-qubit `X` or `Z` named by [`RotationAxis`].
3
4use num_complex::Complex64;
5use rayon::prelude::*;
6
7use crate::pauli_sum::{PartitionRows, PauliSum};

9/// The single-qubit generator `G_q` of an echo perturbation `V = ⊗_{q ∈ sites} exp(-iδ G_q)`.
10#[derive(Clone, Copy, Debug, PartialEq, Eq)]
11pub enum RotationAxis {
12    /// `G_q = X_q`.
13    X,
14    /// `G_q = Z_q`.
15    Z,
16}
17
18impl RotationAxis {
19    /// The key coordinates that conjugation by `V` can change, as `(x-mask, z-mask)`: the `z`-bits of `sites` for [`Z`](Self::Z), the `x`-bits for [`X`](Self::X).
20    ///
21    /// # Panics
22    ///
23    /// If a site is repeated or is not below `64 · W`.
24    pub fn flip_mask<const W: usize>(self, sites: &[usize]) -> ([u64; W], [u64; W]) {
25        let mask = site_mask::<W>(sites, 64 * W);
26        match self {
27            RotationAxis::Z => ([0; W], mask),
28            RotationAxis::X => (mask, [0; W]),
29        }
30    }
31
32    /// `(selector, flip)` halves of a key: `G_q` anticommutes with the term iff the selector bit is set on `q`, and `V` then flips the other half's bit there.
33    fn split<'a, const W: usize>(
34        self,
35        x: &'a [u64; W],
36        z: &'a [u64; W],
37    ) -> (&'a [u64; W], &'a [u64; W]) {
38        match self {
39            RotationAxis::Z => (x, z),
40            RotationAxis::X => (z, x),
41        }
42    }
43}
44
45/// The diagonal echo `Σₙ wₙ cos(2δ)ⁿ / Σₙ wₙ` from an [`anticommute_histogram`](PauliSum::anticommute_histogram) `hist`.
46///
47/// It keeps the class-diagonal terms of [`rotated_overlap`](PauliSum::rotated_overlap), normalized by `Σ|c|²`; an all-zero histogram gives `NaN`.
48pub fn diagonal_echo(hist: &[f64], delta: f64) -> f64 {
49    let cos = (2.0 * delta).cos();
50    let mut weight = 1.0;
51    let mut numerator = 0.0;
52    for &w in hist {
53        numerator += w * weight;
54        weight *= cos;
55    }
56    numerator / hist.iter().sum::<f64>()
57}
58
59/// The bit mask of `qubits`, each asserted below `num_qubits`.
60pub(crate) fn qubit_mask<const W: usize>(
61    qubits: impl IntoIterator<Item = usize>,
62    num_qubits: usize,
63) -> [u64; W] {
64    let mut mask = [0u64; W];
65    for q in qubits {
66        assert!(
67            q < num_qubits,
68            "qubit {q} is outside the {num_qubits}-qubit register"
69        );
70        mask[q / 64] |= 1u64 << (q % 64);
71    }
72    mask
73}
74
75/// [`qubit_mask`] of `sites`, which must also be distinct.
76fn site_mask<const W: usize>(sites: &[usize], num_qubits: usize) -> [u64; W] {
77    let mask = qubit_mask(sites.iter().copied(), num_qubits);
78    assert_eq!(
79        popcount(&mask),
80        sites.len(),
81        "echo: a site is listed twice in {sites:?}"
82    );
83    mask
84}
85
86fn and<const W: usize>(a: &[u64; W], b: &[u64; W]) -> [u64; W] {
87    std::array::from_fn(|w| a[w] & b[w])
88}
89
90fn popcount<const W: usize>(a: &[u64; W]) -> usize {
91    a.iter().map(|w| w.count_ones() as usize).sum()
92}

94impl<const W: usize> PartitionRows<W> {
95    /// `true` if no row reads a coordinate [`RotationAxis::flip_mask`] of `sites` names, so every [`PauliSum::rotated_overlap`] class lies in one partition.
96    pub fn keeps_flip_classes(&self, sites: &[usize], axis: RotationAxis) -> bool {
97        let (mask_x, mask_z) = axis.flip_mask(sites);
98        self.avoids(&mask_x, &mask_z)
99    }
100}

102/// One term that anticommutes with some generator, keyed by its class.
103struct Member<const W: usize> {
104    /// The key with the flippable bits cleared: equal for exactly the strings `V† · V` mixes.
105    class: ([u64; W], [u64; W]),
106    /// The flippable bits themselves, on the anticommuting sites `K`.
107    flip: [u64; W],
108    coeff: Complex64,
109}
110
111impl<const W: usize> PauliSum<W> {
112    /// `w[n] = Σ |c_P|²` over the strings `P` that anticommute with exactly `n` of the generators `G_q`, `q ∈ sites`; length `sites.len() + 1`.
113    ///
114    /// Feed it to [`diagonal_echo`]; histograms of disjoint shares of a sum add.
115    ///
116    /// # Panics
117    ///
118    /// If a site is repeated or is not below [`num_qubits`](Self::num_qubits).
119    pub fn anticommute_histogram(&self, sites: &[usize], axis: RotationAxis) -> Vec<f64> {
120        let mask = site_mask::<W>(sites, self.num_qubits());
121        let bins = sites.len() + 1;
122        (0..self.num_buckets())
123            .into_par_iter()
124            .fold(
125                || vec![0.0f64; bins],
126                |mut hist, b| {
127                    let (xs, zs, cs) = self.bucket(b);
128                    for ((x, z), c) in xs.iter().zip(zs).zip(cs) {
129                        hist[popcount(&and(axis.split(x, z).0, &mask))] += c.norm_sqr();
130                    }
131                    hist
132                },
133            )
134            .reduce(
135                || vec![0.0f64; bins],
136                |mut a, b| {
137                    for (x, y) in a.iter_mut().zip(b) {
138                        *x += y;
139                    }
140                    a
141                },
142            )
143    }
144
145    /// The operator Loschmidt echo `S_δ = 2⁻ⁿ Tr(A V† A V)` of this sum `A`, with `V = ⊗_{q ∈ sites} exp(-iδ G_q)`, computed without materializing `V† A V`.
146    ///
147    /// The value is the real part of `2⁻ⁿ Tr(A† V† A V)`, which is `S_δ` itself for a Hermitian `A`.
148    /// Work is `O(k²)` per class of `k` strings agreeing off the anticommuting sites' flipped bits, so one large class dominates.
149    ///
150    /// # Panics
151    ///
152    /// If a site is repeated or is not below [`num_qubits`](Self::num_qubits).
153    pub fn rotated_overlap(&self, sites: &[usize], delta: f64, axis: RotationAxis) -> f64 {
154        self.rotated_overlap_complex(sites, delta, axis).re
155    }
156
157    /// `2⁻ⁿ Tr(A† V† A V)` in full, the sign convention included; [`Self::rotated_overlap`] is its real part.
158    fn rotated_overlap_complex(
159        &self,
160        sites: &[usize],
161        delta: f64,
162        axis: RotationAxis,
163    ) -> Complex64 {
164        let mask = site_mask::<W>(sites, self.num_qubits());
165        let (sin, cos) = (2.0 * delta).sin_cos();
166        let cos_pow: Vec<f64> = (0..=sites.len()).map(|k| cos.powi(k as i32)).collect();
167        let sin_pow: Vec<f64> = (0..=sites.len()).map(|k| sin.powi(k as i32)).collect();
168
169        // A string commuting with every generator is its own class, and V leaves it alone.
170        let (fixed, members): (Vec<f64>, Vec<Vec<Member<W>>>) = (0..self.num_buckets())
171            .into_par_iter()
172            .fold(
173                || (0.0, Vec::new()),
174                |(mut fixed, mut members), b| {
175                    let (xs, zs, cs) = self.bucket(b);
176                    for ((x, z), &coeff) in xs.iter().zip(zs).zip(cs) {
177                        let (selector, flip) = axis.split(x, z);
178                        let k = and(selector, &mask);
179                        if popcount(&k) == 0 {
180                            fixed += coeff.norm_sqr();
181                            continue;
182                        }
183                        let cleared: [u64; W] = std::array::from_fn(|w| flip[w] & !k[w]);
184                        let class = match axis {
185                            RotationAxis::Z => (*x, cleared),
186                            RotationAxis::X => (cleared, *z),
187                        };
188                        let flip = and(flip, &k);
189                        members.push(Member { class, flip, coeff });
190                    }
191                    (fixed, members)
192                },
193            )
194            .unzip();
195        let mut members: Vec<Member<W>> = members.into_iter().flatten().collect();
196        members.par_sort_unstable_by(|a, b| a.class.cmp(&b.class));
197
198        let mixed: Complex64 = members
199            .par_chunk_by(|a, b| a.class == b.class)
200            .map(|class| {
201                let (x, z) = &class[0].class;
202                let sites_k = popcount(&and(axis.split(x, z).0, &mask));
203                let mut partial = Complex64::new(0.0, 0.0);
204                for p in class {
205                    for q in class {
206                        let d: [u64; W] = std::array::from_fn(|i| p.flip[i] ^ q.flip[i]);
207                        let num_flipped = popcount(&d);
208                        // R[P, Q] is `−sin 2δ` where P holds Y against Q's X (Z axis), or P holds Z against Q's Y (X axis).
209                        let negative = match axis {
210                            RotationAxis::Z => popcount(&and(&p.flip, &d)),
211                            RotationAxis::X => popcount(&and(&q.flip, &d)),
212                        } % 2
213                            == 1;
214                        let magnitude = cos_pow[sites_k - num_flipped] * sin_pow[num_flipped];
215                        partial += p.coeff.conj()
216                            * q.coeff
217                            * if negative { -magnitude } else { magnitude };
218                    }
219                }
220                partial
221            })
222            .sum();
223
224        mixed + fixed.iter().sum::<f64>()
225    }
226}
