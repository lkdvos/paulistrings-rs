14impl<const W: usize> PauliSum<W> {
15    /// `Σ f(x, z, c)` over the terms, buckets in parallel; the summation order is unspecified.
16    pub(crate) fn sum_over_terms(
17        &self,
18        f: impl Fn(&[u64; W], &[u64; W], Complex64) -> Complex64 + Sync,
19    ) -> Complex64 {
20        self.buckets()
21            .par_iter()
22            .map(|columns| {
23                let mut partial = Complex64::new(0.0, 0.0);
24                for i in 0..columns.len() {
25                    partial += f(&columns.x[i], &columns.z[i], columns.coeff[i]);
26                }
27                partial
28            })
29            .sum()
30    }
31}
