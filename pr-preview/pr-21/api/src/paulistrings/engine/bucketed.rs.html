1//! One layer of the bucketed engine, applied coset by coset, and the reusable `LayerScratch`.
2//!
3//! A coset task swaps its bucket columns into worker scratch and writes its outputs back into the emptied columns, so the layer runs in place.
4//! Identity-delta rows keep their keys, so a run's identity stream inherits the source bucket's order and is never sorted; every other row, received rows included, joins the rest stream.

28/// Reusable layer working set for [`propagate_with`](crate::propagate_with), which keeps its capacity across layers and calls and carries the opt-in traces.
29///
30/// A task's output cannot depend on which worker slot it drew: every write destination is cleared before use.
31#[derive(Debug, Default)]
32pub struct LayerScratch<const W: usize> {
33    /// The serial path's coset working set.
34    pub(super) task: CosetScratch<W>,
35    /// The layer's coset order, with its inverse only under [`ExtraRows::NEEDS_BETA`]; a partitioned layer builds it before the export and passes it back in [`LayerKnobs::order`].
36    pub(super) order: CosetOrder,
37    /// Bucket handles in coset-contiguous order while a layer runs.
38    pub(super) staging: Vec<BucketColumns<W>>,
39    /// One coset working set per Rayon worker, indexed by `rayon::current_thread_index()`, so each mutex is uncontended.
40    pub(super) workers: Vec<Mutex<CosetScratch<W>>>,
41    /// Layer-level wall-clock counters; per-coset busy time lives in each `CosetScratch`.
42    #[cfg(feature = "phase-timing")]
43    pub(crate) stats: PhaseStats,
44    /// `None` unless [`Self::enable_trace`] was called.
45    pub(crate) trace: Option<PartitionTrace>,
46}
47
48impl<const W: usize> LayerScratch<W> {
49    /// An empty scratch.
50    pub fn new() -> Self {
51        Self::default()
52    }
53
54    /// Drain the phase counters since the last drain, zeroing them; `per_partition` has one entry.
55    #[cfg(feature = "phase-timing")]
56    pub fn take_stats(&mut self) -> PartitionPhaseStats {
57        let stats = self.take_phase_stats();
58        PartitionPhaseStats {
59            layers: stats.layers,
60            per_partition: vec![stats],
61            scatter_ns: 0,
62            gather_ns: 0,
63        }
64    }
65
66    /// Drain this scratch's own counters, one partition's entry of a driver's [`PartitionPhaseStats`].
67    #[cfg(feature = "phase-timing")]
68    pub(crate) fn take_phase_stats(&mut self) -> PhaseStats {
69        let mut total = std::mem::take(&mut self.stats);
70        total.absorb_coset(&std::mem::take(&mut self.task.stats));
71        for slot in &self.workers {
72            let mut worker_scratch = slot.lock().unwrap();
73            total.absorb_coset(&std::mem::take(&mut worker_scratch.stats));
74        }
75        total
76    }
77
78    /// Start recording a one-partition [`PartitionTrace`] on every later propagation driven by this scratch; idempotent.
79    pub fn enable_trace(&mut self) {
80        self.trace.get_or_insert_with(PartitionTrace::default);
81    }
82
83    /// Drain the per-layer records, or `None` if tracing was never enabled; tracing stays enabled.
84    pub fn take_trace(&mut self) -> Option<PartitionTrace> {
85        self.trace.as_mut().map(std::mem::take)
86    }
87}

89/// A prepared channel's delta set annotated with each entry's coset coordinate, computed once per layer.
90pub(super) enum DeltaPlan<'p, const W: usize> {
91    /// Tabulated deltas; `coordinates[e]` pairs with `ptm.deltas()[e]`.
92    Local {
93        ptm: &'p LocalPtm<W>,
94        coordinates: Vec<u32>,
95        /// Whether `deltas()[0]` is the identity delta, whose stream goes to the run's pre-sorted `id` columns.
96        has_identity: bool,
97        /// Whether the identity amplitude is nonzero on every active support pattern, so the identity stream is one-to-one with the source bucket and the merge borrows its key columns.
98        dense_identity: bool,
99        /// This layer's per-run sort kernel.
100        sort_kernel: SortKernel,
101    },
102    /// Wide rotation: two implicit entries, the identity pass and the generator pass.
103    Rotation {
104        rotation: &'p PreparedRotation<W>,
105        coordinate_identity: u32,
106        generator_coordinate: u32,
107        /// Whether the generator pass emits here ([`LayerKnobs::generator_local`]); `generator_coordinate` is 0 when not.
108        generator_local: bool,
109    },
110}
111
112/// Per-layer overrides the partitioned engine hands the coset loop; [`Default`] is the unpartitioned answer.
113#[derive(Clone, Copy, Debug)]
114pub(crate) struct LayerKnobs<'k> {
115    /// Bucket deltas to build the coset span from, instead of [`Prepared::bucket_deltas`].
116    pub bucket_deltas: Option<&'k [u32]>,
117    /// The channel's total rest-stream count, so every partition picks the sort kernel the unpartitioned run would.
118    pub rest_streams: Option<usize>,
119    /// The unrestricted channel's [`rest_rows_per_key`], since a PTM cut down to local entries can look more disjoint than the channel is.
120    pub rows_per_key: Option<f64>,
121    /// Whether a wide rotation's generator pass emits here; false when every generator row belongs to a partner.
122    pub generator_local: bool,
123    /// The layer's coset order, already built `with_inverse` over the span of `bucket_deltas`; built here when `None`.
124    pub order: Option<&'k CosetOrder>,
125}
126
127impl Default for LayerKnobs<'_> {
128    fn default() -> Self {
129        Self {
130            bucket_deltas: None,
131            rest_streams: None,
132            rows_per_key: None,
133            generator_local: true,
134            order: None,
135        }
136    }
137}

139impl<'p, const W: usize> DeltaPlan<'p, W> {
140    pub(super) fn new(prepared: &'p Prepared<W>, span: &Gf2Span, knobs: LayerKnobs<'_>) -> Self {
141        match prepared {
142            Prepared::Local(ptm) => {
143                let coordinates: Vec<u32> = ptm
144                    .deltas()
145                    .iter()
146                    .map(|d| span.coordinate_of(d.bucket_delta))
147                    .collect();
148                let has_identity = ptm.has_identity();
149                debug_assert!(!has_identity || coordinates[0] == 0);
150                let dense_identity = has_identity && ptm.entry_is_dense(0);
151                let rest_streams = knobs.rest_streams.unwrap_or(ptm.rest_streams());
152                let sort_kernel = SortKernel::for_layer(rest_streams, || {
153                    knobs.rows_per_key.unwrap_or_else(|| rest_rows_per_key(ptm))
154                });
155                DeltaPlan::Local {
156                    ptm,
157                    coordinates,
158                    has_identity,
159                    dense_identity,
160                    sort_kernel,
161                }
162            }
163            Prepared::Rotation(rotation) => DeltaPlan::Rotation {
164                rotation,
165                coordinate_identity: span.coordinate_of(rotation.bucket_delta_identity),
166                // `coordinate_of` demands its argument be in the span, and a remote generator's bucket delta is not.
167                generator_coordinate: if knobs.generator_local {
168                    span.coordinate_of(rotation.bucket_delta_generator)
169                } else {
170                    0
171                },
172                generator_local: knobs.generator_local,
173            },
174        }
175    }
176}

178/// Rows received from other partitions, appended to an output bucket's rest stream before the sort so `keep_term` sees the complete sum; never the identity stream, which the merge requires strictly ascending.
179/// An implementation that can return rows must set `NEEDS_BETA = true`; under `false` every call site compiles away.
180pub(crate) trait ExtraRows<const W: usize> {
181    /// Whether the engine recovers original bucket indices and calls this source at all.
182    const NEEDS_BETA: bool = false;
183
184    /// Rows destined for original bucket `beta`; must equal what [`append_into`](Self::append_into) pushes.
185    fn count(&self, beta: u32) -> usize {
186        let _ = beta;
187        0
188    }
189
190    /// Append original bucket `beta`'s rows onto the run's rest columns.
191    fn append_into(&self, beta: u32, rest: &mut BucketColumns<W>) {
192        let _ = (beta, rest);
193    }
194}

196/// The unpartitioned engine's [`ExtraRows`]: no rows.
197pub(crate) struct NoExtra;

199impl<const W: usize> ExtraRows<W> for NoExtra {}

201/// Apply one prepared channel to a bucketed sum, folding `keep_term` into the merge; the caller runs `finalize_layer`.
202pub fn apply_layer_bucketed<const W: usize, T>(
203    sum: &mut PauliSum<W>,
204    prepared: &Prepared<W>,
205    policy: &T,
206    scratch: &mut LayerScratch<W>,
207) where
208    T: TruncationPolicy<W> + ?Sized,
209{
210    apply_layer_bucketed_with(
211        sum,
212        prepared,
213        policy,
214        scratch,
215        &NoExtra,
216        LayerKnobs::default(),
217    )
218}

220/// [`apply_layer_bucketed`] with the partitioned engine's [`ExtraRows`] source and [`LayerKnobs`].
221pub(crate) fn apply_layer_bucketed_with<const W: usize, T, X>(
222    sum: &mut PauliSum<W>,
223    prepared: &Prepared<W>,
224    policy: &T,
225    scratch: &mut LayerScratch<W>,
226    extra: &X,
227    knobs: LayerKnobs<'_>,
228) where
229    T: TruncationPolicy<W> + ?Sized,
230    X: ExtraRows<W> + Sync,
231{
232    stamp!(stamp);
233
234    // A partitioned layer's local table can look key-preserving while received rows still need merging.
235    if let Prepared::Local(ptm) = prepared {
236        if !X::NEEDS_BETA && ptm.is_key_preserving() {
237            rescale_in_place(sum, ptm, policy);
238            lap!(stamp, scratch.stats.rescale_ns);
239            return;
240        }
241    }
242
243    let own_deltas;
244    let deltas: &[u32] = match knobs.bucket_deltas {
245        Some(deltas) => deltas,
246        None => {
247            own_deltas = prepared.bucket_deltas();
248            &own_deltas
249        }
250    };
251    let span = Gf2Span::new(deltas, sum.hash().bits());
252    let plan = DeltaPlan::new(prepared, &span, knobs);
253    let coset_size = span.coset_size();
254    let num_cosets = span.num_cosets();
255    lap!(stamp, scratch.stats.span_plan_ns);
256
257    // Coset `c` owns `staging[c·2^r .. (c+1)·2^r]`; at `r = 0` the order is the identity and the handle passes are skipped.
258    let order: &CosetOrder = match knobs.order {
259        Some(order) => order,
260        None => {
261            scratch.order.rebuild(&span, X::NEEDS_BETA);
262            &scratch.order
263        }
264    };
265    debug_assert_eq!(order.positions(), sum.num_buckets());
266    debug_assert_eq!(order.dimension(), span.dimension());
267    let identity_order = order.is_identity();
268    if !identity_order {
269        let buckets = sum.buckets_mut();
270        scratch
271            .staging
272            .resize_with(buckets.len(), BucketColumns::default);
273        for (beta, columns) in buckets.iter_mut().enumerate() {
274            scratch.staging[order.position_of(beta as u32) as usize] = std::mem::take(columns);
275        }
276    }
277    lap!(stamp, scratch.stats.permute_ns);
278
279    // Each coset reads and writes only its own chunk (engine/coset.rs); research/FINDINGS.md §Static coset→worker placement.
280    {
281        if num_cosets >= MIN_COSETS_FOR_PARALLEL {
282            let pool = rayon::current_num_threads().max(1);
283            if scratch.workers.len() < pool {
284                scratch.workers.resize_with(pool, Mutex::default);
285            }
286        }
287        let workers = &scratch.workers;
288        let chunks: &mut [BucketColumns<W>] = if identity_order {
289            sum.buckets_mut()
290        } else {
291            scratch.staging.as_mut_slice()
292        };
293        if num_cosets < MIN_COSETS_FOR_PARALLEL {
294            for (coset_index, chunk) in chunks.chunks_mut(coset_size).enumerate() {
295                let base = coset_index * coset_size;
296                fill_coset::<W, T, X>(chunk, &plan, policy, &mut scratch.task, extra, base, order);
297            }
298        } else {
299            chunks
300                .par_chunks_mut(coset_size)
301                .enumerate()
302                .for_each(|(coset_index, chunk)| {
303                    let base = coset_index * coset_size;
304                    // `workers` was sized to `current_num_threads()` above, and this closure runs on that pool.
305                    let worker = rayon::current_thread_index()
306                        .expect("the coset loop runs on a Rayon pool worker");
307                    let mut worker_scratch = workers[worker].lock().unwrap();
308                    fill_coset::<W, T, X>(
309                        chunk,
310                        &plan,
311                        policy,
312                        &mut worker_scratch,
313                        extra,
314                        base,
315                        order,
316                    );
317                });
318        }
319    }
320    lap!(stamp, scratch.stats.coset_loop_ns);
321
322    if !identity_order {
323        let buckets = sum.buckets_mut();
324        for (beta, columns) in buckets.iter_mut().enumerate() {
325            *columns =
326                std::mem::take(&mut scratch.staging[order.position_of(beta as u32) as usize]);
327        }
328    }
329    lap!(stamp, scratch.stats.unpermute_ns);
330    sum.recount();
331    lap!(stamp, scratch.stats.recount_ns);
332
333    #[cfg(debug_assertions)]
334    sum.assert_invariants();
335}

337/// In-place coefficient rescale for a key-preserving channel, with the general path's zero drop and `keep_term`.
338fn rescale_in_place<const W: usize, T>(sum: &mut PauliSum<W>, ptm: &LocalPtm<W>, policy: &T)
339where
340    T: TruncationPolicy<W> + ?Sized,
341{
342    let amplitude = &ptm.deltas()[0].amplitude;
343    sum.update_per_bucket(|_| {
344        |x: &[u64; W], z: &[u64; W], c: Complex64| {
345            let coeff = c * amplitude[ptm.support_bits(x, z)];
346            (coeff != ZERO && policy.keep_term(x, z, coeff)).then_some(coeff)
347        }
348    });
349
350    #[cfg(debug_assertions)]
351    sum.assert_invariants();
352}
