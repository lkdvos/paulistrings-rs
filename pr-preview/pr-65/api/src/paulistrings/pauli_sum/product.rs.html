1//! Operator products of two [`PauliSum`]s: [`PauliSum::product`], [`PauliSum::liouvillian`] and [`PauliSum::anticommutator`].
2//!
3//! A string product XORs keys and the bucket hash is GF(2)-linear, so output bucket `β′` reads only the larger operand's buckets `β′ ⊕ h(s)` over the smaller operand's terms `s`.
4//! Each output bucket is therefore one task that accumulates into its own [`Table`], and no two tasks write the same key.
5//! A Liouvillian visits only the pairs whose supports meet, through a [`support::SupportIndex`] of the larger operand.
6

21/// Which pairs a product keeps and the factor it puts on them.
22#[derive(Clone, Copy, Debug, PartialEq, Eq)]
23pub(crate) enum Pairing {
24    /// `A·B`: every pair, at `i^k`.
25    Product,
26    /// `i[A, B]`: anticommuting pairs, at `2·i^(k+1)`, which is real.
27    Liouvillian,
28    /// `{A, B}`: commuting pairs, at `2·i^k`, which is real.
29    Anticommutator,
30}
31
32impl Pairing {
33    /// The term `(a·P_a)·(b·P_b)` contributes, or `None` when the pairing drops the pair.
34    #[inline(always)]
35    fn term<const W: usize, C: Coefficient>(
36        self,
37        left: &Entry<W, C>,
38        right: &Entry<W, C>,
39    ) -> Option<Entry<W, C>> {
40        let mut key = PauliString { x: left.x, z: left.z };
41        let other = PauliString { x: right.x, z: right.z };
42        let (keep, extra) = match self {
43            Pairing::Product => (true, Phase::ONE),
44            Pairing::Liouvillian => (!key.commutes_with(&other), Phase::I),
45            Pairing::Anticommutator => (key.commutes_with(&other), Phase::ONE),
46        };
47        if !keep {
48            return None;
49        }
50        let phase = key.mul_assign(&other) + extra;
51        let mut c = left.c * right.c;
52        if self != Pairing::Product {
53            c = c * C::real(2.0);
54        }
55        Some(Entry { x: key.x, z: key.z, c: c.mul_phase(phase) })
56    }
57}

59/// A generator's terms sorted by their bucket under the output hash, with the run boundaries.
60struct Groups<const W: usize, C: Coefficient> {
61    terms: Vec<Entry<W, C>>,
62    /// The bucket of each run, ascending.
63    buckets: Vec<u32>,
64    /// Run `g` is `terms[starts[g]..starts[g + 1]]`.
65    starts: Vec<usize>,
66}
67
68impl<const W: usize, C: Coefficient> Groups<W, C> {
69    fn new(terms: impl Iterator<Item = Entry<W, C>>, hash: &Gf2Hash<W>) -> Self {
70        let mut keyed: Vec<(u32, Entry<W, C>)> =
71            terms.map(|e| (hash.bucket_of(&e.x, &e.z), e)).collect();
72        keyed.sort_unstable_by_key(|(bucket, _)| *bucket);
73        let mut buckets = Vec::new();
74        let mut starts = Vec::new();
75        for (i, (bucket, _)) in keyed.iter().enumerate() {
76            if buckets.last() != Some(bucket) {
77                buckets.push(*bucket);
78                starts.push(i);
79            }
80        }
81        starts.push(keyed.len());
82        Self { terms: keyed.into_iter().map(|(_, e)| e).collect(), buckets, starts }
83    }
84}

86/// One generator against one operand, as an output pass reads them.
87pub(crate) struct Job<'a, const W: usize, C: Coefficient> {
88    groups: Groups<W, C>,
89    operand: &'a PauliSum<W, C>,
90    /// Built for a Liouvillian only, since disjoint supports always commute.
91    index: Option<SupportIndex>,
92    /// Whether the generator is the left factor of each pair.
93    generator_left: bool,
94}
95
96impl<'a, const W: usize, C: Coefficient> Job<'a, W, C> {
97    pub(crate) fn new(
98        generator: impl Iterator<Item = Entry<W, C>>,
99        operand: &'a PauliSum<W, C>,
100        pairing: Pairing,
101        generator_left: bool,
102    ) -> Self {
103        let index = (pairing == Pairing::Liouvillian).then(|| SupportIndex::new(operand));
104        Self::with_index(generator, operand, index, generator_left)
105    }
106
107    /// [`Job::new`] with `operand`'s support index already built, which must be `Some` exactly for a Liouvillian.
108    pub(crate) fn with_index(
109        generator: impl Iterator<Item = Entry<W, C>>,
110        operand: &'a PauliSum<W, C>,
111        index: Option<SupportIndex>,
112        generator_left: bool,
113    ) -> Self {
114        Self { groups: Groups::new(generator, &operand.hash), operand, index, generator_left }
115    }
116
117    /// Every pair of this job landing in output bucket `out`, accumulated into `table`.
118    #[inline]
119    fn accumulate_into(
120        &self,
121        out: usize,
122        pairing: Pairing,
123        table: &mut Table<W, C>,
124        candidates: &mut Vec<u64>,
125    ) {
126        for (g, &delta) in self.groups.buckets.iter().enumerate() {
127            let b = out ^ delta as usize;
128            let input = &self.operand.buckets[b];
129            let mut emit = |i: usize, s: &Entry<W, C>| {
130                let l = Entry { x: input.x[i], z: input.z[i], c: input.coeff[i] };
131                let term =
132                    if self.generator_left { pairing.term(s, &l) } else { pairing.term(&l, s) };
133                if let Some(t) = term {
134                    table.accumulate(t.x, t.z, t.c);
135                }
136            };
137            for s in &self.groups.terms[self.groups.starts[g]..self.groups.starts[g + 1]] {
138                let support: [u64; W] = std::array::from_fn(|w| s.x[w] | s.z[w]);
139                let weight: u32 = support.iter().map(|w| w.count_ones()).sum();
140                match &self.index {
141                    Some(index) if weight <= SCAN_ABOVE_WEIGHT => {
142                        index.candidates(b, &support, candidates);
143                        for (w, &word) in candidates.iter().enumerate() {
144                            let mut bits = word;
145                            while bits != 0 {
146                                emit(w * 64 + bits.trailing_zeros() as usize, s);
147                                bits &= bits - 1;
148                            }
149                        }
150                    }
151                    _ => (0..input.len()).for_each(|i| emit(i, s)),
152                }
153            }
154        }
155    }
156}

158/// The factors of `dest ← alpha · pairs + beta · dest`.
159#[derive(Clone, Copy, Debug)]
160pub(crate) struct Weights<C> {
161    pub(crate) alpha: C,
162    pub(crate) beta: C,
163}
164
165impl<C: Coefficient> Weights<C> {
166    /// A fresh result: `alpha = 1`, `beta = 0`.
167    pub(crate) fn fresh() -> Self {
168        Self { alpha: C::ONE, beta: C::ZERO }
169    }
170}

172/// The table's terms merged with `beta · dest`, key-sorted, keeping the nonzero sums `keep` accepts.
173fn drained_into<const W: usize, C: Coefficient>(
174    table: &Table<W, C>,
175    dest: &BucketColumns<W, C>,
176    beta: C,
177    keep: &impl Fn(&[u64; W], &[u64; W], C) -> bool,
178) -> BucketColumns<W, C> {
179    let mut entries: Vec<&Entry<W, C>> = table.entries().iter().collect();
180    entries.sort_unstable_by(|a, b| (&a.x, &a.z).cmp(&(&b.x, &b.z)));
181    let held = if beta == C::ZERO { 0 } else { dest.len() };
182    merge_summed(
183        entries.iter().map(|e| (e.x, e.z, e.c)),
184        scaled_terms(dest, beta).take(held),
185        entries.len() + held,
186        keep,
187    )
188}

190/// `dest ← alpha · Σ pairs + beta · dest` over every pair of every job, where the jobs' generators already carry `alpha` and `dest` is partitioned exactly as the operands; one bucket of `dest` is replaced at a time, keeping the summed terms `keep` accepts.
191///
192/// # Panics
193///
194/// If `dest` or an operand is partitioned differently from `hash`, or the qubit counts differ.
195pub(crate) fn accumulate_into<const W: usize, C: Coefficient>(
196    dest: &mut PauliSum<W, C>,
197    beta: C,
198    jobs: &[Job<'_, W, C>],
199    pairing: Pairing,
200    keep: &(impl Fn(&[u64; W], &[u64; W], C) -> bool + Sync),
201) {
202    for job in jobs {
203        assert!(
204            job.operand.hash.same_rows_as(&dest.hash) && job.operand.hash.bits() == dest.hash.bits(),
205            "accumulate_into: operands partitioned differently from the destination",
206        );
207        assert_eq!(job.operand.num_qubits, dest.num_qubits, "accumulate_into: num_qubits mismatch");
208    }
209    dest.buckets.par_iter_mut().enumerate().for_each_init(
210        || (Table::<W, C>::default(), Vec::new()),
211        |(table, candidates), (out, columns)| {
212            table.clear();
213            for job in jobs {
214                job.accumulate_into(out, pairing, table, candidates);
215            }
216            *columns = drained_into(table, columns, beta, keep);
217        },
218    );
219    dest.recount();
220}

222impl<const W: usize, C: Coefficient> PauliSum<W, C> {
223    /// `self ← weights.alpha · pairing(generator, operand) + weights.beta · self`, keeping the summed terms `keep` accepts; `self` ends partitioned under the larger operand's hash rows.
224    pub(crate) fn pair_into(
225        &mut self,
226        generator: &Self,
227        operand: &Self,
228        pairing: Pairing,
229        weights: Weights<C>,
230        keep: &(impl Fn(&[u64; W], &[u64; W], C) -> bool + Sync),
231    ) {
232        assert!(
233            generator.num_qubits == operand.num_qubits && self.num_qubits == operand.num_qubits,
234            "PauliSum::{pairing:?}: num_qubits mismatch (generator {}, operand {}, destination {})",
235            generator.num_qubits,
236            operand.num_qubits,
237            self.num_qubits,
238        );
239        let generator_is_large = generator.len > operand.len;
240        let (large, small) =
241            if generator_is_large { (generator, operand) } else { (operand, generator) };
242
243        // The large operand's own bucket count, raised to the parallelism floor when the pair count warrants it; only a sum below that floor is refined, so the copy is small.
244        let pairs = large.len.saturating_mul(small.len);
245        let floor = desired_bits(pairs, DEFAULT_TARGET_BUCKET_LEN, DEFAULT_MIN_BUCKETS)
246            .min(DEFAULT_MIN_BUCKETS.trailing_zeros() as u8)
247            .min(large.hash.max_bits());
248        let refined;
249        let large = if floor > large.hash.bits() {
250            let mut copy = large.clone();
251            while copy.hash.bits() < floor {
252                copy.refine();
253            }
254            refined = copy;
255            &refined
256        } else {
257            large
258        };
259
260        self.align_destination(&large.hash, weights.beta);
261        let alpha = weights.alpha;
262        let scaled = small.iter().map(|(x, z, c)| Entry { x: *x, z: *z, c: c * alpha });
263        let job = Job::new(scaled, large, pairing, !generator_is_large);
264        accumulate_into(self, weights.beta, &[job], pairing, keep);
265        self.rebucket(DEFAULT_TARGET_BUCKET_LEN, DEFAULT_MIN_BUCKETS);
266    }
267
268    /// `pairing(self, other)` as a new sum, keeping every summed term.
269    fn paired(&self, other: &Self, pairing: Pairing) -> Self {
270        let mut out = Self::empty_with_hash(self.num_qubits, other.hash.clone());
271        out.pair_into(self, other, pairing, Weights::fresh(), &|_, _, _| true);
272        out
273    }
274
275    /// The operator product `self · other`; terms summing to exactly zero are dropped.
276    ///
277    /// The result is partitioned under the larger operand's hash rows, at the bucket count its size asks for.
278    ///
279    /// # Panics
280    ///
281    /// If the two sums disagree about `num_qubits`, or if `C` is real: a product of anticommuting Hermitian strings carries an odd phase that no real coefficient holds.
282    pub fn product(&self, other: &Self) -> Self {
283        assert!(
284            !C::IS_REAL,
285            "PauliSum::product: a product is not Hermitian, so a real coefficient type ({}) cannot hold it; cast to a complex type first",
286            C::NAME,
287        );
288        self.paired(other, Pairing::Product)
289    }
290
291    /// The Liouvillian `i[self, other] = i(self · other − other · self)`, Hermitian whenever both operands are, so a real coefficient type stays exact.
292    ///
293    /// Only anticommuting string pairs contribute; terms summing to exactly zero are dropped, and the result is partitioned as [`Self::product`]'s.
294    ///
295    /// # Panics
296    ///
297    /// If the two sums disagree about `num_qubits`.
298    pub fn liouvillian(&self, other: &Self) -> Self {
299        self.paired(other, Pairing::Liouvillian)
300    }
301
302    /// The anticommutator `{self, other} = self · other + other · self`, Hermitian whenever both operands are.
303    ///
304    /// Only commuting string pairs contribute; terms summing to exactly zero are dropped, and the result is partitioned as [`Self::product`]'s.
305    ///
306    /// # Panics
307    ///
308    /// If the two sums disagree about `num_qubits`.
309    pub fn anticommutator(&self, other: &Self) -> Self {
310        self.paired(other, Pairing::Anticommutator)
311    }
312}
