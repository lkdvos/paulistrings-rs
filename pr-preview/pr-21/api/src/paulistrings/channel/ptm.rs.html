13const ZERO: Complex64 = Complex64::new(0.0, 0.0);
14const ONE: Complex64 = Complex64::new(1.0, 0.0);
15
16/// A one- or two-qubit gate or noise channel, stored as its Pauli-transfer matrix.
17///
18/// Built by the named constructors: the Cliffords [`hadamard`](Self::hadamard), [`phase_gate`](Self::phase_gate), [`pauli_x`](Self::pauli_x), [`pauli_y`](Self::pauli_y), [`pauli_z`](Self::pauli_z), [`cnot`](Self::cnot), [`cz`](Self::cz), [`swap`](Self::swap);
19/// the general unitaries [`unitary_1q`](Self::unitary_1q), [`unitary_2q`](Self::unitary_2q);
20/// and the noise channels [`depolarizing`](Self::depolarizing), [`dephasing`](Self::dephasing), [`pauli_channel`](Self::pauli_channel), [`depolarizing_2q`](Self::depolarizing_2q), [`amplitude_damping`](Self::amplitude_damping).
21///
22/// On the support, Paulis are indexed `x | (z << 1)` per qubit (`I = 0, X = 1, Z = 2, Y = 3`), the constructor's first qubit in the low two bits.
23/// [`Channel::apply`] maps input Pauli `s` to `Σ_t ptm[s][t] · P_t`, and [`Channel::apply_adjoint`] reads the transpose.
24#[derive(Clone, Debug)]
25pub struct PtmChannel {
26    /// Support qubits in constructor order; only the first `num_support_qubits` are meaningful.
27    qubits: [usize; 2],
28    num_support_qubits: u8,
29    storage: PtmStorage,
30    /// What [`Channel::debug_name`] reports.
31    name: &'static str,
32}
33
34/// The nonzero PTM entries, in the cheapest form the constructor knows to be exact.
35#[derive(Clone, Debug)]
36enum PtmStorage {
37    /// A Clifford: input `s` goes to `image[s]` with coefficient `-1` where bit `s` of `negated` is set, `+1` otherwise.
38    SignedPermutation {
39        image: [u8; LOCAL_DIM],
40        negated: u16,
41    },
42    /// A Pauli-diagonal channel: input `s` keeps its key and is scaled by `scale[s]`.
43    Diagonal { scale: Box<[f64]> },
44    /// Everything else: `amplitudes[i]` takes input `positions[i].0` to output `positions[i].1`, ascending by position.
45    Sparse {
46        positions: Box<[(u8, u8)]>,
47        amplitudes: Box<[Complex64]>,
48    },
49}
50
51impl PtmChannel {
52    /// A channel on `qubits` (one or two, distinct) from its storage.
53    fn new(qubits: &[usize], storage: PtmStorage, name: &'static str) -> Self {
54        debug_assert!(qubits.len() == 1 || qubits.len() == 2);
55        let mut padded = [qubits[0]; 2];
56        padded[..qubits.len()].copy_from_slice(qubits);
57        Self {
58            qubits: padded,
59            num_support_qubits: qubits.len() as u8,
60            storage,
61            name,
62        }
63    }
64
65    /// The support qubits, in constructor order.
66    pub fn qubits(&self) -> &[usize] {
67        &self.qubits[..self.num_support_qubits as usize]
68    }
69
70    /// The adjoint channel, whose PTM is this one's transpose.
71    pub fn adjoint(&self) -> Self {
72        let storage = match &self.storage {
73            PtmStorage::SignedPermutation { image, negated } => {
74                let mut inverse = [0u8; LOCAL_DIM];
75                let mut inverse_negated = 0u16;
76                for (s, &t) in image[..self.dimension()].iter().enumerate() {
77                    let t = t as usize;
78                    inverse[t] = s as u8;
79                    inverse_negated |= ((negated >> s) & 1) << t;
80                }
81                PtmStorage::SignedPermutation {
82                    image: inverse,
83                    negated: inverse_negated,
84                }
85            }
86            PtmStorage::Diagonal { scale } => PtmStorage::Diagonal {
87                scale: scale.clone(),
88            },
89            PtmStorage::Sparse {
90                positions,
91                amplitudes,
92            } => {
93                let mut entries: Vec<((u8, u8), Complex64)> = positions
94                    .iter()
95                    .map(|&(s, t)| (t, s))
96                    .zip(amplitudes.iter().copied())
97                    .collect();
98                entries.sort_by_key(|&(position, _)| position);
99                sparse_storage(entries)
100            }
101        };
102        Self {
103            storage,
104            ..self.clone()
105        }
106    }
107
108    /// `4^k` for `k` support qubits.
109    fn dimension(&self) -> usize {
110        1 << (2 * self.num_support_qubits)
111    }
112
113    /// The local Pauli index of the input's restriction to the support.
114    fn local_index<const W: usize>(&self, x: &[u64; W], z: &[u64; W]) -> usize {
115        self.qubits()
116            .iter()
117            .enumerate()
118            .map(|(j, &q)| {
119                let (word, bit, _) = qubit_loc(q);
120                read_pauli(x, z, word, bit) << (2 * j)
121            })
122            .fold(0, |index, part| index | part)
123    }
124
125    /// The input with its support overwritten by local Pauli index `t`.
126    fn with_local_index<const W: usize>(
127        &self,
128        x: &[u64; W],
129        z: &[u64; W],
130        t: usize,
131    ) -> ([u64; W], [u64; W]) {
132        let mut new_x = *x;
133        let mut new_z = *z;
134        for (j, &q) in self.qubits().iter().enumerate() {
135            let (word, bit, mask) = qubit_loc(q);
136            write_pauli(&mut new_x, &mut new_z, word, bit, mask, (t >> (2 * j)) & 3);
137        }
138        (new_x, new_z)
139    }
140
141    /// Body of `apply` (`transpose == false`) and `apply_adjoint`.
142    fn apply_rows<const W: usize>(
143        &self,
144        transpose: bool,
145        x: &[u64; W],
146        z: &[u64; W],
147        coeff: Complex64,
148        out: &mut OutputBuffer<W>,
149    ) {
150        let s = self.local_index(x, z);
151        match &self.storage {
152            PtmStorage::SignedPermutation { image, negated } => {
153                let (source, t) = if transpose {
154                    let source = image[..self.dimension()]
155                        .iter()
156                        .position(|&image| image as usize == s)
157                        .expect("a signed permutation is a bijection");
158                    (source, source)
159                } else {
160                    (s, image[s] as usize)
161                };
162                let (new_x, new_z) = self.with_local_index(x, z, t);
163                let signed = if (negated >> source) & 1 == 1 {
164                    -coeff
165                } else {
166                    coeff
167                };
168                out.push(new_x, new_z, signed);
169            }
170            PtmStorage::Diagonal { scale } => out.push(*x, *z, coeff * scale[s]),
171            PtmStorage::Sparse {
172                positions,
173                amplitudes,
174            } => {
175                for (&(input, output), &amplitude) in positions.iter().zip(amplitudes.iter()) {
176                    let (from, to) = if transpose {
177                        (output, input)
178                    } else {
179                        (input, output)
180                    };
181                    if from as usize == s {
182                        let (new_x, new_z) = self.with_local_index(x, z, to as usize);
183                        out.push(new_x, new_z, coeff * amplitude);
184                    }
185                }
186            }
187        }
188    }
189
190    /// Calls `visit(s, t, amplitude)` for every stored entry.
191    fn for_each_entry(&self, mut visit: impl FnMut(usize, usize, Complex64)) {
192        match &self.storage {
193            PtmStorage::SignedPermutation { image, negated } => {
194                for (s, &t) in image[..self.dimension()].iter().enumerate() {
195                    let amplitude = if (negated >> s) & 1 == 1 { -ONE } else { ONE };
196                    visit(s, t as usize, amplitude);
197                }
198            }
199            PtmStorage::Diagonal { scale } => {
200                for (s, &value) in scale.iter().enumerate() {
201                    visit(s, s, Complex64::new(value, 0.0));
202                }
203            }
204            PtmStorage::Sparse {
205                positions,
206                amplitudes,
207            } => {
208                for (&(s, t), &amplitude) in positions.iter().zip(amplitudes.iter()) {
209                    visit(s as usize, t as usize, amplitude);
210                }
211            }
212        }
213    }
214
215    /// The dense `table[s][t]` (transposed when `adjoint`) in ascending-qubit local indices, the order [`Prepared`] uses.
216    fn transfer_matrix(&self, adjoint: bool) -> [[Complex64; LOCAL_DIM]; LOCAL_DIM] {
217        let reversed = self.num_support_qubits == 2 && self.qubits[0] > self.qubits[1];
218        let reorder = |index: usize| {
219            if reversed {
220                ((index & 3) << 2) | (index >> 2)
221            } else {
222                index
223            }
224        };
225        let mut table = [[ZERO; LOCAL_DIM]; LOCAL_DIM];
226        self.for_each_entry(|s, t, amplitude| {
227            let (row, column) = if adjoint { (t, s) } else { (s, t) };
228            table[reorder(row)][reorder(column)] += amplitude;
229        });
230        table
231    }
232}
233
234/// Sparse storage from `(position, amplitude)` pairs ascending by position, exact zeros dropped.
235fn sparse_storage(entries: Vec<((u8, u8), Complex64)>) -> PtmStorage {
236    let (positions, amplitudes): (Vec<(u8, u8)>, Vec<Complex64>) = entries
237        .into_iter()
238        .filter(|&(_, amplitude)| amplitude != ZERO)
239        .unzip();
240    PtmStorage::Sparse {
241        positions: positions.into_boxed_slice(),
242        amplitudes: amplitudes.into_boxed_slice(),
243    }
244}

246/// Panics unless the two support qubits differ.
247fn assert_distinct(constructor: &str, q0: usize, q1: usize) {
248    assert!(
249        q0 != q1,
250        "PtmChannel::{constructor}: the two qubits must differ (both are {q0})"
251    );
252}
253
254/// Panics unless `value` lies in `[0, 1]`.
255fn assert_probability(constructor: &str, name: &str, value: f64) {
256    assert!(
257        (0.0..=1.0).contains(&value),
258        "PtmChannel::{constructor}: {name} must lie in [0, 1] (got {value})"
259    );
260}

262impl<const W: usize> Channel<W> for PtmChannel {
263    fn support(&self) -> [u64; W] {
264        support_mask(self.qubits())
265    }
266
267    fn debug_name(&self) -> &'static str {
268        self.name
269    }
270
271    fn apply(
272        &self,
273        input_x: &[u64; W],
274        input_z: &[u64; W],
275        coeff: Complex64,
276        out: &mut OutputBuffer<W>,
277    ) {
278        self.apply_rows(false, input_x, input_z, coeff, out);
279    }
280
281    fn apply_adjoint(
282        &self,
283        input_x: &[u64; W],
284        input_z: &[u64; W],
285        coeff: Complex64,
286        out: &mut OutputBuffer<W>,
287    ) {
288        self.apply_rows(true, input_x, input_z, coeff, out);
289    }
290
291    /// Tabulated straight from the stored PTM, without probing `apply`.
292    fn prepare_for_layer(
293        &self,
294        hash: &Gf2Hash<W>,
295        adjoint: bool,
296        _: PrepareToken,
297    ) -> Option<Prepared<W>> {
298        Some(Prepared::from_transfer_matrix(
299            &self.support(),
300            &self.transfer_matrix(adjoint),
301            hash,
302        ))
303    }
304}
