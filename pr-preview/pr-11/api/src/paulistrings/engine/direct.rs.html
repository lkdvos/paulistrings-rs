1//! The direct-apply small-sum layer path: a hash map instead of the bucketed sort-merge pipeline, for sums small enough that the pipeline's per-layer fixed cost dominates.
2//!
3//! This is **not** the canonical engine. [`bucketed`](super::bucketed) is, at every term count, and it is what [`propagate`](crate::propagate) uses unless the caller opts into [`EngineSelection::Auto`](super::EngineSelection).
4//! See `research/FINDINGS.md` for the threshold's justification.
5//!
6//! # Why a second path exists at all
7//!
8//! The bucketed layer has a per-layer fixed cost that does not shrink with the term count — dominated by `Channel::prepare`'s PTM derivation — so at small enough resident term counts that cost is most of the layer.
9//! This path removes it: no `prepare` (no PTM derivation, no delta plan), no rebucket, no coset span, no permute/unpermute, no sort.
10//! One [`Channel::apply`] call per resident term, straight into an [`FxBuildHasher`] map keyed by the Pauli string, exactly as `test_support::naive_apply_layer` does — that oracle *is* this algorithm, which is why the differential tests here are a real check and not a tautology.
11//!
12//! # The representation is the map, between layers too
13//!
14//! The win requires that consecutive small layers do **not** round-trip through a [`PauliSum`]: materializing one costs a sort of the whole sum, which is more than the fixed cost being saved.
15//! So a [`DirectSum`] holds the terms in its map across layers and materializes exactly once — when the sum outgrows the threshold, or when the propagation ends.
16//!
17//! Two consequences, both deliberate:
18//!
19//! - A [`TruncationPolicy`]'s `keep_term` is applied per layer, on summed coefficients, in the same place the merge applies it.
20//!   Its `finalize_layer` needs a real `PauliSum`, so it costs a materialize → finalize → re-ingest round trip; [`TruncationPolicy::finalizes_layer`] is how a policy says that round trip is pointless.
21//! - This path needs only [`Channel::apply`], never `Channel::prepare`, so it applies channels of **any** support width — including the > 2-qubit channels that make the bucketed path panic.
22//!   It is a strictly wider fallback, not a narrower fast path (see `super::propagate_with_options` for what that means for a circuit that mixes the two).
23//!
24//! GPU-readiness (ARCHITECTURE.md §GPU-Readiness) is the bucketed path's story and this path makes no claim on it: a hash map is not a device buffer, and at these term counts there is nothing to offload.

39/// A Pauli sum held as a hash map, for the direct-apply path.
40///
41/// Owns the map, one double-buffer for the layer's output, and the [`OutputBuffer`] columns a `Channel::apply` writes into.
42/// Capacity in all of them is retained across layers, so the steady state of a run of small layers allocates nothing.
43///
44/// The [`Gf2Hash`] is carried, not used: the map needs no partition, but [`Self::to_sum`] must hand back a `PauliSum` under the *same* hash rows and seed the caller entered with, and with at least as many bucket bits (the engine's partition is grow-only — see [`PauliSum::rebucket`]).
45pub(crate) struct DirectSum<const W: usize> {
46    /// The resident terms. Never holds an exact-zero coefficient: every layer filters them out, and an entering `PauliSum` cannot contain one.
47    live: HashMap<PauliString<W>, Complex64, FxBuildHasher>,
48    /// The layer's output, swapped into `live` at the end of the layer.
49    /// Output keys can collide with input keys not yet visited, so the accumulation cannot be done in place.
50    next: HashMap<PauliString<W>, Complex64, FxBuildHasher>,
51    /// `Channel::apply`'s output columns, sized to the widest `max_fanout` seen.
52    buf_x: Vec<[u64; W]>,
53    buf_z: Vec<[u64; W]>,
54    buf_c: Vec<Complex64>,
55    /// The partition to hand back under, at the bit count entered with.
56    hash: Gf2Hash<W>,
57    num_qubits: usize,
58}
59
60impl<const W: usize> DirectSum<W> {
61    /// Ingest a bucketed sum: one map insert per term, `O(n)`.
62    ///
63    /// Consumes the sum — its bucket columns are dead once the terms are in the map, and freeing them here keeps the direct path's footprint to the map.
64    pub(crate) fn from_sum(sum: PauliSum<W>) -> Self {
65        let n = sum.len();
66        let mut live = HashMap::with_capacity_and_hasher(n, FxBuildHasher);
67        for (x, z, c) in sum.iter() {
68            live.insert(PauliString::<W> { x: *x, z: *z }, c);
69        }
70        Self {
71            live,
72            next: HashMap::with_capacity_and_hasher(n, FxBuildHasher),
73            buf_x: Vec::new(),
74            buf_z: Vec::new(),
75            buf_c: Vec::new(),
76            hash: sum.hash().clone(),
77            num_qubits: sum.num_qubits(),
78        }
79    }
80
81    /// Resident term count — the same quantity [`PauliSum::len`] reports, so the engine's per-layer term counts and `TermTrace` are unaffected by which path produced them.
82    pub(crate) fn len(&self) -> usize {
83        self.live.len()
84    }
85
86    /// Apply one channel layer in place.
87    ///
88    /// `Channel::apply` (or `apply_adjoint`) once per resident term into the fanout-sized buffer, accumulate every emitted row into the output map, then one filtering pass: drop exact-zero sums and apply [`TruncationPolicy::keep_term`] to the *summed* coefficient.
89    /// That order is the merge phase's order (`engine::merge`), not a variant of it — a row whose coefficient is an exact zero still participates in its key's sum, since pre-filtering it would flip the sign of a zero sum.
90    ///
91    /// Equal-key contributions are summed in map iteration order, which is unspecified, so results agree with the bucketed path to floating-point tolerance and not bitwise (ARCHITECTURE.md §Determinism).
92    pub(crate) fn apply_layer<T>(&mut self, ch: &dyn Channel<W>, policy: &T, adjoint: bool)
93    where
94        T: TruncationPolicy<W> + ?Sized,
95    {
96        let Self {
97            live,
98            next,
99            buf_x,
100            buf_z,
101            buf_c,
102            ..
103        } = self;
104
105        let fanout = ch.max_fanout().max(1);
106        if buf_x.len() < fanout {
107            buf_x.resize(fanout, [0u64; W]);
108            buf_z.resize(fanout, [0u64; W]);
109            buf_c.resize(fanout, ZERO);
110        }
111
112        next.clear();
113        next.reserve(live.len());
114
115        for (p, &c) in live.iter() {
116            let mut len = 0usize;
117            {
118                let mut out = OutputBuffer::<W> {
119                    x: buf_x,
120                    z: buf_z,
121                    coeff: buf_c,
122                    len: &mut len,
123                };
124                if adjoint {
125                    ch.apply_adjoint(&p.x, &p.z, c, &mut out);
126                } else {
127                    ch.apply(&p.x, &p.z, c, &mut out);
128                }
129            }
130            for i in 0..len {
131                let key = PauliString::<W> {
132                    x: buf_x[i],
133                    z: buf_z[i],
134                };
135                *next.entry(key).or_insert(ZERO) += buf_c[i];
136            }
137        }
138
139        next.retain(|p, c| *c != ZERO && policy.keep_term(&p.x, &p.z, *c));
140        std::mem::swap(live, next);
141    }
142
143    /// Materialize a bucketed [`PauliSum`], leaving the map intact.
144    ///
145    /// One sort by key plus the key-sorted scatter.
146    /// The partition is the entering hash's rows and seed at `max(desired_bits(len), entering bits)` — the same clamp [`PauliSum::rebucket`] applies, so a sum handed back to the bucketed path is partitioned exactly as that path would have partitioned it, and the grow-only invariant on the bucket count survives the detour.
147    ///
148    /// Borrowing rather than consuming because the `finalize_layer` round trip needs the map back afterwards ([`Self::reload`]).
149    pub(crate) fn to_sum(&self) -> PauliSum<W> {
150        let mut entries: Vec<(PauliString<W>, Complex64)> =
151            self.live.iter().map(|(p, c)| (*p, *c)).collect();
152        entries.sort_unstable_by(|a, b| (&a.0.x, &a.0.z).cmp(&(&b.0.x, &b.0.z)));
153        let n = entries.len();
154        let mut x = Vec::with_capacity(n);
155        let mut z = Vec::with_capacity(n);
156        let mut coeff = Vec::with_capacity(n);
157        for (p, c) in entries {
158            x.push(p.x);
159            z.push(p.z);
160            coeff.push(c);
161        }
162        let bits =
163            desired_bits(n, DEFAULT_TARGET_BUCKET_LEN, DEFAULT_MIN_BUCKETS).max(self.hash.bits());
164        let hash = if bits == self.hash.bits() {
165            self.hash.clone()
166        } else {
167            Gf2Hash::new(self.num_qubits, bits, self.hash.seed())
168        };
169        PauliSum::from_key_sorted(&x, &z, &coeff, hash, self.num_qubits)
170    }
171
172    /// Re-ingest a sum that left this path for one operation and came back — the materialize → [`TruncationPolicy::finalize_layer`] → re-ingest round trip.
173    /// Retains the map's capacity; the carried hash is left as it was (`to_sum` only ever grows it, and `rebucket` would too).
174    pub(crate) fn reload(&mut self, sum: &PauliSum<W>) {
175        self.live.clear();
176        self.live.reserve(sum.len());
177        for (x, z, c) in sum.iter() {
178            self.live.insert(PauliString::<W> { x: *x, z: *z }, c);
179        }
180    }
181}

183/// Run the leading layers of `circuit` on the direct path, returning the materialized sum and **how many layers were applied** — the `k` the caller's sorting loop resumes at.
184///
185/// Stops after the layer that leaves the sum above [`PropagateOptions::small_sum_threshold`], or when the circuit runs out.
186/// The caller has already decided this path applies ([`PropagateOptions::starts_direct`]); nothing here re-decides, and nothing here can hand control back mid-circuit.
187///
188/// `#[inline(never)]` on purpose: it must not land inside `propagate_with_scratch_and_options`'s body, whose layer loop inlines `apply_layer_bucketed` and its merge kernels, which are sensitive to a few bytes of code motion (CLAUDE.md §Performance discipline).
189/// The default `SortedOnly` path must be able to reach this function's call site and not its code.
190///
191/// # Per-layer records
192///
193/// The `TermTrace`/`GateTrace` pushes and the `DEBUG` progress line are emitted here in the same order, with the same fields and the same format string as the sorting loop's epilogue, because downstream tooling parses them: the cross-engine head-to-head driver reads per-layer `terms_in -> terms_out` counts out of exactly these `DEBUG` records to gate term-count parity against PauliPropagation.jl.
194#[inline(never)]
195pub(crate) fn run_direct_prefix<const W: usize, T>(
196    circuit: &crate::circuit::Circuit<W>,
197    sum: PauliSum<W>,
198    policy: &T,
199    direction: super::Direction,
200    scratch: &mut super::LayerScratch<W>,
201    options: super::PropagateOptions,
202) -> (PauliSum<W>, usize)
203where
204    T: TruncationPolicy<W> + ?Sized,
205{
206    let n = circuit.channels.len();
207    let adjoint = matches!(direction, super::Direction::Heisenberg);
208    let tracing = scratch.term_trace.is_some();
209    let gate_tracing = scratch.gate_trace.is_some();
210    // Read once: a policy cannot change its answer mid-circuit, and the branch it guards costs a materialize plus a re-ingest.
211    let finalizes = policy.finalizes_layer();
212
213    let mut direct = DirectSum::from_sum(sum);
214    let mut applied = 0usize;
215
216    while applied < n {
217        let idx = match direction {
218            super::Direction::Forward => applied,
219            super::Direction::Heisenberg => n - 1 - applied,
220        };
221        let ch: &dyn Channel<W> = circuit.channels[idx].as_ref();
222        let application_index = applied;
223
224        let debug_on = log::log_enabled!(target: super::LOG_TARGET, log::Level::Debug);
225        let want_timer = gate_tracing || debug_on;
226        let layer_t0 = want_timer.then(std::time::Instant::now);
227        let terms_before = direct.len();
228
229        direct.apply_layer(ch, policy, adjoint);
230
231        // A layer pass needs a real `PauliSum`, so it costs a round trip.
232        // The policy machinery itself is untouched — this is the same `finalize_layer` call the sorting loop makes, on the same type, at the same point in the layer.
233        if finalizes {
234            let mut materialized = direct.to_sum();
235            policy.finalize_layer(&mut materialized);
236            direct.reload(&materialized);
237        }
238
239        let terms_after = direct.len();
240        applied += 1;
241
242        if tracing {
243            super::record_layer_terms(scratch, terms_before, terms_after);
244        }
245        if want_timer {
246            let dt = layer_t0
247                .expect("want_timer implies layer_t0 is Some")
248                .elapsed();
249            if gate_tracing {
250                super::record_gate_trace(
251                    scratch,
252                    idx as u32,
253                    application_index as u32,
254                    ch.debug_name(),
255                    terms_before,
256                    terms_after,
257                    dt,
258                );
259            }
260            if debug_on {
261                log::debug!(
262                    target: super::LOG_TARGET,
263                    "layer {}/{} [{}]: {} -> {} terms, {:.1} ms",
264                    applied,
265                    n,
266                    ch.debug_name(),
267                    terms_before,
268                    terms_after,
269                    dt.as_secs_f64() * 1e3,
270                );
271            }
272        }
273
274        if terms_after > options.small_sum_threshold {
275            break;
276        }
277    }
278
279    (direct.to_sum(), applied)
280}

348    /// `keep_term` sees the *summed* coefficient, so two rows that cancel to below a threshold are dropped as one term and not kept as two.
349    #[test]
350    fn keep_term_sees_summed_coefficients() {
351        struct Above(f64);
352        impl<const W: usize> TruncationPolicy<W> for Above {
353            fn keep_term(&self, _x: &[u64; W], _z: &[u64; W], c: Complex64) -> bool {
354                c.norm() > self.0
355            }
356            fn finalizes_layer(&self) -> bool {
357                false
358            }
359        }
360
361        // A π/2 Z-rotation on X₀ emits cos(π/2)·X₀ (an exact-ish zero) plus sin(π/2)·Y₀.
362        // Seeding both X₀ and Y₀ makes the Y row a two-row sum.
363        let mut acc = BuildAccumulator::<1>::new(1);
364        acc.add_term(PauliString::<1>::x(0), Phase::ONE, Complex64::new(1.0, 0.0));
365        acc.add_term(
366            PauliString::<1>::y(0),
367            Phase::ONE,
368            Complex64::new(-1.0, 0.0),
369        );
370        let sum = acc.finalize();
371        let ch = PauliRotation::new(PauliString::<1>::z(0), std::f64::consts::FRAC_PI_2);
372
373        let loose = apply_one(sum.clone(), &ch, &Above(1e-12), false);
374        let strict = apply_one(sum, &ch, &Above(0.5), false);
375        // Y₀'s two contributions (+1 from X₀'s sin, −1·cos ≈ 0 from Y₀) sum to ≈ 1, X₀'s to ≈ −1: both survive the loose threshold.
376        assert_eq!(loose.len(), 2);
377        // The strict threshold keeps both too — but only because the sums are ≈ 1, which is the point: a per-row filter would have dropped the ≈ 0 rows and changed the sums.
378        assert_eq!(strict.len(), 2);
379        let y = strict.get(&[1], &[1]).expect("Y term");
380        assert!((y.norm() - 1.0).abs() < 1e-12, "Y coeff {y}");
381    }

383    /// The differential oracle, over the channel zoo at both widths. The oracle shares this path's algorithm, so this pins the plumbing (buffer sizing, adjoint dispatch, zero-drop, re-materialization) rather than the algebra; the cross-path property tests in `tests/small_sum_path.rs` are what pin the algebra against the bucketed engine.
384    fn differential<const W: usize>(num_qubits: usize, seed: u64) {
385        let sum = rand_sum::<W>(300, num_qubits, seed);
386        let hi = num_qubits.saturating_sub(1) as u32;
387
388        let mut matrix = [[Complex64::new(0.0, 0.0); 4]; 4];
389        for (r, row) in matrix.iter_mut().enumerate() {
390            row[r] = Complex64::new(1.0, 0.0);
391        }
392        // A real 4×4 rotation in the (0,1) block: unitary, and dense enough in the PTM to fan a term out widely.
393        let (c, s) = (0.6f64, 0.8f64);
394        matrix[0][0] = Complex64::new(c, 0.0);
395        matrix[0][1] = Complex64::new(-s, 0.0);
396        matrix[1][0] = Complex64::new(s, 0.0);
397        matrix[1][1] = Complex64::new(c, 0.0);
398
399        let channels: Vec<Box<dyn Channel<W>>> = vec![
400            Box::new(Clifford1Q::h(0)),
401            Box::new(Clifford1Q::s(hi)),
402            Box::new(Clifford2Q::cnot(0, hi)),
403            Box::new(PauliRotation::new(
404                PauliString::<W>::z(0),
405                std::f64::consts::FRAC_PI_8,
406            )),
407            // Generator weight 4 — above MAX_LOCAL_SUPPORT, the case that makes `Prepared::derive_local` bail and `PauliRotation` override `prepare`.
408            Box::new(PauliRotation::new(
409                {
410                    let mut g = PauliString::<W>::z(0);
411                    g.x[0] |= 0b0110;
412                    g.z[0] |= 0b1000;
413                    g
414                },
415                0.37,
416            )),
417            Box::new(Depolarizing {
418                support: [0],
419                p: 0.15,
420            }),
421            Box::new(AmplitudeDamping {
422                support: [1],
423                gamma: 0.25,
424            }),
425            Box::new(GeneralUnitary2Q::from_matrix(0, 1, matrix)),
426        ];
427
428        for ch in &channels {
429            for adjoint in [false, true] {
430                let got = apply_one(sum.clone(), ch.as_ref(), &KeepAll, adjoint);
431                let want = naive_apply_layer(&sum, ch.as_ref(), &KeepAll, adjoint);
432                assert_terms_close(&got, &want, 1e-12, "direct vs naive");
433                got.assert_invariants();
434            }
435        }
436    }

448    /// A channel with support on three qubits: `Channel::prepare` declines it and the bucketed path panics, but this path only ever calls `apply`, so it applies it correctly.
449    /// Documented capability, tested here so it cannot regress into a silent wrong answer.
450    #[test]
451    fn applies_a_channel_wider_than_the_bucketed_path_can_prepare() {
452        struct RotateThree;
453        impl<const W: usize> Channel<W> for RotateThree {
454            fn max_fanout(&self) -> usize {
455                1
456            }
457            fn support(&self) -> [u64; W] {
458                support_mask(&[0, 1, 2])
459            }
460            /// Cyclically shifts the x-bits of qubits 0,1,2 — support-bounded, key-changing, and not expressible as a ≤ 2-qubit PTM.
461            fn apply(
462                &self,
463                input_x: &[u64; W],
464                input_z: &[u64; W],
465                coeff: Complex64,
466                out: &mut OutputBuffer<'_, W>,
467            ) {
468                let mut x = *input_x;
469                let low = input_x[0] & 0b111;
470                x[0] = (input_x[0] & !0b111) | ((low << 1) & 0b111) | (low >> 2);
471                out.push(x, *input_z, coeff);
472            }
473        }
474
475        let sum = rand_sum::<1>(64, 8, 0x3B);
476        let got = apply_one(sum.clone(), &RotateThree, &KeepAll, false);
477        let want = naive_apply_layer(&sum, &RotateThree, &KeepAll, false);
478        assert_terms_close(&got, &want, 1e-12, "wide-support direct layer");
479        assert_eq!(got.len(), sum.len());
480    }
