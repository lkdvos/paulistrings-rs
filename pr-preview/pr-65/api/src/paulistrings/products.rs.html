1//! Operator products accumulated into a destination under a [`TruncationPolicy`]: [`PauliSum::add_product`], [`PauliSum::add_liouvillian`] and [`PauliSum::add_anticommutator`], with `*_with` shorthands into a fresh sum.
2//! One call is one layer to the policy: `keep_term` sees each output key's summed coefficient, the destination's own share included, as its bucket is drained, and the layer pass runs once on the result.
3
4use crate::coefficient::Coefficient;
5use crate::pauli_sum::product::{Pairing, Weights};
6use crate::pauli_sum::storage::{DEFAULT_MIN_BUCKETS, DEFAULT_TARGET_BUCKET_LEN};
7use crate::pauli_sum::PauliSum;
8use crate::truncation::TruncationPolicy;
9
10impl<const W: usize, C: Coefficient> PauliSum<W, C> {
11    fn add_paired<P: TruncationPolicy<W, C> + ?Sized>(
12        &mut self,
13        generator: &Self,
14        operand: &Self,
15        pairing: Pairing,
16        weights: Weights<C>,
17        policy: &P,
18    ) {
19        self.pair_into(generator, operand, pairing, weights, &|x, z, c| policy.keep_term(x, z, c));
20        if policy.finalizes_layer() {
21            policy.finalize_layer(self);
22            self.rebucket(DEFAULT_TARGET_BUCKET_LEN, DEFAULT_MIN_BUCKETS);
23        }
24    }
25
26    /// `self ← alpha · generator · operand + beta · self`, truncated by `policy` as one layer; `beta = 0` ignores `self`'s terms.
27    ///
28    /// The pairs are summed straight into `self`'s buckets, one bucket at a time, and `self` ends partitioned under the larger operand's hash rows.
29    ///
30    /// # Panics
31    ///
32    /// If the three sums disagree about `num_qubits`, or if `C` is real: a product of anticommuting Hermitian strings carries an odd phase that no real coefficient holds.
33    pub fn add_product<P: TruncationPolicy<W, C> + ?Sized>(
34        &mut self,
35        generator: &Self,
36        operand: &Self,
37        alpha: C,
38        beta: C,
39        policy: &P,
40    ) {
41        assert!(
42            !C::IS_REAL,
43            "PauliSum::add_product: a product is not Hermitian, so a real coefficient type ({}) cannot hold it; cast to a complex type first",
44            C::NAME,
45        );
46        self.add_paired(generator, operand, Pairing::Product, Weights { alpha, beta }, policy);
47    }
48
49    /// `self ← alpha · i[generator, operand] + beta · self`, as [`Self::add_product`] accumulates; only anticommuting pairs whose supports meet are visited.
50    ///
51    /// # Panics
52    ///
53    /// If the three sums disagree about `num_qubits`.
54    pub fn add_liouvillian<P: TruncationPolicy<W, C> + ?Sized>(
55        &mut self,
56        generator: &Self,
57        operand: &Self,
58        alpha: C,
59        beta: C,
60        policy: &P,
61    ) {
62        self.add_paired(generator, operand, Pairing::Liouvillian, Weights { alpha, beta }, policy);
63    }
64
65    /// `self ← alpha · {generator, operand} + beta · self`, as [`Self::add_product`] accumulates.
66    ///
67    /// # Panics
68    ///
69    /// If the three sums disagree about `num_qubits`.
70    pub fn add_anticommutator<P: TruncationPolicy<W, C> + ?Sized>(
71        &mut self,
72        generator: &Self,
73        operand: &Self,
74        alpha: C,
75        beta: C,
76        policy: &P,
77    ) {
78        self.add_paired(
79            generator,
80            operand,
81            Pairing::Anticommutator,
82            Weights { alpha, beta },
83            policy,
84        );
85    }
86
87    /// `pairing(self, other)` under `policy` into an empty destination.
88    fn fresh<P: TruncationPolicy<W, C> + ?Sized>(
89        &self,
90        other: &Self,
91        pairing: Pairing,
92        policy: &P,
93    ) -> Self {
94        let mut out = Self::empty_with_hash(self.num_qubits(), other.hash().clone());
95        out.add_paired(self, other, pairing, Weights::fresh(), policy);
96        out
97    }
98
99    /// [`Self::product`], truncated by `policy` as one layer.
100    ///
101    /// # Panics
102    ///
103    /// As [`Self::add_product`].
104    pub fn product_with<P: TruncationPolicy<W, C> + ?Sized>(
105        &self,
106        other: &Self,
107        policy: &P,
108    ) -> Self {
109        let mut out = Self::empty_with_hash(self.num_qubits(), other.hash().clone());
110        out.add_product(self, other, C::ONE, C::ZERO, policy);
111        out
112    }
113
114    /// [`Self::liouvillian`], truncated by `policy` as one layer.
115    ///
116    /// # Panics
117    ///
118    /// As [`Self::add_liouvillian`].
119    pub fn liouvillian_with<P: TruncationPolicy<W, C> + ?Sized>(
120        &self,
121        other: &Self,
122        policy: &P,
123    ) -> Self {
124        self.fresh(other, Pairing::Liouvillian, policy)
125    }
126
127    /// [`Self::anticommutator`], truncated by `policy` as one layer.
128    ///
129    /// # Panics
130    ///
131    /// As [`Self::add_anticommutator`].
132    pub fn anticommutator_with<P: TruncationPolicy<W, C> + ?Sized>(
133        &self,
134        other: &Self,
135        policy: &P,
136    ) -> Self {
137        self.fresh(other, Pairing::Anticommutator, policy)
138    }
139}
