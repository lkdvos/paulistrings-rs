1//! The partitioned propagation driver: [`PartitionedSum`] and the [`propagate_partitioned`] front doors.
2//!
3//! A [`PartitionedSum`] is one [`PauliSum`] split across the partitions of a [`PartitionRuntime`]; the layer loop mirrors the unpartitioned one in [`engine`](crate::engine) — rebucket → prepare → layer → finalize, once per channel, on every partition in lock-step (ARCHITECTURE.md §Partitioning).
4//!
5//! Three things are collective and differ from the unpartitioned loop: the bucket count is agreed on a schedule rather than computed every layer (see [`BITS_AGREE_EVERY`]), a layer may exchange rows when [`PartitionPlan::has_remote`](super::plan::PartitionPlan::has_remote), and layer finalization goes through [`PartitionedTruncation::finalize_layer_partitioned`] rather than the unpartitioned policy directly.
6//!
7//! [`PropagateOptions`] is reused unchanged except that [`EngineSelection`](crate::EngineSelection) is ignored: the partitioned path is always the bucketed layer.
8

32/// `log` target for the partitioned engine's progress events — the same target
33/// the unpartitioned [`propagate`](crate::propagate) uses, so one filter

35const LOG_TARGET: &str = "paulistrings::propagate";

37/// Layers between two bucket-count agreements, and the length of the opening ramp that precedes them (ARCHITECTURE.md §Partitioning).
38///
39/// A partitioned layer that exchanges nothing has no other reason to communicate, so the bucket-count all-reduce is the whole cost of a local layer; agreeing every `BITS_AGREE_EVERY`-th layer amortizes that.
40///
41/// Two regimes justify the value. In steady state under a truncation policy the term count moves by a few percent per layer, so a lag of 16 layers is far short of the factor of two that would cost a bucket bit at all.
42/// A run starting from a small operator can double every layer for a while, so the first `BITS_AGREE_EVERY` layers of every call agree unconditionally rather than run the growth phase under-bucketed.
43///
44/// Between agreements a partition keeps the bucket count it has even if its own [`desired_bits`] is higher: nobody refines off-schedule, so the group's counts stay equal by construction and an exchange can always index a partner's blocks.
45/// The lag is bounded by `BITS_AGREE_EVERY` layers.
46pub const BITS_AGREE_EVERY: usize = 16;
47
48/// Whether layer `k` of a call agrees the bucket count with the group.
49///
50/// A pure function of the layer index — the same answer on every partition, with nothing exchanged to reach it.
51/// See [`BITS_AGREE_EVERY`] for the two terms.
52#[inline]
53fn agrees_bucket_bits(k: usize) -> bool {
54    k < BITS_AGREE_EVERY || k.is_multiple_of(BITS_AGREE_EVERY)
55}

57/// A [`Collectives`] view that counts the calls made through it.
58///
59/// Wrapped around the transport for the *policy's* collective finalization only, so the trace's `collectives` figure counts what a composite policy does inside `finalize_layer_partitioned` rather than guessing.
60/// The layer's own exchange never goes through here — it is point-to-point — so the extra indirection is one virtual call per collective.
61struct CountingCollectives<'a> {
62    inner: &'a dyn Collectives,
63    calls: &'a AtomicU32,
64}
65
66impl Collectives for CountingCollectives<'_> {
67    fn rank(&self) -> u32 {
68        self.inner.rank()
69    }
70    fn size(&self) -> u32 {
71        self.inner.size()
72    }
73    fn allreduce_max_u8(&self, v: u8) -> u8 {
74        self.calls.fetch_add(1, Ordering::Relaxed);
75        self.inner.allreduce_max_u8(v)
76    }
77    fn allreduce_sum_u64(&self, buf: &mut [u64]) {
78        self.calls.fetch_add(1, Ordering::Relaxed);
79        self.inner.allreduce_sum_u64(buf)
80    }
81    fn barrier(&self) {
82        self.calls.fetch_add(1, Ordering::Relaxed);
83        self.inner.barrier()
84    }
85}

111/// A [`PauliSum`] split across the partitions of a [`PartitionRuntime`] (ARCHITECTURE.md §Partitioning).
112///
113/// Held across calls: the split, the partition rows, the pools and the per-partition scratch all persist, so a driver stepping an observable through many Trotter steps scatters once and gathers once ([`PartitionedSum::propagate`] per step).
114///
115/// # Invariants
116///
117/// Between calls — and asserted by [`PartitionedSum::assert_invariants`] — every partition's sum satisfies [`PauliSum`]'s own invariants, holds only keys of its own partition, and shares one hash family and bucket count with its peers.
118///
119/// # Examples
120///
121/// ```
122/// use paulistrings::channel::Clifford1Q;
123/// use paulistrings::engine::partitioned::{
124///     PartitionConfig, PartitionRuntime, PartitionedSum, Placement,
125/// };
126/// use paulistrings::{
127///     BuildAccumulator, Circuit, Direction, PartitionedTruncation, PauliString, Phase,
128///     TruncationPolicy,
129/// };
130/// use num_complex::Complex64;
131///
132/// struct KeepAll;
133/// impl<const W: usize> TruncationPolicy<W> for KeepAll {
134///     // `finalizes_layer`'s default is the conservative `true`; the
135///     // `PartitionedTruncation` default body rejects that, since it cannot
136///     // know a layer pass it would be skipping is a no-op.
137///     fn finalizes_layer(&self) -> bool { false }
138/// }
139/// impl<const W: usize> PartitionedTruncation<W> for KeepAll {}
140///
141/// let config = PartitionConfig {
142///     placement: Placement::Unpinned { partitions: 2, threads_per_partition: Some(1) },
143///     bind_memory: false,
144///     partition_row_seed: None,
145/// };
146/// let runtime = PartitionRuntime::new(&config).expect("topology resolves");
147///
148/// let mut acc = BuildAccumulator::<1>::new(2);
149/// acc.add_term(PauliString::<1>::z(0), Phase::ONE, Complex64::new(1.0, 0.0));
150///
151/// let mut circuit = Circuit::<1>::new(2);
152/// circuit.push(Clifford1Q::h(0));
153///
154/// let mut split = PartitionedSum::scatter(acc.finalize(), runtime, &config);
155/// split.propagate(&circuit, &KeepAll, Direction::Heisenberg);
156/// // H conjugates Z to X, wherever the term happened to live.
157/// assert_eq!(split.gather().get(&[1], &[0]), Some(Complex64::new(1.0, 0.0)));
158/// ```
159pub struct PartitionedSum<const W: usize> {
160    /// One sum and its layer/export scratch per partition, in rank order.
161    parts: Vec<HostPartition<W>>,
162    /// The rows that decide which partition a key belongs to.
163    rows: PartitionRows<W>,
164    /// The placement and pools this sum runs on.
165    runtime: Arc<PartitionRuntime>,
166    /// The opt-in per-layer trace, `None` unless
167    /// [`enable_trace`](Self::enable_trace) was called.
168    trace: Option<PartitionTrace>,
169    /// Driver-level scatter time (the layer phases live in each partition's
170    /// own `LayerScratch`).
171    #[cfg(feature = "phase-timing")]
172    scatter_ns: u64,
173    /// Driver-level gather time. An atomic because [`gather`](Self::gather)
174    /// takes `&self`; nothing contends for it.
175    #[cfg(feature = "phase-timing")]
176    gather_ns: std::sync::atomic::AtomicU64,
177    /// Layers driven, summed over calls since the counters were drained.
178    #[cfg(feature = "phase-timing")]
179    layers: u64,
180}

182impl<const W: usize> PartitionedSum<W> {
183    /// Splits `sum` across `runtime`'s partitions, deriving the partition rows from `config`.
184    ///
185    /// The rows come from [`PartitionRows::from_seed`] with `config.partition_row_seed`, falling back to the sum's own hash seed — a different draw from the bucket hash's, so the two are independent with high probability.
186    ///
187    /// Each partition filters its own share **on its own pool**, so the columns are first-touched in the domain that will read them.
188    ///
189    /// # Panics
190    ///
191    /// If `runtime`'s partition count and the derived rows disagree (they cannot, both being `log2(P)` rows), and in debug builds if the partition rows are not independent of the sum's hash rows.
192    pub fn scatter(
193        sum: PauliSum<W>,
194        runtime: Arc<PartitionRuntime>,
195        config: &PartitionConfig,
196    ) -> Self {
197        let seed = config
198            .partition_row_seed
199            .unwrap_or_else(|| sum.hash().seed());
200        let rows = PartitionRows::<W>::from_seed(sum.num_qubits(), runtime.partition_bits(), seed);
201        Self::scatter_with_rows(sum, rows, runtime)
202    }
203
204    /// Splits `sum` across `runtime`'s partitions using caller-supplied rows.
205    ///
206    /// # Bucket counts
207    ///
208    /// Each partition takes the bucket count its *own* share wants ([`desired_bits`] on the default bucket policy, all-reduced to a maximum so the group agrees), but sheds at most `log2(P)` bits of the count the unpartitioned sum arrived with — so the bucket count summed over partitions is the one the sum already had, and at `P = 1` the scatter changes nothing at all.
209    /// The layer loop then re-normalizes upward against the caller's own [`PropagateOptions`].
210    ///
211    /// # Panics
212    ///
213    /// If `rows.num_partitions()` is not the runtime's partition count.
214    /// In debug builds, if the rows are not independent of the sum's hash rows — a partition row inside the hash's row space correlates partition with bucket, which costs load balance (not correctness).
215    /// The check is made here only: the hash gains rows as the sum grows, and independence from *future* rows cannot be checked up front; random rows stay independent with high probability.
216    pub fn scatter_with_rows(
217        sum: PauliSum<W>,
218        rows: PartitionRows<W>,
219        runtime: Arc<PartitionRuntime>,
220    ) -> Self {
221        let size = runtime.num_partitions();
222        assert_eq!(
223            rows.num_partitions(),
224            size,
225            "partition rows name {} partitions but the runtime has {size}",
226            rows.num_partitions(),
227        );
228        assert_eq!(
229            rows.num_qubits(),
230            sum.num_qubits(),
231            "partition rows are for {} qubits, the sum for {}",
232            rows.num_qubits(),
233            sum.num_qubits(),
234        );
235        debug_assert!(
236            rows.is_independent_of(sum.hash()),
237            "partition rows are dependent on the bucket hash rows — the split \
238             will correlate with the bucket partition and load-balance badly",
239        );
240
241        let started = Instant::now();
242        let locals = {
243            let sum = &sum;
244            let rows = &rows;
245            runtime.map_partitions((0..size).collect(), |rank, _, transport| {
246                scatter_local(sum, rows, rank as u32, transport)
247            })
248        };
249        log::info!(
250            target: LOG_TARGET,
251            "scatter: {} terms over {size} partitions, {} bucket bits, {:.3} s",
252            sum.len(),
253            locals[0].hash().bits(),
254            started.elapsed().as_secs_f64(),
255        );
256
257        Self {
258            parts: locals.into_iter().map(HostPartition::new).collect(),
259            rows,
260            runtime,
261            trace: None,
262            #[cfg(feature = "phase-timing")]
263            scatter_ns: started.elapsed().as_nanos() as u64,
264            #[cfg(feature = "phase-timing")]
265            gather_ns: std::sync::atomic::AtomicU64::new(0),
266            #[cfg(feature = "phase-timing")]
267            layers: 0,
268        }
269    }

271    /// Propagates through `circuit` under `policy`, in place, with [`PropagateOptions::default()`].
272    ///
273    /// See [`propagate_with_options`](Self::propagate_with_options) for the non-default knobs and for what the loop does per layer.
274    pub fn propagate<T>(&mut self, circuit: &Circuit<W>, policy: &T, direction: Direction)
275    where
276        T: PartitionedTruncation<W> + ?Sized,
277    {
278        self.propagate_with_options(circuit, policy, direction, PropagateOptions::default())
279    }

281    /// Propagates through `circuit` under `policy` with explicit [`PropagateOptions`].
282    ///
283    /// `direction` means what it means in [`propagate`](crate::propagate): [`Direction::Forward`] applies the channels in order, [`Direction::Heisenberg`] in reverse through [`Channel::apply_adjoint`](crate::Channel::apply_adjoint).
284    /// [`EngineSelection`](crate::EngineSelection) is ignored — the partitioned path is always the bucketed layer (see the module docs).
285    ///
286    /// # Progress logging
287    ///
288    /// Target `paulistrings::propagate`, as in the unpartitioned engine: one `INFO` line on entry and exit, on the calling thread, and one `DEBUG` line per layer **per partition**, tagged `partition r/P`.
289    /// Unlike [`propagate_with_scratch`](crate::propagate_with_scratch) the per-layer lines come from each partition's own driving thread between layers rather than the calling thread; every site is behind `log_enabled!`, so a disabled logger reads no clock.
290    ///
291    /// # Panics
292    ///
293    /// If a channel's [`Channel::prepare`] declines (support wider than `MAX_LOCAL_SUPPORT`), exactly as the unpartitioned engine does — there is no fallback path.
294    /// If `policy` reports [`finalizes_layer`](crate::TruncationPolicy::finalizes_layer) without overriding [`finalize_layer_partitioned`](PartitionedTruncation::finalize_layer_partitioned).
295    /// A panic in any partition is re-raised on the calling thread.
296    pub fn propagate_with_options<T>(
297        &mut self,
298        circuit: &Circuit<W>,
299        policy: &T,
300        direction: Direction,
301        options: PropagateOptions,
302    ) where
303        T: PartitionedTruncation<W> + ?Sized,
304    {
305        let n = circuit.channels.len();
306        let size = self.num_partitions();
307        let terms_in = self.len();
308        let started = Instant::now();
309        log::info!(
310            target: LOG_TARGET,
311            "propagate_partitioned: {terms_in} terms through {n} channels ({direction:?}) \
312             on {size} partitions [{}]",
313            self.runtime.placement_summary(),
314        );
315
316        if n > 0 {
317            // Hoisted out of every partition's layer loop: nothing inside one
318            // can turn tracing on or off.
319            let tracing = self.trace.is_some();
320            let items: Vec<PartitionWork<HostPartition<W>>> = self
321                .parts
322                .iter_mut()
323                .map(|part| PartitionWork::take(part, n, tracing))
324                .collect();
325
326            let runtime = Arc::clone(&self.runtime);
327            let rows = &self.rows;
328            let done = runtime.map_partitions(items, |rank, mut work, transport| {
329                let ctx = PartitionCtx {
330                    rows,
331                    rank,
332                    size,
333                    tracing,
334                };
335                run_layers(
336                    circuit, policy, direction, options, ctx, &mut work, transport,
337                );
338                work
339            });
340            let mut traced = Vec::with_capacity(if tracing { size } else { 0 });
341            for (rank, work) in done.into_iter().enumerate() {
342                self.parts[rank] = work.local;
343                if tracing {
344                    traced.push(work.rows);
345                }
346            }
347            if let Some(trace) = self.trace.as_mut() {
348                assemble(trace, traced);
349            }
350            #[cfg(feature = "phase-timing")]
351            {
352                self.layers += n as u64;
353            }
354        }
355
356        log::info!(
357            target: LOG_TARGET,
358            "propagate_partitioned: {n} layers applied, {terms_in} -> {} terms, {:.3} s",
359            self.len(),
360            started.elapsed().as_secs_f64(),
361        );
362    }

364    /// Merges the partitions back into one sum, leaving `self` intact.
365    ///
366    /// Runs on the calling thread; merging is `log2(P)` passes over the payload.
367    /// A caller that only needs a scalar should prefer [`len`](Self::len) or [`expectation_product_state`](Self::expectation_product_state), which read the partitions in place.
368    pub fn gather(&self) -> PauliSum<W> {
369        #[cfg(feature = "phase-timing")]
370        let started = Instant::now();
371        let out = PauliSum::merge_partitions(self.parts.iter().map(|p| p.sum.clone()).collect());
372        #[cfg(feature = "phase-timing")]
373        self.gather_ns.fetch_add(
374            started.elapsed().as_nanos() as u64,
375            std::sync::atomic::Ordering::Relaxed,
376        );
377        out
378    }
379
380    /// [`gather`](Self::gather) by value — no clone of the parts.
381    pub fn into_gathered(self) -> PauliSum<W> {
382        PauliSum::merge_partitions(self.parts.into_iter().map(|p| p.sum).collect())
383    }

450    /// Drain and return the per-layer records, or `None` if tracing was never enabled (`Some` iff tracing is on).
451    ///
452    /// Draining leaves tracing *enabled* with no records, so a sum reused across calls reports each call separately without re-enabling; records accumulate across layers and calls until drained.
453    pub fn take_trace(&mut self) -> Option<PartitionTrace> {
454        self.trace.as_mut().map(std::mem::take)

457    /// Drain and return the per-phase timing counters: one [`PhaseStats`] per partition, plus the driver's own scatter/gather time and layer count.
458    ///
459    /// Every counter is zeroed afterwards, so a probe can read one measured region at a time.
460    /// `gather_ns` covers [`Self::gather`] calls only — [`Self::into_gathered`] consumes the sum, and with it the counters.
461    #[cfg(feature = "phase-timing")]
462    pub fn take_stats(&mut self) -> PartitionPhaseStats {
463        PartitionPhaseStats {
464            per_partition: self
465                .parts
466                .iter_mut()
467                .map(|part| part.state.layer.take_stats())
468                .collect(),
469            scatter_ns: std::mem::take(&mut self.scatter_ns),
470            gather_ns: self.gather_ns.swap(0, std::sync::atomic::Ordering::Relaxed),
471            layers: std::mem::take(&mut self.layers),
472        }
473    }

475    /// Debug helper: checks every invariant a partitioned sum is supposed to
476    /// hold between calls.
477    ///
478    /// `O(terms)` — a full scan per partition — so it belongs in a test or an
479    /// assertion, not in a loop. The per-partition structural check is
480    /// `PauliSum`'s own `assert_invariants`, which exists only in test and
481    /// debug builds; in a release build this checks the cross-partition
482    /// properties alone.
483    ///
484    /// # Panics
485    ///
486    /// If a partition's sum is internally inconsistent, holds a key belonging
487    /// to another partition, or disagrees with partition 0 about the hash rows,
488    /// the bucket count or the qubit count.
489    pub fn assert_invariants(&self) {
490        assert_eq!(
491            self.parts.len(),
492            self.rows.num_partitions(),
493            "partition count disagrees with the rows",
494        );
495        let head = &self.parts[0].sum;
496        for (rank, local) in self.parts.iter().map(|p| &p.sum).enumerate() {
497            // `PauliSum::assert_invariants` itself exists only in test and
498            // debug builds; the cross-partition checks below are always
499            // compiled, so this method's signature does not change with the
500            // profile.
501            #[cfg(any(test, debug_assertions))]
502            local.assert_invariants();
503            assert_eq!(
504                local.num_qubits(),
505                head.num_qubits(),
506                "partition {rank}: qubit count differs from partition 0",
507            );
508            assert!(
509                local.hash().same_rows_as(head.hash()),
510                "partition {rank}: hash rows differ from partition 0",
511            );
512            assert_eq!(
513                local.hash().bits(),
514                head.hash().bits(),
515                "partition {rank}: bucket bits differ from partition 0",
516            );
517            let held = local.partition_rank_of_all(&self.rows);
518            assert!(
519                held.is_none() || held == Some(rank as u32),
520                "partition {rank} holds keys of partition {held:?}",
521            );
522        }
523    }

526/// The partitioned engine's phase breakdown: one [`PhaseStats`] per partition plus the driver's own counters.
527///
528/// Drained by [`PartitionedSum::take_stats`].
529/// Each partition's `PhaseStats` was measured on that partition's own driving thread and its own pool, so the wall-clock fields are **concurrent**, not additive: the partitions ran at the same time, and the spread between them is the group's load imbalance.
530/// `exchange_ns` in particular absorbs the wait for a partner, so a partition that finishes its own share early pays for the imbalance there.
531#[cfg(feature = "phase-timing")]
532#[derive(Clone, Debug, Default, PartialEq, Eq)]
533pub struct PartitionPhaseStats {
534    /// One breakdown per partition, in rank order.
535    pub per_partition: Vec<PhaseStats>,
536    /// Wall time of the scatter that built this sum (filter + coarsen, on the
537    /// partitions' own pools).
538    pub scatter_ns: u64,
539    /// Wall time summed over [`PartitionedSum::gather`] calls.
540    pub gather_ns: u64,
541    /// Layers driven since the counters were last drained.
542    pub layers: u64,
543}

545/// One partition's share of `sum`, at the bucket count the group agrees on — **the** scatter body, shared by the in-process and distributed drivers.
546///
547/// Runs on the partition's own pool (the caller is inside `install`), so every column is first-touched in the domain that will read it.
548/// One collective: the per-partition [`desired_bits`] maximum, which is what makes the group agree on a count before the first layer.
549pub(crate) fn scatter_local<const W: usize>(
550    sum: &PauliSum<W>,
551    rows: &PartitionRows<W>,
552    rank: u32,
553    coll: &dyn Collectives,
554) -> PauliSum<W> {
555    let mut local = sum.filter_partition(rows, rank);
556    let want = desired_bits(local.len(), DEFAULT_TARGET_BUCKET_LEN, DEFAULT_MIN_BUCKETS);
557    let want = coll.allreduce_max_u8(want);
558    local.coarsen_to(scatter_bits(local.hash().bits(), rows.bits(), want));
559    local
560}

562/// The bucket bits a partition takes on at scatter: at most `pbits` shed from the count the unpartitioned sum arrived with, and never below what this partition's own share wants.
563///
564/// `bits` is the incoming count, `pbits = log2(P)`, `want` the all-reduced per-partition [`desired_bits`].
565/// Shedding exactly `pbits` keeps the bucket count *summed over partitions* equal to the unpartitioned one; the `want` floor stops a sum that arrived under-bucketed from being coarsened at all.
566/// At `P = 1`, `pbits = 0`, so this is `bits` — the scatter is the identity and the partitioned run is the unpartitioned one.
567fn scatter_bits(bits: u8, pbits: u8, want: u8) -> u8 {
568    want.max(bits.saturating_sub(pbits)).min(bits)
569}

571/// What a partition knows about itself while it walks the layers: which keys are its own, where it sits in the group, and whether it is recording.
572///
573/// The two drivers fill this differently — `rank` and `size` come from `map_partitions` in one and from the transport in the other — and nothing in the loop below cares which.
574pub(crate) struct PartitionCtx<'a, const W: usize> {
575    /// The rows deciding which partition a key belongs to.
576    pub(crate) rows: &'a PartitionRows<W>,
577    /// This partition's index, for the per-layer log line.
578    pub(crate) rank: usize,
579    /// Partitions in the group, likewise.
580    pub(crate) size: usize,
581    /// Whether to append a [`PartitionLayerRow`] per layer. Hoisted out of the
582    /// loop: nothing inside one can turn tracing on or off.
583    pub(crate) tracing: bool,
584}

586/// [`Channel::prepare`] or the engine's one hard error.
587///
588/// Called twice on a layer that refines (the bucket count is settled between the two), so the panic — which is the unpartitioned engine's, with the partition and layer named — lives here rather than inline.
589fn prepare_or_panic<const W: usize>(
590    ch: &dyn Channel<W>,
591    hash: &Gf2Hash<W>,
592    adjoint: bool,
593    rank: usize,
594    idx: usize,
595) -> crate::channel::prepared::Prepared<W> {
596    ch.prepare(hash, adjoint).unwrap_or_else(|| {
597        // Same hard error as the unpartitioned engine: no whole-sum fallback
598        // exists to absorb a channel the engine cannot tabulate.
599        let weight: u32 = ch.support().iter().map(|w| w.count_ones()).sum();
600        panic!(
601            "partition {rank}, layer {idx}: Channel::prepare declined, so this channel \
602             cannot be propagated. The engine tabulates channels of support ≤ \
603             {MAX_LOCAL_SUPPORT} qubits (this one declares {weight}), and a channel must \
604             not write outside its declared support. See \
605             research/FINDINGS.md",
606        )
607    })
608}

610/// One partition's whole layer loop — **the** layer loop, shared by the in-process driver ([`PartitionedSum`]) and the distributed one ([`DistributedSum`](super::DistributedSum)).
611///
612/// Runs on the partition's driving thread inside its own pool, in lock-step with its peers: the same channels in the same order, the same collectives per layer.
613/// The two drivers differ only in *what a partition is* — a NUMA domain and an in-process endpoint, or a whole process and an MPI rank — which is entirely the transport's business, so the body below is generic over it and there is exactly one copy of the per-layer sequence.
614/// Every touch of the partition's storage goes through [`PartitionStorage`] and [`PartitionBackend`], so the loop is generic over where the partition lives as well.
615pub(crate) fn run_layers<const W: usize, T, X, B>(
616    circuit: &Circuit<W>,
617    policy: &T,
618    direction: Direction,
619    options: PropagateOptions,
620    ctx: PartitionCtx<'_, W>,
621    work: &mut PartitionWork<B>,
622    transport: &X,
623) where
624    T: PartitionedTruncation<W> + ?Sized,
625    X: Transport,
626    B: PartitionBackend<W, T>,
627{
628    let PartitionCtx {
629        rows,
630        rank,
631        size,
632        tracing,
633    } = ctx;
634    let n = circuit.channels.len();
635    let adjoint = matches!(direction, Direction::Heisenberg);
636    let size32 = transport.size();
637    // Both hoisted out of the loop: neither can change inside one, and `finalizes_layer` is a property of the policy *type*, hence the same answer on every partition — which is what makes skipping the call collective-safe.
638    let finalizes = policy.finalizes_layer();
639    let policy_calls = AtomicU32::new(0);
640    let local = &mut work.local;
641
642    for k in 0..n {
643        let idx = match direction {
644            Direction::Forward => k,
645            Direction::Heisenberg => n - 1 - k,
646        };
647        let ch: &dyn Channel<W> = circuit.channels[idx].as_ref();
648
649        // As in the unpartitioned engine: the per-layer DEBUG log and the opt-in gate trace share one clock read.
650        let debug_on = log::log_enabled!(target: LOG_TARGET, log::Level::Debug);
651        let want_timer = tracing || debug_on;
652        let layer_t0 = want_timer.then(Instant::now);
653        let terms_before = local.len();
654
655        #[cfg(feature = "phase-timing")]
656        let mut st = Stamp::now();
657        #[cfg(feature = "phase-timing")]
658        {
659            let stats = local.stats();
660            stats.layers += 1;
661            stats.terms_in += terms_before as u64;
662        }
663
664        // Prepared *before* the bucket count is settled, because the plan is what says whether this layer needs a collective at all: a delta's remoteness is a property of its mask and the partition rows, neither of which the bucket count touches.
665        // Only `bucket_delta` depends on it, so the prepared form is re-derived below on the rare layer that actually refines.
666        let mut prep = prepare_or_panic(ch, local.hash(), adjoint, rank, idx);
667        let mut plan = PartitionPlan::new(&prep, rows, rank as u32);
668        #[cfg(feature = "phase-timing")]
669        st.lap(&mut local.stats().prepare_ns);
670
671        // The bucket count, on the BITS_AGREE_EVERY schedule: on any layer that exchanges (both sides index the blocks by it), and otherwise every `BITS_AGREE_EVERY`-th.
672        // At `P = 1` there is no group, so the local answer is the agreed one and the loop rebuckets every layer exactly as `propagate` does.
673        let mut collectives = 0u32;
674        let solo = size32 == 1;
675        if solo || plan.has_remote() || agrees_bucket_bits(k) {
676            let mut want =
677                local.proposed_bits(&prep, options.target_bucket_len, options.min_buckets);
678            if !solo {
679                want = transport.allreduce_max_u8(want);
680                collectives += 1;
681            }
682            #[cfg(feature = "phase-timing")]
683            st.lap(&mut local.stats().collective_ns);
684            if want > local.hash().bits() {
685                while local.hash().bits() < want {
686                    local.refine();
687                }
688                #[cfg(feature = "phase-timing")]
689                st.lap(&mut local.stats().rebucket_ns);
690                // The hash moved, so every `bucket_delta` in the prepared form did too.
691                // Rare — the count is grow-only and settles — and this is the only reason a layer prepares twice.
692                prep = prepare_or_panic(ch, local.hash(), adjoint, rank, idx);
693                plan = PartitionPlan::new(&prep, rows, rank as u32);
694                #[cfg(feature = "phase-timing")]
695                st.lap(&mut local.stats().prepare_ns);
696            }
697        }
698
699        // The layer times its own export, exchange and coset loop into the same `LayerScratch`.
700        let counts = local.apply_layer(&prep, &plan, rows, policy, transport);
701        #[cfg(feature = "phase-timing")]
702        st.rearm();
703
704        // Collective, so it runs on every partition or on none — and `finalizes_layer` is the same answer on all of them (see `PartitionedTruncation`).
705        // A policy with no layer pass costs nothing.
706        if finalizes {
707            policy_calls.store(0, Ordering::Relaxed);
708            let counting = CountingCollectives {
709                inner: transport,
710                calls: &policy_calls,
711            };
712            local.finalize_layer(policy, &counting);
713            collectives += policy_calls.load(Ordering::Relaxed);
714        }
715        #[cfg(feature = "phase-timing")]
716        {
717            st.lap(&mut local.stats().finalize_ns);
718            let terms_out = local.len() as u64;
719            local.stats().terms_out += terms_out;
720        }
721
722        // Opt-in trace, behind a hoisted flag and a `#[cold]` callee for the same reason as the unpartitioned engine's term trace: this loop inlines the bucketed layer, whose merge kernels are sensitive to code motion (CLAUDE.md §Performance discipline).
723        // The counts are moved, not copied — the exchange already allocated them.
724        let remote_deltas = counts.remote_deltas;
725        let rows_received = counts.rows_received;
726        let dt = layer_t0.map(|t0| t0.elapsed());
727        if tracing {
728            record_layer_row(
729                &mut work.rows,
730                local.hash().bits(),
731                collectives,
732                idx as u32,
733                k as u32,
734                ch.debug_name(),
735                terms_before,
736                local.len(),
737                counts,
738                dt.unwrap_or_default().as_nanos() as u64,
739            );
740        }
741
742        if debug_on {
743            if let Some(dt) = dt {
744                log::debug!(
745                    target: LOG_TARGET,
746                    "partition {}/{} layer {}/{} [{}]: {} -> {} terms, {} remote deltas, \
747                     {} rows in, {:.1} ms",
748                    rank,
749                    size,
750                    k + 1,
751                    n,
752                    ch.debug_name(),
753                    terms_before,
754                    local.len(),
755                    remote_deltas,
756                    rows_received,
757                    dt.as_secs_f64() * 1e3,
758                );
759            }
760        }
761    }
762}

764/// Propagates `sum` through `circuit` on a partitioned engine built from `config`, and gathers the result.
765///
766/// One-shot convenience: it builds a [`PartitionRuntime`], scatters, runs and gathers.
767/// A caller propagating repeatedly (a Trotter driver stepping an observable) should hold the runtime and a [`PartitionedSum`] instead, so the pools, the split and the scratch survive between calls.
768///
769/// # Errors
770///
771/// [`TopologyError`] if `config` cannot be resolved into slots or a pool cannot be built.
772/// Everything else is a panic, as in [`propagate`](crate::propagate).
773///
774/// # Examples
775///
776/// ```
777/// use paulistrings::channel::Clifford1Q;
778/// use paulistrings::engine::partitioned::{propagate_partitioned, PartitionConfig, Placement};
779/// use paulistrings::{
780///     BuildAccumulator, Circuit, Direction, PartitionedTruncation, PauliString, Phase,
781///     TruncationPolicy,
782/// };
783/// use num_complex::Complex64;
784///
785/// struct KeepAll;
786/// impl<const W: usize> TruncationPolicy<W> for KeepAll {
787///     // `finalizes_layer`'s default is the conservative `true`; the
788///     // `PartitionedTruncation` default body rejects that, since it cannot
789///     // know a layer pass it would be skipping is a no-op.
790///     fn finalizes_layer(&self) -> bool { false }
791/// }
792/// impl<const W: usize> PartitionedTruncation<W> for KeepAll {}
793///
794/// let mut acc = BuildAccumulator::<1>::new(2);
795/// acc.add_term(PauliString::<1>::z(0), Phase::ONE, Complex64::new(1.0, 0.0));
796/// let mut circuit = Circuit::<1>::new(2);
797/// circuit.push(Clifford1Q::h(0));
798///
799/// let config = PartitionConfig {
800///     placement: Placement::Unpinned { partitions: 2, threads_per_partition: Some(1) },
801///     bind_memory: false,
802///     partition_row_seed: None,
803/// };
804/// let out = propagate_partitioned(
805///     &circuit, acc.finalize(), &KeepAll, Direction::Heisenberg, &config,
806/// ).expect("topology resolves");
807/// assert_eq!(out.get(&[1], &[0]), Some(Complex64::new(1.0, 0.0)));
808/// ```
809pub fn propagate_partitioned<const W: usize, T>(
810    circuit: &Circuit<W>,
811    sum: PauliSum<W>,
812    policy: &T,
813    direction: Direction,
814    config: &PartitionConfig,
815) -> Result<PauliSum<W>, TopologyError>
816where
817    T: PartitionedTruncation<W> + ?Sized,
818{
819    propagate_partitioned_with_options(
820        circuit,
821        sum,
822        policy,
823        direction,
824        config,
825        PropagateOptions::default(),
826    )
827}
828
829/// [`propagate_partitioned`] with explicit [`PropagateOptions`].
830///
831/// [`EngineSelection`](crate::EngineSelection) is ignored; see
832/// [`PartitionedSum::propagate_with_options`].
833///
834/// # Errors
835///
836/// [`TopologyError`], as [`propagate_partitioned`].
837pub fn propagate_partitioned_with_options<const W: usize, T>(
838    circuit: &Circuit<W>,
839    sum: PauliSum<W>,
840    policy: &T,
841    direction: Direction,
842    config: &PartitionConfig,
843    options: PropagateOptions,
844) -> Result<PauliSum<W>, TopologyError>
845where
846    T: PartitionedTruncation<W> + ?Sized,
847{
848    let runtime = PartitionRuntime::new(config)?;
849    let mut split = PartitionedSum::scatter(sum, runtime, config);
850    split.propagate_with_options(circuit, policy, direction, options);
851    Ok(split.into_gathered())
852}

854#[cfg(test)]

858    /// `P = 1` sheds nothing, so a scatter cannot change the bucket count and the partitioned run starts exactly where `propagate` would.
859    #[test]
860    fn scatter_bits_is_the_identity_at_one_partition() {
861        for bits in 0u8..12 {
862            for want in 0u8..12 {
863                assert_eq!(scatter_bits(bits, 0, want), bits, "bits={bits} want={want}");
864            }
865        }
866    }
867
868    /// With `P` partitions the count sheds `log2(P)` bits, so the bucket count summed over partitions is the unpartitioned one — unless a partition's own share wants more.
869    #[test]
870    fn scatter_bits_sheds_at_most_log2_p() {
871        // Incoming 10 bits, 4 partitions each wanting 8: shed exactly 2.
872        assert_eq!(scatter_bits(10, 2, 8), 8);
873        // Wanting more than the split leaves: the want wins, capped by what is there (the layer loop grows past it).
874        assert_eq!(scatter_bits(10, 2, 9), 9);
875        assert_eq!(scatter_bits(10, 2, 12), 10);
876        // Wanting less than the split leaves: never shed more than log2(P).
877        assert_eq!(scatter_bits(10, 2, 3), 8);
878        // Fewer incoming bits than there are partitions: the floor saturates at a single bucket per partition, so the per-share want decides.
879        assert_eq!(scatter_bits(1, 2, 0), 0);
880        assert_eq!(scatter_bits(1, 2, 1), 1);
881        assert_eq!(scatter_bits(0, 4, 0), 0);
882    }
