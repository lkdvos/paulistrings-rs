8/// Merge the strictly ascending `partners` into the strictly ascending `bucket`, summing equal keys bucket-first and keeping each output iff it is nonzero and `keep_term` passes; returns whether it ran in place.
9///
10/// In place needs spare capacity for every partner that is a new key: with room for every partner the hits go uncounted and the merge starts at that bound, else they are counted; without room the merge runs into `merged`, which is swapped in, as the general path does.
11/// `dirty`, when given, lists (as the scan packs them) the only rows whose coefficients changed, every other row already passing `keep_term`; a bucket with few such rows and partners merges by [`merge_sparse_in_place`].
12pub(super) fn merge_partners<const W: usize, C: Coefficient, T>(
13    bucket: &mut BucketColumns<W, C>,
14    partners: ColumnsRef<'_, W, C>,
15    policy: &T,
16    merged: &mut BucketColumns<W, C>,
17    dirty: Option<&[u32]>,
18) -> bool
19where
20    T: TruncationPolicy<W, C> + ?Sized,
21{
22    let bound = bucket.len() + partners.coeff.len();
23    let spare = bucket.x.capacity().min(bucket.z.capacity()).min(bucket.coeff.capacity());
24    if let Some(dirty) = dirty {
25        let events = partners.coeff.len() + dirty.len();
26        if spare >= bound && events * SPARSE_MERGE_SEGMENT <= bucket.len() {
27            merge_sparse_in_place(bucket, partners, dirty, policy, bound);
28            return true;
29        }
30    }
31    let total = if spare >= bound { bound } else { bound - count_hits(bucket.view(), partners) };
32    if spare >= total {
33        merge_backward_in_place(bucket, partners, policy, total);
34        true
35    } else {
36        merged.clear();
37        merge_identity_and_rest(bucket.view(), partners, merged, policy);
38        std::mem::swap(bucket, merged);
39        merged.clear();
40        false
41    }
42}
43
44/// Fewest rows per partner or dirty row at which [`merge_sparse_in_place`] beats the per-row merge.
45const SPARSE_MERGE_SEGMENT: usize = 8;
46
47/// [`merge_backward_in_place`] for a bucket whose rows outside `dirty` keep their coefficients and already pass `keep_term`: the clean rows between consecutive events move up as blocks, and only partners and dirty rows take the per-row step; rows below the lowest event are not touched.
48pub(super) fn merge_sparse_in_place<const W: usize, C: Coefficient, T>(
49    bucket: &mut BucketColumns<W, C>,
50    partners: ColumnsRef<'_, W, C>,
51    dirty: &[u32],
52    policy: &T,
53    total: usize,
54) where
55    T: TruncationPolicy<W, C> + ?Sized,
56{
57    let ColumnsRef { x: p_x, z: p_z, coeff: p_c } = partners;
58    let (len, extra) = (bucket.len(), p_c.len());
59    debug_assert_eq!(total, len + extra);
60    debug_assert!(bucket.x.capacity() >= total);
61    debug_assert!(bucket.z.capacity() >= total);
62    debug_assert!(bucket.coeff.capacity() >= total);
63    debug_assert!(dirty.windows(2).all(|pair| pair[0] >> 4 < pair[1] >> 4));
64    let x = bucket.x.as_mut_ptr();
65    let z = bucket.z.as_mut_ptr();
66    let c = bucket.coeff.as_mut_ptr();
67    // As in `merge_backward_in_place`; `d` counts the unread dirty rows, each below `i`, and the gap `w − i` never falls below the unread partners `j`.
68    let (mut i, mut j, mut d, mut w) = (len, extra, dirty.len(), total);
69    // SAFETY: every read is below `i ≤ len`; every write and block destination lies in `[i, total)`, and `ptr::copy` allows the overlap of a block with its destination.
70    unsafe {
71        let key = |row: usize| (*x.add(row), *z.add(row));
72        let emit = |w: &mut usize, key_x: [u64; W], key_z: [u64; W], sum: C| {
73            if sum != C::ZERO && policy.keep_term(&key_x, &key_z, sum) {
74                *w -= 1;
75                x.add(*w).write(key_x);
76                z.add(*w).write(key_z);
77                c.add(*w).write(sum);
78            }
79        };
80        let dirty_row = |d: usize| (dirty[d - 1] >> 4) as usize;
81        while j > 0 || d > 0 {
82            // Rows above the next dirty row and above the next partner's key are clean and already ordered.
83            let mut low = if d > 0 { dirty_row(d) + 1 } else { 0 };
84            if j > 0 {
85                let target = (p_x[j - 1], p_z[j - 1]);
86                let mut high = i;
87                while low < high {
88                    let middle = low + (high - low) / 2;
89                    if key(middle) <= target {
90                        low = middle + 1;
91                    } else {
92                        high = middle;
93                    }
94                }
95            }
96            let block = i - low;
97            if block > 0 && w != i {
98                std::ptr::copy(x.add(low), x.add(w - block), block);
99                std::ptr::copy(z.add(low), z.add(w - block), block);
100                std::ptr::copy(c.add(low), c.add(w - block), block);
101            }
102            w -= block;
103            i = low;
104            let order = match (i > 0, j > 0) {
105                (true, true) => key(i - 1).cmp(&(p_x[j - 1], p_z[j - 1])),
106                (true, false) => std::cmp::Ordering::Greater,
107                _ => std::cmp::Ordering::Less,
108            };
109            match order {
110                std::cmp::Ordering::Less => {
111                    j -= 1;
112                    emit(&mut w, p_x[j], p_z[j], p_c[j]);
113                }
114                std::cmp::Ordering::Equal => {
115                    i -= 1;
116                    j -= 1;
117                    d -= usize::from(d > 0 && dirty_row(d) == i);
118                    let (key_x, key_z) = key(i);
119                    emit(&mut w, key_x, key_z, *c.add(i) + p_c[j]);
120                }
121                std::cmp::Ordering::Greater => {
122                    debug_assert!(d > 0 && dirty_row(d) == i - 1);
123                    i -= 1;
124                    d -= 1;
125                    let (key_x, key_z) = key(i);
126                    emit(&mut w, key_x, key_z, *c.add(i));
127                }
128            }
129        }
130        let out = total - w;
131        if w != i {
132            std::ptr::copy(x.add(w), x.add(i), out);
133            std::ptr::copy(z.add(w), z.add(i), out);
134            std::ptr::copy(c.add(w), c.add(i), out);
135        }
136        bucket.x.set_len(i + out);
137        bucket.z.set_len(i + out);
138        bucket.coeff.set_len(i + out);
139    }
140}

142/// Keys `partners` shares with `bucket`, both strictly ascending.
143fn count_hits<const W: usize, C: Coefficient>(
144    bucket: ColumnsRef<'_, W, C>,
145    partners: ColumnsRef<'_, W, C>,
146) -> usize {
147    let (len, extra) = (bucket.x.len(), partners.x.len());
148    let (mut i, mut j, mut hits) = (0usize, 0usize, 0usize);
149    while i < len && j < extra {
150        let order = (bucket.x[i], bucket.z[i]).cmp(&(partners.x[j], partners.z[j]));
151        // Branchless steps: which side advances is the merge's coin flip.
152        i += usize::from(order.is_le());
153        j += usize::from(order.is_ge());
154        hits += usize::from(order.is_eq());
155    }
156    hits
157}

159/// [`merge_partners`] in the bucket's own buffer: written from the back of `total`, at least the length before drops, then shifted to the front over the slots hits and drops left.
160pub(super) fn merge_backward_in_place<const W: usize, C: Coefficient, T>(
161    bucket: &mut BucketColumns<W, C>,
162    partners: ColumnsRef<'_, W, C>,
163    policy: &T,
164    total: usize,
165) where
166    T: TruncationPolicy<W, C> + ?Sized,
167{
168    let ColumnsRef { x: p_x, z: p_z, coeff: p_c } = partners;
169    let (len, extra) = (bucket.len(), p_c.len());
170    debug_assert!(total >= len + extra - count_hits(bucket.view(), partners));
171    debug_assert!(bucket.x.capacity() >= total);
172    debug_assert!(bucket.z.capacity() >= total);
173    debug_assert!(bucket.coeff.capacity() >= total);
174    debug_assert!(p_x.windows(2).zip(p_z.windows(2)).all(|(x, z)| (x[0], z[0]) < (x[1], z[1])));
175    let x = bucket.x.as_mut_ptr();
176    let z = bucket.z.as_mut_ptr();
177    let c = bucket.coeff.as_mut_ptr();
178    // `i` and `j` count the unread rows of each input, `w` the first written slot, and `h` the hits among the unread rows.
179    // A step writes at most one slot per output key it consumes, so `w ≥ i + j − h ≥ i`, since each unread hit pairs an unread partner: a write never lands on an unread bucket row.
180    let (mut i, mut j, mut w) = (len, extra, total);
181    // SAFETY: every read is below `i ≤ len`, the initialized prefix; every write is at `w - 1 ≥ i` and below `total ≤ capacity`.
182    unsafe {
183        let emit = |w: &mut usize, key_x: [u64; W], key_z: [u64; W], sum: C| {
184            if sum != C::ZERO && policy.keep_term(&key_x, &key_z, sum) {
185                *w -= 1;
186                x.add(*w).write(key_x);
187                z.add(*w).write(key_z);
188                c.add(*w).write(sum);
189            }
190        };
191        while i > 0 && j > 0 {
192            let (b_x, b_z) = (*x.add(i - 1), *z.add(i - 1));
193            let (q_x, q_z) = (p_x[j - 1], p_z[j - 1]);
194            match (b_x, b_z).cmp(&(q_x, q_z)) {
195                std::cmp::Ordering::Greater => {
196                    i -= 1;
197                    emit(&mut w, b_x, b_z, *c.add(i));
198                }
199                std::cmp::Ordering::Less => {
200                    j -= 1;
201                    emit(&mut w, q_x, q_z, p_c[j]);
202                }
203                std::cmp::Ordering::Equal => {
204                    i -= 1;
205                    j -= 1;
206                    emit(&mut w, b_x, b_z, *c.add(i) + p_c[j]);
207                }
208            }
209        }
210        while j > 0 {
211            j -= 1;
212            emit(&mut w, p_x[j], p_z[j], p_c[j]);
213        }
214        // Below the last write the bucket's rows may already sit in their slots, and are then not rewritten.
215        while i > 0 {
216            i -= 1;
217            let (key_x, key_z, sum) = (*x.add(i), *z.add(i), *c.add(i));
218            if sum != C::ZERO && policy.keep_term(&key_x, &key_z, sum) {
219                w -= 1;
220                if w != i {
221                    x.add(w).write(key_x);
222                    z.add(w).write(key_z);
223                    c.add(w).write(sum);
224                }
225            }
226        }
227        let out = total - w;
228        if w > 0 {
229            std::ptr::copy(x.add(w), x, out);
230            std::ptr::copy(z.add(w), z, out);
231            std::ptr::copy(c.add(w), c, out);
232        }
233        bucket.x.set_len(out);
234        bucket.z.set_len(out);
235        bucket.coeff.set_len(out);
236    }
