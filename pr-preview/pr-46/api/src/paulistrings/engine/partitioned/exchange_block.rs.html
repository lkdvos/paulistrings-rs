1//! The exchange wire unit: [`ExchangeBlock`], one per remote delta, and the [`PartnerPayload`] that carries a partner's blocks.
2
3use std::mem::size_of;
4
5use num_complex::Complex64;
6
7use crate::engine::transport::{ChunkMap, Payload};
8
9/// Fixed-size wire prefix of one [`ExchangeBlock`]: four `u32`s, no padding, so it casts to bytes.
10#[repr(C)]
11#[derive(Clone, Copy, Debug, Default, PartialEq, Eq, bytemuck::Pod, bytemuck::Zeroable)]
12pub(crate) struct BlockHeader {
13    /// Destination positions, the group's agreed bucket count; `offsets` has one more entry.
14    pub num_buckets: u32,
15    /// Live rows, `offsets[num_buckets]`.
16    pub rows: u32,
17    /// The width `W` the block was built at, checked on receive.
18    pub width: u32,
19    /// The layer plan's remote-delta index this block carries.
20    pub entry: u32,
21}
22
23/// The rows one remote delta moves to one partner, CSR-indexed by the receiver's destination position ([`CosetOrder`](crate::engine::coset::CosetOrder)).
24///
25/// The columns are grow-only and may be longer than `header.rows`, which alone says how much is live: read through `columns` or `segment`, never `x.len()`.
26#[derive(Clone, Debug, Default)]
27pub(crate) struct ExchangeBlock<const W: usize> {
28    pub header: BlockHeader,
29    /// CSR offsets by destination position, `num_buckets + 1` entries.
30    pub offsets: Vec<u32>,
31    pub x: Vec<[u64; W]>,
32    pub z: Vec<[u64; W]>,
33    pub coeff: Vec<Complex64>,
34}
35
36/// Compares the live rows only, so a reused block equals the same block built fresh.
37impl<const W: usize> PartialEq for ExchangeBlock<W> {
38    fn eq(&self, other: &Self) -> bool {
39        self.header == other.header
40            && self.offsets == other.offsets
41            && self.columns() == other.columns()
42    }
43}
44
45/// Everything one partner receives from this partition for one layer: one block per remote delta, ascending by [`BlockHeader::entry`].
46#[derive(Clone, Debug, Default, PartialEq)]
47pub(crate) struct PartnerPayload<const W: usize> {
48    pub blocks: Vec<ExchangeBlock<W>>,
49}
50
51impl<const W: usize> ExchangeBlock<W> {
52    /// `set_counts` on a fresh block.
53    #[cfg(test)]
54    pub fn with_counts(entry: u32, counts: &[u32]) -> Self {
55        let mut block = Self::default();
56        block.set_counts(entry, counts);
57        block
58    }
59
60    /// Re-aim the block at `counts[p]` rows per destination position under remote-delta index `entry`, growing but never shrinking or re-zeroing the columns.
61    pub(crate) fn set_counts(&mut self, entry: u32, counts: &[u32]) {
62        self.offsets.clear();
63        self.offsets.reserve(counts.len() + 1);
64        self.offsets.push(0u32);
65        let mut rows = 0u32;
66        for &count in counts {
67            rows = rows.checked_add(count).expect("exchange block exceeds u32::MAX rows");
68            self.offsets.push(rows);
69        }
70        self.header =
71            BlockHeader { num_buckets: counts.len() as u32, rows, width: W as u32, entry };
72        self.grow_columns(rows as usize);
73    }
74
75    fn grow_columns(&mut self, rows: usize) {
76        if self.coeff.len() < rows {
77            self.x.resize(rows, [0u64; W]);
78            self.z.resize(rows, [0u64; W]);
79            self.coeff.resize(rows, Complex64::new(0.0, 0.0));
80        }
81    }
82
83    /// The live rows of the three columns.
84    pub(crate) fn columns(&self) -> (&[[u64; W]], &[[u64; W]], &[Complex64]) {
85        let rows = self.rows();
86        (&self.x[..rows], &self.z[..rows], &self.coeff[..rows])
87    }
88
89    /// The rows destined for position `p`; output bucket `β′` reads `segment(order.position_of(β′))`.
90    pub(crate) fn segment(&self, p: u32) -> (&[[u64; W]], &[[u64; W]], &[Complex64]) {
91        let lo = self.offsets[p as usize] as usize;
92        let hi = self.offsets[p as usize + 1] as usize;
93        (&self.x[lo..hi], &self.z[lo..hi], &self.coeff[lo..hi])
94    }
95
96    /// Live rows.
97    pub(crate) fn rows(&self) -> usize {
98        self.header.rows as usize
99    }
100
101    /// Destination positions the block is indexed by.
102    #[cfg(any(test, debug_assertions))]
103    pub(crate) fn num_buckets(&self) -> u32 {
104        self.header.num_buckets
105    }
106
107    /// Wire footprint in bytes, exactly what `byte_parts` hands out.
108    pub(crate) fn bytes(&self) -> usize {
109        size_of::<BlockHeader>()
110            + self.offsets.len() * size_of::<u32>()
111            + 2 * self.rows() * W * size_of::<u64>()
112            + self.rows() * size_of::<Complex64>()
113    }
114}
115
116/// Parts per block on the wire: header, offsets, x, z, coeff.
117pub(super) const PARTS_PER_BLOCK: usize = 5;
118
119/// The CSR `offsets` read at `map`'s chunk boundaries, `chunks + 1` entries.
120pub(crate) fn chunk_rows_of(offsets: &[u32], map: &ChunkMap) -> Vec<usize> {
121    debug_assert_eq!(
122        offsets.len(),
123        map.positions() + 1,
124        "the chunk map is built for a different bucket count than the block",
125    );
126    (0..=map.chunks()).map(|k| offsets[map.bound(k) as usize] as usize).collect()
127}
128
129/// Cut `column` at `bounds` (row indices) and view each piece as bytes.
130fn chunk_slices<'s, T, F>(column: &'s [T], bounds: &[usize], as_bytes: F) -> Vec<&'s [u8]>
131where
132    F: Fn(&'s [T]) -> &'s [u8],
133{
134    bounds.windows(2).map(|window| as_bytes(&column[window[0]..window[1]])).collect()
135}
136
137/// [`chunk_slices`] for the receive side: disjoint mutable pieces, in order.
138fn chunk_slices_mut<'s, T, F>(
139    column: &'s mut [T],
140    bounds: &[usize],
141    as_bytes: F,
142) -> Vec<&'s mut [u8]>
143where
144    F: Fn(&'s mut [T]) -> &'s mut [u8],
145{
146    let mut rest = column;
147    let mut out = Vec::with_capacity(bounds.len().saturating_sub(1));
148    let mut at = bounds[0];
149    for &edge in &bounds[1..] {
150        let (head, tail) = rest.split_at_mut(edge - at);
151        out.push(as_bytes(head));
152        rest = tail;
153        at = edge;
154    }
155    out
156}

158/// Panic unless a declared part length is a whole number of `stride`-byte elements.
159pub(super) fn check_stride(len: usize, stride: usize, what: &str) {
160    assert_eq!(
161        len % stride,
162        0,
163        "exchange block {what}: {len} bytes is not a whole number of {stride}-byte entries",
164    );
165}

167impl<const W: usize> Payload for PartnerPayload<W> {
168    fn byte_parts(&self) -> Vec<&[u8]> {
169        let mut parts = Vec::with_capacity(PARTS_PER_BLOCK * self.blocks.len());
170        for block in &self.blocks {
171            let (x, z, coeff) = block.columns();
172            parts.push(bytemuck::bytes_of(&block.header));
173            parts.push(bytemuck::cast_slice(&block.offsets));
174            parts.push(bytemuck::cast_slice(x.as_flattened()));
175            parts.push(bytemuck::cast_slice(z.as_flattened()));
176            parts.push(bytemuck::cast_slice(coeff));
177        }
178        parts
179    }
180
181    fn receive_into(&mut self, lens: &[usize]) -> Vec<&mut [u8]> {
182        assert_eq!(
183            lens.len() % PARTS_PER_BLOCK,
184            0,
185            "partner payload: {} parts is not a whole number of {PARTS_PER_BLOCK}-part blocks",
186            lens.len(),
187        );
188        let block_count = lens.len() / PARTS_PER_BLOCK;
189        let key_stride = W * size_of::<u64>();
190        self.blocks.truncate(block_count);
191        if self.blocks.len() < block_count {
192            self.blocks.resize_with(block_count, ExchangeBlock::<W>::default);
193        }
194        let mut parts = Vec::with_capacity(lens.len());
195        for (block, lens) in self.blocks.iter_mut().zip(lens.chunks_exact(PARTS_PER_BLOCK)) {
196            assert_eq!(
197                lens[0],
198                size_of::<BlockHeader>(),
199                "partner payload: block header is {} bytes, expected {}",
200                lens[0],
201                size_of::<BlockHeader>(),
202            );
203            check_stride(lens[1], size_of::<u32>(), "offsets");
204            check_stride(lens[2], key_stride, "x column");
205            check_stride(lens[3], key_stride, "z column");
206            check_stride(lens[4], size_of::<Complex64>(), "coeff column");
207            assert_eq!(
208                lens[2], lens[3],
209                "partner payload: x column is {} bytes and z column {}",
210                lens[2], lens[3],
211            );
212            let rows = lens[2] / key_stride;
213            assert_eq!(
214                lens[4] / size_of::<Complex64>(),
215                rows,
216                "partner payload: coeff column carries {} rows where the keys carry {rows}",
217                lens[4] / size_of::<Complex64>(),
218            );
219            block.offsets.clear();
220            block.offsets.resize(lens[1] / size_of::<u32>(), 0);
221            block.grow_columns(rows);
222            // Overwritten by the arriving header, which `finish_receive` checks against the columns.
223            block.header.rows = rows as u32;
224            let ExchangeBlock { header, offsets, x, z, coeff } = block;
225            parts.push(bytemuck::bytes_of_mut(header));
226            parts.push(bytemuck::cast_slice_mut(&mut offsets[..]));
227            parts.push(bytemuck::cast_slice_mut(x[..rows].as_flattened_mut()));
228            parts.push(bytemuck::cast_slice_mut(z[..rows].as_flattened_mut()));
229            parts.push(bytemuck::cast_slice_mut(&mut coeff[..rows]));
230        }
231        parts
232    }
233
234    fn finish_receive(&mut self) {
235        for block in &self.blocks {
236            assert_eq!(
237                block.header.width as usize, W,
238                "partner payload: block encoded at width W={} decoded at W={W}",
239                block.header.width,
240            );
241            // A garbled header must fail here rather than as an out-of-bounds read.
242            assert!(
243                block.header.rows as usize <= block.coeff.len(),
244                "partner payload: a block header claims {} rows but only {} arrived",
245                block.header.rows,
246                block.coeff.len(),
247            );
248            assert_eq!(
249                block.offsets.len(),
250                block.header.num_buckets as usize + 1,
251                "partner payload: a block indexed by {} buckets arrived with {} offsets",
252                block.header.num_buckets,
253                block.offsets.len(),
254            );
255            assert_eq!(
256                block.offsets.last().copied().unwrap_or(0),
257                block.header.rows,
258                "partner payload: offsets end at {:?}, the columns carry {} rows",
259                block.offsets.last(),
260                block.header.rows,
261            );
262        }
263    }
264
265    /// The header and the CSR offsets of every block.
266    fn early_parts(&self) -> Vec<&[u8]> {
267        let mut parts = Vec::with_capacity(2 * self.blocks.len());
268        for block in &self.blocks {
269            parts.push(bytemuck::bytes_of(&block.header));
270            parts.push(bytemuck::cast_slice(&block.offsets));
271        }
272        parts
273    }
274
275    /// The three columns of each block, each cut at the chunk boundaries.
276    fn bulk_parts(&self, map: &ChunkMap) -> Vec<Vec<&[u8]>> {
277        let mut parts = Vec::with_capacity(3 * self.blocks.len());
278        for block in &self.blocks {
279            let rows = block.rows();
280            let (x, z, coeff) = (&block.x[..rows], &block.z[..rows], &block.coeff[..rows]);
281            let bounds = chunk_rows_of(&block.offsets, map);
282            parts.push(chunk_slices(x, &bounds, |s| bytemuck::cast_slice(s.as_flattened())));
283            parts.push(chunk_slices(z, &bounds, |s| bytemuck::cast_slice(s.as_flattened())));
284            parts.push(chunk_slices(coeff, &bounds, bytemuck::cast_slice));
285        }
286        parts
287    }
288
289    fn early_receive_into(&mut self, lens: &[usize]) -> Vec<&mut [u8]> {
290        let mut parts = self.receive_into(lens);
291        let mut early = Vec::with_capacity(2 * parts.len() / PARTS_PER_BLOCK);
292        for (i, view) in parts.drain(..).enumerate() {
293            if i % PARTS_PER_BLOCK < 2 {
294                early.push(view);
295            }
296        }
297        early
298    }
299
300    fn bulk_receive_into(&mut self, map: &ChunkMap) -> Vec<Vec<&mut [u8]>> {
301        let mut parts = Vec::with_capacity(3 * self.blocks.len());
302        for block in &mut self.blocks {
303            let rows = block.header.rows as usize;
304            let bounds = chunk_rows_of(&block.offsets, map);
305            let ExchangeBlock { x, z, coeff, .. } = block;
306            parts.push(chunk_slices_mut(&mut x[..rows], &bounds, |s| {
307                bytemuck::cast_slice_mut(s.as_flattened_mut())
308            }));
309            parts.push(chunk_slices_mut(&mut z[..rows], &bounds, |s| {
310                bytemuck::cast_slice_mut(s.as_flattened_mut())
311            }));
312            parts.push(chunk_slices_mut(&mut coeff[..rows], &bounds, bytemuck::cast_slice_mut));
313        }
314        parts
315    }
316}
