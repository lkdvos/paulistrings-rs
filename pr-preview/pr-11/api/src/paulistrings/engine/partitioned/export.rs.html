1//! The export pass: rows a layer generates that belong to another partition, gathered into one [`ExchangeBlock`] per remote delta.
2//!
3//! A remote delta (see [`PartitionPlan`]) moves every row it produces to one partner, so export is a pure per-delta gather with no per-term routing decision.
4//! Each block is CSR in the receiver's destination-coset order ([`ChunkMap`](super::transport::ChunkMap)), which is what makes a receiving coset's rows contiguous so the transfer can be cut into chunks.
5//! Two parallel passes over the local buckets: count rows per (remote delta, source bucket), then fill each block position by position.
6//! The row arithmetic itself is not reimplemented here: see [`DeltaEntry::emit`](crate::channel::prepared::DeltaEntry::emit) and [`RotationPrep::emit_gen`](crate::channel::prepared::RotationPrep::emit_gen).
7

17const ZERO: Complex64 = Complex64::new(0.0, 0.0);

19/// Rows below which a block's fill stays on one thread.
20///
21/// The fill splits a block's position range in two and hands the halves to `rayon::join`; this threshold is purely about not paying task overhead for a handful of rows.
22const FILL_PARALLEL_MIN_ROWS: usize = 4096;
23
24/// Reusable scratch for [`export_layer`], held by the caller across layers so a layer allocates nothing after the first.
25///
26/// [`Self::counts`] is the pass-1 count table in bucket-major order (`counts[β * K + k]` is remote delta `k`'s row count from source bucket `β`), the layout that lets pass 1 run as one disjoint `par_chunks_mut(K)` over buckets.
27/// [`Self::block_counts`] is that layout transposed into a per-block column for [`ExchangeBlock::set_counts`].
28#[derive(Debug)]
29pub(crate) struct ExportScratch<const W: usize> {
30    /// Pass-1 counts, bucket-major: `counts[β * K + k]`.
31    counts: Vec<u32>,
32    /// One block's counts by destination position, gathered out of
33    /// [`Self::counts`].
34    block_counts: Vec<u32>,
35    /// One block's source bucket per destination position:
36    /// `src_of[p] = map.bucket_at(p) ^ bd`. Built once per block and read by
37    /// both the count transpose and the fill.
38    src_of: Vec<u32>,
39    /// Payloads not currently in flight, with their block columns intact.
40    ///
41    /// The layer's own pool ([`Transport::exchange_layer`](super::transport::Transport::exchange_layer)): both outgoing and incoming payloads are drawn from here and returned when the layer is done with them, so a steady-state remote layer allocates nothing at all.
42    pub(crate) pool: Vec<PartnerPayload<W>>,
43}
44
45// Hand-written because `#[derive(Default)]` would demand `W: Default`.
46impl<const W: usize> Default for ExportScratch<W> {
47    fn default() -> Self {
48        Self {
49            counts: Vec::new(),
50            block_counts: Vec::new(),
51            src_of: Vec::new(),
52            pool: Vec::new(),
53        }
54    }
55}

57/// What one layer's export costs, by partner rank (length is the group size).
58///
59/// Rows and bytes are what this partition *sent*; a partner it sent nothing to has zeros.
60/// The bytes are the wire footprint ([`ExchangeBlock::bytes`]) of the blocks as they stand, so an in-process transport reports the same number an MPI one would move.
61#[derive(Clone, Debug, Default, PartialEq, Eq)]
62pub(crate) struct ExportCounts {
63    /// Rows sent to each partner.
64    pub rows_to: Vec<u64>,
65    /// Wire bytes sent to each partner.
66    pub bytes_to: Vec<u64>,
67}

69/// One remote delta's row-level emitter: the tabulated entry, or the wide rotation's generator pass.
70///
71/// Exists so the two prepared forms share the count and fill loops.
72enum RowEmitter<'p, const W: usize> {
73    /// A tabulated delta of a [`Prepared::Local`] layer.
74    Tabulated {
75        ptm: &'p LocalPtm<W>,
76        entry: &'p DeltaEntry<W>,
77    },
78    /// The generator pass of a [`Prepared::Rotation`] layer; the identity pass is never remote.
79    Generator { prep: &'p RotationPrep<W> },
80}
81
82impl<const W: usize> RowEmitter<'_, W> {
83    /// Whether every term emits a row, so a bucket's count is its length.
84    ///
85    /// Dense over the *active* support patterns only: `amp` is sized `LOCAL_DIM` but the channel populates `4^k` entries.
86    /// A rotation's generator pass is never dense, since a commuting term emits nothing.
87    fn is_dense(&self) -> bool {
88        match self {
89            RowEmitter::Tabulated { ptm, entry } => {
90                let dim = 1usize << (2 * ptm.k());
91                entry.amp[..dim].iter().all(|a| *a != ZERO)
92            }
93            RowEmitter::Generator { .. } => false,
94        }
95    }
96
97    /// The row this delta emits for one term, or `None` when it emits none.
98    #[inline]
99    fn emit(
100        &self,
101        x: &[u64; W],
102        z: &[u64; W],
103        c: Complex64,
104    ) -> Option<([u64; W], [u64; W], Complex64)> {
105        match self {
106            RowEmitter::Tabulated { ptm, entry } => entry.emit(ptm.support_bits(x, z), x, z, c),
107            RowEmitter::Generator { prep } => prep.emit_gen(x, z, c),
108        }
109    }
110}

112/// Build this partition's outgoing payloads for one layer.
113///
114/// Returns one slot per partner rank: `Some(payload)` for every partner the plan names, `None` for the rest (including this partition's own slot).
115/// A partner's [`PartnerPayload::blocks`] is one block per remote delta destined for it, in ascending remote-delta index, *including blocks with no rows*: the receiver indexes blocks positionally.
116pub(crate) fn export_layer<const W: usize>(
117    local: &PauliSum<W>,
118    prep: &Prepared<W>,
119    plan: &PartitionPlan,
120    size: u32,
121    map: &ChunkMap,
122    scratch: &mut ExportScratch<W>,
123) -> (Vec<Option<PartnerPayload<W>>>, ExportCounts) {
124    let k = plan.remote.len();
125    let nb = local.num_buckets();
126    let mut send: Vec<Option<PartnerPayload<W>>> = (0..size).map(|_| None).collect();
127    let mut counts = ExportCounts {
128        rows_to: vec![0; size as usize],
129        bytes_to: vec![0; size as usize],
130    };
131    if k == 0 {
132        return (send, counts);
133    }
134
135    let emitters: Vec<RowEmitter<'_, W>> = plan
136        .remote
137        .iter()
138        .map(|r| match prep {
139            Prepared::Local(ptm) => RowEmitter::Tabulated {
140                ptm,
141                entry: &ptm.deltas()[r.entry],
142            },
143            Prepared::Rotation(p) => {
144                debug_assert_eq!(
145                    r.entry, 1,
146                    "a rotation's only remote entry is the generator pass",
147                );
148                RowEmitter::Generator { prep: p }
149            }
150        })
151        .collect();
152    let dense: Vec<bool> = emitters.iter().map(RowEmitter::is_dense).collect();
153
154    // Pass 1. Bucket-major counts, so each bucket owns one contiguous chunk and the pass needs no synchronization.
155    scratch.counts.clear();
156    scratch.counts.resize(nb * k, 0);
157    scratch
158        .counts
159        .par_chunks_mut(k)
160        .enumerate()
161        .for_each(|(b, slot)| count_bucket(local, prep, plan, &dense, b, slot));
162
163    // Pass 2, one block per remote delta, written into payloads taken from the pool: the fill writes by index into storage that is neither allocated nor zeroed here.
164    debug_assert_eq!(
165        map.positions(),
166        nb,
167        "the chunk map is built for a different bucket count than the sum has",
168    );
169    scratch.block_counts.clear();
170    scratch.block_counts.resize(nb, 0);
171    scratch.src_of.clear();
172    scratch.src_of.resize(nb, 0);
173    let mut blocks_used = vec![0usize; size as usize];
174    for (i, r) in plan.remote.iter().enumerate() {
175        // Destination-coset order: position `p` of the receiver is filled from this partition's bucket `bucket_at(p) ^ bd`.
176        for p in 0..nb {
177            let src = map.bucket_at(p as u32) ^ r.bucket_delta;
178            scratch.src_of[p] = src;
179            scratch.block_counts[p] = scratch.counts[src as usize * k + i];
180        }
181        let q = r.partner as usize;
182        let payload = send[q].get_or_insert_with(|| scratch.pool.pop().unwrap_or_default());
183        let j = blocks_used[q];
184        blocks_used[q] += 1;
185        if payload.blocks.len() <= j {
186            payload
187                .blocks
188                .resize_with(j + 1, ExchangeBlock::<W>::default);
189        }
190        let block = &mut payload.blocks[j];
191        block.set_counts(r.entry as u32, &scratch.block_counts);
192        let rows = block.rows();
193        {
194            let ExchangeBlock {
195                ref offsets,
196                ref mut x,
197                ref mut z,
198                ref mut coeff,
199                ..
200            } = *block;
201            let cols = BlockCols {
202                x: &mut x[..rows],
203                z: &mut z[..rows],
204                c: &mut coeff[..rows],
205            };
206            fill_range(local, &emitters[i], offsets, &scratch.src_of, 0, nb, cols);
207        }
208        counts.rows_to[q] += rows as u64;
209        counts.bytes_to[q] += block.bytes() as u64;
210    }
211    // A payload out of the pool may have carried more blocks than this layer has remote deltas for it; the extras must not travel.
212    for (q, payload) in send.iter_mut().enumerate() {
213        if let Some(payload) = payload {
214            payload.blocks.truncate(blocks_used[q]);
215        }
216    }
217
218    (send, counts)
219}

221/// Count one source bucket's exported rows, one slot per remote delta.
222///
223/// The support pattern is computed once per term and reused across every sparse entry; dense entries are filled from the bucket length without touching a term at all.
224fn count_bucket<const W: usize>(
225    local: &PauliSum<W>,
226    prep: &Prepared<W>,
227    plan: &PartitionPlan,
228    dense: &[bool],
229    b: usize,
230    out: &mut [u32],
231) {
232    let (bx, bz, bc) = local.bucket(b);
233    let n = bc.len();
234    match prep {
235        Prepared::Local(ptm) => {
236            let mut all_dense = true;
237            for (k, &d) in dense.iter().enumerate() {
238                if d {
239                    out[k] = n as u32;
240                } else {
241                    out[k] = 0;
242                    all_dense = false;
243                }
244            }
245            if all_dense {
246                return;
247            }
248            for t in 0..n {
249                let s = ptm.support_bits(&bx[t], &bz[t]);
250                for (k, r) in plan.remote.iter().enumerate() {
251                    if !dense[k] && ptm.deltas()[r.entry].amp[s] != ZERO {
252                        out[k] += 1;
253                    }
254                }
255            }
256        }
257        Prepared::Rotation(p) => {
258            debug_assert_eq!(out.len(), 1, "a rotation has at most one remote delta");
259            let mut anti = 0u32;
260            for t in 0..n {
261                let v = PauliString::<W> { x: bx[t], z: bz[t] };
262                if !v.commutes_with(&p.gen) {
263                    anti += 1;
264                }
265            }
266            out[0] = anti;
267        }
268    }
269}

271/// Fill destination positions `lo..hi` of one block.
272///
273/// `src_of[p]` is the source bucket position `p` draws from.
274/// Splitting the range at `mid` splits the columns at `offsets[mid] - offsets[lo]`, and the two halves are disjoint by construction, so a block's contents are independent of the thread count.
275fn fill_range<const W: usize>(
276    local: &PauliSum<W>,
277    emitter: &RowEmitter<'_, W>,
278    offsets: &[u32],
279    src_of: &[u32],
280    lo: usize,
281    hi: usize,
282    cols: BlockCols<'_, W>,
283) {
284    debug_assert_eq!(cols.len(), (offsets[hi] - offsets[lo]) as usize);
285    if hi - lo > 1 && cols.len() > FILL_PARALLEL_MIN_ROWS {
286        let mid = lo + (hi - lo) / 2;
287        let (head, tail) = cols.split_at((offsets[mid] - offsets[lo]) as usize);
288        rayon::join(
289            || fill_range(local, emitter, offsets, src_of, lo, mid, head),
290            || fill_range(local, emitter, offsets, src_of, mid, hi, tail),
291        );
292        return;
293    }
294    let BlockCols { x, z, c } = cols;
295    let mut w = 0usize;
296    for &src in &src_of[lo..hi] {
297        let (bx, bz, bc) = local.bucket(src as usize);
298        for t in 0..bc.len() {
299            if let Some((kx, kz, kc)) = emitter.emit(&bx[t], &bz[t], bc[t]) {
300                x[w] = kx;
301                z[w] = kz;
302                c[w] = kc;
303                w += 1;
304            }
305        }
306    }
307    debug_assert_eq!(
308        w,
309        x.len(),
310        "export: pass 2 filled {w} rows where pass 1 counted {}",
311        x.len(),
312    );
313}

315/// One block's three columns, restricted to a source-bucket range.
316///
317/// A bundle, not an abstraction: keeps [`fill_range`]'s recursive signature inside clippy's argument budget.
318struct BlockCols<'a, const W: usize> {
319    x: &'a mut [[u64; W]],
320    z: &'a mut [[u64; W]],
321    c: &'a mut [Complex64],
322}
323
324impl<'a, const W: usize> BlockCols<'a, W> {
325    /// Rows in the range.
326    fn len(&self) -> usize {
327        self.c.len()
328    }
329
330    /// Split every column at the same row, consuming the borrow.
331    fn split_at(self, at: usize) -> (BlockCols<'a, W>, BlockCols<'a, W>) {
332        let (x0, x1) = self.x.split_at_mut(at);
333        let (z0, z1) = self.z.split_at_mut(at);
334        let (c0, c1) = self.c.split_at_mut(at);
335        (
336            BlockCols {
337                x: x0,
338                z: z0,
339                c: c0,
340            },
341            BlockCols {
342                x: x1,
343                z: z1,
344                c: c1,
345            },
346        )
347    }
348}

350/// Every exported row belongs to the partner it is addressed to.
351///
352/// Debug builds only, `O(exported rows)`.
353/// A failure means either the plan misclassified a delta or the caller's `local` sum was not the pure partition it claims to be.
354#[cfg(debug_assertions)]
355pub(crate) fn debug_assert_exported_partitions<const W: usize>(
356    send: &[Option<PartnerPayload<W>>],
357    rows: &crate::bucket::hash::PartitionRows<W>,
358) {
359    for (q, payload) in send.iter().enumerate() {
360        let Some(payload) = payload else { continue };
361        for block in &payload.blocks {
362            for i in 0..block.rows() {
363                let part = rows.partition_of(&block.x[i], &block.z[i]);
364                debug_assert_eq!(
365                    part, q as u32,
366                    "export: row {i} of the block for entry {} is addressed to partition {q} but \
367                     belongs to partition {part}",
368                    block.header.entry,
369                );
370            }
371        }
372    }
373}

391    /// The destination-coset order for `plan` over `num_buckets` buckets, in one chunk: the layout every export test compares against.
392    fn map_for(plan: &PartitionPlan, num_buckets: usize) -> ChunkMap {
393        let mut map = ChunkMap::default();
394        map.rebuild(
395            &crate::engine::coset::Gf2Span::new(
396                &plan.local_bucket_deltas,
397                num_buckets.trailing_zeros() as u8,
398            ),
399            num_buckets,
400            1,
401        );
402        map
403    }

405    /// One partition's export of one layer.
406    struct Exported<const W: usize> {
407        rank: u32,
408        send: Vec<Option<PartnerPayload<W>>>,
409        counts: ExportCounts,

412    /// Split `input` across `rows`'s partitions and export one layer from each.
413    fn export_all<const W: usize>(
414        input: &PauliSum<W>,
415        ch: &dyn Channel<W>,
416        adjoint: bool,
417        bits: u8,
418        seed: u64,
419        rows: &PartitionRows<W>,
420    ) -> Vec<Exported<W>> {
421        let hash = Gf2Hash::<W>::new(input.num_qubits(), bits, seed);
422        let whole = input.clone().with_hash(hash);
423        let prep = ch
424            .prepare(whole.hash(), adjoint)
425            .expect("channel could not be prepared");
426        let size = rows.num_partitions() as u32;
427        (0..size)
428            .map(|rank| {
429                let local = whole.filter_partition(rows, rank);
430                let plan = PartitionPlan::new(&prep, rows, rank);
431                let mut scratch = ExportScratch::default();
432                let map = map_for(&plan, local.num_buckets());
433                let (send, counts) = export_layer(&local, &prep, &plan, size, &map, &mut scratch);
434                assert!(
435                    send[rank as usize].is_none(),
436                    "a partition exports to itself"
437                );
438                assert_eq!(send.len(), size as usize);
439                Exported { rank, send, counts }
440            })
441            .collect()
442    }

459    /// Group rows by key into `(count, sum)`.
460    fn by_key<const W: usize>(
461        rows: impl IntoIterator<Item = ([u64; W], [u64; W], Complex64)>,
462    ) -> HashMap<([u64; W], [u64; W]), (usize, Complex64)> {

464        for (x, z, c) in rows {

472    /// The **row-level** oracle: for every input term, the rows [`Channel::apply`] emits whose output key lands in a different partition than the term itself.
473    ///
474    /// Deliberately not `test_support::naive_apply_layer`, which sums a key's contributions from *every* source; the export carries only the partition-crossing ones.
475    /// Rows are summed per (input term, output key) first, and exactly-zero results are dropped the way a zero amplitude emits nothing.
476    fn crossing_rows<const W: usize>(
477        input: &PauliSum<W>,
478        ch: &dyn Channel<W>,
479        adjoint: bool,
480        rows: &PartitionRows<W>,
481    ) -> Vec<([u64; W], [u64; W], Complex64)> {
482        let mf = ch.max_fanout().max(1);
483        let mut buf_x = vec![[0u64; W]; mf];
484        let mut buf_z = vec![[0u64; W]; mf];
485        let mut buf_c = vec![ZERO; mf];
486        let mut out = Vec::new();
487        for (x, z, c) in input.iter() {
488            let src = rows.partition_of(x, z);
489            let mut len = 0usize;
490            {
491                let mut buf = OutputBuffer::<W> {
492                    x: &mut buf_x,
493                    z: &mut buf_z,
494                    coeff: &mut buf_c,
495                    len: &mut len,
496                };
497                if adjoint {
498                    ch.apply_adjoint(x, z, c, &mut buf);
499                } else {
500                    ch.apply(x, z, c, &mut buf);
501                }
502            }
503            let mut per_term: HashMap<([u64; W], [u64; W]), Complex64> = HashMap::new();
504            for i in 0..len {
505                *per_term.entry((buf_x[i], buf_z[i])).or_insert(ZERO) += buf_c[i];
506            }
507            for ((kx, kz), kc) in per_term {
508                if kc != ZERO && rows.partition_of(&kx, &kz) != src {
509                    out.push((kx, kz, kc));
510                }
511            }
512        }
513        out
514    }

516    /// Two terms, one bucket, one remote delta: the export is hand-checkable row for row.
517    ///
518    /// `H` on qubit 0 sends `X₀ → Z₀` and `Z₀ → X₀`, so its non-identity delta is the mask `x₀ z₀`.
519    /// Under a partition row that reads the `x` bit of qubit 0, that mask has `part = 1`, so the delta is remote and `X₀`/`Z₀` sit in different partitions; each exports its single image to the other.
520    #[test]
521    fn a_single_bucket_h_layer_exports_the_swapped_keys() {
522        let mut acc = BuildAccumulator::<1>::with_capacity(8, 2);
523        acc.add_term(PauliString::<1>::x(0), Phase::ONE, Complex64::new(1.0, 0.0));
524        acc.add_term(PauliString::<1>::z(0), Phase::ONE, Complex64::new(2.0, 0.0));
525        let input = acc.finalize();
526        let rows = PartitionRows::<1>::from_rows(8, vec![[1u64]], vec![[0u64]]);
527        assert_eq!(rows.partition_of(&[1], &[0]), 1, "X₀ is in partition 1");
528        assert_eq!(rows.partition_of(&[0], &[1]), 0, "Z₀ is in partition 0");
529
530        // `bits = 0`: one bucket, so every block is a single segment.
531        let exports = export_all(&input, &Clifford1Q::h(0), false, 0, 0x51, &rows);
532        assert_eq!(exports.len(), 2);
533
534        // Partition 0 holds Z₀ (coeff 2) and ships X₀ to partition 1.
535        // Partition 1 holds X₀ (coeff 1) and ships Z₀ to partition 0.
536        let expected: [(u32, [u64; 1], [u64; 1], f64); 2] =
537            [(1, [1], [0], 2.0), (0, [0], [1], 1.0)];
538        for (e, (partner, kx, kz, kc)) in exports.iter().zip(expected) {
539            assert_eq!(e.counts.rows_to[partner as usize], 1);
540            assert_eq!(e.counts.rows_to.iter().sum::<u64>(), 1, "one row only");
541            assert!(e.counts.bytes_to[partner as usize] > 0);
542            let payload = e.send[partner as usize]
543                .as_ref()
544                .unwrap_or_else(|| panic!("rank {} sent nothing to {partner}", e.rank));
545            assert_eq!(payload.blocks.len(), 1, "one remote delta");
546            let block = &payload.blocks[0];
547            // `deltas()` is ascending by `local_delta`, so entry 0 is the identity and the X↔Z swap (local delta `0b11`) is entry 1.
548            assert_eq!(block.header.entry, 1);
549            assert_eq!(block.num_buckets(), 1);
550            assert_eq!(block.rows(), 1);
551            let (sx, sz, sc) = block.segment(0);
552            assert_eq!((sx, sz), (&[kx][..], &[kz][..]));
553            assert!((sc[0] - Complex64::new(kc, 0.0)).norm() < TOL);
554        }
555    }

557    /// A rotation every term commutes with produces no rows — but still
558    /// produces its block, because the receiver indexes blocks positionally.
559    #[test]
560    fn an_all_commuting_rotation_exports_empty_blocks() {
561        // Generator Z₀X₂X₄X₆, weight 4 > MAX_LOCAL_SUPPORT (the Rotation arm).
562        let gen = {
563            let mut g = PauliString::<1>::z(0);
564            for q in [2u32, 4, 6] {
565                g.mul_assign(&PauliString::<1>::x(q));
566            }
567            g
568        };
569        let mut acc = BuildAccumulator::<1>::with_capacity(8, 3);
570        for p in [gen, PauliString::<1>::z(1), PauliString::<1>::x(3)] {
571            assert!(p.commutes_with(&gen), "fixture term must commute");
572            acc.add_term(p, Phase::ONE, Complex64::new(1.5, 0.0));
573        }
574        let input = acc.finalize();
575        // A row on the `x` bit of qubit 2 makes `part(gen) = 1`: the generator pass is remote.
576        let rows = PartitionRows::<1>::from_rows(8, vec![[1u64 << 2]], vec![[0u64]]);
577        assert_eq!(rows.partition_of(&gen.x, &gen.z), 1);
578
579        let rot = PauliRotation::new(gen, 0.41);
580        let exports = export_all(&input, &rot, false, 2, 0x52, &rows);
581        assert!(
582            all_rows(&exports).is_empty(),
583            "commuting terms emit nothing"
584        );
585        for e in &exports {
586            assert!(e.counts.rows_to.iter().all(|&r| r == 0));
587            let blocks: Vec<_> = e.send.iter().flatten().flat_map(|p| &p.blocks).collect();
588            assert_eq!(blocks.len(), 1, "the empty block is still exported");
589            assert_eq!(blocks[0].rows(), 0);
590        }

593    /// The CSR is in the *receiver's* order: every row of segment `p` lands in the receiver's bucket `map.bucket_at(p)`.
594    ///
595    /// This is the contract `RecvRows` reads the block through, so it is checked directly against the hash rather than only end to end through the differential nets.
596    /// The matrix covers a non-trivial permutation and a non-zero bucket delta on the remote entry.
597    #[test]
598    fn every_segment_holds_the_rows_of_its_destination_bucket() {
599        let input = rand_sum::<1>(700, 8, 0x9C7);
600        for (_name, ch) in &differential_channels_w1() {
601            for &adjoint in &[false, true] {
602                for &bits in &[1u8, 3, 4] {
603                    for &pbits in &[1u8, 2] {
604                        let rows = PartitionRows::<1>::from_seed(8, pbits, 0x1234);
605                        let hash = Gf2Hash::<1>::new(8, bits, 0xAB);
606                        let whole = input.clone().with_hash(hash);
607                        let prep = ch.prepare(whole.hash(), adjoint).expect("prepare");
608                        let size = rows.num_partitions() as u32;
609                        for rank in 0..size {
610                            let local = whole.filter_partition(&rows, rank);
611                            let plan = PartitionPlan::new(&prep, &rows, rank);
612                            let map = map_for(&plan, local.num_buckets());
613                            let mut scratch = ExportScratch::default();
614                            let (send, _) =
615                                export_layer(&local, &prep, &plan, size, &map, &mut scratch);
616                            for payload in send.iter().flatten() {
617                                for block in &payload.blocks {
618                                    for p in 0..block.num_buckets() {
619                                        let want = map.bucket_at(p);
620                                        let (sx, sz, _) = block.segment(p);
621                                        for (x, z) in sx.iter().zip(sz) {
622                                            assert_eq!(
623                                                whole.hash().bucket_of(x, z),
624                                                want,
625                                                "segment {p} of entry {} carries a row for                                                  another bucket",
626                                                block.header.entry,
627                                            );
628                                        }
629                                    }
630                                }
631                            }
632                        }
633                    }
634                }
635            }
636        }
637    }

639    /// Pass 1 sized every block exactly: the counted row total is what pass 2 wrote, in every block, for every channel class.
640    ///
641    /// The two passes agreeing is `fill_range`'s own `debug_assert` (this suite runs in debug).
642    /// What is checked here is the shape around it: the CSR offsets end at `header.rows`, the grow-only columns hold at least that many rows, and the reported per-partner totals are the blocks' own.
643    #[test]
644    fn the_counted_rows_are_the_filled_rows() {
645        let input = rand_sum::<1>(700, 8, 0x9C0);
646        for (name, ch) in &differential_channels_w1() {
647            for &adjoint in &[false, true] {
648                for &bits in &[0u8, 3] {
649                    for &pbits in &[1u8, 2] {
650                        let rows = PartitionRows::<1>::from_seed(8, pbits, 0x1234);
651                        let exports = export_all(&input, ch.as_ref(), adjoint, bits, 0xAB, &rows);
652                        for e in &exports {
653                            let mut rows_to = vec![0u64; rows.num_partitions()];
654                            for (q, payload) in e.send.iter().enumerate() {
655                                let Some(payload) = payload else { continue };
656                                for block in &payload.blocks {
657                                    let what = format!(
658                                        "{name} adjoint={adjoint} bits={bits} p={pbits} \
659                                         rank={} entry={}",
660                                        e.rank, block.header.entry,
661                                    );
662                                    // Grow-only columns: at least the rows the counts sized, and the CSR offsets end exactly there.
663                                    assert!(block.coeff.len() >= block.rows(), "{what}: rows");
664                                    assert!(block.x.len() >= block.rows(), "{what}: x");
665                                    assert!(block.z.len() >= block.rows(), "{what}: z");
666                                    assert_eq!(
667                                        block.offsets.last().copied(),
668                                        Some(block.header.rows),
669                                        "{what}: offsets",
670                                    );
671                                    rows_to[q] += block.rows() as u64;
672                                }
673                            }
674                            assert_eq!(e.counts.rows_to, rows_to, "{name}: reported rows");
675                        }
676                    }
677                }
678            }
679        }
680    }

781    /// The scratch is reusable: a layer through a scratch a wider layer has
782    /// already grown gives the same payload as one through a fresh scratch.
783    #[test]
784    fn a_reused_scratch_gives_the_same_export() {
785        let input = rand_sum::<1>(500, 8, 0x9C5);
786        let rows = PartitionRows::<1>::from_seed(8, 2, 0x5555);
787        let hash = Gf2Hash::<1>::new(8, 4, 0xAB);
788        let whole = input.with_hash(hash);
789        let mut scratch = ExportScratch::default();
790
791        // A wide-fanout layer first, so the scratch is left large.
792        let channels = differential_channels_w1();
793        let (_, wide) = channels
794            .iter()
795            .find(|(n, _)| *n == "haar_su4")
796            .expect("the dense SU(4) cell");
797        let prep = wide.prepare(whole.hash(), false).unwrap();
798        let plan = PartitionPlan::new(&prep, &rows, 0);
799        let wide_local = whole.filter_partition(&rows, 0);
800        let map = map_for(&plan, wide_local.num_buckets());
801        let _ = export_layer(&wide_local, &prep, &plan, 4, &map, &mut scratch);
802
803        let h = Clifford1Q::h(3);
804        let prep = Channel::<1>::prepare(&h, whole.hash(), false).unwrap();
805        let plan = PartitionPlan::new(&prep, &rows, 0);
806        let local = whole.filter_partition(&rows, 0);
807        let map = map_for(&plan, local.num_buckets());
808        let (reused, counts_reused) = export_layer(&local, &prep, &plan, 4, &map, &mut scratch);
809        let (fresh, counts_fresh) =
810            export_layer(&local, &prep, &plan, 4, &map, &mut ExportScratch::default());
811        assert_eq!(counts_reused, counts_fresh);
812        assert_eq!(reused.len(), fresh.len());
813        for (a, b) in reused.iter().zip(&fresh) {
814            assert_eq!(a, b, "a reused scratch changed the payload");
815        }
