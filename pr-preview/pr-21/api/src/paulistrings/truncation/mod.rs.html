13/// A truncation strategy: a per-term filter, a per-layer pass, or both.
14///
15/// Built-ins: [`CoefficientThreshold`] and [`WeightCutoff`] filter terms, [`TopN`], [`ApproxTopN`] and [`CollapseSample`] run a layer pass, [`And`] and [`Or`] compose, and [`BuiltinTruncation`] holds any of them as one value.
16///
17/// Above one partition every partition sees a disjoint slice of the layer, so a layer pass must be collective: a policy whose `finalizes_layer()` is `true` runs partitioned only if it overrides [`finalize_layer_partitioned`](Self::finalize_layer_partitioned) and [`supports_partitioned`](Self::supports_partitioned).
18/// Every built-in does except exact [`TopN`], which is not yet supported partitioned; [`ApproxTopN`] is the partitioned alternative.
19///
20/// ```
21/// use paulistrings::TruncationPolicy;
22/// use num_complex::Complex64;
23///
24/// /// Drop the imaginary half: keep only terms whose coefficient is real.
25/// struct RealOnly;
26/// impl<const W: usize> TruncationPolicy<W> for RealOnly {
27///     fn keep_term(&self, _x: &[u64; W], _z: &[u64; W], c: Complex64) -> bool {
28///         c.im == 0.0
29///     }
30/// }
31/// ```
32pub trait TruncationPolicy<const W: usize>: Send + Sync {
33    /// Per-term filter, run inside the merge on every output, so it must be cheap.
34    ///
35    /// `c` is the coefficient summed over every contribution to the key `(x, z)`.
36    fn keep_term(&self, _x: &[u64; W], _z: &[u64; W], _c: Complex64) -> bool {
37        true
38    }
39
40    /// Pass over the whole sum after each layer; [`PauliSum::retain`] is the easy way to write a filter here.
41    fn finalize_layer(&self, _sum: &mut PauliSum<W>) {}
42
43    /// Whether [`finalize_layer`](Self::finalize_layer) does anything, a hint that lets the engine skip it.
44    ///
45    /// Return `false` only if `finalize_layer` is a no-op: the engine may then skip it, which is a wrong answer otherwise.
46    fn finalizes_layer(&self) -> bool {
47        true
48    }
49
50    /// The layer pass over one partition's slice of the layer, run on every layer on every partition in lock-step.
51    ///
52    /// The default runs [`finalize_layer`](Self::finalize_layer) at one partition and nothing above it.
53    /// An override must issue the same collective calls on every partition, never skipping one on a partition with nothing to do, and decide only from all-reduced values.
54    ///
55    /// # Panics
56    ///
57    /// Above one partition, if the policy reports [`finalizes_layer()`](Self::finalizes_layer) and has not overridden this method.
58    fn finalize_layer_partitioned(&self, local: &mut PauliSum<W>, collectives: &dyn Collectives) {
59        if collectives.size() == 1 {
60            self.finalize_layer(local);
61            return;
62        }
63        assert!(
64            !self.finalizes_layer(),
65            "{} has a layer pass but no partitioned one: every partition sees only a disjoint \
66             slice of the layer, so the pass has to be collective. Override \
67             `finalize_layer_partitioned` and `supports_partitioned` (see `ApproxTopN`, which \
68             all-reduces its octave histogram), or use `ApproxTopN` itself.",
69            std::any::type_name_of_val(self),
70        );
71    }
72
73    /// Whether [`finalize_layer_partitioned`](Self::finalize_layer_partitioned) runs above one partition; every partitioned driver checks it before the first layer.
74    ///
75    /// The default is `true` exactly when there is no layer pass; a policy that overrides `finalize_layer_partitioned` with a collective pass returns `true`.
76    fn supports_partitioned(&self) -> bool {
77        !self.finalizes_layer()
78    }
79}
80
81/// Panics unless `policy` can run on `partitions` partitions.
82pub(crate) fn assert_supports_partitions<const W: usize, T>(policy: &T, partitions: usize)
83where
84    T: TruncationPolicy<W> + ?Sized,
85{
86    assert!(
87        partitions == 1 || policy.supports_partitioned(),
88        "{} cannot run on {partitions} partitions: its layer pass has no collective form. \
89         Partitioned exact `TopN` is not yet supported; `ApproxTopN` is the partitioned \
90         alternative.",
91        std::any::type_name_of_val(policy),
92    );
93}
