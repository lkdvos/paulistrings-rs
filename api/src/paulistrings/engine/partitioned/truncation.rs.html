1//! [`PartitionedTruncation`] — truncation whose layer pass is collective.
2//!
3//! In partitioned mode each partition holds a *disjoint subset* of the terms
4//! (ARCHITECTURE.md §Partitioning), so a [`TruncationPolicy`] splits cleanly
5//! in two:
6//!
7//! - [`keep_term`](TruncationPolicy::keep_term) is per term and needs nothing: it runs inside the merge, on the complete summed coefficient of a key, and a key lives on exactly one partition.
8//! - [`finalize_layer`](TruncationPolicy::finalize_layer) may be *global*, and a partition cannot see the global layer.
9//!   [`ApproxTopN`] chooses an octave edge from the histogram of the whole layer — exactly the sum of the per-partition histograms — so one `allreduce` makes every partition choose the same edge, and the union of the retained sets is bit for bit the single-partition answer.
10//!
11//! Exact [`TopN`](crate::truncation::TopN) has no such reduction: the `n`-th
12//! largest magnitude is a distributed *k*-th selection, not a sum. It is
13//! rejected at compile time rather than approximated — see the trait docs.
14

26/// A truncation policy usable in partitioned propagation: its layer finalization is collective.
27///
28/// # The contract
29///
30/// A truncation policy that finalizes a layer (`finalizes_layer() == true`) must implement [`PartitionedTruncation`]; the trait's default panics otherwise, since finalizing a layer with no collective form would let partitions diverge silently.
31/// [`finalize_layer_partitioned`](Self::finalize_layer_partitioned) is called on every layer, on every partition, in lock-step: a collective is only well-defined if every partition issues the same calls in the same order, so an implementation must not skip a call on a partition that happens to have nothing to do, and must derive its decision only from all-reduced values so every partition applies the identical predicate to its own terms.
32/// Under those two rules the retained set is exactly what a single partition holding the whole sum would have retained.
33///
34/// # Which policies implement it
35///
36/// [`CoefficientThreshold`] and [`WeightCutoff`] are per-term filters with no layer pass, so they take the default no-op body.
37/// [`ApproxTopN`] all-reduces its octave histogram every layer, regardless of what the partition rows do.
38/// [`CollapseSample`] all-reduces the global length every layer, and the per-partition norms on a layer that collapses.
39/// [`And`] runs both sides in order, like [`And::finalize_layer`](TruncationPolicy::finalize_layer); [`Or`] runs neither, because its unpartitioned `finalize_layer` is the trait's no-op default rather than either child's, and the two must agree.
40///
41/// ```
42/// use paulistrings::truncation::{And, ApproxTopN, CoefficientThreshold};
43/// use paulistrings::PartitionedTruncation;
44///
45/// fn propagate_partitioned<T: PartitionedTruncation<1>>(_policy: T) {}
46///
47/// propagate_partitioned(And(CoefficientThreshold(1e-12), ApproxTopN(1_000)));
48/// ```
49///
50/// # Why [`TopN`](crate::truncation::TopN) is rejected
51///
52/// ```compile_fail
53/// use paulistrings::truncation::TopN;
54/// use paulistrings::PartitionedTruncation;
55///
56/// fn propagate_partitioned<T: PartitionedTruncation<1>>(_policy: T) {}
57///
58/// // error[E0277]: the trait bound `TopN: PartitionedTruncation<1>` is not
59/// // satisfied — exact top-n has no collective form yet.
60/// propagate_partitioned(TopN(10));
61/// ```
62///
63/// `TopN` needs the exact `n`-th largest `|c|²` of the whole layer — a distributed *k*-th selection, not a reduction — so it is rejected at compile time rather than approximated, which would silently change what `TopN` means.
64/// Use [`ApproxTopN`] when `n` is a memory budget.
65pub trait PartitionedTruncation<const W: usize>: TruncationPolicy<W> {
66    /// The collective layer pass: `local` is this partition's slice of the layer, `coll` its view of the group.
67    ///
68    /// Called on every layer on every partition, in lock-step, for a policy whose [`finalizes_layer`](TruncationPolicy::finalizes_layer) is `true` — the driver skips the call entirely on one that answers `false` (see the trait docs).
69    /// The default is no layer pass at all, which is correct exactly for the policies that have none, so it asserts that this is one of them rather than silently dropping a `finalize_layer` a caller was relying on.
70    ///
71    /// # Panics
72    ///
73    /// If the policy reports [`finalizes_layer()`](TruncationPolicy::finalizes_layer) and has not overridden this method.
74    fn finalize_layer_partitioned(&self, _local: &mut PauliSum<W>, _coll: &dyn Collectives) {
75        assert!(
76            !self.finalizes_layer(),
77            "{} has a layer finalization but no partitioned one. A layer pass \
78             in partitioned mode has to be collective — every partition sees \
79             only a disjoint slice of the layer — so the default no-op body \
80             cannot stand in for it. Override `finalize_layer_partitioned` \
81             (see `ApproxTopN`, which all-reduces its octave histogram), or \
82             use `ApproxTopN` itself.",
83            std::any::type_name_of_val(self),
84        );
85    }
86}

94impl<const W: usize, A, B> PartitionedTruncation<W> for And<A, B>
95where
96    A: PartitionedTruncation<W>,
97    B: PartitionedTruncation<W>,
98{
99    /// Both sides, first then second — the order [`And::finalize_layer`](TruncationPolicy::finalize_layer) uses, and the same order on every partition, so the collectives stay in lock-step.
100    fn finalize_layer_partitioned(&self, local: &mut PauliSum<W>, coll: &dyn Collectives) {
101        self.0.finalize_layer_partitioned(local, coll);
102        self.1.finalize_layer_partitioned(local, coll);
103    }
104}

106impl<const W: usize, A, B> PartitionedTruncation<W> for Or<A, B>
107where
108    A: PartitionedTruncation<W>,
109    B: PartitionedTruncation<W>,
110{
111    /// Neither side — `Or`'s unpartitioned [`finalize_layer`](TruncationPolicy::finalize_layer) is the trait's no-op default rather than either child's, and the two paths have to agree.
112    ///
113    /// The children are still bounded by this trait: a composition is partitioned-safe only if its parts are, which keeps the bound meaningful if `Or` ever grows a layer pass.
114    fn finalize_layer_partitioned(&self, _local: &mut PauliSum<W>, _coll: &dyn Collectives) {}
115}

117impl<const W: usize> PartitionedTruncation<W> for ApproxTopN {
118    /// One `allreduce_sum_u64` of `[len, hist…]`, then the single-partition edge walk and retain.
119    ///
120    /// Octave populations are additive across a disjoint partition of the terms, and so is the term count, so the reduced buffer is exactly the histogram and length the single-partition path would have computed.
121    /// `octave_edge` is a pure function of those two, so every partition reaches the same edge decision and retains its own terms against it, with no second round of communication.
122    ///
123    /// The length rides in slot 0 of the same buffer rather than in its own reduction.
124    /// Neither early exit of the single-partition path is taken here, because neither test can be answered before the reduction — a partition that returned early would desynchronize the group.
125    fn finalize_layer_partitioned(&self, local: &mut PauliSum<W>, coll: &dyn Collectives) {
126        let hist = octave_histogram(local);
127
128        // [len, bin 0, bin 1, …]. `u32` counters widen to `u64` for the reduction: `P` partitions of up to `u32::MAX` terms each can overflow a `u32` bin, and the transport's reduction is `u64`.
129        let mut packed = [0u64; 1 + APPROX_BINS];
130        packed[0] = local.len() as u64;
131        for (slot, &count) in packed[1..].iter_mut().zip(hist.iter()) {
132            *slot = u64::from(count);
133        }
134        coll.allreduce_sum_u64(&mut packed);
135
136        let total = packed[0] as usize;
137        let edge = octave_edge(&packed[1..], total, self.0);
138        retain_at_or_above(local, edge);
139    }
140}

142impl<const W: usize> PartitionedTruncation<W> for CollapseSample {
143    /// Two reductions, then a pick every partition makes identically: `[len, pass]` as integers, where only rank 0 contributes the pass index, and the per-partition `Σ|c|²` as a vector in which each partition fills its own slot, so every partition holds the same exact weights.
144    /// The shared uniform picks a partition by weight and the offset left over picks within it, exactly as [`finalize_layer`](TruncationPolicy::finalize_layer) picks a bucket and then a term.
145    fn finalize_layer_partitioned(&self, local: &mut PauliSum<W>, coll: &dyn Collectives) {
146        let rank = coll.rank() as usize;
147        let call = if rank == 0 { self.next_call() } else { 0 };
148        let mut head = [local.len() as u64, call];
149        coll.allreduce_sum_u64(&mut head);
150        let [len, call] = head;
151        if len as usize <= self.cache {
152            return;
153        }
154
155        let norms = bucket_norms(local);
156        let mut weights = vec![0.0f64; coll.size() as usize];
157        weights[rank] = norms.iter().sum();
158        coll.allreduce_sum_f64(&mut weights);
159        let total: f64 = weights.iter().sum();
160        let target = self.target(call, total, len as usize);
161        let (chosen, rest) = pick_slot(weights.iter().copied(), target)
162            .expect("CollapseSample: a positive total has a positive partition");
163        if rank == chosen {
164            collapse_to_target(local, &norms, rest);
165        } else {
166            local.clear();
167        }
168        if rank == 0 {
169            let n = self.count_collapse();
170            log::debug!(
171                target: LOG_TARGET,
172                "collapse_sample: {len} terms over {} partitions, sum |c|^2 = {total:.6e}, \
173                 collapsed to one on partition {chosen} (collapse {n})",
174                coll.size(),
175            );
176        }
177    }
178}

191    /// Split `sum` into `1 << pbits` partitions, run the policy's collective layer pass on each in its own thread, and gather the result.
192    ///
193    /// One `std::thread::scope` per call with one thread per partition, which is what the in-process transport's blocking collectives need: every rank has to be able to make progress independently.
194    fn partitioned_finalize<const W: usize, T>(
195        policy: &T,
196        sum: &PauliSum<W>,
197        pbits: u8,
198    ) -> PauliSum<W>
199    where
200        T: PartitionedTruncation<W>,
201    {
202        let rows = PartitionRows::<W>::from_seed(sum.num_qubits(), pbits, PSEED);
203        let p = rows.num_partitions() as u32;
204        let parts: Vec<PauliSum<W>> = (0..p).map(|r| sum.filter_partition(&rows, r)).collect();
205        assert_eq!(
206            parts.iter().map(PauliSum::len).sum::<usize>(),
207            sum.len(),
208            "the split must be a partition of the terms",
209        );
210
211        let group = InProcessTransport::group(p);
212        let gathered: Vec<PauliSum<W>> = std::thread::scope(|scope| {
213            let handles: Vec<_> = parts
214                .into_iter()
215                .zip(group)
216                .map(|(mut local, transport)| {
217                    scope.spawn(move || {
218                        policy.finalize_layer_partitioned(&mut local, &transport);
219                        (transport.rank(), local)
220                    })
221                })
222                .collect();
223            let mut slots: Vec<Option<PauliSum<W>>> = (0..p).map(|_| None).collect();
224            for handle in handles {
225                let (rank, local) = handle.join().expect("partition thread panicked");
226                slots[rank as usize] = Some(local);
227            }
228            slots.into_iter().map(Option::unwrap).collect()
229        });
230
231        for local in &gathered {
232            local.assert_invariants();
233        }
234        let merged = PauliSum::<W>::merge_partitions(gathered);
235        merged.assert_invariants();
236        merged
237    }

239    /// The whole point: partitioned finalization is *exactly* the single-partition one, for every `P`.
240    fn assert_matches_single_partition<const W: usize, T>(
241        policy: &T,
242        input: &PauliSum<W>,
243        pbits: u8,
244        what: &str,
245    ) where
246        T: PartitionedTruncation<W>,
247    {
248        let mut want = input.clone();
249        policy.finalize_layer(&mut want);
250
251        let got = partitioned_finalize(policy, input, pbits);
252        assert_eq!(
253            got.len(),
254            want.len(),
255            "{what}: partitioned kept {} terms, single-partition {}",
256            got.len(),
257            want.len(),
258        );
259        assert_same_terms(&got, &want, what);
260    }
