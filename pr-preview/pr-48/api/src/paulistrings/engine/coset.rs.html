1//! `Gf2Span`, the GF(2) span of a channel's bucket deltas and the coset index algebra over it, and `CosetOrder`, the coset-contiguous bucket order a layer runs in.
2//!
3//! The bucket hash is linear, so output bucket `β′` reads only the input buckets `β′ ⊕ h(D)` for a channel's key deltas `D`.
4//! A coset of the span of `h(D)` is therefore a closed task: its output buckets read only its own input buckets and no other coset writes them, which is why a layer needs no synchronization inside it.
5//! The span rather than `h(D)` itself, since a custom channel's delta set need not be XOR-closed.
6//! Where several deltas share a bucket delta the gather walks them all, so correctness never depends on the draw of the hash rows.
7
8use rayon::prelude::*;
9
10use crate::pauli_sum::hash::B_MAX_BITS;

12/// Positions from which [`CosetOrder::rebuild`] tabulates on the Rayon pool: the serial table costs tens of ns per position, a fork-join tens of µs.
13pub(crate) const PARALLEL_REBUILD_MIN_POSITIONS: usize = 4096;
14
15/// Highest set bit of a nonzero word.
16fn highest_bit(v: u32) -> u32 {
17    debug_assert!(v != 0, "highest_bit: zero has no highest set bit");
18    31 - v.leading_zeros()
19}
20
21/// Software `pext`, since BMI2 is not assumed and this runs off any inner loop.
22fn pext(value: u32, mask: u32) -> u32 {
23    let mut out = 0u32;
24    let mut out_bit = 0u32;
25    let mut remaining = mask;
26    while remaining != 0 {
27        let bit = remaining & remaining.wrapping_neg();
28        if value & bit != 0 {
29            out |= 1 << out_bit;
30        }
31        out_bit += 1;
32        remaining ^= bit;
33    }
34    out
35}
36
37/// Software `pdep`, the inverse of [`pext`] on `mask`.
38fn pdep(value: u32, mask: u32) -> u32 {
39    let mut out = 0u32;
40    let mut in_bit = 0u32;
41    let mut remaining = mask;
42    while remaining != 0 {
43        let bit = remaining & remaining.wrapping_neg();
44        if value & (1 << in_bit) != 0 {
45            out |= bit;
46        }
47        in_bit += 1;
48        remaining ^= bit;
49    }
50    out
51}

53/// The span of a channel's bucket deltas, which need not be XOR-closed, and the map between a bucket index and its `(coset, member)` coordinates.
54// `basis` is reduced row echelon, ascending by pivot (its highest set bit), and each pivot bit is set in exactly one vector; bit `j` of a member index selects `basis[j]`.
55#[derive(Clone, Debug)]
56pub(crate) struct Gf2Span {
57    /// Reduced echelon basis, ascending by pivot bit.
58    basis: Vec<u32>,
59    /// OR of the pivot bits — one bit per basis vector.
60    pivot_mask: u32,
61    /// Bucket-index width: indices live in `0..2^bits`.
62    bits: u8,
63    /// The bucket-space bits outside `pivot_mask`: the free bits a representative ranges over.
64    nonpivot_mask: u32,
65}
66
67impl Gf2Span {
68    /// The span of `deltas` in a `bits`-wide index space; panics if `bits > B_MAX_BITS` or a delta lies outside it.
69    pub(crate) fn new(deltas: &[u32], bits: u8) -> Self {
70        assert!(bits <= B_MAX_BITS, "Gf2Span: bits {bits} exceeds B_MAX_BITS {B_MAX_BITS}");
71        let space_mask = if bits == 0 { 0 } else { (1u32 << bits) - 1 };
72
73        let mut basis: Vec<u32> = Vec::new();
74        let mut pivot_mask = 0u32;
75
76        for &delta in deltas {
77            assert!(
78                delta & !space_mask == 0,
79                "Gf2Span: delta {delta} has bits outside the {bits}-bit bucket space"
80            );
81            // Pivot bits are disjoint across basis vectors, so one pass in any order clears them all.
82            let mut reduced = delta;
83            for &vector in &basis {
84                if reduced & (1 << highest_bit(vector)) != 0 {
85                    reduced ^= vector;
86                }
87            }
88            if reduced == 0 {
89                continue;
90            }
91            let pivot = highest_bit(reduced);
92            // Back-substitute so the new pivot is unique to one vector; only vectors with a higher pivot can carry bit `p`.
93            for vector in basis.iter_mut() {
94                if *vector & (1 << pivot) != 0 {
95                    *vector ^= reduced;
96                }
97            }
98            let position = basis.partition_point(|&vector| highest_bit(vector) < pivot);
99            basis.insert(position, reduced);
100            pivot_mask |= 1 << pivot;
101        }
102
103        Self { basis, pivot_mask, bits, nonpivot_mask: space_mask & !pivot_mask }
104    }
105
106    /// Dimension `r` of the span.
107    pub(crate) fn dimension(&self) -> usize {
108        self.basis.len()
109    }
110
111    /// Buckets per coset, `2^r`.
112    pub(crate) fn coset_size(&self) -> usize {
113        1usize << self.basis.len()
114    }
115
116    /// Number of cosets, `2^bits / 2^r`.
117    pub(crate) fn num_cosets(&self) -> usize {
118        (1usize << self.bits) >> self.basis.len()
119    }
120
121    /// Whether `beta` is its coset's representative: the unique member with every pivot bit clear, which is also the coset's minimum.
122    fn is_representative(&self, beta: u32) -> bool {
123        beta & self.pivot_mask == 0
124    }
125
126    /// The representative of `beta`'s coset.
127    // Not `beta & !pivot_mask`: basis vectors carry non-pivot bits too, so masking can leave the coset.
128    pub(crate) fn representative_of(&self, beta: u32) -> u32 {
129        debug_assert!(
130            beta >> self.bits == 0,
131            "Gf2Span::representative_of: beta outside the bucket space"
132        );
133        let mut reduced = beta;
134        for &vector in &self.basis {
135            if reduced & (1 << highest_bit(vector)) != 0 {
136                reduced ^= vector;
137            }
138        }
139        reduced
140    }
141
142    /// The member index of `delta` in the span, read off its pivot bits since the basis is reduced.
143    pub(crate) fn coordinate_of(&self, delta: u32) -> u32 {
144        debug_assert!(
145            self.representative_of(delta) == 0,
146            "Gf2Span::coordinate_of: delta {delta} is not in the span"
147        );
148        pext(delta, self.pivot_mask)
149    }
150
151    /// The position of `representative` among all representatives in ascending order.
152    fn rank_of_representative(&self, representative: u32) -> u32 {
153        debug_assert!(
154            self.is_representative(representative),
155            "Gf2Span::rank_of_representative: {representative} is not a representative"
156        );
157        pext(representative, self.nonpivot_mask)
158    }
159
160    /// The representative of coset `coset`, the `coset`-th smallest, so its members are positions `coset << r ..` of [`Self::permuted_index`].
161    pub(crate) fn representative_at(&self, coset: u32) -> u32 {
162        debug_assert!(
163            (coset as usize) < self.num_cosets(),
164            "Gf2Span::representative_at: coset {coset} out of range"
165        );
166        pdep(coset, self.nonpivot_mask)
167    }
168
169    /// Member `i` of the coset with representative `rep`; bit `j` of `i` selects `basis[j]`.
170    pub(crate) fn member(&self, rep: u32, i: u32) -> u32 {
171        debug_assert!(
172            (i as usize) < self.coset_size(),
173            "Gf2Span::member: index {i} beyond coset size"
174        );
175        let mut v = rep;
176        for (j, &b) in self.basis.iter().enumerate() {
177            if (i >> j) & 1 == 1 {
178                v ^= b;
179            }
180        }
181        v
182    }
183
184    /// `beta` renumbered so coset `c` owns `c << r .. (c + 1) << r`, with the member coordinate in the low `r` bits.
185    pub(crate) fn permuted_index(&self, beta: u32) -> u32 {
186        let representative = self.representative_of(beta);
187        (self.rank_of_representative(representative) << self.basis.len())
188            | self.coordinate_of(beta ^ representative)
189    }
190}

192/// A layer's coset-contiguous bucket order: coset `c` owns positions `c << r .. (c + 1) << r` ([`Gf2Span::permuted_index`]).
193///
194/// Built once per layer and shared by the coset loop, the export pass and the exchange's chunking; empty, i.e. the identity, when `r = 0`.
195#[derive(Clone, Debug, Default)]
196pub(crate) struct CosetOrder {
197    /// `permutation[β]` is bucket `β`'s position.
198    permutation: Vec<u32>,
199    /// `inverse[p]` is the bucket at position `p`; built only on request.
200    inverse: Vec<u32>,
201    positions: u32,
202    dimension: u32,
203}
204
205/// Refill `table` with `f(0..len)`, on the Rayon pool when `parallel`.
206fn tabulate(table: &mut Vec<u32>, len: u32, parallel: bool, f: impl Fn(u32) -> u32 + Sync + Send) {
207    if parallel {
208        (0..len)
209            .into_par_iter()
210            .with_min_len(PARALLEL_REBUILD_MIN_POSITIONS / 8)
211            .map(f)
212            .collect_into_vec(table);
213    } else {
214        table.clear();
215        table.extend((0..len).map(f));
216    }
217}
218
219impl CosetOrder {
220    /// Re-aim the order at `span`, keeping the allocations; the inverse is built only when `with_inverse`.
221    pub(crate) fn rebuild(&mut self, span: &Gf2Span, with_inverse: bool) {
222        self.positions = 1u32 << span.bits;
223        self.dimension = span.dimension() as u32;
224        self.permutation.clear();
225        self.inverse.clear();
226        if self.dimension == 0 {
227            return;
228        }
229        let parallel = self.positions as usize >= PARALLEL_REBUILD_MIN_POSITIONS;
230        tabulate(&mut self.permutation, self.positions, parallel, |beta| span.permuted_index(beta));
231        if with_inverse {
232            // Read off the span rather than scattered from `permutation`, so the table fills in parallel.
233            let r = self.dimension;
234            let member_mask = (1u32 << r) - 1;
235            tabulate(&mut self.inverse, self.positions, parallel, |position| {
236                span.member(span.representative_at(position >> r), position & member_mask)
237            });
238        }
239    }
240
241    /// Bucket `beta`'s position.
242    pub(crate) fn position_of(&self, beta: u32) -> u32 {
243        if self.permutation.is_empty() {
244            beta
245        } else {
246            self.permutation[beta as usize]
247        }
248    }
249
250    /// The bucket at position `position`, the inverse of [`Self::position_of`]; the order must have been built `with_inverse`.
251    pub(crate) fn bucket_at(&self, position: u32) -> u32 {
252        debug_assert!(
253            self.inverse.len() == self.permutation.len(),
254            "CosetOrder::bucket_at: the inverse was not built",
255        );
256        if self.inverse.is_empty() {
257            position
258        } else {
259            self.inverse[position as usize]
260        }
261    }
262
263    /// Positions, i.e. buckets.
264    pub(crate) fn positions(&self) -> usize {
265        self.positions as usize
266    }
267
268    /// `r`, the `log2` of the coset size.
269    pub(crate) fn dimension(&self) -> usize {
270        self.dimension as usize
271    }
272
273    /// Capacity of the permutation, for the footprint tests.
274    #[cfg(test)]
275    pub(crate) fn capacity(&self) -> usize {
276        self.permutation.capacity() + self.inverse.capacity()
277    }
278}
