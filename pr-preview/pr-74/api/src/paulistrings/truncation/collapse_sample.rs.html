15/// Whenever the sum holds more than `cache` terms after a layer, replace it by `samples` systematic draws in proportion to `|c|`, an unbiased estimate of it.
16///
17/// With `‖O‖₁ = Σ|c|`, the terms are laid end to end in iteration order, term `i` owning the half-open stretch `[S_i, S_i + |c_i|)` of `[0, ‖O‖₁)`; one `u ∈ [0, 1)` per collapse places the points `t_k = (u + k)·h` for `k < samples`, with `h = ‖O‖₁ / samples`.
18/// A term holding `m ≥ 1` points survives with coefficient `m·h·c/|c|`, every other term is dropped, and so the expectation of the result is the sum.
19/// A point on a boundary belongs to the later stretch, a point that rounding puts at or past `‖O‖₁` to the last term with a nonzero stretch, and a zero coefficient is never drawn.
20/// Every term is drawn `⌊samples·|c|/‖O‖₁⌋` or `⌈…⌉` times up to rounding, so the result holds at most `min(samples, len)` terms; a collapse resamples even when `samples` is at least the length.
21/// `‖O‖₁`, the stretches and the points are `f64` whatever the coefficient type.
22///
23/// One seed is one Monte Carlo trajectory: `u` at the `k`-th layer pass is a function of `(seed, k)` only, and the pass counter lives in the policy, so reusing a policy continues its sequence.
24/// Partitioned, the stretches run in rank order and then bucket order, and each partition takes exactly the points that fall in its own: the group draws `samples` points, every term's multiplicity has its unpartitioned law, and a partition with no point clears; passes and collapses count on rank 0's policy only.
25/// Inside a [`PerPartition`] each partition instead draws `samples` points from its own slice whenever that slice exceeds `cache`.
26/// A collapse panics if `‖O‖₁` is zero or not finite.
27///
28/// A lone partition may apply a run of local layers (consecutive one-to-one channels, or rotations the engine joins) as one pass and check the cache once, at the run's end, while the run's worst case, its input length times the product of its layers' fanouts, stays within [`RUN_GROWTH`](Self::RUN_GROWTH)` × cache`.
29/// Every collapse is unbiased given the sum it draws from, so every schedule of them is, and the run's estimate keeps the expectation of the layer-by-layer one; fewer passes take fewer pass indices, so the trajectory a seed names depends on the runs taken.
30#[derive(Debug)]
31pub struct CollapseSample {
32    /// Largest sum left untouched.
33    pub cache: usize,
34    /// Points drawn per collapse, at least one.
35    pub samples: usize,
36    /// Trajectory seed.
37    pub seed: u64,
38    /// Layer passes seen.
39    calls: AtomicU64,
40    /// Collapses performed.
41    collapses: AtomicU64,
42}
43
44impl CollapseSample {
45    /// How far past `cache` a run of local layers may grow, worst case, before its one collapse check.
46    pub const RUN_GROWTH: usize = 8;
47
48    /// A fresh trajectory: above `cache` terms, collapse to `samples` draws keyed by `seed`.
49    ///
50    /// # Panics
51    ///
52    /// If `samples` is zero.
53    pub fn new(cache: usize, samples: usize, seed: u64) -> Self {
54        assert!(samples >= 1, "CollapseSample: samples must be at least 1");
55        Self { cache, samples, seed, calls: AtomicU64::new(0), collapses: AtomicU64::new(0) }
56    }
57
58    /// Collapses performed so far (rank 0's count in partitioned mode).
59    pub fn collapses(&self) -> u64 {
60        self.collapses.load(Ordering::Relaxed)
61    }
62
63    /// Claim the next pass index.
64    pub(crate) fn next_call(&self) -> u64 {
65        self.calls.fetch_add(1, Ordering::Relaxed)
66    }
67
68    pub(crate) fn count_collapse(&self) -> u64 {
69        self.collapses.fetch_add(1, Ordering::Relaxed) + 1
70    }
71
72    /// The offset `u` of pass `call`'s points.
73    pub(crate) fn uniform(&self, call: u64) -> f64 {
74        Rng::from_key(&[self.seed, call]).uniform()
75    }
76}
77
78/// Panics unless `total`, the `‖O‖₁` of `len` terms, is positive and finite.
79pub(crate) fn assert_drawable(total: f64, len: usize) {
80    assert!(
81        total.is_finite() && total > 0.0,
82        "CollapseSample: cannot draw by |c| from {len} terms with Σ|c| = {total}",
83    );
84}
85
86/// `values` summed left to right from `0.0`, the order every stretch boundary of a collapse is accumulated in.
87pub(crate) fn running_sum(values: impl IntoIterator<Item = f64>) -> f64 {
88    values.into_iter().fold(0.0, |total, value| total + value)
89}

91/// `Σ|c|` of each bucket, in bucket order.
92pub(crate) fn bucket_l1_norms<const W: usize, C: Coefficient>(sum: &PauliSum<W, C>) -> Vec<f64> {
93    (0..sum.num_buckets())
94        .into_par_iter()
95        .map(|b| running_sum(sum.bucket(b).2.iter().map(|c| c.magnitude())))
96        .collect()
97}

99/// The points `t_k = (u + k)·h`, `k < samples`, of one collapse over `[0, total)`.
100#[derive(Clone, Copy, Debug)]
101pub(crate) struct SystematicPoints {
102    pub(crate) u: f64,
103    pub(crate) samples: usize,
104    pub(crate) total: f64,
105}
106
107impl SystematicPoints {
108    fn step(&self) -> f64 {
109        self.total / self.samples as f64
110    }
111
112    fn point(&self, k: usize) -> f64 {
113        (self.u + k as f64) * self.step()
114    }
115
116    /// How many points lie below `boundary`, all of them once it reaches `total`; the points are nondecreasing in `k`.
117    fn below(&self, boundary: f64) -> usize {
118        if boundary >= self.total {
119            return self.samples;
120        }
121        let (mut lo, mut hi) = (0, self.samples);
122        while lo < hi {
123            let mid = lo + (hi - lo) / 2;
124            if self.point(mid) < boundary {
125                lo = mid + 1;
126            } else {
127                hi = mid;
128            }
129        }
130        lo
131    }
132}

134/// `weight · c/|c|` in `f64`, given `magnitude = |c| > 0`.
135fn drawn<C: Coefficient>(c: C, magnitude: f64, weight: f64) -> C {
136    C::from_complex64(c.to_complex64() / magnitude * weight, f64::INFINITY)
137        .expect("a real coefficient scaled by a real factor stays real")
138}

140/// The systematic resample [`CollapseSample`] documents, of a sum whose stretches start at `offset` of `points.total`; `norms` is [`bucket_l1_norms`] of `sum`.
141pub(crate) fn systematic_resample<const W: usize, C: Coefficient>(
142    sum: &mut PauliSum<W, C>,
143    norms: &[f64],
144    offset: f64,
145    points: SystematicPoints,
146) {
147    let mut starts = Vec::with_capacity(norms.len() + 1);
148    let mut prefix = 0.0f64;
149    for &norm in norms {
150        starts.push(prefix);
151        prefix += norm;
152    }
153    starts.push(prefix);
154    // Both passes form a boundary as `offset + (bucket start + running Σ|c|)`, so a bucket's last stretch ends exactly where the next bucket's first begins.
155    let first: Vec<usize> = starts.par_iter().map(|&start| points.below(offset + start)).collect();
156    let step = points.step();
157    sum.update_per_bucket(|b| {
158        let (bucket_start, end) = (starts[b], first[b + 1]);
159        let mut next = first[b];
160        let mut running = 0.0f64;
161        move |_: &[u64; W], _: &[u64; W], c: C| {
162            if next == end {
163                return None;
164            }
165            let magnitude = c.magnitude();
166            running += magnitude;
167            let boundary = offset + (bucket_start + running);
168            let mut stop = end;
169            if boundary < points.total {
170                stop = next;
171                while stop < end && points.point(stop) < boundary {
172                    stop += 1;
173                }
174            }
175            let multiplicity = stop - next;
176            next = stop;
177            (multiplicity > 0).then(|| drawn(c, magnitude, multiplicity as f64 * step))
178        }
179    });
180}

182/// Equal when `cache`, `samples`, `seed` and both counters are.
183impl PartialEq for CollapseSample {
184    fn eq(&self, other: &Self) -> bool {
185        self.cache == other.cache
186            && self.samples == other.samples
187            && self.seed == other.seed
188            && self.calls.load(Ordering::Relaxed) == other.calls.load(Ordering::Relaxed)
189            && self.collapses() == other.collapses()
190    }
191}

193impl<const W: usize, C: Coefficient> TruncationPolicy<W, C> for CollapseSample {
194    fn finalize_layer(&self, sum: &mut PauliSum<W, C>) {
195        let call = self.next_call();
196        let len = sum.len();
197        if len <= self.cache {
198            return;
199        }
200        let norms = bucket_l1_norms(sum);
201        let total = running_sum(norms.iter().copied());
202        assert_drawable(total, len);
203        let points = SystematicPoints { u: self.uniform(call), samples: self.samples, total };
204        systematic_resample(sum, &norms, 0.0, points);
205        let n = self.count_collapse();
206        log::debug!(
207            target: LOG_TARGET,
208            "collapse_sample: {len} terms, sum |c| = {total:.6e}, {} draws kept {} (collapse {n})",
209            self.samples,
210            sum.len(),
211        );
212    }
213
214    /// Reduces `[len, pass]` (the pass from rank 0) and the per-partition `Σ|c|`, then resamples this partition's stretch of the group's points, as [`finalize_layer`](TruncationPolicy::finalize_layer) resamples each bucket's.
215    fn finalize_layer_partitioned(
216        &self,
217        local: &mut PauliSum<W, C>,
218        collectives: &dyn Collectives,
219    ) {
220        if collectives.size() == 1 {
221            self.finalize_layer(local);
222            return;
223        }
224        let rank = collectives.rank() as usize;
225        let call = if rank == 0 { self.next_call() } else { 0 };
226        let mut head = [local.len() as u64, call];
227        collectives.allreduce_sum_u64(&mut head);
228        let [len, call] = head;
229        if len as usize <= self.cache {
230            return;
231        }
232
233        let norms = bucket_l1_norms(local);
234        let mut weights = vec![0.0f64; collectives.size() as usize];
235        weights[rank] = running_sum(norms.iter().copied());
236        collectives.allreduce_sum_f64(&mut weights);
237        let offset = running_sum(weights[..rank].iter().copied());
238        let total = running_sum(weights.iter().copied());
239        assert_drawable(total, len as usize);
240        let points = SystematicPoints { u: self.uniform(call), samples: self.samples, total };
241        systematic_resample(local, &norms, offset, points);
242        if rank == 0 {
243            let collapses = self.count_collapse();
244            log::debug!(
245                target: LOG_TARGET,
246                "collapse_sample: {len} terms over {} partitions, sum |c| = {total:.6e}, \
247                 {} draws (collapse {collapses})",
248                collectives.size(),
249                self.samples,
250            );
251        }
252    }
253
254    fn supports_partitioned(&self) -> bool {
255        true
256    }
257
258    fn run_length_limit(&self) -> usize {
259        self.cache.saturating_mul(Self::RUN_GROWTH)
260    }
261}
