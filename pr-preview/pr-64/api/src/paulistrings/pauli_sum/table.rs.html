1//! [`Table`], an accumulating hash table of terms: a dense entry list plus a linear-probing index of positions into it, and the in-bucket key hash.
2//!
3//! Entries are dense in `0..len` in insertion order and a key is never removed, so a position stays valid until [`Table::clear`].
4
5use crate::coefficient::Coefficient;
6
7/// One term.
8#[derive(Clone, Copy, Debug)]
9pub(crate) struct Entry<const W: usize, C: Coefficient> {
10    pub(crate) x: [u64; W],
11    pub(crate) z: [u64; W],
12    pub(crate) c: C,
13}
14
15/// The 64×64→128 multiply folded to 64 bits.
16#[inline(always)]
17fn folded_multiply(a: u64, b: u64) -> u64 {
18    let product = (a as u128) * (b as u128);
19    (product as u64) ^ ((product >> 64) as u64)
20}
21
22const KEY_SEEDS: [u64; 4] =
23    [0x243F_6A88_85A3_08D3, 0x1319_8A2E_0370_7344, 0xA409_3822_299F_31D0, 0x082E_FA98_EC4E_6C89];
24
25/// The in-bucket key hash; the bucket index is a GF(2)-linear function of the key, so this one must be nonlinear in every bit.
26#[inline(always)]
27pub(crate) fn key_hash<const W: usize>(x: &[u64; W], z: &[u64; W]) -> u64 {
28    let mut h = 0u64;
29    for w in 0..W {
30        h = h.rotate_left(29)
31            ^ folded_multiply(x[w] ^ KEY_SEEDS[(2 * w) % 4], z[w] ^ KEY_SEEDS[(2 * w + 1) % 4]);
32    }
33    h
34}
35
36const EMPTY: u8 = 0;
37
38/// Live entries may fill at most `MAX_LOAD_NUMERATOR / 8` of the slots before a rebuild.
39const MAX_LOAD_NUMERATOR: usize = 7;
40
41/// Tags probed at once; `tags` mirrors its first `GROUP` bytes past the end, so a group read never wraps.
42const GROUP: usize = 16;
43
44const MIN_CAPACITY: usize = GROUP;
45
46/// Bit `k` set iff byte `k` of the group at `tags[0..GROUP]` equals `value`.
47#[inline(always)]
48fn match_byte(tags: &[u8], value: u8) -> u32 {
49    debug_assert!(tags.len() >= GROUP);
50    #[cfg(target_arch = "x86_64")]
51    {
52        use core::arch::x86_64::{
53            _mm_cmpeq_epi8, _mm_loadu_si128, _mm_movemask_epi8, _mm_set1_epi8,
54        };
55        // SAFETY: SSE2 is part of the x86_64 baseline, and the slice holds `GROUP` readable bytes.
56        unsafe {
57            let group = _mm_loadu_si128(tags.as_ptr().cast());
58            _mm_movemask_epi8(_mm_cmpeq_epi8(group, _mm_set1_epi8(value as i8))) as u32
59        }
60    }
61    #[cfg(not(target_arch = "x86_64"))]
62    {
63        (0..GROUP).fold(0, |bits, k| bits | (((tags[k] == value) as u32) << k))
64    }
65}
66
67#[inline(always)]
68fn tag_of(hash: u64) -> u8 {
69    0x80 | (hash >> 57) as u8
70}
71
72/// An accumulating table of terms with pairwise distinct keys.
73#[derive(Clone, Debug, Default)]
74pub(crate) struct Table<const W: usize, C: Coefficient> {
75    entries: Vec<Entry<W, C>>,
76    tags: Vec<u8>,
77    positions: Vec<u32>,
78}
79
80impl<const W: usize, C: Coefficient> Table<W, C> {
81    fn capacity_for(n: usize) -> usize {
82        (n * 8).div_ceil(MAX_LOAD_NUMERATOR).next_power_of_two().max(MIN_CAPACITY)
83    }
84
85    pub(crate) fn entries(&self) -> &[Entry<W, C>] {
86        &self.entries
87    }
88
89    /// Forget every entry, keeping the allocations.
90    pub(crate) fn clear(&mut self) {
91        self.entries.clear();
92        self.tags.fill(EMPTY);
93    }
94
95    #[inline(always)]
96    fn mask(&self) -> usize {
97        self.positions.len() - 1
98    }
99
100    /// Make room for `additional` pushes without an index rebuild in between.
101    #[inline(always)]
102    pub(crate) fn reserve(&mut self, additional: usize) {
103        let needed = self.entries.len() + additional;
104        if needed > self.positions.len() * MAX_LOAD_NUMERATOR / 8 {
105            self.rebuild(Self::capacity_for(needed + needed / 8));
106        }
107        self.entries.reserve(additional);
108    }
109
110    /// Re-index every entry at `capacity` index slots.
111    #[cold]
112    fn rebuild(&mut self, capacity: usize) {
113        debug_assert!(capacity.is_power_of_two());
114        self.tags.clear();
115        self.tags.resize(capacity + GROUP, EMPTY);
116        self.positions.clear();
117        self.positions.resize(capacity, 0);
118        for position in 0..self.entries.len() {
119            let e = &self.entries[position];
120            let hash = key_hash(&e.x, &e.z);
121            let slot = self.free_slot(hash);
122            self.claim(slot, hash, position);
123        }
124    }
125
126    /// The first empty index slot on `hash`'s probe sequence.
127    #[inline(always)]
128    fn free_slot(&self, hash: u64) -> usize {
129        let mask = self.mask();
130        let mut base = hash as usize & mask;
131        loop {
132            let free = match_byte(&self.tags[base..base + GROUP], EMPTY);
133            if free != 0 {
134                return (base + free.trailing_zeros() as usize) & mask;
135            }
136            base = (base + GROUP) & mask;
137        }
138    }
139
140    #[inline(always)]
141    fn claim(&mut self, slot: usize, hash: u64, position: usize) {
142        debug_assert_eq!(self.tags[slot], EMPTY);
143        let tag = tag_of(hash);
144        self.tags[slot] = tag;
145        // The mirror of slots `0..GROUP`; for any other slot this rewrites the slot itself.
146        let mirror = ((slot.wrapping_sub(GROUP)) & self.mask()) + GROUP;
147        self.tags[mirror] = tag;
148        self.positions[slot] = position as u32;
149    }
150
151    /// Add `c` to the coefficient of key `(x, z)`, inserting the key when absent.
152    #[inline(always)]
153    pub(crate) fn accumulate(&mut self, x: [u64; W], z: [u64; W], c: C) {
154        self.reserve(1);
155        let hash = key_hash(&x, &z);
156        let tag = tag_of(hash);
157        let mask = self.mask();
158        let mut base = hash as usize & mask;
159        loop {
160            let group = &self.tags[base..base + GROUP];
161            let empty = match_byte(group, EMPTY);
162            // Linear probing with no removal stops at the first empty slot, so only matches before it count.
163            let mut candidates =
164                match_byte(group, tag) & (empty & empty.wrapping_neg()).wrapping_sub(1);
165            while candidates != 0 {
166                let slot = (base + candidates.trailing_zeros() as usize) & mask;
167                let e = &mut self.entries[self.positions[slot] as usize];
168                if e.x == x && e.z == z {
169                    e.c += c;
170                    return;
171                }
172                candidates &= candidates - 1;
173            }
174            if empty != 0 {
175                let slot = (base + empty.trailing_zeros() as usize) & mask;
176                let position = self.entries.len();
177                self.claim(slot, hash, position);
178                self.entries.push(Entry { x, z, c });
179                return;
180            }
181            base = (base + GROUP) & mask;
182        }
183    }
184}
