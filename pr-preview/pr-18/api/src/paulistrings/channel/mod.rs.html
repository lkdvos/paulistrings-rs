1//! [`Channel<W>`] — unified abstraction for gates and noise.
2//!
3//! Built-ins: [`Clifford1Q`], [`Clifford2Q`], [`PauliRotation`], [`GeneralUnitary1Q`], [`GeneralUnitary2Q`], [`Depolarizing`], [`Dephasing`], [`PauliChannel`], [`Depolarizing2Q`], [`AmplitudeDamping`], [`IdentityChannel`]. See ARCHITECTURE.md §Channels.
4//!
5//! # Implementing a custom channel
6//!
7//! Implement the trait directly; the engine treats your type as just another `Box<dyn `[`Channel<W>`]`>` inside a [`Circuit`].
8//!
9//! ```
10//! use paulistrings::{Channel, OutputBuffer};
11//! use num_complex::Complex64;
12//!
13//! /// Multiplies every input coefficient by a complex factor, with no
14//! /// support and `MAX_FANOUT = 1`.
15//! struct GlobalPhase {
16//!     factor: Complex64,
17//! }
18//!
19//! impl<const W: usize> Channel<W> for GlobalPhase {
20//!     fn max_fanout(&self) -> usize { 1 }
21//!     fn support(&self) -> [u64; W] { [0; W] }
22//!     fn apply(
23//!         &self,
24//!         input_x: &[u64; W],
25//!         input_z: &[u64; W],
26//!         coeff: Complex64,
27//!         out: &mut OutputBuffer<'_, W>,
28//!     ) {
29//!         out.push(*input_x, *input_z, coeff * self.factor);
30//!     }
31//! }
32//!
33//! let ch = GlobalPhase {
34//!     factor: Complex64::new(0.0, 1.0),
35//! };
36//! let _: Box<dyn Channel<1>> = Box::new(ch);
37//! ```
38//!
39//! [`PauliSum`]: crate::PauliSum
40//! [`engine`]: crate::engine
41//! [`Circuit`]: crate::Circuit

60/// Pre-allocated, fixed-capacity SoA scratch buffer for channel outputs.
61///
62/// Sized by the engine to `n_in · channel.max_fanout()` so `apply` writes without dynamic growth. Channel impls write via [`OutputBuffer::push`].
63pub struct OutputBuffer<'a, const W: usize> {
64    /// X-part column. Length equals the buffer's capacity.
65    pub x: &'a mut [[u64; W]],
66    /// Z-part column. Length equals the buffer's capacity.
67    pub z: &'a mut [[u64; W]],
68    /// Coefficient column. Length equals the buffer's capacity.
69    pub coeff: &'a mut [Complex64],
70    /// Cursor into the slices; `apply` writes at `len` and advances.
71    pub len: &'a mut usize,
72}
73
74impl<'a, const W: usize> OutputBuffer<'a, W> {
75    /// Append one term to the buffer at the current cursor.
76    ///
77    /// Capacity is `self.x.len()`; a `Channel::apply` body must not push more than its declared `max_fanout`. Out-of-range writes are caught by slice bounds-checking (and, in debug builds, an explicit assertion).
78    #[inline]
79    pub fn push(&mut self, x: [u64; W], z: [u64; W], c: Complex64) {
80        debug_assert!(
81            *self.len < self.x.len(),
82            "OutputBuffer overflow: {} pushes into a buffer of capacity {}",
83            *self.len + 1,
84            self.x.len()
85        );
86        let i = *self.len;
87        self.x[i] = x;
88        self.z[i] = z;
89        self.coeff[i] = c;
90        *self.len = i + 1;
91    }
92
93    /// Reset the cursor to zero so the same backing storage can be reused for the next input term without reallocation.
94    #[inline]
95    pub fn clear(&mut self) {
96        *self.len = 0;
97    }
98}

100/// Pack a list of qubit indices into a [`Channel::support`] bitmask.
101///
102/// Bit `q % 64` of word `q / 64` is set for each `q` in `qubits`; order and duplicates do not matter.
103#[inline]
104pub fn support_mask<const W: usize>(qubits: &[u32]) -> [u64; W] {
105    let mut mask = [0u64; W];
106    for &q in qubits {
107        debug_assert!((q as usize) < 64 * W, "qubit {q} out of range for W={W}");
108        mask[q as usize / 64] |= 1u64 << (q % 64);
109    }
110    mask
111}

113// ---- Shared per-qubit bit extract/insert idiom ----
114//
115// Every built-in `Channel::apply`/`apply_adjoint` reads and/or overwrites the two bit-planes (`x`, `z`) at one or two support qubits; these four helpers are the common core, private to this module and its descendants.
116
117/// Decompose qubit index `q` into `(word, bit, mask)`: the index into a `[u64; W]` array, the bit position within that word, and `1u64 << bit`.
118#[inline(always)]
119fn qubit_loc(q: usize) -> (usize, usize, u64) {
120    let word = q / 64;
121    let bit = q % 64;
122    (word, bit, 1u64 << bit)
123}

125/// Read the packed single-qubit Pauli index (`x | (z << 1)`, i.e. `I=0, X=1, Z=2, Y=3` — the convention [`Clifford1Q`] and [`GeneralUnitary1Q`] both document) of the qubit at `(word, bit)`.
126#[inline(always)]
127fn read_pauli<const W: usize>(x: &[u64; W], z: &[u64; W], word: usize, bit: usize) -> usize {
128    let x_bit = (x[word] >> bit) & 1;
129    let z_bit = (z[word] >> bit) & 1;
130    (x_bit | (z_bit << 1)) as usize
131}
132
133/// Overwrite the qubit at `(word, bit, mask)` of `(nx, nz)` with the packed Pauli index `p` (same `x | (z << 1)` encoding as [`read_pauli`]).
134#[inline(always)]
135fn write_pauli<const W: usize>(
136    nx: &mut [u64; W],
137    nz: &mut [u64; W],
138    word: usize,
139    bit: usize,
140    mask: u64,
141    p: usize,
142) {
143    let ox = (p & 1) as u64;
144    let oz = ((p >> 1) & 1) as u64;
145    nx[word] = (nx[word] & !mask) | (ox << bit);
146    nz[word] = (nz[word] & !mask) | (oz << bit);
147}
148
149/// Set (`value = true`) or clear (`value = false`) the single bit at `(word, mask)` of one bit-plane array. Used where only one of `x`/`z` changes, e.g. amplitude damping's `I ↔ Z` fan-out.
150#[inline(always)]
151fn set_bit<const W: usize>(arr: &mut [u64; W], word: usize, mask: u64, value: bool) {
152    if value {
153        arr[word] |= mask;
154    } else {
155        arr[word] &= !mask;
156    }
157}

159/// Anything that maps a Pauli string to a small weighted sum of Pauli strings.
160///
161/// [`Channel::max_fanout`] is a method, not an associated `const`, so the trait stays `dyn`-compatible — [`Circuit`](crate::Circuit) stores `Box<dyn Channel<W>>` to keep the channel set open for user extensions.
162///
163/// See the [module-level docs](self) for an `impl Channel` example.
164pub trait Channel<const W: usize>: Send + Sync {
165    /// Maximum number of output terms produced per input term. Used by the engine to size the scratch buffer up-front.
166    fn max_fanout(&self) -> usize;
167
168    /// Qubits this channel acts on, packed as one combined per-qubit bitmask (bit `q` set iff qubit `q` is in the support), one word per `W`.
169    /// Outputs differ from inputs only at these bit positions; the engine uses this for bucket layout (ARCHITECTURE.md §Bucketing). Build one with [`support_mask`].
170    fn support(&self) -> [u64; W];
171
172    /// Short human-readable name, used only in the engine's per-layer progress log (see [`propagate`](crate::propagate)). Never parsed, never part of any output.
173    ///
174    /// The default returns [`core::any::type_name`] of the concrete type, trimmed to its last path segment — `Clifford1Q`, `PauliRotation`, `Depolarizing`.
175    /// The trim is textual (cut at the first `<`, keep everything after the last `::`), so a type whose name is not a plain (possibly generic) named path — a tuple, a closure, a `Box<...>` — trims to something unhelpful; override if you need a name you can rely on.
176    fn debug_name(&self) -> &'static str {
177        let full = core::any::type_name::<Self>();
178        let head = match full.find('<') {
179            Some(i) => &full[..i],
180            None => full,
181        };
182        match head.rfind("::") {
183            Some(i) => &head[i + 2..],
184            None => head,
185        }
186    }
187
188    /// Apply the channel to a single input term, writing outputs to `out`.
189    fn apply(
190        &self,
191        input_x: &[u64; W],
192        input_z: &[u64; W],
193        coeff: Complex64,
194        out: &mut OutputBuffer<'_, W>,
195    );
196
197    /// Apply the channel's adjoint to a single input term, writing outputs to `out`. Used by the engine in `Direction::Heisenberg` mode for backpropagating observables.
198    ///
199    /// The default is `self.apply(...)`, i.e. assumes the channel is self-adjoint. Channels that are not (`PauliRotation`, `Clifford1Q::s`) override this.
200    fn apply_adjoint(
201        &self,
202        input_x: &[u64; W],
203        input_z: &[u64; W],
204        coeff: Complex64,
205        out: &mut OutputBuffer<'_, W>,
206    ) {
207        self.apply(input_x, input_z, coeff, out);
208    }
209
210    /// Prepare this channel for one layer of the bucketed engine.
211    ///
212    /// The default derives a dense local Pauli-transfer matrix by probing `apply` on the `4^|support|` local basis Paulis, so a channel that implements `apply` gets the bucketed engine for free.
213    /// Override only when the support is wider than [`prepared::MAX_LOCAL_SUPPORT`] and a tighter description exists (only `PauliRotation` above generator weight 2 does among the built-ins).
214    /// `None` means "cannot be bucketed": `propagate` panics rather than proceed with an unsound preparation (ARCHITECTURE.md §Prepared-Channels).
215    ///
216    /// # Contract
217    ///
218    /// Implementors must honour bounded support: the output amplitude may depend on the input only through its bits at [`Channel::support`] positions.
219    /// Deriving assumes it; debug builds check it against an all-ones background, and a property test checks it against randomized full-width inputs.
220    fn prepare(&self, hash: &Gf2Hash<W>, adjoint: bool) -> Option<Prepared<W>> {
221        Prepared::derive_local(self, hash, adjoint)
222    }
223}

371    #[test]
372    fn clear_resets_cursor() {
373        let (mut x, mut z, mut c, mut len) = alloc_bufs::<1>(4);
374        {
375            let mut buf = OutputBuffer::<1> {
376                x: &mut x,
377                z: &mut z,
378                coeff: &mut c,
379                len: &mut len,
380            };
381            buf.push([0xAA], [0xBB], Complex64::new(1.0, 0.0));
382            buf.push([0xCC], [0xDD], Complex64::new(2.0, 0.0));
383            assert_eq!(*buf.len, 2);
384            buf.clear();
385            assert_eq!(*buf.len, 0);
386            buf.push([0xEE], [0xFF], Complex64::new(3.0, 0.0));
387            assert_eq!(*buf.len, 1);
388        }
389        // The post-clear push lands at slot 0, overwriting the prior contents.
390        assert_eq!(x[0], [0xEE]);
391        assert_eq!(z[0], [0xFF]);
392        assert_eq!(c[0], Complex64::new(3.0, 0.0));
393        // Slot 1 was written before the clear and is left as-is.
394        assert_eq!(x[1], [0xCC]);
395        assert_eq!(z[1], [0xDD]);
396    }

398    #[test]
399    fn reuse_does_not_grow_backing_vecs() {
400        let cap = 4;
401        let mut x: Vec<[u64; 1]> = vec![[0u64; 1]; cap];
402        let mut z: Vec<[u64; 1]> = vec![[0u64; 1]; cap];
403        let mut c: Vec<Complex64> = vec![Complex64::new(0.0, 0.0); cap];
404        assert_eq!(x.capacity(), cap);
405        assert_eq!(z.capacity(), cap);
406        assert_eq!(c.capacity(), cap);
407        let mut len: usize;
408        for i in 0..100u64 {
409            len = 0;
410            let mut buf = OutputBuffer::<1> {
411                x: &mut x,
412                z: &mut z,
413                coeff: &mut c,
414                len: &mut len,
415            };
416            buf.push([i], [0], Complex64::new(i as f64, 0.0));
417            buf.push([i + 1], [0], Complex64::new((i + 1) as f64, 0.0));
418        }
419        assert_eq!(x.capacity(), cap);
420        assert_eq!(z.capacity(), cap);
421        assert_eq!(c.capacity(), cap);
422    }
