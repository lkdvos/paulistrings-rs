1//! One layer of the partitioned engine: export -> exchange -> coset loop.
2//!
3//! A partition holds only the terms with `rows.partition_of(v) == rank`.
4//! [`export_layer`] builds one exchange block per remote delta, [`Transport::exchange_layer`] runs the all-to-all, and the bucketed coset loop merges local deltas with received rows via [`ExtraRows`] before `keep_term` runs (ARCHITECTURE.md §Truncation).
5//! Whether a delta is remote depends only on its mask, so every partition reaches the same verdict on whether to exchange.
6//! This function does not rebucket and does not call [`TruncationPolicy::finalize_layer`]; the driver owns both.

22/// Chunks a layer's bulk transfer is cut into, by default.
23///
24/// The receiver consumes chunk `k` as soon as it lands, so the pipeline depth is the chunk count, traded off against the per-message MPI overhead of a small chunk.
25/// Not derived from the thread count: both sides of an exchange must cut the same block the same way, and two ranks need not have equal-sized pools.
26pub(crate) const DEFAULT_EXCHANGE_CHUNKS: usize = 8;
27
28/// The chunk count every partition cuts this layer's transfer into.
29///
30/// [`DEFAULT_EXCHANGE_CHUNKS`], unless `PAULISTRINGS_EXCHANGE_CHUNKS` names another (`1` is the un-pipelined layout).
31/// Read once per process, so every rank in a group launched with the same environment agrees, which it must since both sides cut the same block the same way.
32pub(crate) fn exchange_chunks() -> usize {
33    static CHUNKS: std::sync::OnceLock<usize> = std::sync::OnceLock::new();
34    *CHUNKS.get_or_init(|| {
35        std::env::var("PAULISTRINGS_EXCHANGE_CHUNKS")
36            .ok()
37            .and_then(|v| v.parse::<usize>().ok())
38            .filter(|&k| k > 0)
39            .unwrap_or(DEFAULT_EXCHANGE_CHUNKS)
40    })
41}

43/// The rows this partition received, as an [`ExtraRows`] source for the coset loop.
44///
45/// One entry per remote delta, in ascending remote-delta index.
46/// Output bucket `β′` reads `segment(map.position_of(β′))` of each, per the receive rule in [`transport`](super::transport)'s module docs.
47/// Received rows always join the **rest** stream: [`NEEDS_BETA`](ExtraRows::NEEDS_BETA) is `true`.
48pub(crate) struct RecvRows<'a, const W: usize> {
49    /// The block per remote delta, ascending by entry.
50    blocks: Vec<Option<&'a ExchangeBlock<W>>>,
51    /// The destination-coset order the blocks are laid out in.
52    map: &'a ChunkMap,
53    /// What a coset task blocks on before it reads a chunk's rows; a no-op under a blocking transport.
54    wait: &'a dyn ChunkWait,
55    /// Nanoseconds spent in [`append_into`](ExtraRows::append_into), summed across coset tasks. Measurement only.
56    #[cfg(feature = "phase-timing")]
57    append_ns: std::sync::atomic::AtomicU64,
58    /// The part of [`append_ns`](Self::append_ns) spent blocked in [`ChunkWait::wait_chunk`]. Measurement only.
59    #[cfg(feature = "phase-timing")]
60    chunk_wait_ns: std::sync::atomic::AtomicU64,
61}
62
63impl<'a, const W: usize> RecvRows<'a, W> {
64    /// Pair each of `plan`'s remote deltas with the block that carries it.
65    ///
66    /// A delta is classified from its mask, so partner `q`'s remote deltas addressed here are the same entries, in the same order, as this partition's remote deltas addressed to `q`: the `j`-th of `remote_for_partner(q)` is the `j`-th block of `recv[q]`.
67    fn new(
68        plan: &PartitionPlan,
69        recv: &'a [Option<PartnerPayload<W>>],
70        map: &'a ChunkMap,
71        wait: &'a dyn ChunkWait,
72    ) -> Self {
73        let mut blocks = Vec::with_capacity(plan.remote.len());
74        for r in &plan.remote {
75            let j = plan
76                .remote_for_partner(r.partner)
77                .position(|other| other.entry == r.entry)
78                .expect("a remote delta is in its own partner's list");
79            let block = recv[r.partner as usize]
80                .as_ref()
81                .and_then(|payload| payload.blocks.get(j));
82            debug_assert!(
83                block.is_some(),
84                "partition {} sent no block for remote delta {} (entry {}) — the partitions \
85                 disagree about the layer's delta set",
86                r.partner,
87                j,
88                r.entry,
89            );
90            debug_assert!(
91                block.is_none_or(|b| b.header.entry == r.entry as u32),
92                "partition {} sent entry {:?} where entry {} was expected",
93                r.partner,
94                block.map(|b| b.header.entry),
95                r.entry,
96            );
97            blocks.push(block);
98        }
99        Self {
100            blocks,
101            map,
102            wait,
103            #[cfg(feature = "phase-timing")]
104            append_ns: std::sync::atomic::AtomicU64::new(0),
105            #[cfg(feature = "phase-timing")]
106            chunk_wait_ns: std::sync::atomic::AtomicU64::new(0),
107        }
108    }
109}

111impl<const W: usize> ExtraRows<W> for RecvRows<'_, W> {
112    const NEEDS_BETA: bool = true;
113
114    fn count(&self, beta: u32) -> usize {
115        let p = self.map.position_of(beta);
116        let mut n = 0usize;
117        for block in self.blocks.iter().flatten() {
118            n += block.segment(p).2.len();
119        }
120        n
121    }
122
123    fn append_into(
124        &self,
125        beta: u32,
126        x: &mut Vec<[u64; W]>,
127        z: &mut Vec<[u64; W]>,
128        c: &mut Vec<Complex64>,
129    ) {
130        #[cfg(feature = "phase-timing")]
131        let t0 = std::time::Instant::now();
132        let p = self.map.position_of(beta);
133        // Every member of a coset is in one chunk (`ChunkMap`), so this is one wait per task, not one per member.
134        self.wait.wait_chunk(self.map.chunk_of_position(p));
135        #[cfg(feature = "phase-timing")]
136        self.chunk_wait_ns.fetch_add(
137            t0.elapsed().as_nanos() as u64,
138            std::sync::atomic::Ordering::Relaxed,
139        );
140        for block in &self.blocks {
141            let Some(block) = block else { continue };
142            let (sx, sz, sc) = block.segment(p);
143            x.extend_from_slice(sx);
144            z.extend_from_slice(sz);
145            c.extend_from_slice(sc);
146        }
147        #[cfg(feature = "phase-timing")]
148        self.append_ns.fetch_add(
149            t0.elapsed().as_nanos() as u64,
150            std::sync::atomic::Ordering::Relaxed,
151        );
152    }
153}

155/// One partition's reusable per-layer scratch: the coset loop's and the export pass's, held together so the driver carries one value per partition.
156#[derive(Debug, Default)]
157pub(crate) struct PartitionState<const W: usize> {
158    /// The bucketed engine's layer scratch.
159    pub layer: LayerScratch<W>,
160    /// The export pass's count buffers and its pool of exchange payloads.
161    pub export: ExportScratch<W>,
162    /// The destination-coset order this layer's blocks are laid out in, and the chunks its bulk transfer is cut into.
163    pub chunks: ChunkMap,
164}

166/// What one layer's exchange moved, from this partition's point of view.
167///
168/// `rows_sent` and `bytes_sent` are indexed by partner rank; `rows_received` is a total, since a received row's provenance stops mattering once merged.
169/// A layer with no remote delta reports all zeros and issues no transport call.
170#[derive(Clone, Debug, Default, PartialEq, Eq)]
171pub(crate) struct LayerExchangeCounts {
172    /// Remote deltas the layer had, i.e. blocks sent per partner-delta pair.
173    pub remote_deltas: usize,
174    /// Rows sent to each partner rank.
175    pub rows_sent: Vec<u64>,
176    /// Wire bytes sent to each partner rank.
177    pub bytes_sent: Vec<u64>,
178    /// Rows received from all partners together.
179    pub rows_received: u64,
180}

184    fn none(size: u32) -> Self {
185        Self {
186            remote_deltas: 0,
187            rows_sent: vec![0; size as usize],
188            bytes_sent: vec![0; size as usize],
189            rows_received: 0,
190        }
191    }

194/// Apply one prepared channel to this partition's share of a sum.
195///
196/// `local` must hold exactly the terms with `rows.partition_of(v) == transport.rank()`, under a hash and bucket count every partition agrees on; it comes back holding this partition's share of the layer's output, merged, deduplicated and filtered through `policy`'s `keep_term`.
197/// Neither rebuckets nor calls `finalize_layer`: both are collective decisions the driver makes with the counts this returns.
198/// Classifies `prep`'s deltas itself; the driver needs that classification before it decides whether the layer takes a collective, so it holds the plan and calls [`apply_layer_partitioned_with_plan`] instead.
199#[cfg(test)]
200pub(crate) fn apply_layer_partitioned<const W: usize, T, X>(
201    local: &mut PauliSum<W>,
202    prep: &Prepared<W>,
203    rows: &PartitionRows<W>,
204    policy: &T,
205    state: &mut PartitionState<W>,
206    transport: &X,
207) -> LayerExchangeCounts
208where
209    T: TruncationPolicy<W> + ?Sized,
210    X: Transport,
211{
212    let plan = PartitionPlan::new(prep, rows, transport.rank());
213    apply_layer_partitioned_with_plan(local, prep, &plan, rows, policy, state, transport)
214}

216/// [`apply_layer_partitioned`] with the delta classification already made.
217///
218/// `plan` must be `PartitionPlan::new(prep, rows, transport.rank())`; the driver builds it a step earlier, because `has_remote()` is what decides whether the layer agrees the bucket count with the group (ARCHITECTURE.md §Partitioning).
219pub(crate) fn apply_layer_partitioned_with_plan<const W: usize, T, X>(
220    local: &mut PauliSum<W>,
221    prep: &Prepared<W>,
222    plan: &PartitionPlan,
223    #[cfg_attr(not(debug_assertions), allow(unused_variables))] rows: &PartitionRows<W>,
224    policy: &T,
225    state: &mut PartitionState<W>,
226    transport: &X,
227) -> LayerExchangeCounts
228where
229    T: TruncationPolicy<W> + ?Sized,
230    X: Transport,
231{
232    let size = transport.size();
233
234    // Nothing crosses: the ordinary engine, and — crucially — *no* transport
235    // call. Every partition took this branch, because `part(d)` is a function
236    // of the delta alone.
237    if !plan.has_remote() {
238        apply_layer_bucketed(local, prep, policy, &mut state.layer);
239        return LayerExchangeCounts::none(size);
240    }
241
242    #[cfg(feature = "phase-timing")]
243    let mut st = crate::engine::stats::Stamp::now();
244    // Both sides lay the blocks out in the *receiver's* coset order, and every
245    // partition derives it from the same local bucket deltas and the same
246    // collectively agreed bucket count — so the sender can permute without a
247    // word of negotiation. `apply_layer_bucketed_with` rebuilds the identical
248    // span below from the same two inputs.
249    let span = Gf2Span::new(&plan.local_bucket_deltas, local.hash().bits());
250    state
251        .chunks
252        .rebuild(&span, local.num_buckets(), exchange_chunks());
253    let (send, export) = export_layer(local, prep, plan, size, &state.chunks, &mut state.export);
254    #[cfg(feature = "phase-timing")]
255    {
256        st.lap(&mut state.layer.stats.export_ns);
257        state.layer.stats.rows_exported += export.rows_to.iter().sum::<u64>();
258    }
259    #[cfg(debug_assertions)]
260    {
261        super::export::debug_assert_exported_partitions(&send, rows);
262        for r in &plan.remote {
263            debug_assert!(
264                send[r.partner as usize].is_some(),
265                "no payload for partner {}, which the plan names",
266                r.partner,
267            );
268        }
269    }
270    // The local delta table, the local bucket deltas, the channel's total stream count, and — for a wide rotation — whether the generator pass emits here at all.
271    let retained;
272    let local_prep: &Prepared<W> = match prep {
273        Prepared::Local(ptm) => {
274            retained = Prepared::Local(ptm.retain_entries(&plan.local_entries));
275            &retained
276        }
277        // The identity entry of a rotation is always local, so `has_remote()` means the generator crosses; `gen_local` switches its pass off instead.
278        Prepared::Rotation(_) => prep,
279    };
280    let knobs = LayerKnobs {
281        bucket_deltas: Some(&plan.local_bucket_deltas),
282        rest_streams: Some(plan.rest_streams_total),
283        // From `prep`, not `local_prep`: asking the restricted PTM could put one partition on a different sort kernel than the unpartitioned run — see `LayerKnobs::rows_per_key`.
284        rows_per_key: match prep {
285            Prepared::Local(ptm) => Some(rest_rows_per_key(ptm)),
286            Prepared::Rotation(_) => None,
287        },
288        gen_local: match prep {
289            // Entry 1 is the generator pass (`plan`'s numbering).
290            Prepared::Rotation(_) => plan.local_entries[1],
291            // Meaningless for a tabulated channel.
292            Prepared::Local(_) => true,
293        },
294    };
295
296    // Split the scratch so the exchange can borrow the payload pool while the coset loop inside it borrows the layer scratch and the chunk map.
297    let PartitionState {
298        layer: layer_scratch,
299        export: export_scratch,
300        chunks: map,
301    } = state;
302    #[cfg(feature = "phase-timing")]
303    let body_ns = std::cell::Cell::new(0u64);
304    #[cfg(feature = "phase-timing")]
305    let exchange_start = std::time::Instant::now();
306
307    // `exchange_layer` returns once the block headers and CSR offsets are here; the rows themselves may still be in flight, and each coset task waits for its own chunk at the top of `append_into`.
308    let (recv, rows_received) =
309        transport.exchange_layer(send, &mut export_scratch.pool, map, |recv, wait| {
310            #[cfg(feature = "phase-timing")]
311            let body_start = std::time::Instant::now();
312            #[cfg(debug_assertions)]
313            for block in recv.iter().flatten().flat_map(|payload| &payload.blocks) {
314                // The bucket count is a collective decision the driver makes before the layer; a block indexed by a different one would be read at the wrong offsets.
315                debug_assert_eq!(
316                    block.num_buckets() as usize,
317                    local.num_buckets(),
318                    "a partner sent a block indexed by {} buckets where this partition has {}: \
319                     the partitions disagree about the bucket count",
320                    block.num_buckets(),
321                    local.num_buckets(),
322                );
323            }
324            let rows_received: u64 = recv
325                .iter()
326                .flatten()
327                .flat_map(|payload| &payload.blocks)
328                .map(|block| block.rows() as u64)
329                .sum();
330            let recv_rows = RecvRows::new(plan, recv, map, wait);
331            apply_layer_bucketed_with(local, local_prep, policy, layer_scratch, &recv_rows, knobs);
332            #[cfg(feature = "phase-timing")]
333            {
334                // The coset loop has joined, so the counters are quiescent.
335                layer_scratch.stats.recv_rows += rows_received;
336                layer_scratch.stats.append_ns += recv_rows
337                    .append_ns
338                    .load(std::sync::atomic::Ordering::Relaxed);
339                layer_scratch.stats.chunk_wait_ns += recv_rows
340                    .chunk_wait_ns
341                    .load(std::sync::atomic::Ordering::Relaxed);
342                body_ns.set(body_start.elapsed().as_nanos() as u64);
343            }
344            rows_received
345        });
346    #[cfg(feature = "phase-timing")]
347    {
348        // `exchange_ns` excludes the layer work it wraps, so it isolates whatever transfer the coset loop did not manage to hide before the closing wait.
349        layer_scratch.stats.exchange_ns +=
350            exchange_start.elapsed().as_nanos() as u64 - body_ns.get();
351        st.rearm();
352    }
353    // The payloads that carried the received rows go back into the pool with their columns intact, for the next export or receive to reuse.
354    export_scratch.pool.extend(recv.into_iter().flatten());
355
356    LayerExchangeCounts {
357        remote_deltas: plan.remote.len(),
358        rows_sent: export.rows_to,
359        bytes_sent: export.bytes_to,
360        rows_received,
361    }
362}

398    /// `count` and `append_into` read the block at the *destination position*, and each waits for exactly the chunk that position falls in.
399    ///
400    /// This is the whole receive-side contract of the pipelined exchange: get the position wrong and rows land in the wrong bucket; get the chunk wrong and a task reads a column MPI is still writing.
401    /// The oracle is hand-built, one row per position carrying that position's index, so a misread shows up as a wrong number rather than a wrong sum.
402    #[test]
403    fn received_rows_are_read_by_position_and_wait_for_their_own_chunk() {
404        const BITS: u8 = 5;
405        let n = 1u32 << BITS;
406        for deltas in [vec![0u32], vec![0, 4], vec![0, 1, 2, 3]] {
407            let span = crate::engine::coset::Gf2Span::new(&deltas, BITS);
408            for chunks in [1usize, 2, 4, 8, 64] {
409                let mut map = ChunkMap::default();
410                map.rebuild(&span, n as usize, chunks);
411
412                // One row per position, coefficient = the position index.
413                let counts: Vec<u32> = vec![1; n as usize];
414                let mut block = ExchangeBlock::<1>::with_counts(1, &counts);
415                block.x = (0..n).map(|p| [u64::from(p)]).collect();
416                block.z = (0..n).map(|_| [0u64]).collect();
417                block.coeff = (0..n).map(|p| Complex64::new(f64::from(p), 0.0)).collect();
418
419                let payload = PartnerPayload::<1> {
420                    blocks: vec![block],
421                };
422                let recv = vec![None, Some(payload)];
423                let mut plan_deltas = deltas.clone();
424                plan_deltas.sort_unstable();
425                let plan = PartitionPlan {
426                    local_entries: vec![true, false],
427                    local_bucket_deltas: plan_deltas,
428                    remote: vec![super::super::plan::RemoteDelta {
429                        entry: 1,
430                        partner: 1,
431                        bucket_delta: 0,
432                        partition_delta: 1,
433                    }],
434                    rest_streams_total: 1,
435                };
436
437                let spy = WaitSpy(std::sync::Mutex::new(Vec::new()));
438                let rows = RecvRows::new(&plan, &recv, &map, &spy);
439                for beta in 0..n {
440                    let p = map.position_of(beta);
441                    assert_eq!(rows.count(beta), 1, "one row per bucket");
442                    let (mut x, mut z, mut c) = (Vec::new(), Vec::new(), Vec::new());
443                    rows.append_into(beta, &mut x, &mut z, &mut c);
444                    assert_eq!(x, vec![[u64::from(p)]], "bucket {beta} read position {p}");
445                    assert_eq!(z, vec![[0u64]]);

447                }

449                assert_eq!(waited.len(), n as usize, "one wait per append_into");
450                for (beta, &k) in waited.iter().enumerate() {

454                        "bucket {beta} waited for the wrong chunk at {chunks} chunks",
455                    );
456                    assert!(k < map.chunks(), "chunk {k} is outside the map");
457                }
458            }
459        }

462    /// Run one layer on every partition of `rows`, each on its own thread with its own transport, and give back the parts in rank order with their exchange counts.
463    ///
464    /// `threads`, when given, runs each partition inside a Rayon pool of that size, the knob the byte-identity test turns.
465    /// Every part is checked on the way out: the bucketed invariants hold, and it holds keys of its own partition only.
466    fn run_parts<const W: usize, T>(
467        whole: &PauliSum<W>,
468        prep: &Prepared<W>,
469        rows: &PartitionRows<W>,
470        policy: &T,
471        threads: Option<usize>,
472    ) -> (Vec<PauliSum<W>>, Vec<LayerExchangeCounts>)
473    where
474        T: TruncationPolicy<W> + ?Sized,
475    {
476        let size = rows.num_partitions() as u32;
477        let transports = InProcessTransport::group(size);
478        let parts: Vec<PauliSum<W>> = (0..size).map(|r| whole.filter_partition(rows, r)).collect();
479        let results: Vec<(PauliSum<W>, LayerExchangeCounts)> = std::thread::scope(|scope| {
480            let handles: Vec<_> = parts
481                .into_iter()
482                .zip(transports)
483                .map(|(mut local, transport)| {
484                    scope.spawn(move || {
485                        let mut state = PartitionState::<W>::default();
486                        let mut run = || {
487                            apply_layer_partitioned(
488                                &mut local, prep, rows, policy, &mut state, &transport,
489                            )
490                        };
491                        let counts = match threads {
492                            Some(n) => rayon::ThreadPoolBuilder::new()
493                                .num_threads(n)
494                                .build()
495                                .expect("pool")
496                                .install(run),
497                            None => run(),
498                        };
499                        (local, counts)
500                    })
501                })
502                .collect();
503            handles
504                .into_iter()
505                .map(|h| h.join().expect("a partition panicked"))
506                .collect()
507        });
508
509        let mut sums = Vec::with_capacity(results.len());
510        let mut counts = Vec::with_capacity(results.len());
511        for (rank, (part, count)) in results.into_iter().enumerate() {
512            part.assert_invariants();
513            let held = part.partition_rank_of_all(rows);
514            assert!(
515                held.is_none() || held == Some(rank as u32),
516                "part {rank} holds keys of partition {held:?}",
517            );
518            sums.push(part);
519            counts.push(count);
520        }
521        (sums, counts)
522    }

539    /// `Z₀X₂X₄X₆` — weight 4, so `prepare` gives the `Prepared::Rotation` arm.
540    fn wide_gen() -> PauliString<1> {

617    /// A transport that counts the exchanges it is asked for.
618    struct CountingTransport {
619        inner: InProcessTransport,
620        exchanges: AtomicUsize,
621    }
622
623    impl Collectives for CountingTransport {
624        fn rank(&self) -> u32 {
625            self.inner.rank()
626        }
627        fn size(&self) -> u32 {
628            self.inner.size()
629        }
630        fn allreduce_max_u8(&self, v: u8) -> u8 {
631            self.inner.allreduce_max_u8(v)
632        }
633        fn allreduce_sum_u64(&self, buf: &mut [u64]) {
634            self.inner.allreduce_sum_u64(buf)
635        }
636        fn barrier(&self) {
637            self.inner.barrier()
638        }
639    }

641    impl Transport for CountingTransport {
642        fn exchange_layer<P, F, R>(
643            &self,
644            send: Vec<Option<P>>,
645            spare: &mut Vec<P>,
646            map: &ChunkMap,
647            body: F,
648        ) -> (Vec<Option<P>>, R)
649        where
650            P: Payload,
651            F: FnOnce(&[Option<P>], &dyn ChunkWait) -> R,
652        {
653            self.exchanges.fetch_add(1, Ordering::Relaxed);
654            self.inner.exchange_layer(send, spare, map, body)
655        }
656    }

658    /// A layer whose every delta stays inside its partition issues **no** transport call, not an empty one.
659    ///
660    /// That is what lets a partitioned run skip the collective entirely on layers that do not cross: the verdict comes from the plan, which every partition computes identically.
661    #[test]
662    fn no_remote_deltas_means_no_exchange() {
663        let input = rand_sum::<1>(300, 8, 0x9D2);
664        // `h(3)`'s only non-identity delta is the mask `x₃ z₃`; a partition
665        // row reading qubit 0 alone cannot see it.
666        let rows = PartitionRows::<1>::from_rows(8, vec![[1u64]], vec![[0u64]]);
667        let h = Clifford1Q::h(3);
668        let hash = Gf2Hash::<1>::new(8, 4, 0x9D);
669        let whole = input.clone().with_hash(hash);
670        let prep = Channel::<1>::prepare(&h, whole.hash(), false).expect("prepare");
671        assert!(
672            !PartitionPlan::new(&prep, &rows, 0).has_remote(),
673            "fixture must have no remote delta",
674        );
675
676        let transports: Vec<CountingTransport> = InProcessTransport::group(2)
677            .into_iter()
678            .map(|inner| CountingTransport {
679                inner,
680                exchanges: AtomicUsize::new(0),
681            })
682            .collect();
683        let parts: Vec<PauliSum<1>> = (0..2).map(|r| whole.filter_partition(&rows, r)).collect();
684        let prep = &prep;
685        let rows = &rows;
686        let results: Vec<(PauliSum<1>, LayerExchangeCounts, usize)> = std::thread::scope(|scope| {
687            let handles: Vec<_> = parts
688                .into_iter()
689                .zip(transports)
690                .map(|(mut local, transport)| {
691                    scope.spawn(move || {
692                        let mut state = PartitionState::<1>::default();
693                        let counts = apply_layer_partitioned(
694                            &mut local,
695                            prep,
696                            rows,
697                            &AlwaysKeep,
698                            &mut state,
699                            &transport,
700                        );
701                        let seen = transport.exchanges.load(Ordering::Relaxed);
702                        (local, counts, seen)
703                    })
704                })
705                .collect();
706            handles.into_iter().map(|h| h.join().unwrap()).collect()
707        });
708
709        let mut got_parts = Vec::new();
710        for (part, counts, exchanges) in results {
711            assert_eq!(exchanges, 0, "a local-only layer called the transport");
712            assert_eq!(counts, LayerExchangeCounts::none(2));
713            got_parts.push(part);
714        }
715        let got = PauliSum::merge_partitions(got_parts);
716        let want = naive_apply_layer(&input, &h, &AlwaysKeep, false);
717        assert_terms_close(&got, &want, TOL, "local-only layer");
718    }

720    /// A wide rotation whose generator crosses: every generator row is exported, and the exported total is exactly the number of terms that anticommute with the generator.
721    #[test]
722    fn all_remote_rotation_generator() {
723        let gen = wide_gen();
724        let rot = PauliRotation::new(gen, 0.41);
725        let rows = rows_seeing_qubit_2_x();
726        assert_eq!(rows.partition_of(&gen.x, &gen.z), 1);
727        let input = rand_sum::<1>(500, 8, 0x9D3);
728        let hash = Gf2Hash::<1>::new(8, 3, 0x9E);

731        let plan = PartitionPlan::new(&prep, &rows, 0);

733        assert!(!plan.local_entries[1], "entry 1 is the generator pass");
734        assert_eq!(plan.local_bucket_deltas, vec![0], "only the identity stays");

737            .iter()

759    /// A key whose contributions live on two partitions, and the amplitudes that bring each of them to it.
760    ///
761    /// `w` anticommutes with the generator, so it contributes to itself through the identity pass with amplitude `alpha = cos θ`, and its generator image `u = w · gen` contributes through the generator pass with amplitude `beta = i^k sin θ`.
762    /// `part(u)` differs from `part(w)`, so the two contributions *must* meet across the exchange or not at all.
763    /// Both amplitudes are measured off the oracle rather than assumed, so the caller can pick coefficients whose sum at `w` is exactly `beta·alpha − alpha·beta`.
764    fn cancelling_pair(
765        rot: &PauliRotation<1>,
766        rows: &PartitionRows<1>,
767    ) -> (PauliString<1>, PauliString<1>, Complex64, Complex64) {
768        let gen = wide_gen();
769        let w = PauliString::<1>::x(0);
770        assert!(!w.commutes_with(&gen));
771        let mut u = w;
772        u.mul_assign(&gen);
773        assert_ne!(
774            rows.partition_of(&w.x, &w.z),
775            rows.partition_of(&u.x, &u.z),
776            "the pair must straddle the partition boundary",
777        );
778
779        let probe = |term: PauliString<1>| -> Complex64 {
780            let mut acc = BuildAccumulator::<1>::with_capacity(8, 1);
781            acc.add_term(term, Phase::ONE, Complex64::new(1.0, 0.0));
782            naive_apply_layer(&acc.finalize(), rot, &AlwaysKeep, false)
783                .get(&w.x, &w.z)
784                .unwrap_or(ZERO)
785        };
786        let alpha = probe(w);
787        let beta = probe(u);
788        assert!(alpha.norm() > 0.1 && beta.norm() > 0.1, "degenerate probe");
789        (w, u, alpha, beta)
790    }

792    /// The layer's two-term fixture: `w` with coefficient `beta` and `u` with
793    /// coefficient `-alpha · scale`, so `w`'s output is `beta·alpha` plus
794    /// `-scale · alpha·beta`.
795    fn cancelling_input(
796        w: PauliString<1>,
797        u: PauliString<1>,
798        alpha: Complex64,
799        beta: Complex64,
800        scale: f64,
801    ) -> PauliSum<1> {
802        let mut acc = BuildAccumulator::<1>::with_capacity(8, 2);
803        acc.add_term(w, Phase::ONE, beta);
804        acc.add_term(u, Phase::ONE, -alpha * scale);
805        acc.finalize()
806    }

808    /// `keep_term` sees the sum across the partition boundary: two contributions that each clear the threshold by five orders of magnitude cancel to below it, and the term is dropped, exactly as the oracle (which sums before it filters) drops it.
809    ///
810    /// Were the received rows merged *after* the policy ran, the term would survive with the wrong coefficient.
811    #[test]
812    fn keep_term_sees_the_sum_across_partitions() {
813        let rot = PauliRotation::new(wide_gen(), 0.41);
814        let rows = rows_seeing_qubit_2_x();
815        let (w, u, alpha, beta) = cancelling_pair(&rot, &rows);
816        let input = cancelling_input(w, u, alpha, beta, 1.0 - 1e-7);
817        let policy = CoefficientThreshold(1e-6);
818
819        for bits in [0u8, 3] {
820            let hash = Gf2Hash::<1>::new(8, bits, 0x9F);
821            let whole = input.clone().with_hash(hash);
822            let prep = Channel::<1>::prepare(&rot, whole.hash(), false).expect("prepare");
823            let (parts, _) = run_parts(&whole, &prep, &rows, &policy, None);
824            let got = PauliSum::merge_partitions(parts);
825            assert!(
826                got.get(&w.x, &w.z).is_none(),
827                "bits={bits}: a term that only survives unsummed was kept ({:?})",
828                got.get(&w.x, &w.z),
829            );
830            let want = naive_apply_layer(&input, &rot, &policy, false);
831            assert_terms_close(&got, &want, TOL, &format!("threshold bits={bits}"));
832
833            // Without the threshold the residue is there, and it is the sum.
834            let (parts, _) = run_parts(&whole, &prep, &rows, &AlwaysKeep, None);
835            let kept = PauliSum::merge_partitions(parts);
836            let residue = kept.get(&w.x, &w.z).expect("the residue survives");
837            assert!(
838                residue.norm() < 1e-7 && residue.norm() > 1e-9,
839                "bits={bits}: residue {residue} is not the near-cancellation",
840            );
841            let want = naive_apply_layer(&input, &rot, &AlwaysKeep, false);
842            assert_terms_close(&kept, &want, TOL, &format!("no threshold bits={bits}"));
843        }
844    }
