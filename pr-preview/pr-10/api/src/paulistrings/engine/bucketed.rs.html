26/// Reusable per-layer scratch, held by the caller across layers so a layer allocates nothing after the first call: every field retains its high-water capacity across cosets and layers.
27/// The serial path uses this instance directly; the parallel path gives each Rayon worker its own slot in `workers`, so capacity is bounded by `threads × coset working set`.
28///
29/// A task's output cannot depend on which scratch slot it drew: the swap site clears every write destination before use, so worker-to-slot assignment varying run to run is unobservable — which is what keeps output byte-identical across thread counts.
30#[derive(Debug, Default)]
31pub struct LayerScratch<const W: usize> {
32    /// The per-coset working set (serial path).
33    pub(super) task: CosetScratch<W>,
34    /// The layer's handle permutation, `perm[β] = span.perm_index(β)`.
35    pub(super) perm: Vec<u32>,
36    /// The inverse of [`Self::perm`], `inv_perm[perm[β]] = β`, so a coset member's *original* bucket index is recoverable from its permuted position. Filled only when the layer's [`ExtraRows`] source asks for it ([`ExtraRows::NEEDS_BETA`]); left empty otherwise, which [`fill_coset`] reads as "the permutation is the identity".
37    pub(super) inv_perm: Vec<u32>,
38    /// Staging area the bucket handles are permuted into. Holds handles only while a layer runs; its elements carry no capacity of their own.
39    pub(super) staging: Vec<BucketCols<W>>,
40    /// Worker-persistent coset working sets for the parallel path, one slot per Rayon worker, indexed by `rayon::current_thread_index()`. Each worker locks only its own slot, so the mutexes are uncontended.
41    pub(super) workers: Vec<Mutex<CosetScratch<W>>>,
42    /// Layer-level (wall-clock) phase counters; the per-coset busy-time counters live in each `CosetScratch`.
43    #[cfg(feature = "phase-timing")]
44    pub(crate) stats: PhaseStats,
45    /// The opt-in per-layer term-count trace, `None` unless [`Self::enable_term_trace`] was called.
46    pub(crate) term_trace: Option<TermTrace>,
47    /// The opt-in per-layer gate trace, `None` unless [`Self::enable_gate_trace`] was called.
48    pub(crate) gate_trace: Option<GateTrace>,
49}

51impl<const W: usize> LayerScratch<W> {
52    /// An empty scratch.
53    pub fn new() -> Self {

57    /// Drain and return the accumulated phase counters (layer-level wall-clock fields plus every worker's busy-time counters), zeroing them.
58    /// Counters accumulate across layers and `propagate_with_scratch` calls until drained.
59    #[cfg(feature = "phase-timing")]
60    pub fn take_stats(&mut self) -> PhaseStats {
61        let mut total = std::mem::take(&mut self.stats);
62        total.absorb_coset(&std::mem::take(&mut self.task.stats));
63        for slot in &self.workers {
64            let mut ws = slot.lock().unwrap();
65            total.absorb_coset(&std::mem::take(&mut ws.stats));
66        }
67        total
68    }

95/// Both vectors have one entry per layer applied, in application order (so *reverse* circuit order under [`Direction::Heisenberg`](crate::Direction)). Always compiled — the `phase-timing` feature gates only the timing counters.

132    pub gate_name: Vec<&'static str>,
133    /// Resident term count before the layer.
134    pub terms_in: Vec<usize>,

136    pub terms_out: Vec<usize>,
137    /// Elapsed wall-clock nanoseconds for the complete gate application.

155/// One output member's gather run: key columns and coefficients.
156/// Equal-key summation order is not pinned by a sort tiebreak — see the module doc and `merge::sort_rows_with_scratch`.
157#[derive(Clone, Debug, Default)]
158pub(super) struct GatherRun<const W: usize> {
159    /// The identity-delta stream: keys untouched, so it inherits the source bucket's strictly-ascending, duplicate-free order and is never sorted. Under a dense identity plan only `id_coeff` is populated, aligned 1:1 with `old[j]` whose keys the merge borrows in place; under a sparse plan all three columns are filled.
160    id_x: Vec<[u64; W]>,
161    id_z: Vec<[u64; W]>,
162    id_coeff: Vec<Complex64>,
163    /// Every other delta's rows — keys XOR'd by a constant mask, so generally unsorted; canonicalized per run by `sort_rows_with_scratch`.
164    x: Vec<[u64; W]>,
165    z: Vec<[u64; W]>,
166    coeff: Vec<Complex64>,
167}
168
169/// `a != ZERO`, without the `||` short-circuit — one flag, no branch.
170#[inline(always)]
171fn nonzero(a: Complex64) -> bool {
172    (a.re != 0.0) | (a.im != 0.0)
173}
174
175impl<const W: usize> GatherRun<W> {
176    #[inline]
177    fn reset(&mut self, cap_id_keys: usize, cap_id_coeff: usize, cap_rest: usize) {
178        self.id_x.clear();
179        self.id_z.clear();
180        self.id_coeff.clear();
181        self.x.clear();
182        self.z.clear();
183        self.coeff.clear();
184        // One slot past the exact capacity: `push_if` writes a discarded row at `len` before
185        // deciding not to publish it via `set_len`, so the write must stay in bounds even when
186        // every countable row is kept.
187        self.id_x.reserve(cap_id_keys + 1);
188        self.id_z.reserve(cap_id_keys + 1);
189        self.id_coeff.reserve(cap_id_coeff + 1);
190        self.x.reserve(cap_rest + 1);
191        self.z.reserve(cap_rest + 1);
192        self.coeff.reserve(cap_rest + 1);
193    }
194
195    /// Write one row at `len` in a three-column stream and publish it iff `keep`.
196    /// The single unsafe block behind every gather append, so there is one invariant to audit
197    /// rather than several copies of it that drift apart.
198    ///
199    /// # Safety
200    ///
201    /// `xs.len() < xs.capacity()` and likewise for `zs` and `cs`. The caller gets this from
202    /// [`GatherRun::reset`], which reserves one slot past the plan's exact per-run capacity.
203    #[inline]
204    unsafe fn append_if(
205        xs: &mut Vec<[u64; W]>,
206        zs: &mut Vec<[u64; W]>,
207        cs: &mut Vec<Complex64>,
208        keep: bool,
209        x: [u64; W],
210        z: [u64; W],
211        c: Complex64,
212    ) {
213        let n = xs.len();
214        debug_assert!(n < xs.capacity());
215        debug_assert!(n < zs.capacity());
216        debug_assert!(n < cs.capacity());
217        unsafe {
218            xs.as_mut_ptr().add(n).write(x);
219            zs.as_mut_ptr().add(n).write(z);
220            cs.as_mut_ptr().add(n).write(c);
221            let m = n + keep as usize;
222            xs.set_len(m);
223            zs.set_len(m);
224            cs.set_len(m);
225        }
226    }
227
228    /// Branchless filtered append to the rest stream.
229    #[inline]
230    fn push_if(&mut self, keep: bool, x: [u64; W], z: [u64; W], c: Complex64) {
231        // SAFETY: `reset` reserved `cap_rest + 1` on all three columns, and `cap_rest` bounds
232        // every row the plan's deltas can emit into this run.
233        unsafe { Self::append_if(&mut self.x, &mut self.z, &mut self.coeff, keep, x, z, c) }
234    }
235
236    /// Unconditional append to the rest stream — [`Self::push_if`] with the predicate known true, which is what the rotation generator pass wants.
237    #[inline]
238    fn push_row(&mut self, x: [u64; W], z: [u64; W], c: Complex64) {
239        self.push_if(true, x, z, c);
240    }
241
242    /// Branchless filtered append to the identity stream.
243    #[inline]
244    fn push_id_if(&mut self, keep: bool, x: [u64; W], z: [u64; W], c: Complex64) {
245        // SAFETY: as `push_if`, with `cap_id_keys` / `cap_id_coeff` bounding the identity stream's rows.
246        unsafe {
247            Self::append_if(
248                &mut self.id_x,
249                &mut self.id_z,
250                &mut self.id_coeff,
251                keep,
252                x,
253                z,
254                c,
255            )
256        }
257    }
258
259    /// Unchecked unconditional append to the identity *coefficient* column.
260    /// The dense-identity plans emit exactly one identity row per source row and borrow the
261    /// keys from the source bucket, so this is the only column materialized — hence its own
262    /// helper rather than [`Self::push_id_if`].
263    #[inline]
264    fn push_id_coeff(&mut self, c: Complex64) {
265        let n = self.id_coeff.len();
266        debug_assert!(n < self.id_coeff.capacity());
267        // SAFETY: `reset` reserved `cap_id_coeff + 1`, and `cap_id_coeff` is the source
268        // bucket's length — exactly the number of calls this loop makes.
269        unsafe {
270            self.id_coeff.as_mut_ptr().add(n).write(c);
271            self.id_coeff.set_len(n + 1);
272        }
273    }
274
275    #[cfg(any(test, feature = "phase-timing"))]
276    #[inline]
277    fn len(&self) -> usize {
278        self.id_coeff.len() + self.coeff.len()
279    }
280}

303    },

306/// Per-layer overrides the partitioned engine hands the coset loop.
307/// [`Default`] is the non-partitioned answer to all three, so `apply_layer_bucketed` passes it and nothing about the single-partition path changes.
308#[derive(Clone, Copy, Debug)]
309pub(crate) struct LayerKnobs<'k> {
310    /// Bucket deltas to build the coset span from. `None` (default) means the prepared channel's own [`Prepared::bucket_deltas`]; the partitioned layer passes its plan's local bucket deltas.
311    pub bucket_deltas: Option<&'k [u32]>,
312    /// Rest-stream count driving the sort-kernel choice. `None` (default) derives it from the plan. The partitioned layer passes the channel's total stream count (local + remote), so every partition picks the kernel the unpartitioned run would.
313    pub rest_streams: Option<usize>,
314    /// Rest-stream overlap driving the sort-kernel choice, as [`rest_streams`](Self::rest_streams) does for the count. `None` (default) derives it from the plan's own PTM. 315    pub rows_per_key: Option<f64>,
316    /// Whether a wide rotation's generator pass emits here (see [`DeltaPlan::Rotation::gen_local`]). `true` by default.
317    pub gen_local: bool,
318}

331/// Rest rows landing on one output key, averaged over the output keys that get at least one —
332/// the plan-time estimate of `rows_sorted / distinct keys`, and the second arm of the radix
333/// gate (`merge::RADIX_MAX_REST_ROWS_PER_KEY`). `1.0` means the rest streams are pairwise
334/// disjoint, so the per-run merge's branch is a coin flip; higher means the streams overlap
335/// and the merge steps through them in lock-step, which the branch predictor learns.
336/// Must be asked of the channel's own PTM, never a restricted one: a partitioned layer hands
337/// [`DeltaPlan::new`] a PTM cut down to its local entries, and a cut-down delta set can only
338/// look more disjoint than the channel is — see [`LayerKnobs::rows_per_key`].
339pub(super) fn rest_rows_per_key<const W: usize>(ptm: &LocalPtm<W>) -> f64 {
340    // Entry 0 is the identity delta when present, and its stream is the pre-sorted `id` columns — never part of the sorted rest stream.
341    let rest_start = usize::from(ptm.deltas().first().is_some_and(|d| d.local_delta == 0));
342    let dim = 1usize << (2 * ptm.k());
343    let (mut rows, mut keys) = (0u32, 0u32);
344    for o in 0..dim {
345        let n = ptm.deltas()[rest_start..]
346            .iter()
347            .filter(|d| d.amp[o ^ d.local_delta as usize] != ZERO)
348            .count() as u32;
349        rows += n;
350        keys += u32::from(n > 0);
351    }
352    if keys == 0 {
353        0.0
354    } else {
355        f64::from(rows) / f64::from(keys)
356    }
357}

359impl<'p, const W: usize> DeltaPlan<'p, W> {
360    pub(super) fn new(prep: &'p Prepared<W>, span: &Gf2Span, knobs: LayerKnobs<'_>) -> Self {
361        match prep {
362            Prepared::Local(ptm) => {
363                let coords: Vec<u32> = ptm
364                    .deltas()
365                    .iter()
366                    .map(|d| span.coord_of(d.bucket_delta))
367                    .collect();
368                let has_identity = ptm.deltas().first().is_some_and(|d| d.local_delta == 0);
369                // The identity delta hashes to bucket delta 0, so the id stream stays in its own member.
370                debug_assert!(!has_identity || coords[0] == 0);
371                // Dense over the active patterns only: `amp` is sized LOCAL_DIM but the channel populates `4^k` entries.
372                let dim = 1usize << (2 * ptm.k());
373                let dense_identity =
374                    has_identity && ptm.deltas()[0].amp[..dim].iter().all(|a| *a != ZERO);
375                // `deltas()` is the realized delta set (§Bucketing), so its length minus the identity entry is the stream count the sort-kernel crossover turns on.
376                let rest_streams = knobs
377                    .rest_streams
378                    .unwrap_or(ptm.deltas().len() - has_identity as usize);
379                // Both arms read channel-wide quantities, never this partition's view of them, so every partition picks the kernel the unpartitioned run would.
380                let radix_sort = rest_streams >= RADIX_MIN_REST_STREAMS
381                    || (rest_streams >= RADIX_MIN_DISJOINT_STREAMS
382                        && knobs.rows_per_key.unwrap_or_else(|| rest_rows_per_key(ptm))
383                            < RADIX_MAX_REST_ROWS_PER_KEY);
384                DeltaPlan::Local {
385                    ptm,
386                    coords,
387                    has_identity,
388                    dense_identity,
389                    radix_sort,
390                }
391            }
392            Prepared::Rotation(r) => DeltaPlan::Rotation {
393                prep: r,
394                coord_identity: span.coord_of(r.bucket_delta_identity),
395                // `coord_of` demands its argument be in the span, and a
396                // remote generator's bucket delta is not.
397                coord_gen: if knobs.gen_local {
398                    span.coord_of(r.bucket_delta_gen)
399                } else {
400                    0
401                },
402                gen_local: knobs.gen_local,
403            },
404        }
405    }
406}

408/// Extra rest-stream rows for an output bucket, supplied by the partitioned engine.
409/// A partition generates rows whose output bucket lives on another partition; they arrive here
410/// and are appended to that bucket's gather run after the local gather and before the per-run
411/// sort. Going into the rest stream (never the pre-sorted id stream) is what makes them safe:
412/// it is sorted anyway, so `merge2_into` sees the complete sum before `keep_term` runs
413/// (ARCHITECTURE.md §Truncation).
414/// Zero-cost when [`NoExtra`]: [`NEEDS_BETA`](Self::NEEDS_BETA) is a `const false` that deletes
415/// every call site, along with the inverse-permutation pass that exists only to answer them.
416///
417/// # Contract
418///
419/// An implementation that can return rows must set `NEEDS_BETA = true`.
420pub(crate) trait ExtraRows<const W: usize> {
421    /// Whether the engine must recover each coset member's original bucket index before calling this source. `false` (default) also means the engine never calls [`count`](Self::count) or [`append_into`](Self::append_into).
422    const NEEDS_BETA: bool = false;
423
424    /// How many rows are destined for output bucket `beta` (its original index). Used to size the gather run exactly, so it must agree with what [`append_into`](Self::append_into) pushes.
425    #[inline]
426    fn count(&self, beta: u32) -> usize {
427        let _ = beta;
428        0
429    }
430
431    /// Append bucket `beta`'s rows onto the run's rest columns, all three in step. `beta` is the bucket's original index.
432    #[inline]
433    fn append_into(
434        &self,
435        beta: u32,
436        x: &mut Vec<[u64; W]>,
437        z: &mut Vec<[u64; W]>,
438        c: &mut Vec<Complex64>,
439    ) {
440        let _ = (beta, x, z, c);
441    }
442}

450/// Apply one prepared channel to a bucketed sum.
451/// `policy`'s `keep_term` is folded into the per-bucket merge, so it sees fully summed coefficients (ARCHITECTURE.md §Truncation).
452/// `finalize_layer` is not called here; `propagate` owns that.
453pub fn apply_layer_bucketed<const W: usize, T>(
454    sum: &mut PauliSum<W>,
455    prep: &Prepared<W>,
456    policy: &T,
457    scratch: &mut LayerScratch<W>,
458) where
459    T: TruncationPolicy<W> + ?Sized,
460{
461    apply_layer_bucketed_with(sum, prep, policy, scratch, &NoExtra, LayerKnobs::default())
462}
463
464/// [`apply_layer_bucketed`] with an [`ExtraRows`] source feeding each output bucket's rest stream and the partitioned engine's per-layer [`LayerKnobs`] — the entry point the partitioned engine drives.
465/// Under [`NoExtra`] and default knobs it is [`apply_layer_bucketed`], with the hook's call sites and the inverse-permutation pass behind a `const false`.
466pub(crate) fn apply_layer_bucketed_with<const W: usize, T, X>(
467    sum: &mut PauliSum<W>,
468    prep: &Prepared<W>,
469    policy: &T,
470    scratch: &mut LayerScratch<W>,
471    extra: &X,
472    knobs: LayerKnobs<'_>,
473) where
474    T: TruncationPolicy<W> + ?Sized,
475    X: ExtraRows<W> + Sync,
476{
477    #[cfg(feature = "phase-timing")]
478    let mut st = Stamp::now();
479
480    // Key-preserving channels leave every key bitwise unchanged, so the output is already
481    // sorted and duplicate-free: rescaling each coefficient is an in-place filter, no sort needed.
482    // `!X::NEEDS_BETA` must still be asked: a partitioned layer's local delta table can look
483    // key-preserving while received rows still have to be merged in.
484    if let Prepared::Local(ptm) = prep {
485        if !X::NEEDS_BETA && ptm.is_key_preserving() {
486            rescale_in_place(sum, ptm, policy);
487            #[cfg(feature = "phase-timing")]
488            st.lap(&mut scratch.stats.rescale_ns);
489            return;
490        }
491    }
492
493    // The coset structure of this layer's bucket-delta set: `span(h(D))` rather than `h(D)`
494    // itself, since an open-trait channel's delta set need not be XOR-closed.
495    let own_deltas;
496    let deltas: &[u32] = match knobs.bucket_deltas {
497        Some(d) => d,
498        None => {
499            own_deltas = prep.bucket_deltas();
500            &own_deltas
501        }
502    };
503    let span = Gf2Span::new(deltas, sum.hash().bits());
504    let plan = DeltaPlan::new(prep, &span, knobs);
505    let m = span.coset_size();
506    let num_cosets = span.num_cosets();
507    #[cfg(feature = "phase-timing")]
508    st.lap(&mut scratch.stats.span_plan_ns);
509
510    // Permute the bucket handles into coset-contiguous order: coset `c` owns
511    // `staging[c·2^r .. (c+1)·2^r]`, members ascending by basis coordinate. Handles are three
512    // `Vec` headers; the term data never moves. At `r = 0` `perm_index` is the identity, so
513    // both handle passes are skipped and the chunk loop runs on the buckets directly.
514    let identity_perm = span.r() == 0;
515    if !identity_perm {
516        let buckets = sum.buckets_mut();
517        scratch.perm.clear();
518        scratch
519            .perm
520            .extend((0..buckets.len() as u32).map(|beta| span.perm_index(beta)));
521        scratch
522            .staging
523            .resize_with(buckets.len(), BucketCols::default);
524        for (beta, cols) in buckets.iter_mut().enumerate() {
525            scratch.staging[scratch.perm[beta] as usize] = std::mem::take(cols);
526        }
527    }
528    // The inverse handle permutation, so `fill_coset` can name a member's original bucket.
529    // Under `NoExtra` this is a `const false` branch the compiler deletes and `inv_perm` stays
530    // empty, which `fill_coset` reads as "the permutation is the identity".
531    if X::NEEDS_BETA {
532        scratch.inv_perm.clear();
533        if !identity_perm {
534            scratch.inv_perm.resize(scratch.perm.len(), 0);
535            for (beta, &p) in scratch.perm.iter().enumerate() {
536                scratch.inv_perm[p as usize] = beta as u32;
537            }
538        }
539    }
540    #[cfg(feature = "phase-timing")]
541    st.lap(&mut scratch.stats.permute_ns);
542
543    // Each coset is a closed task: it reads and writes only its own chunk, so the chunk loop
544    // needs no atomics, no cross-task locks, and no reconciliation pass, and output is
545    // byte-identical across thread counts (ARCHITECTURE.md §Determinism).
546    {
547        // Size the worker pool before `staging` is borrowed below; keeping existing slots
548        // preserves their high-water capacity.
549        if num_cosets >= MIN_COSETS_FOR_PARALLEL {
550            let pool = rayon::current_num_threads().max(1);
551            if scratch.workers.len() < pool {
552                scratch.workers.resize_with(pool, Mutex::default);
553            }
554        }
555        let workers = &scratch.workers;
556        // Empty unless this layer's `ExtraRows` asked for it; `fill_coset` takes empty to
557        // mean the identity permutation.
558        let inv_perm: &[u32] = &scratch.inv_perm;
559        let chunks: &mut [BucketCols<W>] = if identity_perm {
560            sum.buckets_mut()
561        } else {
562            scratch.staging.as_mut_slice()
563        };
564        if num_cosets < MIN_COSETS_FOR_PARALLEL {
565            for (ci, chunk) in chunks.chunks_mut(m).enumerate() {
566                let base = ci * m;
567                fill_coset::<W, T, X>(
568                    chunk,
569                    &plan,
570                    policy,
571                    &mut scratch.task,
572                    extra,
573                    base,
574                    inv_perm,
575                );
576            }
577        } else {
578            chunks
579                .par_chunks_mut(m)
580                .enumerate()
581                .for_each(|(ci, chunk)| {
582                    let base = ci * m;
583                    // Inside `par_chunks_mut` the body always runs on a pool worker, so the
584                    // index is present and below the pool size; the fresh-scratch arm is a
585                    // defensive fallback only.
586                    match rayon::current_thread_index() {
587                        Some(i) if i < workers.len() => {
588                            let mut ws = workers[i].lock().unwrap();
589                            fill_coset::<W, T, X>(
590                                chunk, &plan, policy, &mut ws, extra, base, inv_perm,
591                            );
592                        }
593                        _ => {
594                            let mut ws = CosetScratch::<W>::default();
595                            fill_coset::<W, T, X>(
596                                chunk, &plan, policy, &mut ws, extra, base, inv_perm,
597                            );
598                        }
599                    }
600                });
601        }
602    }
603    #[cfg(feature = "phase-timing")]
604    st.lap(&mut scratch.stats.coset_loop_ns);
605
606    // Un-permute: every handle goes back to its bucket index, leaving the staging slots as
607    // empty, capacity-free defaults.
608    if !identity_perm {
609        let buckets = sum.buckets_mut();
610        for (beta, cols) in buckets.iter_mut().enumerate() {
611            *cols = std::mem::take(&mut scratch.staging[scratch.perm[beta] as usize]);
612        }
613    }
614    #[cfg(feature = "phase-timing")]
615    st.lap(&mut scratch.stats.unpermute_ns);
616    sum.recount();
617    #[cfg(feature = "phase-timing")]
618    st.lap(&mut scratch.stats.recount_ns);
619
620    #[cfg(debug_assertions)]
621    sum.assert_invariants();
622}

624/// Below this many cosets there is nothing to spread, so skip Rayon entirely — mostly the
625/// `bits = 0` / `r = bits` case, where one coset spans every bucket and the layer degenerates
626/// to a single whole-sum task on the same code path.
627pub(super) const MIN_COSETS_FOR_PARALLEL: usize = 2;
628
629/// Gather, sort and merge one coset, in place. The unit of parallel work.
630/// `chunk` holds the coset's `2^r` bucket columns, members ascending by basis coordinate,
631/// serving as both input source and output destination.
632/// `extra` supplies rows generated elsewhere (the partitioned engine); they join each member's
633/// rest stream between the gather and the sort, so the merge sees the complete sum. Naming
634/// their destination needs the member's original bucket index, recovered from `chunk_base` and
635/// `inv_perm`. Under [`NoExtra`] all of that is behind `X::NEEDS_BETA == false` and compiles away.
636pub(super) fn fill_coset<const W: usize, T, X>(
637    chunk: &mut [BucketCols<W>],
638    plan: &DeltaPlan<'_, W>,
639    policy: &T,
640    ws: &mut CosetScratch<W>,
641    extra: &X,
642    chunk_base: usize,
643    inv_perm: &[u32],
644) where
645    T: TruncationPolicy<W> + ?Sized,
646    X: ExtraRows<W>,
647{
648    let m = chunk.len();
649    // Member `j`'s original bucket index. An empty `inv_perm` means the layer skipped the
650    // handle permutation (`r = 0`), where the permuted slot is the bucket.
651    let beta_of = |j: usize| -> u32 {
652        if inv_perm.is_empty() {
653            (chunk_base + j) as u32
654        } else {
655            inv_perm[chunk_base + j]
656        }
657    };
658    #[cfg(feature = "phase-timing")]
659    let CosetScratch {
660        old,
661        runs,
662        sort,
663        stats,
664    } = ws;
665    #[cfg(not(feature = "phase-timing"))]
666    let CosetScratch { old, runs, sort } = ws;
667    #[cfg(feature = "phase-timing")]
668    let mut st = Stamp::now();
669    old.resize_with(m, BucketCols::default);
670    runs.resize_with(m, GatherRun::default);
671
672    // Swap the coset's columns out. The chunk slots inherit this scratch's cleared,
673    // capacity-retaining columns and become the write destinations — capacities circulate
674    // between buckets across cosets, which holds the steady state allocation-free in aggregate.
675    for (slot, cols) in chunk.iter_mut().zip(old.iter_mut()) {
676        std::mem::swap(slot, cols);
677        slot.clear();
678    }
679    #[cfg(feature = "phase-timing")]
680    st.lap(&mut stats.swap_ns);
681
682    // Exact per-run capacity, counted once per delta entry, split by destination stream: the
683    // identity entry feeds the pre-sorted `id` columns, everything else the sorted rest. Under
684    // a dense identity the id key columns stay empty since the merge borrows the source
685    // bucket's keys, so only `id_coeff` needs capacity.
686    for (j, run) in runs.iter_mut().enumerate() {
687        let (cap_id_keys, cap_id_coeff, mut cap_rest): (usize, usize, usize) = match plan {
688            DeltaPlan::Local {
689                coords,
690                has_identity,
691                dense_identity,
692                ..
693            } => {
694                let mut id = 0usize;
695                let mut rest = 0usize;
696                for (e, &c) in coords.iter().enumerate() {
697                    let l = old[j ^ c as usize].len();
698                    if *has_identity && e == 0 {
699                        id += l;
700                    } else {
701                        rest += l;
702                    }
703                }
704                (if *dense_identity { 0 } else { id }, id, rest)
705            }
706            DeltaPlan::Rotation {
707                coord_identity,
708                coord_gen,
709                gen_local,
710                ..
711            } => (
712                0,
713                old[j ^ *coord_identity as usize].len(),
714                // A remote generator emits nothing here, so the rest stream is whatever
715                // `extra` adds below and nothing else.
716                if *gen_local {
717                    old[j ^ *coord_gen as usize].len()
718                } else {
719                    0
720                },
721            ),
722        };
723        // Received rows land in the rest stream, so they belong to its exact capacity.
724        // `const false` under `NoExtra`.
725        if X::NEEDS_BETA {
726            cap_rest += extra.count(beta_of(j));
727        }
728        run.reset(cap_id_keys, cap_id_coeff, cap_rest);
729    }
730    #[cfg(feature = "phase-timing")]
731    st.lap(&mut stats.size_ns);
732
733    // Gather. Two visit orders produce the same multiset of rows per run, differing only in
734    // arrival order, which the key-only sort erases up to floating-point tolerance (see
735    // `local_gather_orders_agree_to_fp_tolerance`). Which order is faster depends on `r`; see
736    // `GATHER_OUTPUT_MAJOR_MIN_R`.
737    match plan {
738        DeltaPlan::Local {
739            ptm,
740            coords,
741            has_identity,
742            dense_identity,
743            ..
744        } => {
745            if m >= 1 << GATHER_OUTPUT_MAJOR_MIN_R {
746                gather_local_output_major(old, runs, ptm, coords, *has_identity, *dense_identity);
747            } else {
748                gather_local_input_major(old, runs, ptm, coords, *has_identity, *dense_identity);
749            }
750        }
751        DeltaPlan::Rotation {
752            prep,
753            coord_identity,
754            coord_gen,
755            gen_local,
756        } => {
757            // The identity pass and the generator pass. Every term emits exactly one
758            // identity-pass row (full coefficient when it commutes, `cos`-scaled when it
759            // doesn't, kept even when `cos == 0` — see `merge2_into` on signed zeros), so the
760            // id stream is the whole source bucket in order and only the coefficient is
761            // materialized; the merge borrows the keys from the source bucket in place.
762            for (i, src) in old.iter().enumerate() {
763                for t in 0..src.len() {
764                    let v = PauliString::<W> {
765                        x: src.x[t],
766                        z: src.z[t],
767                    };
768                    if v.commutes_with(&prep.gen) {
769                        runs[i ^ *coord_identity as usize].push_id_coeff(src.coeff[t]);
770                    } else {
771                        runs[i ^ *coord_identity as usize].push_id_coeff(src.coeff[t] * prep.cos);
772                        // `true` on every unpartitioned layer; under a partitioning that sees
773                        // the generator, these rows belong to a partner and the export pass
774                        // has already shipped them.
775                        if *gen_local {
776                            let mut prod = v;
777                            let phase = prod.mul_assign(&prep.gen);
778                            let total = Phase::I + phase;
779                            runs[i ^ *coord_gen as usize].push_row(
780                                prod.x,
781                                prod.z,
782                                total.apply(src.coeff[t]) * prep.sin,
783                            );
784                        }
785                    }
786                }
787            }
788        }
789    }
790    // Rows generated on other partitions whose output bucket lives here. The rest stream only:
791    // it is sorted below, so a received row may duplicate a local key and `merge2_into` still
792    // sees the complete sum before `keep_term` runs. `const false` under `NoExtra`.
793    if X::NEEDS_BETA {
794        for (j, run) in runs.iter_mut().enumerate() {
795            extra.append_into(beta_of(j), &mut run.x, &mut run.z, &mut run.coeff);
796        }
797    }
798    #[cfg(feature = "phase-timing")]
799    st.lap(&mut stats.gather_ns);
800
801    // Sort each run's rest stream by key alone, then fuse the two-stream merge with the
802    // segmented reduction into the member's live slot: the id stream never moves through the
803    // sort. Under a dense identity plan the id stream's keys were never materialized either —
804    // they are the source bucket's own key columns, borrowed here.
805    // Which sort kernel this layer uses, hoisted out of the run loop: a per-layer property of
806    // the plan, never a per-run one.
807    let radix = matches!(
808        plan,
809        DeltaPlan::Local {
810            radix_sort: true,
811            ..
812        }
813    );
814    for (j, (run, dst)) in runs.iter_mut().zip(chunk.iter_mut()).enumerate() {
815        #[cfg(feature = "phase-timing")]
816        {
817            stats.rows_gathered += run.len() as u64;
818            stats.rows_sorted += run.coeff.len() as u64;
819        }
820        if radix {
821            sort_rows_radix_with_scratch(&mut run.x, &mut run.z, &mut run.coeff, sort);
822        } else {
823            sort_rows_with_scratch(&mut run.x, &mut run.z, &mut run.coeff, sort);
824        }
825        #[cfg(feature = "phase-timing")]
826        st.lap(&mut stats.sort_ns);
827        let (a_x, a_z): (&[[u64; W]], &[[u64; W]]) = match plan {
828            DeltaPlan::Local {
829                dense_identity: true,
830                ..
831            } => {
832                let src = &old[j];
833                debug_assert_eq!(src.len(), run.id_coeff.len());
834                #[cfg(feature = "phase-timing")]
835                {
836                    stats.rows_id += src.len() as u64;
837                }
838                (&src.x, &src.z)
839            }
840            DeltaPlan::Local { .. } => (&run.id_x, &run.id_z),
841            DeltaPlan::Rotation { coord_identity, .. } => {
842                let src = &old[j ^ *coord_identity as usize];
843                debug_assert_eq!(src.len(), run.id_coeff.len());
844                #[cfg(feature = "phase-timing")]
845                {
846                    stats.rows_id += src.len() as u64;
847                }
848                (&src.x, &src.z)
849            }
850        };
851        merge2_into::<W, T>(
852            a_x,
853            a_z,
854            &run.id_coeff,
855            &run.x,
856            &run.z,
857            &run.coeff,
858            &mut dst.x,
859            &mut dst.z,
860            &mut dst.coeff,
861            policy,
862        );
863        #[cfg(feature = "phase-timing")]
864        st.lap(&mut stats.merge_ns);
865    }
866
867    // Leave `old` cleared so the next coset's swap hands its chunk clean, capacity-retaining
868    // columns. Runs are cleared by their own `reset`.
869    for cols in old.iter_mut() {
870        cols.clear();
871    }
872    #[cfg(feature = "phase-timing")]
873    {
874        st.lap(&mut stats.clear_ns);
875        stats.cosets += 1;
876        stats.runs += m as u64;
877    }
878}

880/// Coset dimension at or above which the gather switches to output-major.
881///
882/// Input-major loads each term once but keeps `2^r` write streams open per task; at large `r`
883/// those streams plus the swapped coset no longer fit L2. Output-major re-reads each input
884/// bucket `2^r` times but keeps a single write stream, trading re-reads (coset-local, cheap)
885/// for write-stream pressure. Below this threshold the extra re-reads cost more than the write
886/// streams save; above it, the reverse. Both orders gather the same multiset of rows, so the
887/// choice is a pure performance knob, never a correctness one — pinned by
888/// `local_gather_orders_agree_to_fp_tolerance`. See `research/FINDINGS.md` for the measurements
889/// behind the current value.
890const GATHER_OUTPUT_MAJOR_MIN_R: u8 = 3;
891
892/// Input-major gather for a tabulated (`Local`) plan: each term is loaded once and its whole
893/// fanout is scattered by `member(i) ⊕ δ = member(i ⊕ coord(δ))`. Rows land in the runs in
894/// (input member, input position, delta) order.
895///
896/// The zero-amplitude filter is branchless: the row is always materialized and
897/// [`GatherRun::push_if`] publishes it only when the amplitude is nonzero, avoiding a
898/// data-dependent branch on which PTM entries vanish for a given support pattern (see
899/// `research/FINDINGS.md`).
900fn gather_local_input_major<const W: usize>(
901    old: &[BucketCols<W>],
902    runs: &mut [GatherRun<W>],
903    ptm: &LocalPtm<W>,
904    coords: &[u32],
905    has_identity: bool,
906    dense_identity: bool,
907) {
908    let rest_start = has_identity as usize;
909    for (i, src) in old.iter().enumerate() {
910        for t in 0..src.len() {
911            let s = ptm.support_bits(&src.x[t], &src.z[t]);
912            if has_identity {
913                // Entry 0 is the identity delta: masks are zero, so the row lands in this
914                // member's own run with its key untouched — the pre-sorted id stream.
915                let a = ptm.deltas()[0].amp[s];
916                if dense_identity {
917                    // Dense: `a` never vanishes and the stream is 1:1 with the source rows,
918                    // so only the coefficient is stored — the merge borrows the keys from `old[i]`.
919                    debug_assert!(a != ZERO);
920                    runs[i].push_id_coeff(src.coeff[t] * a);
921                } else {
922                    runs[i].push_id_if(nonzero(a), src.x[t], src.z[t], src.coeff[t] * a);
923                }
924            }
925            for (e, d) in ptm.deltas().iter().enumerate().skip(rest_start) {
926                let a = d.amp[s];
927                let mut kx = src.x[t];
928                let mut kz = src.z[t];
929                for w in 0..W {
930                    kx[w] ^= d.mask_x[w];
931                    kz[w] ^= d.mask_z[w];
932                }
933                runs[i ^ coords[e] as usize].push_if(nonzero(a), kx, kz, src.coeff[t] * a);
934            }
935        }
936    }
937}

939/// Output-major gather for a tabulated (`Local`) plan: for each output member, stream the one
940/// input bucket per delta entry and append to that member's run only. Rows land in (delta,
941/// input position) order — the same multiset as [`gather_local_input_major`] in a different
942/// order; they agree only up to floating-point tolerance (see
943/// `local_gather_orders_agree_to_fp_tolerance`).
944///
945/// The zero-amplitude filter is branchless, as in [`gather_local_input_major`], and helps here
946/// even though output-major is only reached on a dense PTM with nothing to filter: what
947/// [`GatherRun::push_if`] removes is the `Vec::push` capacity check and length increment per row.
948fn gather_local_output_major<const W: usize>(
949    old: &[BucketCols<W>],
950    runs: &mut [GatherRun<W>],
951    ptm: &LocalPtm<W>,
952    coords: &[u32],
953    has_identity: bool,
954    dense_identity: bool,
955) {
956    let rest_start = has_identity as usize;
957    for (j, run) in runs.iter_mut().enumerate() {
958        if has_identity {
959            // Entry 0: masks zero — the member's own bucket streams into the pre-sorted id
960            // columns; coefficient only when the identity is dense (keys borrowed).
961            let d = &ptm.deltas()[0];
962            let src = &old[j];
963            for t in 0..src.len() {
964                let s = ptm.support_bits(&src.x[t], &src.z[t]);
965                let a = d.amp[s];
966                if dense_identity {
967                    debug_assert!(a != ZERO);
968                    run.push_id_coeff(src.coeff[t] * a);
969                } else {
970                    run.push_id_if(nonzero(a), src.x[t], src.z[t], src.coeff[t] * a);
971                }
972            }
973        }
974        for (e, d) in ptm.deltas().iter().enumerate().skip(rest_start) {
975            let src = &old[j ^ coords[e] as usize];
976            for t in 0..src.len() {
977                let s = ptm.support_bits(&src.x[t], &src.z[t]);
978                let a = d.amp[s];
979                let mut kx = src.x[t];
980                let mut kz = src.z[t];
981                for w in 0..W {
982                    kx[w] ^= d.mask_x[w];
983                    kz[w] ^= d.mask_z[w];
984                }
985                run.push_if(nonzero(a), kx, kz, src.coeff[t] * a);
986            }
987        }
988    }
989}

991/// In-place coefficient rescale for a key-preserving channel.
992///
993/// Keys are untouched, so each bucket stays sorted and duplicate-free and no
994/// gather, sort or merge is needed. `keep_term` still applies, on the rescaled
995/// coefficient, and exact zeros are still dropped — matching the general path.
996fn rescale_in_place<const W: usize, T>(sum: &mut PauliSum<W>, ptm: &LocalPtm<W>, policy: &T)
997where
998    T: TruncationPolicy<W> + ?Sized,
999{
1000    let amp = &ptm.deltas()[0].amp;
1001    sum.buckets_mut().par_iter_mut().for_each(|cols| {
1002        let n = cols.len();
1003        let mut keep = 0usize;
1004        for i in 0..n {
1005            let s = ptm.support_bits(&cols.x[i], &cols.z[i]);
1006            let c = cols.coeff[i] * amp[s];
1007            if c == ZERO || !policy.keep_term(&cols.x[i], &cols.z[i], c) {
1008                continue;
1009            }
1010            // `keep <= i` always, so this never overwrites an unread slot.
1011            cols.x[keep] = cols.x[i];
1012            cols.z[keep] = cols.z[i];
1013            cols.coeff[keep] = c;
1014            keep += 1;
1015        }
1016        cols.x.truncate(keep);
1017        cols.z.truncate(keep);
1018        cols.coeff.truncate(keep);
1019    });
1020    sum.recount();
1021
1022    #[cfg(debug_assertions)]
1023    sum.assert_invariants();
1024}

1498        // the id coefficients and the merge borrows the keys from the source
1499        // bucket — mirrored below in `merge_run`.

1507                let mut cap_rest = 0usize;

1527            runs
1528        };

1530        let b = gather(true);

1532            a.iter().map(GatherRun::len).sum::<usize>() > 0,
1533            "gather produced nothing — the coset assembly is wrong"
1534        );

1536        // comparing, rather than comparing the sorted-but-unmerged rows
1537        // element-wise: a duplicate key's rows can land in either relative
1538        // order under the key-only sort, independent of visit order, so a
1539        // raw element-wise comparison could see a spurious mismatch at a tie
1540        // that has nothing to do with which order gathered it.

1542            |j: usize, run: &GatherRun<1>| -> (Vec<[u64; 1]>, Vec<[u64; 1]>, Vec<Complex64>) {
1543                let src = &old[j];
1544                assert_eq!(src.len(), run.id_coeff.len(), "dense id must be 1:1");
1545                let mut mx = Vec::new();
1546                let mut mz = Vec::new();
1547                let mut mc = Vec::new();

1549                    &src.x,
1550                    &src.z,
1551                    &run.id_coeff,
1552                    &run.x,
1553                    &run.z,
1554                    &run.coeff,
1555                    &mut mx,
1556                    &mut mz,
1557                    &mut mc,
1558                    &AlwaysKeep,
1559                );
1560                (mx, mz, mc)
1561            };
1562        for (j, (ra, rb)) in a.iter().zip(b.iter()).enumerate() {
1563            let (max, maz, mac) = merge_run(j, ra);
1564            let (mbx, mbz, mbc) = merge_run(j, rb);
1565            assert_eq!(max, mbx, "run {j}: merged keys (x) diverge");
1566            assert_eq!(maz, mbz, "run {j}: merged keys (z) diverge");
1567            assert_eq!(mac.len(), mbc.len(), "run {j}: merged term count diverges");
1568            for (i, (ca, cb)) in mac.iter().zip(mbc.iter()).enumerate() {
1569                let d = (ca - cb).norm();
1570                assert!(
1571                    d < TOL,
1572                    "run {j} term {i}: merged coefficients {ca} vs {cb} (delta {d:e})"
1573                );
1574            }
1575        }

1858    /// Goes through [`canonical_triples`] (the public `iter()`), so it is blind to how the sum is partitioned or stored: the digest depends only on the term set and the coefficient bit patterns, moving on any ULP change.

1865            }
1866            h = fnv_fold(h, c.re.to_bits());
1867            h = fnv_fold(h, c.im.to_bits());
1868        }
1869        h
1870    }

2762    /// A direct, deliberately naive transcription of the `TopN` tie-group
2763    /// rule (ARCHITECTURE.md §Truncation), used as an oracle for the
2764    /// production `finalize_layer`.
2765    ///
2766    /// Full sort instead of a selection, `retain` instead of a per-bucket
2767    /// compaction, no parallelism: nothing here shares code with the thing it
2768    /// checks. Returns the surviving terms as canonical triples.
2769    fn top_n_reference<const W: usize>(
2770        sum: &PauliSum<W>,
2771        n: usize,
2772    ) -> Vec<([u64; W], [u64; W], Complex64)> {
2773        let mut triples = canonical_triples(sum);
2774        if triples.len() <= n {
2775            return triples;
2776        }
2777        if n == 0 {
2778            return Vec::new();
2779        }
2780        let mut mags: Vec<f64> = triples.iter().map(|t| t.2.norm()).collect();
2781        mags.sort_by(|a, b| b.partial_cmp(a).expect("no NaN magnitudes"));
2782        // `t` = the n-th largest magnitude.
2783        let t = mags[n - 1];
2784        let count_gt = mags.iter().filter(|&&m| m > t).count();
2785        let count_eq = mags.iter().filter(|&&m| m == t).count();
2786        // Keep the tie group iff it fits entirely.
2787        let keep_tied = count_gt + count_eq <= n;
2788        triples.retain(|(_, _, c)| {
2789            let m = c.norm();
2790            m > t || (keep_tied && m == t)
2791        });
2792        triples
2793    }
