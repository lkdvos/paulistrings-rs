1//! The rotation layer: a two-delta table `{0, g}` applied to each bucket's own columns by a scan, a linear partner order and a merge-join.
2//!
3//! Contracts: the result agrees with the general coset fill to floating-point tolerance, every output passes `keep_term` exactly as there, and on the Clifford arm the identity residues `|a₀| ≤ ε·|a₁|` are dropped where the general path keeps them.

16/// A table shaped like a Pauli rotation about `g`: an identity entry and one generator entry.
17///
18/// On a pattern the generator entry skips, the identity amplitude is exactly one; the patterns it reaches are closed under `⊕ g`, so a partner never lands on a string that keeps its coefficient.
19#[derive(Clone, Debug)]
20pub(super) struct RotationLayer<'p, const W: usize, C: Coefficient = num_complex::Complex64> {
21    ptm: &'p LocalPtm<W, C>,
22    /// `(mask_x, mask_z)` of `g`.
23    mask: ([u64; W], [u64; W]),
24    /// `a₀[s]`, the identity entry's amplitude.
25    identity: [C; LOCAL_DIM],
26    /// `a₁[s]`, the amplitude taking pattern `s` to `s ⊕ g`.
27    partner: [C; LOCAL_DIM],
28    /// Whether every anticommuting pattern has `|a₀| ≤ ε·|a₁|`, so strings move instead of splitting.
29    clifford: bool,
30    order: PartnerOrder<W>,
31}
32
33impl<'p, const W: usize, C: Coefficient> RotationLayer<'p, W, C> {
34    /// The rotation shape of `ptm`, or `None` if it has another.
35    pub(super) fn of(ptm: &'p LocalPtm<W, C>) -> Option<Self> {
36        let [identity, generator] = ptm.deltas() else { return None };
37        if identity.local_delta != 0 {
38            return None;
39        }
40        let dim = 1usize << (2 * ptm.k());
41        let delta = generator.local_delta as usize;
42        let mut clifford = true;
43        for s in 0..dim {
44            let (a0, a1) = (identity.amplitude[s], generator.amplitude[s]);
45            if a1 == C::ZERO {
46                if a0 != C::ONE {
47                    return None;
48                }
49            } else {
50                if generator.amplitude[s ^ delta] == C::ZERO {
51                    return None;
52                }
53                // `cos θ` evaluated at a quarter turn is a rounding residue of `sin θ`, not an amplitude.
54                clifford &= a0.norm_sqr() <= f64::EPSILON * f64::EPSILON * a1.norm_sqr();
55            }
56        }
57        let mask = generator.mask();
58        Some(Self {
59            ptm,
60            mask,
61            identity: identity.amplitude,
62            partner: generator.amplitude,
63            clifford,
64            order: PartnerOrder::new(&mask.0, &mask.1),
65        })
66    }
67
68    /// Append each anticommuting string's partner `(k ⊕ g, c·a₁)` to `partners`, then scale the bucket by `a₀` in place, or on the Clifford arm drop the moved strings.
69    fn scan(&self, bucket: &mut BucketColumns<W, C>, partners: &mut BucketColumns<W, C>) {
70        let len = bucket.len();
71        // One slot past the exact bound, for `append_if`'s discarded write.
72        partners.x.reserve(len + 1);
73        partners.z.reserve(len + 1);
74        partners.coeff.reserve(len + 1);
75        let BucketColumns { x, z, coeff } = bucket;
76        let (mask_x, mask_z) = &self.mask;
77        let mut kept = 0usize;
78        for row in 0..len {
79            let (key_x, key_z, c) = (x[row], z[row], coeff[row]);
80            let pattern = self.ptm.support_bits(&key_x, &key_z);
81            let amplitude = self.partner[pattern];
82            let anticommutes = amplitude.is_nonzero();
83            let partner_x = core::array::from_fn(|w| key_x[w] ^ mask_x[w]);
84            let partner_z = core::array::from_fn(|w| key_z[w] ^ mask_z[w]);
85            // SAFETY: reserved `len + 1` above, and at most `len` rows are published.
86            unsafe {
87                GatherRun::append_if(partners, anticommutes, partner_x, partner_z, c * amplitude)
88            };
89            if self.clifford {
90                x[kept] = key_x;
91                z[kept] = key_z;
92                coeff[kept] = c;
93                kept += usize::from(!anticommutes);
94            } else {
95                // Unconditional: a store only for anticommuting strings mispredicts on every active gate.
96                coeff[row] = c * self.identity[pattern];
97            }
98        }
99        if self.clifford {
100            x.truncate(kept);
101            z.truncate(kept);
102            coeff.truncate(kept);
103        }
104    }
105}

107/// Apply `layer` to one coset, one bucket or the pair `{β, β ⊕ h(g)}`, each partner run merging into the other member.
108pub(super) fn fill_rotation_coset<const W: usize, C: Coefficient, T>(
109    members: &mut [BucketColumns<W, C>],
110    layer: &RotationLayer<'_, W, C>,
111    policy: &T,
112    scratch: &mut CosetScratch<W, C>,
113) where
114    T: TruncationPolicy<W, C> + ?Sized,
115{
116    let size = members.len();
117    debug_assert!(size == 1 || size == 2);
118    // The general path's buffers, idle between its cosets: each member's run holds its partners in `rest` and in key order in `identity`, and `old[0]` is the merge fallback.
119    scratch.runs.resize_with(size.max(scratch.runs.len()), GatherRun::default);
120    if scratch.old.is_empty() {
121        scratch.old.push(BucketColumns::default());
122    }
123    stamp!(stamp);
124    // Every scan precedes every merge, so each partner carries its source's unscaled coefficient.
125    for (j, bucket) in members.iter_mut().enumerate() {
126        let partners = &mut scratch.runs[j ^ (size - 1)].rest;
127        partners.clear();
128        // Counted as the general path's identity rows, whose keys stay put.
129        tally!(scratch.stats.rows_gathered, bucket.len());
130        tally!(scratch.stats.rows_id, bucket.len());
131        layer.scan(bucket, partners);
132    }
133    lap!(stamp, scratch.stats.gather_ns);
134    for (j, bucket) in members.iter_mut().enumerate() {
135        let run = &mut scratch.runs[j];
136        tally!(scratch.stats.rows_gathered, run.rest.len());
137        tally!(scratch.stats.rows_sorted, run.rest.len());
138        layer.order.order(&run.rest, &mut run.identity, scratch.sort.indices());
139        lap!(stamp, scratch.stats.sort_ns);
140        #[cfg_attr(not(test), allow(unused_variables))]
141        let in_place = merge_partners(bucket, run.identity.view(), policy, &mut scratch.old[0]);
142        #[cfg(test)]
143        {
144            scratch.rotation_merges.0 += usize::from(in_place);
145            scratch.rotation_merges.1 += usize::from(!in_place);
146        }
147        lap!(stamp, scratch.stats.merge_ns);
148    }
149    tally!(scratch.stats.cosets, 1);
150    tally!(scratch.stats.runs, size);
151}

153/// Merge the strictly ascending `partners` into the strictly ascending `bucket`, summing equal keys bucket-first and keeping each output iff it is nonzero and `keep_term` passes; returns whether it ran in place.
154///
155/// In place needs spare capacity for every partner that is a new key: with room for every partner the hits go uncounted and the merge starts at that bound, else they are counted; without room the merge runs into `merged`, which is swapped in, as the general path does.
156fn merge_partners<const W: usize, C: Coefficient, T>(
157    bucket: &mut BucketColumns<W, C>,
158    partners: ColumnsRef<'_, W, C>,
159    policy: &T,
160    merged: &mut BucketColumns<W, C>,
161) -> bool
162where
163    T: TruncationPolicy<W, C> + ?Sized,
164{
165    let bound = bucket.len() + partners.coeff.len();
166    let spare = bucket.x.capacity().min(bucket.z.capacity()).min(bucket.coeff.capacity());
167    let total = if spare >= bound { bound } else { bound - count_hits(bucket.view(), partners) };
168    if spare >= total {
169        merge_backward_in_place(bucket, partners, policy, total);
170        true
171    } else {
172        merged.clear();
173        merge_identity_and_rest(bucket.view(), partners, merged, policy);
174        std::mem::swap(bucket, merged);
175        merged.clear();
176        false
177    }
178}

180/// Keys `partners` shares with `bucket`, both strictly ascending.
181fn count_hits<const W: usize, C: Coefficient>(
182    bucket: ColumnsRef<'_, W, C>,
183    partners: ColumnsRef<'_, W, C>,
184) -> usize {
185    let (len, extra) = (bucket.x.len(), partners.x.len());
186    let (mut i, mut j, mut hits) = (0usize, 0usize, 0usize);
187    while i < len && j < extra {
188        let order = (bucket.x[i], bucket.z[i]).cmp(&(partners.x[j], partners.z[j]));
189        // Branchless steps: which side advances is the merge's coin flip.
190        i += usize::from(order.is_le());
191        j += usize::from(order.is_ge());
192        hits += usize::from(order.is_eq());
193    }
194    hits
195}

197/// [`merge_partners`] in the bucket's own buffer: written from the back of `total`, at least the length before drops, then shifted to the front over the slots hits and drops left.
198fn merge_backward_in_place<const W: usize, C: Coefficient, T>(
199    bucket: &mut BucketColumns<W, C>,
200    partners: ColumnsRef<'_, W, C>,
201    policy: &T,
202    total: usize,
203) where
204    T: TruncationPolicy<W, C> + ?Sized,
205{
206    let ColumnsRef { x: p_x, z: p_z, coeff: p_c } = partners;
207    let (len, extra) = (bucket.len(), p_c.len());
208    debug_assert!(total >= len + extra - count_hits(bucket.view(), partners));
209    debug_assert!(bucket.x.capacity() >= total);
210    debug_assert!(bucket.z.capacity() >= total);
211    debug_assert!(bucket.coeff.capacity() >= total);
212    debug_assert!(p_x.windows(2).zip(p_z.windows(2)).all(|(x, z)| (x[0], z[0]) < (x[1], z[1])));
213    let x = bucket.x.as_mut_ptr();
214    let z = bucket.z.as_mut_ptr();
215    let c = bucket.coeff.as_mut_ptr();
216    // `i` and `j` count the unread rows of each input, `w` the first written slot, and `h` the hits among the unread rows.
217    // A step writes at most one slot per output key it consumes, so `w ≥ i + j − h ≥ i`, since each unread hit pairs an unread partner: a write never lands on an unread bucket row.
218    let (mut i, mut j, mut w) = (len, extra, total);
219    // SAFETY: every read is below `i ≤ len`, the initialized prefix; every write is at `w - 1 ≥ i` and below `total ≤ capacity`.
220    unsafe {
221        let emit = |w: &mut usize, key_x: [u64; W], key_z: [u64; W], sum: C| {
222            if sum != C::ZERO && policy.keep_term(&key_x, &key_z, sum) {
223                *w -= 1;
224                x.add(*w).write(key_x);
225                z.add(*w).write(key_z);
226                c.add(*w).write(sum);
227            }
228        };
229        while i > 0 && j > 0 {
230            let (b_x, b_z) = (*x.add(i - 1), *z.add(i - 1));
231            let (q_x, q_z) = (p_x[j - 1], p_z[j - 1]);
232            match (b_x, b_z).cmp(&(q_x, q_z)) {
233                std::cmp::Ordering::Greater => {
234                    i -= 1;
235                    emit(&mut w, b_x, b_z, *c.add(i));
236                }
237                std::cmp::Ordering::Less => {
238                    j -= 1;
239                    emit(&mut w, q_x, q_z, p_c[j]);
240                }
241                std::cmp::Ordering::Equal => {
242                    i -= 1;
243                    j -= 1;
244                    emit(&mut w, b_x, b_z, *c.add(i) + p_c[j]);
245                }
246            }
247        }
248        while j > 0 {
249            j -= 1;
250            emit(&mut w, p_x[j], p_z[j], p_c[j]);
251        }
252        // Below the last write the bucket's rows may already sit in their slots, and are then not rewritten.
253        while i > 0 {
254            i -= 1;
255            let (key_x, key_z, sum) = (*x.add(i), *z.add(i), *c.add(i));
256            if sum != C::ZERO && policy.keep_term(&key_x, &key_z, sum) {
257                w -= 1;
258                if w != i {
259                    x.add(w).write(key_x);
260                    z.add(w).write(key_z);
261                    c.add(w).write(sum);
262                }
263            }
264        }
265        let out = total - w;
266        if w > 0 {
267            std::ptr::copy(x.add(w), x, out);
268            std::ptr::copy(z.add(w), z, out);
269            std::ptr::copy(c.add(w), c, out);
270        }
271        bucket.x.set_len(out);
272        bucket.z.set_len(out);
273        bucket.coeff.set_len(out);
274    }
275}
