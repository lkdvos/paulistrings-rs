8/// Merge the strictly ascending `partners` into the strictly ascending `bucket`, summing equal keys bucket-first and keeping each output iff it is nonzero and `keep_term` passes; returns whether it ran in place.
9///
10/// In place needs spare capacity for every partner that is a new key: with room for every partner the hits go uncounted and the merge starts at that bound, else they are counted; without room the merge runs into `merged`, which is swapped in, as the general path does.
11/// `dirty`, when given, lists (as the scan packs them) the only rows whose coefficients changed, every other row already passing `keep_term`; a bucket with few such rows and partners merges by [`merge_sparse_in_place`].
12/// `new_keys` promises that no partner's key is in the bucket, as after [`fold_hits`].
13pub(super) fn merge_partners<const W: usize, C: Coefficient, T>(
14    bucket: &mut BucketColumns<W, C>,
15    partners: ColumnsRef<'_, W, C>,
16    policy: &T,
17    merged: &mut BucketColumns<W, C>,
18    dirty: Option<&[u32]>,
19    new_keys: bool,
20) -> bool
21where
22    T: TruncationPolicy<W, C> + ?Sized,
23{
24    let bound = bucket.len() + partners.coeff.len();
25    let spare = bucket.x.capacity().min(bucket.z.capacity()).min(bucket.coeff.capacity());
26    if let Some(dirty) = dirty {
27        let events = partners.coeff.len() + dirty.len();
28        if spare >= bound && events * SPARSE_MERGE_SEGMENT <= bucket.len() {
29            merge_sparse_in_place(bucket, partners, dirty, policy, bound);
30            return true;
31        }
32    }
33    let total = if spare >= bound || new_keys {
34        bound
35    } else {
36        bound - count_hits(bucket.view(), partners)
37    };
38    if spare >= total {
39        merge_backward_in_place(bucket, partners, policy, total);
40        true
41    } else {
42        merged.clear();
43        merge_identity_and_rest(bucket.view(), partners, merged, policy);
44        std::mem::swap(bucket, merged);
45        merged.clear();
46        false
47    }
48}

50/// Add each partner whose key is one of the bucket's `dirty` rows (as the scan packs them) to that row's coefficient, bucket-first as the merge sums, and keep the other partners in order, less those `floor` refuses (see [`TruncationPolicy::coefficient_floor`]).
51///
52/// A partner of an anticommuting string anticommutes too, so it can only equal a listed row: `dirty` must list every anticommuting row of `bucket`.
53pub(super) fn fold_hits<const W: usize, C: Coefficient>(
54    bucket: &mut BucketColumns<W, C>,
55    dirty: &[u32],
56    partners: &mut BucketColumns<W, C>,
57    floor: f64,
58) {
59    let floor_sqr = if floor >= 0.0 { floor * floor } else { -1.0 };
60    let BucketColumns { x, z, coeff } = bucket;
61    let BucketColumns { x: p_x, z: p_z, coeff: p_c } = partners;
62    let (mut d, mut kept) = (0usize, 0usize);
63    for j in 0..p_c.len() {
64        let key = (p_x[j], p_z[j]);
65        let mut hit = None;
66        while let Some(&packed) = dirty.get(d) {
67            let candidate = (packed >> 4) as usize;
68            match (x[candidate], z[candidate]).cmp(&key) {
69                std::cmp::Ordering::Less => d += 1,
70                std::cmp::Ordering::Equal => {
71                    hit = Some(candidate);
72                    d += 1;
73                    break;
74                }
75                std::cmp::Ordering::Greater => break,
76            }
77        }
78        let c = p_c[j];
79        if let Some(row) = hit {
80            coeff[row] += c;
81            continue;
82        }
83        p_x[kept] = key.0;
84        p_z[kept] = key.1;
85        p_c[kept] = c;
86        kept += usize::from(c.norm_sqr() > floor_sqr);
87    }
88    p_x.truncate(kept);
89    p_z.truncate(kept);
90    p_c.truncate(kept);
91}

93/// Fewest rows per partner or dirty row at which [`merge_sparse_in_place`] beats the per-row merge.
94const SPARSE_MERGE_SEGMENT: usize = 8;

96/// [`merge_backward_in_place`] for a bucket whose rows outside `dirty` keep their coefficients and already pass `keep_term`: the clean rows between consecutive events move up as blocks, and only partners and dirty rows take the per-row step; rows below the lowest event are not touched.
97pub(super) fn merge_sparse_in_place<const W: usize, C: Coefficient, T>(
98    bucket: &mut BucketColumns<W, C>,
99    partners: ColumnsRef<'_, W, C>,
100    dirty: &[u32],
101    policy: &T,
102    total: usize,
103) where
104    T: TruncationPolicy<W, C> + ?Sized,
105{
106    let ColumnsRef { x: p_x, z: p_z, coeff: p_c } = partners;
107    let (len, extra) = (bucket.len(), p_c.len());
108    debug_assert_eq!(total, len + extra);
109    debug_assert!(bucket.x.capacity() >= total);
110    debug_assert!(bucket.z.capacity() >= total);
111    debug_assert!(bucket.coeff.capacity() >= total);
112    debug_assert!(dirty.windows(2).all(|pair| pair[0] >> 4 < pair[1] >> 4));
113    let x = bucket.x.as_mut_ptr();
114    let z = bucket.z.as_mut_ptr();
115    let c = bucket.coeff.as_mut_ptr();
116    // As in `merge_backward_in_place`; `d` counts the unread dirty rows, each below `i`, and the gap `w − i` never falls below the unread partners `j`.
117    let (mut i, mut j, mut d, mut w) = (len, extra, dirty.len(), total);
118    // SAFETY: every read is below `i ≤ len`; every write and block destination lies in `[i, total)`, and `ptr::copy` allows the overlap of a block with its destination.
119    unsafe {
120        let key = |row: usize| (*x.add(row), *z.add(row));
121        let emit = |w: &mut usize, key_x: [u64; W], key_z: [u64; W], sum: C| {
122            if sum != C::ZERO && policy.keep_term(&key_x, &key_z, sum) {
123                *w -= 1;
124                x.add(*w).write(key_x);
125                z.add(*w).write(key_z);
126                c.add(*w).write(sum);
127            }
128        };
129        let dirty_row = |d: usize| (dirty[d - 1] >> 4) as usize;
130        while j > 0 || d > 0 {
131            // Rows above the next dirty row and above the next partner's key are clean and already ordered.
132            let mut low = if d > 0 { dirty_row(d) + 1 } else { 0 };
133            if j > 0 {
134                let target = (p_x[j - 1], p_z[j - 1]);
135                let mut high = i;
136                while low < high {
137                    let middle = low + (high - low) / 2;
138                    if key(middle) <= target {
139                        low = middle + 1;
140                    } else {
141                        high = middle;
142                    }
143                }
144            }
145            let block = i - low;
146            if block > 0 && w != i {
147                std::ptr::copy(x.add(low), x.add(w - block), block);
148                std::ptr::copy(z.add(low), z.add(w - block), block);
149                std::ptr::copy(c.add(low), c.add(w - block), block);
150            }
151            w -= block;
152            i = low;
153            let order = match (i > 0, j > 0) {
154                (true, true) => key(i - 1).cmp(&(p_x[j - 1], p_z[j - 1])),
155                (true, false) => std::cmp::Ordering::Greater,
156                _ => std::cmp::Ordering::Less,
157            };
158            match order {
159                std::cmp::Ordering::Less => {
160                    j -= 1;
161                    emit(&mut w, p_x[j], p_z[j], p_c[j]);
162                }
163                std::cmp::Ordering::Equal => {
164                    i -= 1;
165                    j -= 1;
166                    d -= usize::from(d > 0 && dirty_row(d) == i);
167                    let (key_x, key_z) = key(i);
168                    emit(&mut w, key_x, key_z, *c.add(i) + p_c[j]);
169                }
170                std::cmp::Ordering::Greater => {
171                    debug_assert!(d > 0 && dirty_row(d) == i - 1);
172                    i -= 1;
173                    d -= 1;
174                    let (key_x, key_z) = key(i);
175                    emit(&mut w, key_x, key_z, *c.add(i));
176                }
177            }
178        }
179        let out = total - w;
180        if w != i {
181            std::ptr::copy(x.add(w), x.add(i), out);
182            std::ptr::copy(z.add(w), z.add(i), out);
183            std::ptr::copy(c.add(w), c.add(i), out);
184        }
185        bucket.x.set_len(i + out);
186        bucket.z.set_len(i + out);
187        bucket.coeff.set_len(i + out);
188    }
189}

191/// Keys `partners` shares with `bucket`, both strictly ascending.
192fn count_hits<const W: usize, C: Coefficient>(
193    bucket: ColumnsRef<'_, W, C>,
194    partners: ColumnsRef<'_, W, C>,
195) -> usize {
196    let (len, extra) = (bucket.x.len(), partners.x.len());
197    let (mut i, mut j, mut hits) = (0usize, 0usize, 0usize);
198    while i < len && j < extra {
199        let order = (bucket.x[i], bucket.z[i]).cmp(&(partners.x[j], partners.z[j]));
200        // Branchless steps: which side advances is the merge's coin flip.
201        i += usize::from(order.is_le());
202        j += usize::from(order.is_ge());
203        hits += usize::from(order.is_eq());
204    }
205    hits
206}

208/// [`merge_partners`] in the bucket's own buffer: written from the back of `total`, at least the length before drops, then shifted to the front over the slots hits and drops left.
209pub(super) fn merge_backward_in_place<const W: usize, C: Coefficient, T>(
210    bucket: &mut BucketColumns<W, C>,
211    partners: ColumnsRef<'_, W, C>,
212    policy: &T,
213    total: usize,
214) where
215    T: TruncationPolicy<W, C> + ?Sized,
216{
217    let ColumnsRef { x: p_x, z: p_z, coeff: p_c } = partners;
218    let (len, extra) = (bucket.len(), p_c.len());
219    debug_assert!(total >= len + extra - count_hits(bucket.view(), partners));
220    debug_assert!(bucket.x.capacity() >= total);
221    debug_assert!(bucket.z.capacity() >= total);
222    debug_assert!(bucket.coeff.capacity() >= total);
223    debug_assert!(p_x.windows(2).zip(p_z.windows(2)).all(|(x, z)| (x[0], z[0]) < (x[1], z[1])));
224    let x = bucket.x.as_mut_ptr();
225    let z = bucket.z.as_mut_ptr();
226    let c = bucket.coeff.as_mut_ptr();
227    // `i` and `j` count the unread rows of each input, `w` the first written slot, and `h` the hits among the unread rows.
228    // A step writes at most one slot per output key it consumes, so `w ≥ i + j − h ≥ i`, since each unread hit pairs an unread partner: a write never lands on an unread bucket row.
229    let (mut i, mut j, mut w) = (len, extra, total);
230    // SAFETY: every read is below `i ≤ len`, the initialized prefix; every write is at `w - 1 ≥ i` and below `total ≤ capacity`.
231    unsafe {
232        let emit = |w: &mut usize, key_x: [u64; W], key_z: [u64; W], sum: C| {
233            if sum != C::ZERO && policy.keep_term(&key_x, &key_z, sum) {
234                *w -= 1;
235                x.add(*w).write(key_x);
236                z.add(*w).write(key_z);
237                c.add(*w).write(sum);
238            }
239        };
240        while i > 0 && j > 0 {
241            let (b_x, b_z) = (*x.add(i - 1), *z.add(i - 1));
242            let (q_x, q_z) = (p_x[j - 1], p_z[j - 1]);
243            match (b_x, b_z).cmp(&(q_x, q_z)) {
244                std::cmp::Ordering::Greater => {
245                    i -= 1;
246                    emit(&mut w, b_x, b_z, *c.add(i));
247                }
248                std::cmp::Ordering::Less => {
249                    j -= 1;
250                    emit(&mut w, q_x, q_z, p_c[j]);
251                }
252                std::cmp::Ordering::Equal => {
253                    i -= 1;
254                    j -= 1;
255                    emit(&mut w, b_x, b_z, *c.add(i) + p_c[j]);
256                }
257            }
258        }
259        while j > 0 {
260            j -= 1;
261            emit(&mut w, p_x[j], p_z[j], p_c[j]);
262        }
263        // Below the last write the bucket's rows may already sit in their slots, and are then not rewritten.
264        while i > 0 {
265            i -= 1;
266            let (key_x, key_z, sum) = (*x.add(i), *z.add(i), *c.add(i));
267            if sum != C::ZERO && policy.keep_term(&key_x, &key_z, sum) {
268                w -= 1;
269                if w != i {
270                    x.add(w).write(key_x);
271                    z.add(w).write(key_z);
272                    c.add(w).write(sum);
273                }
274            }
275        }
276        let out = total - w;
277        if w > 0 {
278            std::ptr::copy(x.add(w), x, out);
279            std::ptr::copy(z.add(w), z, out);
280            std::ptr::copy(c.add(w), c, out);
281        }
282        bucket.x.set_len(out);
283        bucket.z.set_len(out);
284        bucket.coeff.set_len(out);
285    }
