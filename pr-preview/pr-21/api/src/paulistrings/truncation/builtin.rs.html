19/// Drop terms whose coefficient magnitude is at most `epsilon`.
20///
21/// The test is `|c|² > ε²`, so magnitudes below about `1.6e-162` count as zero.
22#[derive(Clone, Copy, Debug, PartialEq)]
23pub struct CoefficientThreshold(
24    /// Magnitude threshold. Terms with `|coeff| <= epsilon` are dropped.
25    pub f64,
26);
27
28impl<const W: usize> TruncationPolicy<W> for CoefficientThreshold {
29    fn keep_term(&self, _x: &[u64; W], _z: &[u64; W], c: Complex64) -> bool {
30        let eps = self.0;
31        eps < 0.0 || c.norm_sqr() > eps * eps
32    }
33
34    fn finalizes_layer(&self) -> bool {
35        false
36    }
37}

39/// Drop terms whose Pauli weight (number of non-identity qubits) exceeds `k`.
40#[derive(Clone, Copy, Debug, PartialEq)]
41pub struct WeightCutoff(
42    /// Maximum allowed Pauli weight. Terms with weight `> k` are dropped.
43    pub u32,
44);

46impl<const W: usize> TruncationPolicy<W> for WeightCutoff {
47    fn keep_term(&self, x: &[u64; W], z: &[u64; W], _c: Complex64) -> bool {
48        let weight: u32 = (0..W).map(|i| (x[i] | z[i]).count_ones()).sum();
49        weight <= self.0
50    }

53        false
54    }

57/// Retain the `n` terms of largest coefficient magnitude.
58///
59/// With `t` the `n`-th largest magnitude, every term above `t` is kept, plus as many terms at exactly `t` as fit in `n`, the first in bucket order; which tied terms survive is not a semantic promise.
60/// A sum of more than `n` terms keeps exactly `n`, and `n == 0` clears; a NaN coefficient is never kept, so a sum with fewer than `n` other terms keeps those.
61/// Magnitudes are compared squared, so those below about `1.6e-162` tie at zero.
62#[derive(Clone, Copy, Debug, PartialEq)]
63pub struct TopN(
64    /// The retained term count.
65    pub usize,
66);
67
68impl<const W: usize> TruncationPolicy<W> for TopN {
69    fn finalize_layer(&self, sum: &mut PauliSum<W>) {
70        let n = self.0;
71        if sum.len() <= n {
72            return;
73        }
74        if n == 0 {
75            sum.clear();
76            return;
77        }
78
79        let total = sum.len();
80        let mut buffer = MAGNITUDES.take();
81        if buffer.len() < total {
82            buffer.resize(total, 0.0);
83        }
84        let magnitudes = &mut buffer[..total];
85        {
86            let view = &*sum;
87            let num_buckets = view.num_buckets();
88            let mut handles: Vec<&mut [f64]> = Vec::with_capacity(num_buckets);
89            let mut rest: &mut [f64] = magnitudes;
90            for b in 0..num_buckets {
91                let (head, tail) = rest.split_at_mut(view.bucket_len(b));
92                handles.push(head);
93                rest = tail;
94            }
95            debug_assert!(rest.is_empty(), "bucket lengths must sum to len()");
96            handles.into_par_iter().enumerate().for_each(|(b, dst)| {
97                for (d, c) in dst.iter_mut().zip(view.bucket(b).2.iter()) {
98                    *d = c.norm_sqr();
99                }
100            });
101        }
102
103        // A NaN sorts below every number, so it becomes the boundary only when fewer than `n` terms are finite.
104        magnitudes.select_nth_unstable_by(n - 1, |a, b| {
105            b.partial_cmp(a)
106                .unwrap_or_else(|| a.is_nan().cmp(&b.is_nan()))
107        });
108        let boundary = magnitudes[n - 1];
109        // Everything above the pivot sits before it after the selection.
110        let above = magnitudes[..n - 1]
111            .par_iter()
112            .filter(|&&m| m > boundary)
113            .count();
114
115        MAGNITUDES.set(buffer);
116
117        retain_with_boundary_quota(
118            sum,
119            |m| {
120                if m.is_nan() {
121                    core::cmp::Ordering::Less
122                } else {
123                    // Unordered only against a NaN boundary, which every finite term is above.
124                    m.partial_cmp(&boundary)
125                        .unwrap_or(core::cmp::Ordering::Greater)
126                }
127            },
128            n - above,
129        );
130    }
131
132    /// Panics above one partition, where the `n`-th largest of a split sum has no collective form yet.
133    fn finalize_layer_partitioned(&self, local: &mut PauliSum<W>, collectives: &dyn Collectives) {
134        assert!(
135            collectives.size() == 1,
136            "TopN({}) on {} partitions: partitioned exact TopN is not yet supported; ApproxTopN \
137             is the partitioned alternative.",
138            self.0,
139            collectives.size(),
140        );
141        self.finalize_layer(local);
142    }
143}

145/// How many boundary terms each bucket keeps: `quota` handed out in bucket order, each bucket taking at most its own count.
146fn bucket_quotas(boundary_counts: &[usize], quota: usize) -> Vec<usize> {
147    let mut remaining = quota;
148    boundary_counts
149        .iter()
150        .map(|&count| {
151            let taken = count.min(remaining);
152            remaining -= taken;
153            taken
154        })
155        .collect()
156}

158/// Keep every term `classify` puts above the boundary (`Greater`) and the first `quota` it puts at it (`Equal`), in bucket order.
159pub(crate) fn retain_with_boundary_quota<const W: usize>(
160    sum: &mut PauliSum<W>,
161    classify: impl Fn(f64) -> core::cmp::Ordering + Sync,
162    quota: usize,
163) {
164    use core::cmp::Ordering;
165    let boundary_counts: Vec<usize> = (0..sum.num_buckets())
166        .into_par_iter()
167        .map(|b| {
168            sum.bucket(b)
169                .2
170                .iter()
171                .filter(|c| classify(c.norm_sqr()) == Ordering::Equal)
172                .count()
173        })
174        .collect();
175    let quotas = bucket_quotas(&boundary_counts, quota);
176    sum.retain_per_bucket(|b| {
177        let mut remaining = quotas[b];
178        let classify = &classify;
179        move |_: &[u64; W], _: &[u64; W], c: Complex64| match classify(c.norm_sqr()) {
180            Ordering::Greater => true,
181            Ordering::Equal if remaining > 0 => {
182                remaining -= 1;
183                true
184            }
185            Ordering::Equal | Ordering::Less => false,
186        }
187    });
188}

191pub(crate) const APPROX_BINS: usize = 2048;

193/// Retain `n` terms, cutting at an octave of `|c|²` instead of selecting exactly.
194///
195/// Every octave of `|c|²` that fits whole is kept from the top down; the first that does not fit, the boundary octave, contributes as many terms as fit in `n`, the first in bucket order (rank order across partitions).
196/// A sum of more than `n` terms keeps exactly `n` (fewer only if coefficients are NaN), and `n == 0` clears.
197/// Which boundary-octave terms survive is not a semantic promise.
198/// Partition-exact: a split sum's octave histogram is the sum of its partitions', so every partition cuts at the boundary one partition would.
199#[derive(Clone, Copy, Debug, PartialEq)]
200pub struct ApproxTopN(
201    /// The retained term count.
202    pub usize,
203);

205/// [`octave_edge`]'s decision, shared by the host, partitioned and device paths.
206#[derive(Clone, Copy, Debug, PartialEq)]
207pub(crate) enum EdgeDecision {
208    /// The whole sum fits within `n`: nothing is dropped.
209    KeepAll,
210    /// `n == 0`.
211    Clear,
212    /// Keep every term above octave `bin` and `quota` terms of octave `bin`, `kept_above + quota == n` in all.
213    Boundary {
214        /// The boundary octave, `norm_sqr().to_bits() >> 52`.
215        bin: usize,
216        /// Terms above the boundary octave, across the whole sum.
217        kept_above: usize,
218        /// Boundary-octave terms kept, across the whole sum.
219        quota: usize,
220    },
221}
222
223/// The octave of `|c|²`, [`APPROX_BINS`] wide.
224fn octave(norm_sqr: f64) -> usize {
225    (norm_sqr.to_bits() >> 52) as usize
226}
227
228/// `|c|²` against the boundary octave `bin`; a NaN is below every octave.
229pub(crate) fn classify_octave(norm_sqr: f64, bin: usize) -> core::cmp::Ordering {
230    if norm_sqr.is_nan() {
231        core::cmp::Ordering::Less
232    } else {
233        octave(norm_sqr).cmp(&bin)
234    }
235}

237/// Population of each octave of `|c|²`, binned by the exponent bits `norm_sqr().to_bits() >> 52`.
238pub(crate) fn octave_histogram<const W: usize>(sum: &PauliSum<W>) -> [u32; APPROX_BINS] {
239    debug_assert!(sum.len() <= u32::MAX as usize, "len exceeds bin counters");
240    let num_buckets = sum.num_buckets();
241    let tasks = (rayon::current_num_threads() * 4).clamp(1, num_buckets);
242    (0..tasks)
243        .into_par_iter()
244        .map(|t| {
245            let mut h = [0u32; APPROX_BINS];
246            for b in (num_buckets * t / tasks)..(num_buckets * (t + 1) / tasks) {
247                for c in sum.bucket(b).2 {
248                    h[octave(c.norm_sqr())] += 1;
249                }
250            }
251            h
252        })
253        .reduce(
254            || [0u32; APPROX_BINS],
255            |mut a, b| {
256                for (x, y) in a.iter_mut().zip(b.iter()) {
257                    *x += y;
258                }
259                a
260            },
261        )
262}

264/// The boundary octave [`ApproxTopN`] cuts at, from a histogram of the whole sum, local or all-reduced.
265pub(crate) fn octave_edge<C>(hist: &[C], total_len: usize, n: usize) -> EdgeDecision
266where
267    C: Copy + Into<u64>,
268{
269    debug_assert_eq!(hist.len(), APPROX_BINS, "octave_edge: wrong histogram size");
270    if total_len <= n {
271        return EdgeDecision::KeepAll;
272    }
273    if n == 0 {
274        return EdgeDecision::Clear;
275    }
276
277    let mut kept_above = 0usize;
278    for bin in (0..APPROX_BINS).rev() {
279        let population = hist[bin].into() as usize;
280        if kept_above + population > n {
281            return EdgeDecision::Boundary {
282                bin,
283                kept_above,
284                quota: n - kept_above,
285            };
286        }
287        kept_above += population;
288    }
289    unreachable!("a histogram of more than n terms overflows n at some octave")
290}

292/// Apply an [`EdgeDecision`] to one sum or partition, keeping `local_quota` boundary-octave terms.
293pub(crate) fn retain_at_edge<const W: usize>(
294    sum: &mut PauliSum<W>,
295    edge: EdgeDecision,
296    local_quota: usize,
297) {
298    match edge {
299        EdgeDecision::KeepAll => {}
300        EdgeDecision::Clear => sum.clear(),
301        EdgeDecision::Boundary { bin, .. } => {
302            retain_with_boundary_quota(sum, |m| classify_octave(m, bin), local_quota);
303        }
304    }
305}

307/// This partition's share of a boundary quota: what is left of `quota` after the lower ranks' boundary terms, at most `local_count`.
308///
309/// One `allreduce_sum_u64` of a length-`P` vector holding each rank's `local_count` in its slot; every rank calls it with the same `quota`.
310pub(crate) fn local_boundary_quota(
311    collectives: &dyn Collectives,
312    local_count: usize,
313    quota: usize,
314) -> usize {
315    let rank = collectives.rank() as usize;
316    let mut counts = vec![0u64; collectives.size() as usize];
317    counts[rank] = local_count as u64;
318    collectives.allreduce_sum_u64(&mut counts);
319    let below: u64 = counts[..rank].iter().sum();
320    (quota as u64).saturating_sub(below).min(local_count as u64) as usize
321}

323impl<const W: usize> TruncationPolicy<W> for ApproxTopN {
324    fn finalize_layer(&self, sum: &mut PauliSum<W>) {
325        let n = self.0;
326        let total = sum.len();
327        if total <= n {
328            return;
329        }
330        if n == 0 {
331            sum.clear();
332            return;
333        }
334
335        let edge = octave_edge(&octave_histogram(sum), total, n);
336        let quota = match edge {
337            EdgeDecision::Boundary { quota, .. } => quota,
338            EdgeDecision::KeepAll | EdgeDecision::Clear => 0,
339        };
340        retain_at_edge(sum, edge, quota);
341    }
342
343    /// One `allreduce_sum_u64` of `[len, histogram…]`, the single-partition edge walk, and at a boundary one more of the per-rank boundary counts for this rank's quota; no early exit, since that would desynchronize the group.
344    fn finalize_layer_partitioned(&self, local: &mut PauliSum<W>, collectives: &dyn Collectives) {
345        if collectives.size() == 1 {
346            self.finalize_layer(local);
347            return;
348        }
349        let histogram = octave_histogram(local);
350
351        // `u64`, since `P` partitions' `u32` bins can overflow a `u32`.
352        let mut packed = [0u64; 1 + APPROX_BINS];
353        packed[0] = local.len() as u64;
354        for (slot, &count) in packed[1..].iter_mut().zip(histogram.iter()) {
355            *slot = u64::from(count);
356        }
357        collectives.allreduce_sum_u64(&mut packed);
358
359        let total = packed[0] as usize;
360        let edge = octave_edge(&packed[1..], total, self.0);
361        let local_quota = match edge {
362            EdgeDecision::Boundary { bin, quota, .. } => {
363                local_boundary_quota(collectives, histogram[bin] as usize, quota)
364            }
365            EdgeDecision::KeepAll | EdgeDecision::Clear => 0,
366        };
367        retain_at_edge(local, edge, local_quota);
368    }
369
370    fn supports_partitioned(&self) -> bool {
371        true
372    }
373}

375/// Replace the sum by one Pauli string, drawn with probability `|c|² / Σ|c|²` and given coefficient `1`, whenever it holds more than `cache` terms.
376///
377/// One seed is one Monte Carlo trajectory: the draw at the `k`-th layer pass is a function of `(seed, k)` only, and the pass counter lives in the policy, so reusing a policy continues its sequence.
378/// Partitioned runs agree with unpartitioned ones in distribution, not string for string, and count passes and collapses on rank 0's policy only.
379/// A collapse panics if the sum's total weight is zero or not finite.
380#[derive(Debug)]
381pub struct CollapseSample {
382    /// Largest sum left untouched.
383    pub cache: usize,
384    /// Trajectory seed.
385    pub seed: u64,
386    /// Layer passes seen.
387    calls: AtomicU64,
388    /// Collapses performed.
389    collapses: AtomicU64,
390}
391
392impl CollapseSample {
393    /// A fresh trajectory: collapse above `cache` terms, draws keyed by `seed`.
394    pub fn new(cache: usize, seed: u64) -> Self {
395        Self {
396            cache,
397            seed,
398            calls: AtomicU64::new(0),
399            collapses: AtomicU64::new(0),
400        }
401    }
402
403    /// Collapses performed so far (rank 0's count in partitioned mode).
404    pub fn collapses(&self) -> u64 {
405        self.collapses.load(Ordering::Relaxed)
406    }
407
408    /// Claim the next pass index.
409    pub(crate) fn next_call(&self) -> u64 {
410        self.calls.fetch_add(1, Ordering::Relaxed)
411    }
412
413    pub(crate) fn count_collapse(&self) -> u64 {
414        self.collapses.fetch_add(1, Ordering::Relaxed) + 1
415    }
416
417    /// The point of the cumulative weight pass `call` draws; panics unless `total` is positive and finite.
418    pub(crate) fn target(&self, call: u64, total: f64, len: usize) -> f64 {
419        assert!(
420            total.is_finite() && total > 0.0,
421            "CollapseSample: cannot draw by weight from {len} terms with Σ|c|² = {total}",
422        );
423        Rng::from_key(&[self.seed, call]).uniform() * total
424    }
425}

427/// `Σ|c|²` of each bucket, in bucket order.
428pub(crate) fn bucket_norms<const W: usize>(sum: &PauliSum<W>) -> Vec<f64> {
429    (0..sum.num_buckets())
430        .into_par_iter()
431        .map(|b| sum.bucket(b).2.iter().map(|c| c.norm_sqr()).sum())
432        .collect()
433}

435/// The slot whose stretch of the running total contains `target`, and the offset into it; past the end, the last positive slot.
436pub(crate) fn pick_slot(
437    weights: impl IntoIterator<Item = f64>,
438    target: f64,
439) -> Option<(usize, f64)> {
440    let mut cumulative = 0.0f64;
441    let mut last = None;
442    for (i, w) in weights.into_iter().enumerate() {
443        if w > 0.0 {
444            if target < cumulative + w {
445                return Some((i, target - cumulative));
446            }
447            last = Some((i, target - cumulative));
448        }
449        cumulative += w;
450    }
451    last
452}

454/// Reduce `sum` to the single string at `target` of its cumulative weight, with coefficient `1`; `norms` is [`bucket_norms`] of `sum`.
455pub(crate) fn collapse_to_target<const W: usize>(
456    sum: &mut PauliSum<W>,
457    norms: &[f64],
458    target: f64,
459) {
460    let (b, rest) = pick_slot(norms.iter().copied(), target)
461        .expect("collapse_to_target: no bucket has positive weight");
462    let (pos, _) = pick_slot(sum.bucket(b).2.iter().map(|c| c.norm_sqr()), rest)
463        .expect("collapse_to_target: a positive bucket has a positive term");
464    let (x, z) = (sum.bucket(b).0[pos], sum.bucket(b).1[pos]);
465    sum.clear();
466    sum.buckets_mut()[b].push(x, z, Complex64::new(1.0, 0.0));
467    sum.recount();
468}

470/// Equal when `cache`, `seed` and both counters are.
471impl PartialEq for CollapseSample {
472    fn eq(&self, other: &Self) -> bool {
473        self.cache == other.cache
474            && self.seed == other.seed
475            && self.calls.load(Ordering::Relaxed) == other.calls.load(Ordering::Relaxed)
476            && self.collapses() == other.collapses()
477    }
478}

480impl<const W: usize> TruncationPolicy<W> for CollapseSample {
481    fn finalize_layer(&self, sum: &mut PauliSum<W>) {
482        let call = self.next_call();
483        let len = sum.len();
484        if len <= self.cache {
485            return;
486        }
487        let norms = bucket_norms(sum);
488        let total: f64 = norms.iter().sum();
489        collapse_to_target(sum, &norms, self.target(call, total, len));
490        let n = self.count_collapse();
491        log::debug!(
492            target: LOG_TARGET,
493            "collapse_sample: {len} terms, sum |c|^2 = {total:.6e}, collapsed to one (collapse {n})",
494        );
495    }
496
497    /// Reduces `[len, pass]` (the pass from rank 0) and the per-partition `Σ|c|²`, then picks a partition by weight and a term within it, as [`finalize_layer`](TruncationPolicy::finalize_layer) picks a bucket and then a term.
498    fn finalize_layer_partitioned(&self, local: &mut PauliSum<W>, collectives: &dyn Collectives) {
499        if collectives.size() == 1 {
500            self.finalize_layer(local);
501            return;
502        }
503        let rank = collectives.rank() as usize;
504        let call = if rank == 0 { self.next_call() } else { 0 };
505        let mut head = [local.len() as u64, call];
506        collectives.allreduce_sum_u64(&mut head);
507        let [len, call] = head;
508        if len as usize <= self.cache {
509            return;
510        }
511
512        let norms = bucket_norms(local);
513        let mut weights = vec![0.0f64; collectives.size() as usize];
514        weights[rank] = norms.iter().sum();
515        collectives.allreduce_sum_f64(&mut weights);
516        let total: f64 = weights.iter().sum();
517        let target = self.target(call, total, len as usize);
518        let (chosen, rest) = pick_slot(weights.iter().copied(), target)
519            .expect("CollapseSample: a positive total has a positive partition");
520        if rank == chosen {
521            collapse_to_target(local, &norms, rest);
522        } else {
523            local.clear();
524        }
525        if rank == 0 {
526            let collapses = self.count_collapse();
527            log::debug!(
528                target: LOG_TARGET,
529                "collapse_sample: {len} terms over {} partitions, sum |c|^2 = {total:.6e}, \
530                 collapsed to one on partition {chosen} (collapse {collapses})",
531                collectives.size(),
532            );
533        }
534    }
535
536    fn supports_partitioned(&self) -> bool {
537        true
538    }
539}

541/// Both policies: a term is kept if both keep it, and both layer passes run, first then second.
542#[derive(Clone, Copy, Debug, PartialEq)]
543pub struct And<A, B>(
544    /// First policy.
545    pub A,
546    /// Second policy.
547    pub B,
548);
549
550impl<const W: usize, A, B> TruncationPolicy<W> for And<A, B>
551where
552    A: TruncationPolicy<W>,
553    B: TruncationPolicy<W>,
554{
555    fn keep_term(&self, x: &[u64; W], z: &[u64; W], c: Complex64) -> bool {
556        self.0.keep_term(x, z, c) && self.1.keep_term(x, z, c)
557    }
558
559    fn finalize_layer(&self, sum: &mut PauliSum<W>) {
560        self.0.finalize_layer(sum);
561        self.1.finalize_layer(sum);
562    }
563
564    fn finalizes_layer(&self) -> bool {
565        self.0.finalizes_layer() || self.1.finalizes_layer()
566    }
567
568    fn finalize_layer_partitioned(&self, local: &mut PauliSum<W>, collectives: &dyn Collectives) {
569        self.0.finalize_layer_partitioned(local, collectives);
570        self.1.finalize_layer_partitioned(local, collectives);
571    }
572
573    fn supports_partitioned(&self) -> bool {
574        self.0.supports_partitioned() && self.1.supports_partitioned()
575    }
576}

578/// Either of two per-term filters: a term is kept if either keeps it.
579///
580/// `Or` combines per-term filters only: an operand with a layer pass ([`TopN`], [`ApproxTopN`], [`CollapseSample`], or an [`And`] containing one) is rejected, since "either keeps" has no meaning for a pass over the whole sum.
581/// Use [`And`] to add a filter to a layer pass.
582#[derive(Clone, Copy, Debug, PartialEq)]
583pub struct Or<A, B>(pub(crate) A, pub(crate) B);
584
585impl<A, B> Or<A, B> {
586    /// Either of `first` and `second`.
587    ///
588    /// The operands are checked in [`finalizes_layer`](TruncationPolicy::finalizes_layer), which every driver calls before the first layer, since a layer pass is a property of the policy at a width `W` and `new` has no `W`.
589    pub fn new(first: A, second: B) -> Self {
590        Self(first, second)
591    }
592}
593
594/// Why an `Or` of these operands is rejected, or `None` if both are per-term filters.
595pub(crate) fn or_operand_error(first_has_pass: bool, second_has_pass: bool) -> Option<String> {
596    let which = match (first_has_pass, second_has_pass) {
597        (false, false) => return None,
598        (true, false) => "its first operand has",
599        (false, true) => "its second operand has",
600        (true, true) => "both operands have",
601    };
602    Some(format!(
603        "Or combines per-term filters only, but {which} a layer pass (TopN, ApproxTopN or \
604         CollapseSample, alone or inside an And); use And to add a filter to a layer pass"
605    ))
606}
607
608impl<const W: usize, A, B> TruncationPolicy<W> for Or<A, B>
609where
610    A: TruncationPolicy<W>,
611    B: TruncationPolicy<W>,
612{
613    fn keep_term(&self, x: &[u64; W], z: &[u64; W], c: Complex64) -> bool {
614        self.0.keep_term(x, z, c) || self.1.keep_term(x, z, c)
615    }
616
617    /// `false`; panics if either operand has a layer pass.
618    fn finalizes_layer(&self) -> bool {
619        if let Some(message) = or_operand_error(self.0.finalizes_layer(), self.1.finalizes_layer())
620        {
621            panic!("{message}");
622        }
623        false
624    }
625}
