1//! Product-state read-outs: [`ProductBasis`] and the [`PauliSum`] expectation value in one.
2
3use num_complex::Complex64;
4
5use crate::pauli_sum::PauliSum;
6
7/// One of the three single-qubit Pauli axes: the axis a [`ProductBasis`] qubit is an eigenstate of.
8#[derive(Clone, Copy, Debug, PartialEq, Eq)]
9pub enum PauliAxis {
10    /// `X`, with eigenstates `|+⟩` (`+1`) and `|-⟩` (`-1`).
11    X,
12    /// `Y`, with eigenstates `|+i⟩` (`+1`) and `|-i⟩` (`-1`).
13    Y,
14    /// `Z`, with eigenstates `|0⟩` (`+1`) and `|1⟩` (`-1`).
15    Z,
16}
17
18impl PauliAxis {
19    /// The axis Pauli's symplectic bits `(x, z)`.
20    const fn bits(self) -> (bool, bool) {
21        match self {
22            PauliAxis::X => (true, false),
23            PauliAxis::Y => (true, true),
24            PauliAxis::Z => (false, true),
25        }
26    }
27}
28
29/// A product of single-qubit stabilizer states, one per qubit: on qubit `q` an axis `A_q ∈ {X, Y, Z}` with a sign `s_q ∈ {+1, -1}`, i.e. `|ψ⟩ = ⊗_q |A_q, s_q⟩`.
30///
31/// Stored as bit masks in the symplectic key layout; build one with [`ProductBasis::uniform`] or [`ProductBasis::from_axes`].
32/// A term `P` has `⟨ψ|P|ψ⟩ = Π_q ⟨P_q⟩`, with `⟨P_q⟩ = 1` for `I`, `s_q` for `A_q`, and `0` for any other Pauli.
33#[derive(Clone, Debug, PartialEq, Eq)]
34pub struct ProductBasis<const W: usize> {
35    /// `x`-bit of each qubit's axis Pauli.
36    axis_x: [u64; W],
37    /// `z`-bit of each qubit's axis Pauli.
38    axis_z: [u64; W],
39    /// Sign bit per qubit: `1` selects the `-1` eigenstate.
40    negative: [u64; W],
41}
42
43impl<const W: usize> ProductBasis<W> {
44    /// The `+1` eigenstate of `axis` on every qubit: `|+…+⟩`, `|+i…+i⟩` or `|0…0⟩`.
45    pub fn uniform(axis: PauliAxis) -> Self {
46        let (axis_x, axis_z) = axis.bits();
47        let all = |b: bool| if b { [!0u64; W] } else { [0u64; W] };
48        Self {
49            axis_x: all(axis_x),
50            axis_z: all(axis_z),
51            negative: [0u64; W],
52        }
53    }
54
55    /// A basis from per-qubit `(axis, minus)` pairs: item `i` describes qubit `i`, and `minus = true` selects that axis's `-1` eigenstate.
56    ///
57    /// Qubits past the end of the iterator match only an identity factor, so supply one pair per qubit the sum uses.
58    ///
59    /// # Panics
60    ///
61    /// Panics if more than `64 * W` pairs are supplied.
62    pub fn from_axes<I>(axes: I) -> Self
63    where
64        I: IntoIterator<Item = (PauliAxis, bool)>,
65    {
66        let mut out = Self {
67            axis_x: [0u64; W],
68            axis_z: [0u64; W],
69            negative: [0u64; W],
70        };
71        for (q, (axis, minus)) in axes.into_iter().enumerate() {
72            assert!(
73                q < 64 * W,
74                "ProductBasis::from_axes: qubit {q} exceeds the {W}-word width",
75            );
76            let word = q / 64;
77            let bit = 1u64 << (q % 64);
78            let (axis_x, axis_z) = axis.bits();
79            if axis_x {
80                out.axis_x[word] |= bit;
81            }
82            if axis_z {
83                out.axis_z[word] |= bit;
84            }
85            if minus {
86                out.negative[word] |= bit;
87            }
88        }
89        out
90    }
91}
92
93impl<const W: usize> PauliSum<W> {
94    /// Expectation value `⟨ψ|O|ψ⟩` in the product state `basis`; complex since `O` need not be Hermitian.
95    pub fn expectation_product(&self, basis: &ProductBasis<W>) -> Complex64 {
96        self.sum_over_terms(|x, z, coeff| {
97            // An equality on both key halves, not a subset test: an X factor on a Y-axis qubit must not match.
98            let mut mismatch = 0u64;
99            let mut sign_bits = 0u32;
100            for w in 0..W {
101                let support = x[w] | z[w];
102                mismatch |=
103                    (x[w] ^ (support & basis.axis_x[w])) | (z[w] ^ (support & basis.axis_z[w]));
104                sign_bits += (support & basis.negative[w]).count_ones();
105            }
106            match (mismatch == 0, sign_bits & 1 == 0) {
107                (false, _) => Complex64::new(0.0, 0.0),
108                (true, true) => coeff,
109                (true, false) => -coeff,
110            }
111        })
112    }
113}
