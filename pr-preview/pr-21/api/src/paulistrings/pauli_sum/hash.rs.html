1//! The GF(2)-linear bucket hash [`Gf2Hash`] and the partition rows [`PartitionRows`] (ARCHITECTURE.md §Hash, §Partitioning).

3use crate::pauli_string::word_mask;
4use crate::rng::{mix64, SPLITMIX_GAMMA};

6/// Maximum number of bucket bits; all rows are drawn up front so the active hash is a prefix of one fixed matrix.
7pub const B_MAX_BITS: u8 = 22;
8
9/// Maximum number of partition bits, i.e. `P ≤ 64` partitions.
10pub const P_MAX_BITS: u8 = 6;
11
12/// Salt keeping [`PartitionRows::from_seed`]'s rows independent of [`Gf2Hash::new`]'s under the same seed.
13const PARTITION_ROW_SALT: u64 = 0xD1B5_4A32_D192_ED03;
14
15/// Word `word` of the `half` (0 = x, 1 = z) of row `row`, draw `attempt`.
16// Not a GF(2)-linear generator such as xorshift: research/FINDINGS.md §`Gf2Hash` rows are splitmix64, not xorshift successors.
17fn row_word(seed: u64, row: usize, attempt: u32, word: usize, half: u64) -> u64 {
18    debug_assert!(row < 1 << 16 && word < 1 << 15);
19    let position = ((attempt as u64) << 32) | ((row as u64) << 16) | ((word as u64) << 1) | half;
20    mix64(seed.wrapping_add(SPLITMIX_GAMMA.wrapping_mul(position.wrapping_add(1))))
21}

23/// A GF(2) matrix over the key space `(x, z)`, each row masked to the live qubit columns; `Gf2Hash` and `PartitionRows` are its two uses.
24#[derive(Clone, Debug, PartialEq, Eq)]
25pub(crate) struct Gf2Matrix<const W: usize> {
26    rows_x: Vec<[u64; W]>,
27    rows_z: Vec<[u64; W]>,
28    num_qubits: usize,
29}
30
31impl<const W: usize> Gf2Matrix<W> {
32    /// `num_rows` rows drawn from `seed` with the `exclude` coordinates cleared, an all-zero row redrawn unless `num_qubits == 0`.
33    pub(crate) fn draw(
34        num_qubits: usize,
35        num_rows: usize,
36        seed: u64,
37        exclude_x: &[u64; W],
38        exclude_z: &[u64; W],
39    ) -> Self {
40        let mut rows_x: Vec<[u64; W]> = Vec::with_capacity(num_rows);
41        let mut rows_z: Vec<[u64; W]> = Vec::with_capacity(num_rows);
42        let has_live_columns = num_qubits > 0;
43        for row in 0..num_rows {
44            let mut attempt = 0u32;
45            let (row_x, row_z) = loop {
46                let mut row_x = [0u64; W];
47                let mut row_z = [0u64; W];
48                let mut any = false;
49                for w in 0..W {
50                    let mask = word_mask(num_qubits, w);
51                    row_x[w] = row_word(seed, row, attempt, w, 0) & mask & !exclude_x[w];
52                    row_z[w] = row_word(seed, row, attempt, w, 1) & mask & !exclude_z[w];
53                    any |= (row_x[w] | row_z[w]) != 0;
54                }
55                if any || !has_live_columns {
56                    break (row_x, row_z);
57                }
58                attempt += 1;
59            };
60            rows_x.push(row_x);
61            rows_z.push(row_z);
62        }
63        Self {
64            rows_x,
65            rows_z,
66            num_qubits,
67        }
68    }
69
70    /// Explicit rows, masked to the live columns.
71    pub(crate) fn from_rows(
72        num_qubits: usize,
73        mut rows_x: Vec<[u64; W]>,
74        mut rows_z: Vec<[u64; W]>,
75    ) -> Self {
76        assert_eq!(rows_x.len(), rows_z.len(), "Gf2Matrix: row count mismatch");
77        for (row_x, row_z) in rows_x.iter_mut().zip(rows_z.iter_mut()) {
78            for w in 0..W {
79                let mask = word_mask(num_qubits, w);
80                row_x[w] &= mask;
81                row_z[w] &= mask;
82            }
83        }
84        Self {
85            rows_x,
86            rows_z,
87            num_qubits,
88        }
89    }
90
91    pub(crate) fn num_rows(&self) -> usize {
92        self.rows_x.len()
93    }
94
95    pub(crate) fn num_qubits(&self) -> usize {
96        self.num_qubits
97    }
98
99    pub(crate) fn row(&self, i: usize) -> KeyRow<W> {
100        (self.rows_x[i], self.rows_z[i])
101    }
102
103    pub(crate) fn rows(&self) -> (&[[u64; W]], &[[u64; W]]) {
104        (&self.rows_x, &self.rows_z)
105    }
106
107    /// Bit `row` of `M·v`, as `0` or `1`.
108    pub(crate) fn row_parity(&self, x: &[u64; W], z: &[u64; W], row: usize) -> u32 {
109        let row_x = &self.rows_x[row];
110        let row_z = &self.rows_z[row];
111        let mut folded: u64 = 0;
112        for w in 0..W {
113            folded ^= (x[w] & row_x[w]) ^ (z[w] & row_z[w]);
114        }
115        folded.count_ones() & 1
116    }
117
118    /// The first `rows` bits of `M·v`, row `i` at bit `i`.
119    pub(crate) fn apply(&self, x: &[u64; W], z: &[u64; W], rows: usize) -> u32 {
120        let mut index: u32 = 0;
121        for i in 0..rows {
122            index |= self.row_parity(x, z, i) << i;
123        }
124        index
125    }
126
127    /// Whether this matrix's first `rows` rows and `other`'s first `other_rows` are jointly GF(2)-independent.
128    pub(crate) fn independent_with(&self, rows: usize, other: &Self, other_rows: usize) -> bool {
129        let stacked: Vec<KeyRow<W>> = (0..rows)
130            .map(|i| self.row(i))
131            .chain((0..other_rows).map(|i| other.row(i)))
132            .collect();
133        gf2_rank_wide(&stacked) == rows + other_rows
134    }
135}

137/// The GF(2)-linear hash `h(v) = H·v` from a Pauli key `v = (x, z)` to a bucket index, for a dense random `H` drawn from a seed.
138///
139/// Linearity gives `h(v ⊕ d) = h(v) ⊕ h(d)`, so a channel's output buckets are predictable from its input buckets.
140#[derive(Clone, Debug)]
141pub struct Gf2Hash<const W: usize> {
142    /// All [`B_MAX_BITS`] rows of `H`.
143    matrix: Gf2Matrix<W>,
144    /// Active prefix length: `B = 1 << bits` buckets.
145    bits: u8,
146    /// Seed the rows were generated from.
147    seed: u64,
148}
149
150impl<const W: usize> Gf2Hash<W> {
151    /// Build a hash over `num_qubits` qubits with `bits` active bucket bits.
152    ///
153    /// The rows are a function of `(num_qubits, seed)` alone, independent of `W`.
154    ///
155    /// # Panics
156    ///
157    /// Panics if `bits > B_MAX_BITS`, or in debug builds if `num_qubits > 64 · W`.
158    pub fn new(num_qubits: usize, bits: u8, seed: u64) -> Self {
159        assert!(
160            bits <= B_MAX_BITS,
161            "Gf2Hash: bits {bits} exceeds B_MAX_BITS {B_MAX_BITS}",
162        );
163        debug_assert!(num_qubits <= 64 * W);
164
165        Self {
166            matrix: Gf2Matrix::draw(num_qubits, B_MAX_BITS as usize, seed, &[0; W], &[0; W]),
167            bits,
168            seed,
169        }
170    }
171
172    /// Number of active bucket bits.
173    pub fn bits(&self) -> u8 {
174        self.bits
175    }
176
177    /// Number of buckets, `1 << bits()`.
178    pub fn num_buckets(&self) -> usize {
179        1usize << self.bits
180    }
181
182    /// The seed the rows were generated from.
183    pub fn seed(&self) -> u64 {
184        self.seed
185    }
186
187    /// The qubit count the rows were masked against.
188    pub fn num_qubits(&self) -> usize {
189        self.matrix.num_qubits()
190    }
191
192    /// `h(v)` for a key given as separate `x` and `z` words.
193    pub fn bucket_of(&self, x: &[u64; W], z: &[u64; W]) -> u32 {
194        self.matrix.apply(x, z, self.bits as usize)
195    }
196
197    /// Bit `row` of `H·v`, as `0` or `1`.
198    pub(super) fn row_parity(&self, x: &[u64; W], z: &[u64; W], row: u8) -> u32 {
199        self.matrix.row_parity(x, z, row as usize)
200    }
201
202    /// Double the bucket count: `B → 2B`.
203    ///
204    /// # Panics
205    ///
206    /// Panics if already at `B_MAX_BITS`.
207    pub fn refine(&mut self) {
208        assert!(
209            self.bits < B_MAX_BITS,
210            "Gf2Hash::refine: already at B_MAX_BITS {B_MAX_BITS}",
211        );
212        self.bits += 1;
213    }
214
215    /// Halve the bucket count: `B → B/2`, merging buckets `b` and `b + B/2`.
216    ///
217    /// # Panics
218    ///
219    /// Panics if already at a single bucket.
220    pub fn coarsen(&mut self) {
221        assert!(
222            self.bits > 0,
223            "Gf2Hash::coarsen: already at a single bucket"
224        );
225        self.bits -= 1;
226    }
227
228    /// `true` if `other` was generated with the same rows, so sums partitioned by the two can be combined (after matching `bits`).
229    pub(crate) fn same_rows_as(&self, other: &Self) -> bool {
230        self.seed == other.seed && self.num_qubits() == other.num_qubits()
231    }
232
233    /// Row `i < B_MAX_BITS` of `H`, which may lie beyond the active prefix.
234    #[cfg(any(test, feature = "cuda"))]
235    pub(crate) fn row(&self, i: usize) -> ([u64; W], [u64; W]) {
236        self.matrix.row(i)
237    }
238}

240/// The `p` GF(2) rows `part(v) = P·v` that split a sum across `2^p` partitions, fixed while the bucket hash refines (ARCHITECTURE.md §Partitioning).
241#[derive(Clone, Debug, PartialEq, Eq)]
242pub struct PartitionRows<const W: usize> {
243    /// One row per partition bit: `P = 1 << rows` partitions.
244    matrix: Gf2Matrix<W>,
245}
246
247impl<const W: usize> PartitionRows<W> {
248    /// Panics unless the rows split a sum over `num_qubits` qubits into `partitions` partitions; debug builds also check them independent of `hash`.
249    pub(crate) fn assert_splits(&self, hash: &Gf2Hash<W>, num_qubits: usize, partitions: usize) {
250        assert_eq!(
251            self.num_partitions(),
252            partitions,
253            "partition rows name {} partitions for a split into {partitions}",
254            self.num_partitions(),
255        );
256        assert_eq!(
257            self.num_qubits(),
258            num_qubits,
259            "partition rows are for {} qubits, the sum for {num_qubits}",
260            self.num_qubits(),
261        );
262        debug_assert!(
263            self.is_independent_of(hash),
264            "partition rows are dependent on the bucket hash rows — the split will correlate \
265             with the bucket partition and load-balance badly",
266        );
267    }
268
269    /// Draw `bits` partition rows from `seed`, unrelated to [`Gf2Hash::new`]'s rows under the same seed.
270    ///
271    /// # Panics
272    ///
273    /// Panics if `bits > P_MAX_BITS`, or in debug builds if `num_qubits > 64 · W`.
274    pub fn from_seed(num_qubits: usize, bits: u8, seed: u64) -> Self {
275        Self::from_seed_excluding(num_qubits, bits, seed, &[0; W], &[0; W])
276    }
277
278    /// [`Self::from_seed`] with the coordinates in `(exclude_x, exclude_z)` cleared from every row, so keys differing only there share a partition.
279    ///
280    /// [`crate::DistributedSum::rotated_overlap`] needs this with [`crate::RotationAxis::flip_mask`].
281    ///
282    /// # Panics
283    ///
284    /// As [`Self::from_seed`], and if `bits > 0` while the exclusion covers every live column.
285    pub fn from_seed_excluding(
286        num_qubits: usize,
287        bits: u8,
288        seed: u64,
289        exclude_x: &[u64; W],
290        exclude_z: &[u64; W],
291    ) -> Self {
292        assert!(
293            bits <= P_MAX_BITS,
294            "PartitionRows: bits {bits} exceeds P_MAX_BITS {P_MAX_BITS}",
295        );
296        debug_assert!(num_qubits <= 64 * W);
297
298        let keeps_a_column = (0..W).any(|w| {
299            let mask = word_mask(num_qubits, w);
300            (mask & !exclude_x[w]) | (mask & !exclude_z[w]) != 0
301        });
302        assert!(
303            bits == 0 || num_qubits == 0 || keeps_a_column,
304            "PartitionRows::from_seed_excluding: the exclusion covers every column",
305        );
306        Self {
307            matrix: Gf2Matrix::draw(
308                num_qubits,
309                bits as usize,
310                mix64(seed) ^ PARTITION_ROW_SALT,
311                exclude_x,
312                exclude_z,
313            ),
314        }
315    }

317    /// Build from explicit rows, masked to the live qubit columns.
318    ///
319    /// # Panics
320    ///
321    /// Panics if `rows_x` and `rows_z` differ in length, if there are more than [`P_MAX_BITS`] rows, or if any row masks to all-zero while `num_qubits > 0`.
322    pub(crate) fn from_rows(
323        num_qubits: usize,
324        rows_x: Vec<[u64; W]>,
325        rows_z: Vec<[u64; W]>,
326    ) -> Self {
327        assert_eq!(
328            rows_x.len(),
329            rows_z.len(),
330            "PartitionRows::from_rows: row count mismatch, {} x-rows vs {} z-rows",
331            rows_x.len(),
332            rows_z.len(),
333        );
334        assert!(
335            rows_x.len() <= P_MAX_BITS as usize,
336            "PartitionRows: bits {} exceeds P_MAX_BITS {P_MAX_BITS}",
337            rows_x.len(),
338        );
339        debug_assert!(num_qubits <= 64 * W);
340
341        let matrix = Gf2Matrix::from_rows(num_qubits, rows_x, rows_z);
342        for i in 0..matrix.num_rows() {
343            let (row_x, row_z) = matrix.row(i);
344            assert!(
345                num_qubits == 0 || row_x.iter().chain(&row_z).any(|&w| w != 0),
346                "PartitionRows::from_rows: row {i} masks to zero",
347            );
348        }
349        Self { matrix }
350    }

352    /// Rows that label a qubit cut: `log2(blocks.len())` z-only rows giving block `b` the label `b`.
353    ///
354    /// A term's label is the XOR of the labels of the blocks it has odd z-weight in, so an x-only generator is local under any cut and a `ZZ(i, j)` bond is remote exactly when it crosses blocks (ARCHITECTURE.md §Partitioning).
355    /// Blocks must be disjoint; qubits in no block behave as if in block 0.
356    ///
357    /// # Panics
358    ///
359    /// Panics if `blocks.len()` is not a power of two or exceeds `2^P_MAX_BITS`, if a qubit is `>= num_qubits` or appears in two blocks, or if some row would be all-zero.
360    pub fn cut(num_qubits: usize, blocks: &[Vec<usize>]) -> Self {
361        assert!(
362            blocks.len().is_power_of_two(),
363            "PartitionRows::cut: block count {} is not a power of two",
364            blocks.len(),
365        );
366        let bits = blocks.len().trailing_zeros() as u8;
367        assert!(
368            bits <= P_MAX_BITS,
369            "PartitionRows: bits {bits} exceeds P_MAX_BITS {P_MAX_BITS}",
370        );
371        debug_assert!(num_qubits <= 64 * W);
372
373        let mut seen = vec![false; num_qubits];
374        let mut rows_z = vec![[0u64; W]; bits as usize];
375        for (b, qubits) in blocks.iter().enumerate() {
376            for &q in qubits {
377                assert!(
378                    q < num_qubits,
379                    "PartitionRows::cut: qubit {q} in block {b} is outside 0..{num_qubits}",
380                );
381                assert!(
382                    !seen[q],
383                    "PartitionRows::cut: blocks must be disjoint, qubit {q} appears twice",
384                );
385                seen[q] = true;
386                for (i, row) in rows_z.iter_mut().enumerate() {
387                    if (b >> i) & 1 == 1 {
388                        row[q / 64] |= 1u64 << (q % 64);
389                    }
390                }
391            }
392        }
393        for (i, row) in rows_z.iter().enumerate() {
394            assert!(
395                row.iter().any(|w| *w != 0),
396                "PartitionRows::cut: row {i} is empty — no qubit lies in a block \
397                 whose label has bit {i} set",
398            );
399        }
400
401        Self::from_rows(num_qubits, vec![[0u64; W]; bits as usize], rows_z)
402    }

404    /// The trivial partitioning: one partition, no rows.
405    pub fn none(num_qubits: usize) -> Self {
406        Self {
407            matrix: Gf2Matrix::from_rows(num_qubits, Vec::new(), Vec::new()),
408        }

413        self.matrix.num_rows() as u8

421    /// The qubit count the rows were masked against.
422    pub fn num_qubits(&self) -> usize {
423        self.matrix.num_qubits()
424    }

426    /// `part(v)` for a key given as separate `x` and `z` words.
427    pub fn partition_of(&self, x: &[u64; W], z: &[u64; W]) -> u32 {
428        self.matrix.apply(x, z, self.matrix.num_rows())
429    }

431    /// The rows as `(x-masks, z-masks)`, already masked to the live columns.
432    pub fn rows(&self) -> (&[[u64; W]], &[[u64; W]]) {
433        self.matrix.rows()
434    }

436    /// `true` if no row reads a coordinate in `(mask_x, mask_z)`, i.e. keys differing only there always share a partition.
437    pub fn avoids(&self, mask_x: &[u64; W], mask_z: &[u64; W]) -> bool {
438        let (rows_x, rows_z) = self.matrix.rows();
439        rows_x.iter().zip(rows_z).all(|(row_x, row_z)| {
440            (0..W).all(|w| row_x[w] & mask_x[w] == 0 && row_z[w] & mask_z[w] == 0)
441        })
442    }
443
444    /// `true` if the partition rows and `hash`'s active rows are jointly GF(2)-independent.
445    pub fn is_independent_of(&self, hash: &Gf2Hash<W>) -> bool {
446        self.matrix
447            .independent_with(self.matrix.num_rows(), &hash.matrix, hash.bits() as usize)
448    }

451/// One row of the key space as `(x-masks, z-masks)`.
452type KeyRow<const W: usize> = ([u64; W], [u64; W]);
453
454/// GF(2) rank of rows over the key space, by Gaussian elimination.
455fn gf2_rank_wide<const W: usize>(rows: &[KeyRow<W>]) -> usize {
456    let mut pivots: Vec<(usize, KeyRow<W>)> = Vec::with_capacity(rows.len());
457    for row in rows {
458        let mut reduced = *row;
459        'reduce: loop {
460            let Some(lead) = leading_column(&reduced) else {
461                break;
462            };
463            for (pivot_column, pivot_row) in pivots.iter() {
464                if *pivot_column == lead {
465                    for w in 0..W {
466                        reduced.0[w] ^= pivot_row.0[w];
467                        reduced.1[w] ^= pivot_row.1[w];
468                    }
469                    continue 'reduce;
470                }
471            }
472            pivots.push((lead, reduced));
473            break;
474        }
475    }
476    pivots.len()
477}

479/// Index of the highest set column of a key-space row, or `None` if it is zero.
480fn leading_column<const W: usize>(row: &KeyRow<W>) -> Option<usize> {
481    for w in (0..W).rev() {
482        if row.1[w] != 0 {
483            return Some(W * 64 + w * 64 + (63 - row.1[w].leading_zeros() as usize));
484        }
485    }
486    for w in (0..W).rev() {
487        if row.0[w] != 0 {
488            return Some(w * 64 + (63 - row.0[w].leading_zeros() as usize));
489        }
490    }
491    None
492}
